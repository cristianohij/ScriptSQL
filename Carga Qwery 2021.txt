		'1' id,

		PROUM1,

	from
		TBS010

		PROCOD,
		PROUM4,
		PROUM4QTD
	from
		TBS010) as a left join
	(select
			'1' id,
			TDPPROCOD,
			case when TDPPROLOJ = 'S' and convert(datetime,convert(char,getdate(),103)) between TDPVALPROI and TDPVALPROF then TDPPREPRO1 else TDPPRELOJ1 end TDPPRELOJ
		from
			TBS031
	union
		select
			'2' id,
			TDPPROCOD,
			case when TDPPROLOJ = 'S' and convert(datetime,convert(char,getdate(),103)) between TDPVALPROI and TDPVALPROF then TDPPREPRO2 else TDPPRELOJ2 end TDPPRELOJ
		from
			TBS031

			'3' id,
			TDPPROCOD,
			case when TDPPROLOJ = 'S' and convert(datetime,convert(char,getdate(),103)) between TDPVALPROI and TDPVALPROF then TDPPREPRO3 else TDPPRELOJ3 end TDPPRELOJ
		from
			TBS031
	union
		select
			'4' id,
			TDPPROCOD,
			case when TDPPROLOJ = 'S' and convert(datetime,convert(char,getdate(),103)) between TDPVALPROI and TDPVALPROF then TDPPREPRO4 else TDPPRELOJ4 end TDPPRELOJ
		from
			TBS031) as b on PROCOD = TDPPROCOD and a.id = b.id left join
	TBS0103 on PROCOD = CBPPROCOD and PROUM1QTD = CBPQTDEMB
where
	PROUM1QTD <> 0 and
	TDPPRELOJ is not null
