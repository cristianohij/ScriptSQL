select PRPCLICOD AS COD_DO_CLIENTE,PRPSIT AS PENDENTE,PRPNUM AS PEDIDO,
PROCOD AS COD_PRODUTO,PRPQTD AS QTD_PRODUTO,PRPUNI AS UNID,PRPVENCOD AS VENDEDOR,
PRPPRODES AS DESCRICAO,PRPMARNOM AS MARCA,PRPDATREG AS DATA
FROM TBS058
WHERE PRPSIT IN ('P') AND PRPDATREG BETWEEN '20150101' AND '20150625'
ORDER BY PRPDATREG

select PRPCLICOD AS COD_DO_CLIENTE,
PRPSIT AS PENDENTE,
rtrim(ltrim(PROCOD)) AS COD_PRODUTO,
PRPNUM AS PEDIDO,
replace(str(isnull(PRPQTD,0),10,4),'.',',') AS QTD_PRODUTO,
PRPUNI AS UNID,
PRPVENCOD AS VENDEDOR,PRPPRODES AS DESCRICAO,PRPMARNOM AS MARCA,PRPDATREG AS DATA
FROM TBS058
WHERE PRPSIT IN ('R') 
ORDER BY PRPDATREG
