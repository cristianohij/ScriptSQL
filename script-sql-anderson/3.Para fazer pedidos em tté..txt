select
PROCOD collate Latin1_General_CI_AS as 'Código', 
PROLOCFIS collate Latin1_General_CI_AS as 'Localização', 
PROSTATUS  collate Latin1_General_CI_AS as 'Status', 
rtrim(ltrim(A.PROCOD))  collate Latin1_General_CI_AS as 'Código', 
PRODES  collate Latin1_General_CI_AS as 'Descrição', 
rtrim(MARCOD) + ' - ' + rtrim(MARNOM) collate Latin1_General_CI_AS as 'Nome Marca', 
PROUM1  collate Latin1_General_CI_AS as 'Unid. Medida',
replace(str(isnull(SaldoEstCD.SDISEST,0),10,4),'.',',') collate Latin1_General_CI_AS as 'Saldo Disponivel Estoque CD', 
replace(str(isnull(SaldoEstCD.SRESEST,0),10,4),'.',',')  collate Latin1_General_CI_AS as 'Saldo Reservado Estoque CD',
replace(str(isnull(SaldoEsttt.SDISEST,0),10,4),'.',',') collate Latin1_General_CI_AS as 'Saldo Disponivel Estoque ND',
replace(str(isnull(SaldoEsttt.SRESEST,0),10,4),'.',',') collate Latin1_General_CI_AS as 'Saldo Reservado Estoque ND',
replace(str(isnull(SDISLOJA,0),10,4),'.',',')  collate Latin1_General_CI_AS as 'Saldo Disponivel Loja'
from tt.SIBD.dbo.TBS010 P

left join (select A.PROCOD  collate Latin1_General_CI_AS as PRO, str(sum(A.ESTQTDATU-A.ESTQTDRES))  collate Latin1_General_CI_AS as SDISLOJA, str(A.ESTQTDRES)  collate Latin1_General_CI_AS as SRESLOJA
from tt.SIBD.dbo.TBS032 A join tt.SIBD.dbo.TBS010 B on A.PROCOD = B.PROCOD
where A.ESTLOC in (2) and (A.ESTQTDATU <> 0 or A.ESTQTDRES <> 0 or A.ESTQTDPEN <> 0 or A.ESTQTDCMP <> 0)
group by A.PROCOD, A.ESTQTDRES) as SaldoLoja on P.PROCOD = SaldoLoja.PRO

left join (select A.PROCOD  collate Latin1_General_CI_AS as PRO, str(sum(A.ESTQTDATU-A.ESTQTDRES)) collate Latin1_General_CI_AS as SDISEST, str(A.ESTQTDRES)  collate Latin1_General_CI_AS as SRESEST
from tt.SIBD.dbo.TBS032 A join tt.SIBD.dbo.TBS010 B on A.PROCOD = B.PROCOD
where A.ESTLOC in (1) and (A.ESTQTDATU <> 0 or A.ESTQTDRES <> 0 or A.ESTQTDPEN <> 0 or A.ESTQTDCMP <> 0)
group by A.PROCOD, A.ESTQTDRES) as SaldoEsttt on P.PROCOD = SaldoEsttt.PRO

left join (select A.PROCOD  collate Latin1_General_CI_AS as PRO, str(sum(A.ESTQTDATU-A.ESTQTDRES)) collate Latin1_General_CI_AS as SDISEST, str(A.ESTQTDRES)  collate Latin1_General_CI_AS as SRESEST
from cd.SIBD.dbo.TBS032 A join cd.SIBD.dbo.TBS010 B on A.PROCOD = B.PROCOD
where A.ESTLOC in (1) and (A.ESTQTDATU <> 0 or A.ESTQTDRES <> 0 or A.ESTQTDPEN <> 0 or A.ESTQTDCMP <> 0)
group by A.PROCOD, A.ESTQTDRES) as SaldoEstCD on P.PROCOD = SaldoEstCD.PRO
where SDISLOJA <> 0 or SaldoEsttt.SDISEST <> 0 or SaldoEstCD.SDISEST <> 0
order by PROCOD
