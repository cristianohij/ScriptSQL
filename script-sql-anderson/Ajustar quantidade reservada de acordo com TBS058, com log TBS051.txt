SELECT 
ISNULL(A.PROCOD,B.PROCOD) AS PROCOD,
ISNULL(A.ESTLOC,B.PRPESTLOC) AS ESTLOC,
'R' AS INFALT,
CASE WHEN ISNULL(QTD,0) > ISNULL(ESTQTDRES,0) THEN 'E' ELSE 'S' END ACAO,
CASE WHEN ISNULL(QTD,0) > ISNULL(ESTQTDRES,0) THEN ISNULL(QTD,0) - ISNULL(ESTQTDRES,0) ELSE ISNULL(ESTQTDRES,0) - ISNULL(QTD,0) END AS MOV,
ISNULL(QTD,0) AS QTD,
ISNULL(ESTQTDRES,0) AS ESTQTDRES,
ISNULL(ESTQTDATU,0) AS ESTQTDATU,
ISNULL(ESTQTDCMP,0) AS ESTQTDCMP,
ISNULL(ESTQTDPEN,0) AS ESTQTDPEN,
ISNULL(ESTQTDDIS,0) AS ESTQTDDIS,
ISNULL(PROUM1,0) AS PROUM1

INTO #TBS051
FROM #TBS032 A 
FULL OUTER JOIN #TBS058 B ON A.PROCOD = B.PROCOD AND A.ESTLOC = B.PRPESTLOC
WHERE 
ISNULL(ESTQTDRES,0) <> ISNULL(QTD,0)
