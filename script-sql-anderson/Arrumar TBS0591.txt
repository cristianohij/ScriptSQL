DECLARE @Data_de DATE , @Data_ate DATE
SET @Data_de = '20160801'
SET @Data_ate = GETDATE()
select B.NFECHAACE, B.NFEDATEMI,B.NFETIPENT , A.NFECALC,A.NFEFATOR, A.*  --,A.NFECALCULO,
FROM TBS0591 A (NOLOCK)
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND  A.SEREMPCOD = B.SEREMPCOD
WHERE --B.NFENUM = 1507489
(B.NFECAN = 'N' OR B.NFECAN = '') AND (B.NFEDATENT BETWEEN @Data_de AND @Data_ate) --AND A.NFENUM = 2853
