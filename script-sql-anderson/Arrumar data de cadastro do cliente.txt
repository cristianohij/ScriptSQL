FROM TBS002 (NOLOCK) 
WHERE 
CLICOD IN (SELECT CLICOD FROM TBS002 (NOLOCK) WHERE CLIDATCAD = '17530101' OR CLIDATCAD IS NULL
			UNION 
			SELECT CLICOD +1 FROM TBS002 (NOLOCK) WHERE CLIDATCAD = '17530101' OR CLIDATCAD IS NULL
			UNION 
			SELECT CLICOD - 1 FROM TBS002 (NOLOCK) WHERE CLIDATCAD = '17530101' OR CLIDATCAD IS NULL)
			
ORDER BY CLICOD
