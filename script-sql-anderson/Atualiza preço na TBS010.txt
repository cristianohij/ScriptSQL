BEGIN TRAN 

UPDATE TBS010 SET
PROPREUM1COR = TDPPRECOR1, 
PROPREUM2COR = TDPPRECOR2, 
PROPREUM1LOJ = TDPPRELOJ1, 
PROPREUM2LOJ = TDPPRELOJ2

FROM TBS010 A (NOLOCK)
INNER JOIN LOGTBS031 B (NOLOCK) ON A.PROCOD COLLATE DATABASE_DEFAULT = B.TDPPROCOD

WHERE 
TDPDAT > CONVERT(DATE,GETDATE()) AND (
TDPPRECOR1 <> PROPREUM1COR OR 
TDPPRECOR2 <> PROPREUM2COR OR 
TDPPRELOJ1 <> PROPREUM1LOJ OR 
TDPPRELOJ2 <> PROPREUM2LOJ ) 

COMMIT TRAN
