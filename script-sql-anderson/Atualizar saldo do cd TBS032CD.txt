-- 1° CRIE UMA TABELA DE SALDO
SELECT 
ESTLOC,
PROCOD,
ESTQTDATU,
ESTQTDRES,
ESTQTDCMP,
ESTQTDPEN,
ESTDATALT,
GETDATE () AS DATULTALT
INTO TBS032CD  
FROM TBS032 
WHERE (ESTQTDATU <> 0 OR ESTQTDPEN <> 0 OR ESTQTDCMP <> 0) AND ESTLOC IN (1,2)





-- 2° FAÇO A ATUALIZAÇÃO EM 15 MIN, ATRAVES DE UM TRABALHO NO SQL AGENT

MERGE TBS032CD  AS A -- DESTINO
USING TBS032 AS B -- ORIGEM 

ON A.PROCOD = B.PROCOD

WHEN MATCHED AND A.ESTDATALT <> B.ESTDATALT AND B.ESTLOC = A.ESTLOC
	
	THEN UPDATE SET
	A.ESTLOC = B.ESTLOC,
	A.ESTQTDATU = B.ESTQTDATU,
	A.ESTQTDRES = B.ESTQTDRES,
	A.ESTQTDCMP = B.ESTQTDCMP,
	A.ESTQTDPEN = B.ESTQTDPEN,
	A.ESTDATALT = B.ESTDATALT
	

-- Quando não há registro no destino e há na origem
	
	WHEN NOT MATCHED AND (B.ESTQTDATU <> 0 OR B.ESTQTDPEN <> 0 OR B.ESTQTDCMP <> 0) AND B.ESTLOC IN (1,2)
	
	THEN INSERT(
	ESTLOC,
	PROCOD,
	ESTQTDATU,
	ESTQTDRES,
	ESTQTDCMP,
	ESTQTDPEN,
	ESTDATALT)
	
	VALUES(
	B.ESTLOC,
	B.PROCOD,
	B.ESTQTDATU,
	B.ESTQTDRES,
	B.ESTQTDCMP,
	B.ESTQTDPEN,
	B.ESTDATALT)
	 
	 -- NAO TEM NA ORIGEM , MAS TEM NO DESTINO 
	 WHEN NOT MATCHED BY SOURCE 
	 THEN
	 DELETE ;
 
	UPDATE TBS032CD
	SET DATULTALT = GETDATE()
