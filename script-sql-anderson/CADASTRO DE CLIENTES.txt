SET @dataDe = '20080101'

WITH 
CADASTRO AS(SELECT 
			RTRIM(LTRIM(CONVERT(CHAR (10) ,CLIDATCAD,103))) AS 'DATA DE CADASTRO',
			CASE WHEN A.CLITIPPES = 'J' 
				THEN 'JURIDICA' 
				ELSE 'FISICA'
				END  AS PESSOA,
			RTRIM(LTRIM(A.CLICGC)) AS CNPJ,
			RTRIM(LTRIM(A.CLICPF)) AS CPF,
			RTRIM(LTRIM(CLISIT)) AS SITUACAO,
			A.CLICOD AS CLIENTE,
			RTRIM(LTRIM(CLINOM)) AS 'RAZÃO SOCIAL',
			RTRIM(LTRIM(CLINOMFAN)) AS 'FANTASIA',
			'FATURAMENTO' AS ENDER,
			isnull(RTRIM(LTRIM(A.CLIEND)),'') + ' , '+ RTRIM(LTRIM(A.CLINUM)) + ' , '+ ISNULL(RTRIM(LTRIM(A.CLIBAI)),'') AS 'CLIEND',
			RTRIM(LTRIM(A.CLICEP)) AS 'CEP',
			ISNULL(RTRIM(LTRIM(C.MUNNOM)),'') + ' , '+ ISNULL(RTRIM(LTRIM(C.UFESIG)),'')  AS 'CLIMUN',
			RTRIM(LTRIM(CLICONTAT)) AS CONTATO	,
			RTRIM(LTRIM(CLITEL)) AS CLITEL,
			RTRIM(LTRIM(CLIRAM1)) AS RAM1,
			RTRIM(LTRIM(CLITEL2)) AS CLITEL2,
			RTRIM(LTRIM(CLIRAM2)) AS RAM2,
			RTRIM(LTRIM(CLITEL3)) AS CLITEL3,
			RTRIM(LTRIM(CLIRAM3)) AS RAM3,
			RTRIM(LTRIM(CLIEMAIL)) AS EMAIL,
			
			RTRIM(LTRIM(ISNULL(B.VENCOD,0))) AS 'COD VEN.' ,
			RTRIM(LTRIM(ISNULL(B.VENNOM,'SEM VENDEDOR'))) AS 'NOME VEN.'
			
			FROM TBS002 A (NOLOCK) 
			LEFT JOIN TBS004 B (NOLOCK) ON A.VENCOD = B.VENCOD
			LEFT JOIN TBS003 C (NOLOCK) ON A.MUNCOD = C.MUNCOD
						
			WHERE  
			CLIDATCAD BETWEEN @dataDe AND @dataAte
			AND A.CLICOD =(CASE WHEN @codcli=0 THEN A.CLICOD ELSE RTRIM(@codcli) END)
			AND CLINOM LIKE(RTRIM(LTRIM(UPPER(@nomcli)))+'%')
			AND ISNULL(B.VENCOD,0) =(CASE WHEN @codven=0 THEN ISNULL(B.VENCOD,0) ELSE RTRIM(@codven) END)
			AND ISNULL(VENNOM,0) LIKE(RTRIM(LTRIM(UPPER(@nomven)))+'%')
			
			UNION 
			
			SELECT 
			RTRIM(LTRIM(CONVERT(CHAR (10) ,CLIDATCAD,103))) AS 'DATA DE CADASTRO',
			CASE WHEN A.CLIENDTIPPES = 'J' 
				THEN 'JURIDICA' 
				ELSE 'FISICA'
				END  AS PESSOA,
			RTRIM(LTRIM(A.CLIENDCGC)) AS CNPJ,
			RTRIM(LTRIM(A.CLIENDCPF)) AS CPF,
			RTRIM(LTRIM(CLISIT)) AS SITUACAO,
			A.CLICOD AS CLIENTE,
			RTRIM(LTRIM(CLINOM)) AS 'RAZÃO SOCIAL',
			RTRIM(LTRIM(CLINOMFAN)) AS 'FANTASIA',
			'ENTREGA ' + RTRIM(ltrim(A.CLIENDCOD)) AS ENDER,
			isnull(RTRIM(LTRIM(A.CLILOG)),'') + ' , '+ RTRIM(LTRIM(A.CLIENDNUM))+ ' , '+ RTRIM(LTRIM(A.CLIENDCPL)) + ' , '+ ISNULL(RTRIM(LTRIM(A.CLIENDBAI)),'')  AS 'CLIEND',
			RTRIM(LTRIM(A.CLIENDCEP)) AS 'CEP',
			ISNULL(RTRIM(LTRIM(C.MUNNOM)),'') + ' , '+ ISNULL(RTRIM(LTRIM(C.UFESIG)),'')  AS 'CLIMUN',

			RTRIM(LTRIM(CLITEL2)) AS CLITEL2,
			RTRIM(LTRIM(CLIRAM2)) AS RAM2,
			RTRIM(LTRIM(CLITEL3)) AS CLITEL3,
			RTRIM(LTRIM(CLIRAM3)) AS RAM3,
			RTRIM(LTRIM(CLIEMAIL)) AS EMAIL,
			
			RTRIM(LTRIM(ISNULL(B.VENCOD,0))) AS 'COD VEN.' ,
			RTRIM(LTRIM(ISNULL(B.VENNOM,'SEM VENDEDOR'))) AS 'NOME VEN.'
			
			FROM TBS0021 A (NOLOCK) 
			LEFT JOIN TBS002 D (NOLOCK) ON A.CLICOD = D.CLICOD
			LEFT JOIN TBS004 B (NOLOCK) ON D.VENCOD = B.VENCOD
			LEFT JOIN TBS003 C (NOLOCK) ON A.CLIENDMUNCOD = C.MUNCOD
			
						
			WHERE  
			CLIDATCAD BETWEEN @dataDe AND @dataAte
			AND A.CLICOD =(CASE WHEN @codcli=0 THEN A.CLICOD ELSE RTRIM(@codcli) END)
			AND CLINOM LIKE(RTRIM(LTRIM(UPPER(@nomcli)))+'%')
			AND ISNULL(B.VENCOD,0) =(CASE WHEN @codven=0 THEN ISNULL(B.VENCOD,0) ELSE RTRIM(@codven) END)
			AND ISNULL(VENNOM,0) LIKE(RTRIM(LTRIM(UPPER(@nomven)))+'%')
			AND A.CLIENDTIP = 'E'
			
			UNION
			
			SELECT 
			RTRIM(LTRIM(CONVERT(CHAR (10) ,CLIDATCAD,103))) AS 'DATA DE CADASTRO',
			CASE WHEN A.CLIENDTIPPES = 'J' 
				THEN 'JURIDICA' 
				ELSE 'FISICA'
				END  AS PESSOA,
			RTRIM(LTRIM(A.CLIENDCGC)) AS CNPJ,
			RTRIM(LTRIM(A.CLIENDCPF)) AS CPF,
			RTRIM(LTRIM(CLISIT)) AS SITUACAO,
			A.CLICOD AS CLIENTE,
			RTRIM(LTRIM(CLINOM)) AS 'RAZÃO SOCIAL',
			RTRIM(LTRIM(CLINOMFAN)) AS 'FANTASIA',
			'COBRANÇA ' + RTRIM(ltrim(A.CLIENDCOD)) AS ENDER,
			isnull(RTRIM(LTRIM(A.CLILOG)),'') + ' , '+ RTRIM(LTRIM(A.CLIENDNUM))+ ' , '+ RTRIM(LTRIM(A.CLIENDCPL)) + ' , '+ ISNULL(RTRIM(LTRIM(A.CLIENDBAI)),'')  AS 'CLIEND',
			RTRIM(LTRIM(A.CLIENDCEP)) AS 'CEP',
			ISNULL(RTRIM(LTRIM(C.MUNNOM)),'') + ' , '+ ISNULL(RTRIM(LTRIM(C.UFESIG)),'')  AS 'CLIMUN',
			RTRIM(LTRIM(CLICONTAT)) AS CONTATO	,
			RTRIM(LTRIM(CLITEL)) AS CLITEL,
			RTRIM(LTRIM(CLIRAM1)) AS RAM1,
			RTRIM(LTRIM(CLITEL2)) AS CLITEL2,
			RTRIM(LTRIM(CLIRAM2)) AS RAM2,
			RTRIM(LTRIM(CLITEL3)) AS CLITEL3,
			RTRIM(LTRIM(CLIRAM3)) AS RAM3,
			RTRIM(LTRIM(CLIEMAIL)) AS EMAIL,
			
			RTRIM(LTRIM(ISNULL(B.VENCOD,0))) AS 'COD VEN.' ,
			RTRIM(LTRIM(ISNULL(B.VENNOM,'SEM VENDEDOR'))) AS 'NOME VEN.'
			
			FROM TBS0021 A (NOLOCK) 
			LEFT JOIN TBS002 D (NOLOCK) ON A.CLICOD = D.CLICOD
			LEFT JOIN TBS004 B (NOLOCK) ON D.VENCOD = B.VENCOD
			LEFT JOIN TBS003 C (NOLOCK) ON A.CLIENDMUNCOD = C.MUNCOD
			
						
			WHERE  
			CLIDATCAD BETWEEN @dataDe AND @dataAte
			AND A.CLICOD =(CASE WHEN @codcli=0 THEN A.CLICOD ELSE RTRIM(@codcli) END)
			AND CLINOM LIKE(RTRIM(LTRIM(UPPER(@nomcli)))+'%')
			AND ISNULL(B.VENCOD,0) =(CASE WHEN @codven=0 THEN ISNULL(B.VENCOD,0) ELSE RTRIM(@codven) END)
			AND ISNULL(VENNOM,0) LIKE(RTRIM(LTRIM(UPPER(@nomven)))+'%')
			AND A.CLIENDTIP = 'C'
			
			
			)
			
SELECT * from CADASTRO

WHERE ENDER LIKE(CASE WHEN @ender = '' THEN ENDER ELSE RTRIM(LTRIM(UPPER(@ender)))+'%' END)

order by 'DATA DE CADASTRO' DESC
