IF @Opcao = 1 -- TUDO
SELECT  A.NFENUM,B.NFECHAACE,A.NFECFOP AS CFOP,

COUNT(A.PROCOD) AS QTD_ITENS, 
	SUM(dbo.NFETOTITE(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE)) AS VALOR_ENTRADA_CFOP, 
SUM(CASE WHEN B.NFETIP NOT IN('T') OR B.NFENOM NOT LIKE('TANBY%') THEN A.NFEBASICMS ELSE 0 END) AS BASE_ICMS, 
SUM(CASE WHEN B.NFETIP NOT IN('T') OR B.NFENOM NOT LIKE('TANBY%') THEN dbo.NFEVALICMS(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE) ELSE 0 END) AS VALOR_ICMS, 
SUM(CASE WHEN B.NFETIP NOT IN('T') OR B.NFENOM NOT LIKE('TANBY%') THEN A.NFEBASIPI ELSE 0 END) AS BASE_IPI, 
SUM(CASE WHEN B.NFETIP NOT IN('T') OR B.NFENOM NOT LIKE('TANBY%') THEN dbo.NFEVALIPI(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE) ELSE 0 END) AS VALOR_IPI, 
SUM(CASE WHEN A.NFESTBPISE >= '6' 
		THEN dbo.NFETOTITE(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE) 
		ELSE 
			CASE WHEN B.NFETIP IN('T') OR B.NFENOM LIKE('TANBY%') 
				THEN dbo.NFETOTITE(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE) 
				ELSE 0 
				END

SUM(CASE WHEN A.NFESTBPISE <= '5' AND (B.NFETIP NOT IN('T') OR B.NFENOM NOT LIKE('TANBY%')) 
		THEN dbo.NFETOTITE(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE) 
		ELSE 0 END) AS BASE_PIS,

		THEN dbo.NFEVALPISENT(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE) 
		ELSE 0 END) AS VALOR_PIS, 
SUM(CASE WHEN A.NFESTBPISE <= '5' AND  (B.NFETIP NOT IN('T') OR B.NFENOM NOT LIKE('TANBY%')) 
		THEN dbo.NFEVALCOFINSE(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE) 
		ELSE 0 END) AS VALOR_COFINS

FROM 
TBS0591 AS A ( nolock) 
LEFT JOIN
TBS059 AS B ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND A.SEREMPCOD = B.SEREMPCOD
LEFT JOIN 
TBS006 C (NOLOCK) ON A.NFECOD = C.FORCOD

WHERE
(B.NFECAN = 'N' OR B.NFECAN = '') AND 
(B.NFEDATENT BETWEEN @Data_de AND @Data_ate) AND 
(A.NFECFOP =(CASE WHEN @CFOP = '' THEN A.NFECFOP ELSE @CFOP END))

GROUP BY A.NFECFOP, A.NFECST, A.NFENUM, B.NFECHAACE

ORDER BY A.NFENUM
