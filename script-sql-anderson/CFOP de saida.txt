FROM            TBS0671 INNER JOIN TBS010 ON TBS0671.PROCOD =TBS010.PROCOD INNER JOIN

						 TBS042 ON TBS0671.TESEMPCOD = TBS042.TESEMPCOD AND TBS0671.TESCOD = TBS042.TESCOD

WHERE        (TBS067.NFSDATEMI BETWEEN '20150901' AND '20150930') AND (TBS067.NFSTIP = 'N') AND (TBS067.NFSCAN = 'N')

AND (TBS042.TESCNTVEN = 'S')

--ORDER BY TBS0671.NFSCFOP

	FROM TBS1172 A INNER JOIN TBS117 B ON 
		A.NFDEMPCOD = B.NFDEMPCOD AND
		A.NFDNUM = B.NFDNUM AND
		A.SNEEMPCOD = A.SNEEMPCOD AND
		A.SNESER = B.SNESER

GROUP BY A.NFDCFOP,A.NFDCSTCSOSN
