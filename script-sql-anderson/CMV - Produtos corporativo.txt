-- VENDAS DO CORPARATIVO GERAL

if object_id('TempDB.dbo.#NFS') is not null
   begin
      drop table #NFS
   end

SELECT 
PROCOD,
SUM(NFSQTD*NFSQTDEMB) AS QTD,
SUM(round(dbo.NFSTOTITEST(B.NFSEMPCOD,B.NFSNUM,B.SNEEMPCOD,B.SNESER,B.NFSITE),2)) AS NFSTOTITEST
		
INTO #NFS
FROM 
TBS0671 B (nolock)
LEFT JOIN TBS067 A (nolock) ON A.NFSEMPCOD = B.NFSEMPCOD and A.NFSNUM = B.NFSNUM and A.SNEEMPCOD = B.SNEEMPCOD and A.SNESER = B.SNESER
LEFT JOIN TBS080 D (NOLOCK) ON B.NFSNUM = D.ENFNUM AND B.SNEEMPCOD = D.SNEEMPCOD AND B.SNESER = D.SNESER
			  
WHERE 
NFSDATEMI between @datade and @dataate and 
NFSTIP IN('N','C') and 
D.ENFSIT = 6 AND 
D.ENFTIPDOC = 1 AND
A.NFSCLINOM NOT LIKE ('TANBY%') AND 		-- TIRO AS TRANSFERENCIAS
B.NFSCFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S')

GROUP BY 
PROCOD

-----------------------------------------------------------------------------------------------------------------------------------------------------------------------

-- VENDAS DO CORPARATIVO ENTRE O GRUPO

if object_id('TempDB.dbo.#NFSGRU') is not null
   begin
      drop table #NFSGRU
   end

SELECT 
PROCOD,
SUM(NFSQTD*NFSQTDEMB) AS QTD,
SUM(round(dbo.NFSTOTITEST(B.NFSEMPCOD,B.NFSNUM,B.SNEEMPCOD,B.SNESER,B.NFSITE),2)) AS NFSTOTITEST
		
INTO #NFSGRU
FROM 
TBS0671 B (nolock)
LEFT JOIN TBS067 A (nolock) ON A.NFSEMPCOD = B.NFSEMPCOD and A.NFSNUM = B.NFSNUM and A.SNEEMPCOD = B.SNEEMPCOD and A.SNESER = B.SNESER
LEFT JOIN TBS080 D (NOLOCK) ON B.NFSNUM = D.ENFNUM AND B.SNEEMPCOD = D.SNEEMPCOD AND B.SNESER = D.SNESER
			  
WHERE 
NFSDATEMI between @datade and @dataate and 
NFSTIP IN('N','C') and 
D.ENFSIT = 6 AND 
D.ENFTIPDOC = 1 AND
A.NFSCLINOM NOT LIKE ('TANBY%') AND 		-- TIRO AS TRANSFERENCIAS
A.NFSCLICOD IN ((SELECT CLICOD FROM #GRUPOCLI)) AND
B.NFSCFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S')

GROUP BY 
PROCOD

-- NOTAS DE ENTRADA, CONTABILIZAR O CUSTO ND

if object_id('TempDB.dbo.#CUSTO') is not null
   drop table #CUSTO

SELECT
PROCOD,
sum(ROUND(NFETOTOPEITE,2)) / SUM(NFEQTD*NFEQTDEMB) as NFEPREMED

INTO #CUSTO

FROM
TBS0591 A (NOLOCK) 
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND 
A.NFENUM = B.NFENUM AND A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND A.SEREMPCOD = B.SEREMPCOD
		
WHERE 
NFEDATEFE between @datade - 120 and @dataate and -- 4 MESES ANTES DA DATA DE INICIO DAS VENDAS ATÉ A DATA FINAL DAS VENDAS
A.NFETIP = 'N' and 
(NFENOSFOR = 'N' OR NFENOSFOR = '') AND 
(B.NFECAN = 'N' OR B.NFECAN = '') AND 
NFEDATEFE <> '' AND
A.NFECFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E') AND
B.NFECOD NOT IN (SELECT FORCOD FROM #GRUPOFOR) 
		
GROUP BY 
PROCOD

IF OBJECT_ID('TempDB.dbo.#TBS010') IS NOT NULL
DROP TABLE #TBS010

SELECT 
CASE WHEN len(A.MARCOD) = 4 
	then rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) 
	else right(('00' + ltrim(str(A.MARCOD))),3) + ' - ' + rtrim(A.MARNOM) 
end as 'MARCA',
A.PROCOD,
PRODES,
PROUM1,
PROCODBAR1,
PROCODBAR2

INTO #TBS010
FROM TBS010 A (NOLOCK)
WHERE 
A.PROCOD = CASE WHEN @PROCOD = '' THEN A.PROCOD ELSE RTRIM(@PROCOD) END AND
A.MARCOD =(CASE WHEN @marcod = 0 THEN A.MARCOD ELSE Upper(RTRIM(@marcod)) END) AND 
A.MARNOM LIKE(CASE WHEN @marcanom = '' THEN '% %' ELSE Upper(RTRIM(@marcanom))+'%' END) AND 
A.PRODES LIKE (CASE WHEN @PRODES = '' THEN '% %' ELSE UPPER(RTRIM(@PRODES))+'%' END)

-- TABELA FINAL

SELECT  
MARCA,
A.PROCOD,
PRODES,
PROUM1,
ISNULL(B.QTD,0) AS VENQTD,
ISNULL(C.QTD,0) AS VENQTDGRU,
ISNULL(B.NFSTOTITEST,0) AS VENVAL,
ISNULL(C.NFSTOTITEST,0) AS VENVALGRU,

-- ROUND((ISNULL(A.NFSTOTITEST,0) - ISNULL(B.NFSTOTITEST,0))/ CASE WHEN ISNULL(A.QTD,0) -  = 0 THEN 1 ELSE ISNULL(A.QTD,0) END ,4) AS VALMED,
ISNULL(isnull(E.NFEPREMED,F.NFEPREMED),(SELECT TDPCUSREA FROM TBS031 D (NOLOCK) WHERE A.PROCOD = D.TDPPROCOD)) AS CUSTO
-- isnull(NFEPREMED,(SELECT TDPCUSREA FROM TBS031 D (NOLOCK) WHERE A.PROCOD = D.TDPPROCOD)) * ISNULL(QTD,0) as CUSVAL,
-- ISNULL(A.NFSTOTITEST,0) - (isnull(NFEPREMED,(SELECT TDPCUSREA FROM TBS031 D (NOLOCK) WHERE A.PROCOD = D.TDPPROCOD)) * ISNULL(QTD,0)) AS LUCVAL

FROM #TBS010 A (NOLOCK)
LEFT JOIN #NFS B 	ON A.PROCOD = B.PROCOD
LEFT JOIN #NFSGRU C ON A.PROCOD = C.PROCOD
LEFT JOIN #CUSTO E 	ON A.PROCOD = E.PROCOD
LEFT JOIN #CUSTOCD F ON A.PROCOD collate Latin1_General_CI_AS = F.PROCOD

WHERE 
ISNULL(B.QTD,0) <> 0 OR ISNULL(C.QTD,0) <> 0
