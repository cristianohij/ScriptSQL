DECLARE @DataDeE date, @DataAteE date, @TituloDe int, @TituloAte int, @VencDe date, 
@VencAte date, @zero int, @NomFor char(50) , @Portador int, @Opcao int , @codfor int

set @DataDeE = '20150101'
set @DataAteE = '20161231'
set @TituloDe = 0
set @TituloAte = 0
set @VencDe = '20150101'
set @VencAte = '20161231'
set @zero = 0
set @NomFor = '' 
set @Portador = 0
set @Opcao = 1
set @codfor = 0


SELECT PFXCOD AS PFX,
CPATIT AS TITULO,
CPAPAR AS PAR,
PORCOD AS PORT,
CPADATEMI AS DATA_EMI,
CPADATVENREA AS VENC_REAL,
CPADATBAI AS BAIXA,
subString(convert(char(8),CPADATVENREA,112),1,4) AS ANO_VEN_REAL,
subString(convert(char(8),CPADATVENREA,112),5,2) AS MES_VEN_REAL,
subString(convert(char(8),CPADATVENREA,112),7,2) AS DIA_VEN_REAL,
CASE WHEN DATEDIFF(DAY,GETDATE(),CPADATVENREA)<0 OR CPADATBAI > '17530101' THEN 0 ELSE DATEDIFF(DAY,GETDATE(),CPADATVENREA) END AS DIAS_PARA_VENCER,
dbo.CPADIAATR(CPAEMPCOD,PFXEMPCOD,A.FOREMPCOD,PFXCOD,CPATIT,CPAPAR,A.FORCOD) AS DIA_ATRA,
CPAVAL AS VALOR_ORIG,
CPAVALRES AS RESIDUO,
CPAVALABT AS ABATIMENTO,
CPAVALACR AS ACRESCIMO,
CPAVALPAG AS VALOR_PAGO,
dbo.CPAVALSDO(CPAEMPCOD,PFXEMPCOD,A.FOREMPCOD,PFXCOD,CPATIT,CPAPAR,A.FORCOD) as SALDO_ABERTO,
RTRIM(LTRIM(CPAOBS)) AS OBS,
RTRIM(A.FORCOD) AS COD_FOR,
RTRIM(A.FORNOM) AS FORNECEDOR,
RTRIM(LTRIM(FOREND)) +' , '+ RTRIM(LTRIM(FORNUM)) + ' , ' + RTRIM(LTRIM(FORBAI)) AS ENDFOR,
RTRIM(LTRIM(FORCEP)) AS CEPFOR,
RTRIM(LTRIM(ISNULL(C.MUNNOM,''))) AS MUNFOR

--DROP TABLE  #CON
INTO #CON
FROM TBS057 A (NOLOCK) LEFT JOIN TBS006 B (NOLOCK) ON A.FORCOD = B.FORCOD
					   LEFT JOIN TBS003 C (NOLOCK) ON B.MUNCOD = C.MUNCOD

	WHERE 
	CPADATEMI BETWEEN @DataDeE AND CASE WHEN @DataAteE = '' THEN GETDATE() ELSE @DataAteE END AND 
	CPATIT between @TituloDe and case when @TituloAte = 0 then 999999999 else @TituloAte end and
	CPADATVENREA BETWEEN @VencDe and case when @VencAte = '' then GETDATE() else @VencAte end AND 
	PORCOD <= CASE WHEN @zero=0 THEN 999 ELSE 0 END  AND
	A.FORNOM LIKE(RTRIM(CASE WHEN @NomFor = '' THEN A.FORNOM ELSE LTRIM(RTRIM(@NomFor)) +'%' END)) AND
	PORCOD = CASE WHEN @Portador = 0 THEN PORCOD ELSE @Portador END AND 
	B.FORCOD = CASE WHEN @codfor = 0 THEN B.FORCOD ELSE @codfor END 



if @Opcao = 1 --TODOS 
	SELECT * FROM #CON

if @Opcao = 2 -- EM ABERTO
	SELECT * FROM #CON
	WHERE BAIXA = '17530101'
	

if @Opcao = 3 --VENCIDOS
	SELECT * FROM #CON
	WHERE BAIXA = '17530101' AND VENC_REAL < GETDATE()-1


if @Opcao = 4  --PAGOS
	SELECT * FROM #CON
	WHERE BAIXA <> '17530101'
	
if @Opcao = 5 -- PAGOS COM ATRASO
	SELECT * FROM #CON
	WHERE BAIXA <> '17530101' AND BAIXA > VENC_REAL
	
if @Opcao = 6 -- PAGOS ANTES DO VENCIMENTO
	SELECT * FROM #CON
	WHERE BAIXA <> '17530101' AND BAIXA < VENC_REAL
	
if @Opcao = 7 -- PAGOS NO VENCIMENTOS
	SELECT * FROM #CON
	WHERE BAIXA <> '17530101' AND BAIXA = VENC_REAL
	
if @Opcao = 8 -- PAGOS COM SALDO ABERTO
	SELECT * FROM #CON
	WHERE BAIXA <> '17530101' AND SALDO_ABERTO > 0

if object_id('sp_CONTAS_PAG') is not null
   drop procedure sp_CONTAS_PAG
go

create procedure sp_CONTAS_PAG  @DataDeE date, @DataAteE date, @TituloDe int, @TituloAte int, @VencDe date, @VencAte date, @zero int, @NomFor char(50) , 
@Portador int, @Opcao int ,@codfor int

as

begin
  
  
  if object_id('TempDB.dbo.#CONTAS_PAG') is not null
   drop table #CONTAS_PAG
   
SELECT PFXCOD AS PFX,
CPATIT AS TITULO,
CPAPAR AS PAR,
PORCOD AS PORT,
CPADATEMI AS DATA_EMI,
CPADATVENREA AS VENC_REAL,
CPADATBAI AS BAIXA,
subString(convert(char(8),CPADATVENREA,112),1,4) AS ANO_VEN_REAL,
subString(convert(char(8),CPADATVENREA,112),5,2) AS MES_VEN_REAL,
subString(convert(char(8),CPADATVENREA,112),7,2) AS DIA_VEN_REAL,
CASE WHEN DATEDIFF(DAY,GETDATE(),CPADATVENREA)<0 OR CPADATBAI > '17530101' THEN 0 ELSE DATEDIFF(DAY,GETDATE(),CPADATVENREA) END AS DIAS_PARA_VENCER,
dbo.CPADIAATR(CPAEMPCOD,PFXEMPCOD,A.FOREMPCOD,PFXCOD,CPATIT,CPAPAR,A.FORCOD) AS DIA_ATRA,
CPAVAL AS VALOR_ORIG,
CPAVALRES AS RESIDUO,
CPAVALABT AS ABATIMENTO,
CPAVALACR AS ACRESCIMO,
CPAVALPAG AS VALOR_PAGO,
dbo.CPAVALSDO(CPAEMPCOD,PFXEMPCOD,A.FOREMPCOD,PFXCOD,CPATIT,CPAPAR,A.FORCOD) as SALDO_ABERTO,
RTRIM(LTRIM(CPAOBS)) AS OBS,
RTRIM(A.FORCOD) AS COD_FOR,
RTRIM(A.FORNOM) AS FORNECEDOR,
RTRIM(LTRIM(FOREND)) +' , '+ RTRIM(LTRIM(FORNUM)) + ' , ' + RTRIM(LTRIM(FORBAI)) AS ENDFOR, -- MUDEI DIA 20/04/2016 FORENDTMP)) +' , '+ RTRIM(LTRIM(FORNUM)) + ' , ' + RTRIM(LTRIM(FORBAI)) AS ENDFOR,
RTRIM(LTRIM(FORCEP)) AS CEPFOR,
RTRIM(LTRIM(ISNULL(C.MUNNOM,''))) AS MUNFOR


INTO #CONTAS_PAG

FROM TBS057 A (NOLOCK) LEFT JOIN TBS006 B (NOLOCK) ON A.FORCOD = B.FORCOD
					   LEFT JOIN TBS003 C (NOLOCK) ON B.MUNCOD = C.MUNCOD



if @Opcao = 1 --TODOS 
	SELECT * FROM #CONTAS_PAG 
	WHERE 
	DATA_EMI BETWEEN @DataDeE AND CASE WHEN @DataAteE = '' THEN GETDATE() ELSE @DataAteE END AND 
	TITULO between @TituloDe and case when @TituloAte = 0 then 9999999999 else @TituloAte end and
	VENC_REAL BETWEEN @VencDe and case when @VencAte = '' then GETDATE() else @VencAte end AND 
	PORT <= CASE WHEN @zero=0 THEN 999 ELSE 0 END  AND
	FORNECEDOR LIKE(RTRIM(CASE WHEN @NomFor = '' THEN FORNECEDOR ELSE LTRIM(RTRIM(@NomFor)) +'%' END)) AND
	PORT = CASE WHEN @Portador = 0 THEN PORT ELSE @Portador END AND 
	COD_FOR = CASE WHEN @codfor = 0 THEN COD_FOR ELSE @codfor END



if @Opcao = 2 -- EM ABERTO
	SELECT * FROM #CONTAS_PAG
	WHERE 
	BAIXA = '17530101' AND 
	DATA_EMI BETWEEN @DataDeE AND CASE WHEN @DataAteE = '' THEN GETDATE() ELSE @DataAteE END AND 
	TITULO between @TituloDe and case when @TituloAte = 0 then 9999999999 else @TituloAte end and
	VENC_REAL BETWEEN @VencDe and case when @VencAte = '' then GETDATE() else @VencAte end AND 
	PORT <= CASE WHEN @zero=0 THEN 999 ELSE 0 END  AND
	FORNECEDOR LIKE(RTRIM(CASE WHEN @NomFor = '' THEN FORNECEDOR ELSE LTRIM(RTRIM(@NomFor)) +'%' END)) AND
	PORT = CASE WHEN @Portador = 0 THEN PORT ELSE @Portador END AND
	COD_FOR = CASE WHEN @codfor = 0 THEN COD_FOR ELSE @codfor END
	

if @Opcao = 3 --VENCIDOS
	SELECT * FROM #CONTAS_PAG
	WHERE 
	BAIXA = '17530101' AND 
	VENC_REAL < GETDATE()-1 AND 
	DATA_EMI BETWEEN @DataDeE AND CASE WHEN @DataAteE = '' THEN GETDATE() ELSE @DataAteE END AND 
	TITULO between @TituloDe and case when @TituloAte = 0 then 9999999999 else @TituloAte end and
	VENC_REAL BETWEEN @VencDe and case when @VencAte = '' then GETDATE() else @VencAte end AND 
	PORT <= CASE WHEN @zero=0 THEN 999 ELSE 0 END  AND
	FORNECEDOR LIKE(RTRIM(CASE WHEN @NomFor = '' THEN FORNECEDOR ELSE LTRIM(RTRIM(@NomFor)) +'%' END)) AND
	PORT = CASE WHEN @Portador = 0 THEN PORT ELSE @Portador END AND
	COD_FOR = CASE WHEN @codfor = 0 THEN COD_FOR ELSE @codfor END


if @Opcao = 4  --PAGOS
	SELECT * FROM #CONTAS_PAG
	WHERE 
	BAIXA <> '17530101' AND 
	DATA_EMI BETWEEN @DataDeE AND CASE WHEN @DataAteE = '' THEN GETDATE() ELSE @DataAteE END AND 
	TITULO between @TituloDe and case when @TituloAte = 0 then 9999999999 else @TituloAte end and
	VENC_REAL BETWEEN @VencDe and case when @VencAte = '' then GETDATE() else @VencAte end AND 
	PORT <= CASE WHEN @zero=0 THEN 999 ELSE 0 END  AND
	FORNECEDOR LIKE(RTRIM(CASE WHEN @NomFor = '' THEN FORNECEDOR ELSE LTRIM(RTRIM(@NomFor)) +'%' END)) AND
	PORT = CASE WHEN @Portador = 0 THEN PORT ELSE @Portador END AND
	COD_FOR = CASE WHEN @codfor = 0 THEN COD_FOR ELSE @codfor END

	
if @Opcao = 5 -- PAGOS COM ATRASO
	SELECT * FROM #CONTAS_PAG
	WHERE 
	BAIXA <> '17530101' AND 
	BAIXA > VENC_REAL AND 
	DATA_EMI BETWEEN @DataDeE AND CASE WHEN @DataAteE = '' THEN GETDATE() ELSE @DataAteE END AND 
	TITULO between @TituloDe and case when @TituloAte = 0 then 9999999999 else @TituloAte end and
	VENC_REAL BETWEEN @VencDe and case when @VencAte = '' then GETDATE() else @VencAte end AND 
	PORT <= CASE WHEN @zero=0 THEN 999 ELSE 0 END  AND
	FORNECEDOR LIKE(RTRIM(CASE WHEN @NomFor = '' THEN FORNECEDOR ELSE LTRIM(RTRIM(@NomFor)) +'%' END)) AND
	PORT = CASE WHEN @Portador = 0 THEN PORT ELSE @Portador END AND
	COD_FOR = CASE WHEN @codfor = 0 THEN COD_FOR ELSE @codfor END


	
if @Opcao = 6 -- PAGOS ANTES DO VENCIMENTO
	SELECT * FROM #CONTAS_PAG
	WHERE 
	BAIXA <> '17530101' AND 
	BAIXA < VENC_REAL AND 
	DATA_EMI BETWEEN @DataDeE AND CASE WHEN @DataAteE = '' THEN GETDATE() ELSE @DataAteE END AND 
	TITULO between @TituloDe and case when @TituloAte = 0 then 9999999999 else @TituloAte end and
	VENC_REAL BETWEEN @VencDe and case when @VencAte = '' then GETDATE() else @VencAte end AND 
	PORT <= CASE WHEN @zero=0 THEN 999 ELSE 0 END  AND
	FORNECEDOR LIKE(RTRIM(CASE WHEN @NomFor = '' THEN FORNECEDOR ELSE LTRIM(RTRIM(@NomFor)) +'%' END)) AND
	PORT = CASE WHEN @Portador = 0 THEN PORT ELSE @Portador END AND
	COD_FOR = CASE WHEN @codfor = 0 THEN COD_FOR ELSE @codfor END

	
if @Opcao = 7 -- PAGOS NO VENCIMENTOS
	SELECT * FROM #CONTAS_PAG
	WHERE 
	BAIXA <> '17530101' AND 
	BAIXA = VENC_REAL AND 
	DATA_EMI BETWEEN @DataDeE AND CASE WHEN @DataAteE = '' THEN GETDATE() ELSE @DataAteE END AND 
	TITULO between @TituloDe and case when @TituloAte = 0 then 9999999999 else @TituloAte end and
	VENC_REAL BETWEEN @VencDe and case when @VencAte = '' then GETDATE() else @VencAte end AND 
	PORT <= CASE WHEN @zero=0 THEN 999 ELSE 0 END  AND
	FORNECEDOR LIKE(RTRIM(CASE WHEN @NomFor = '' THEN FORNECEDOR ELSE LTRIM(RTRIM(@NomFor)) +'%' END)) AND
	PORT = CASE WHEN @Portador = 0 THEN PORT ELSE @Portador END AND 
	COD_FOR = CASE WHEN @codfor = 0 THEN COD_FOR ELSE @codfor END
	
SELECT PFX,
TITULO,
PAR,
PORT,
DATA_EMI,
VENC_REAL,
BAIXA,
ANO_VEN_REAL,
MES_VEN_REAL,
DIA_VEN_REAL,
DIAS_PARA_VENCER,
DIA_ATRA,
VALOR_ORIG,
RESIDUO,
ABATIMENTO,
ACRESCIMO,
VALOR_PAGO,
SALDO_ABERTO,
COD_FOR,
FORNECEDOR,
OBS,
ENDFOR,
CEPFOR,
MUNFOR

from #CONTAS_PAG

END
