SELECT CASE WHEN ISNULL(B.CONSTRAINT_NAME,'') <> '' THEN 'PK' ELSE 'SK' END AS CH , 
A.COLUMN_NAME, A.IS_NULLABLE, A.DATA_TYPE, A.CHARACTER_MAXIMUM_LENGTH , A.COLLATION_NAME,
CASE WHEN A.COLLATION_NAME <> '' 
	THEN A.COLUMN_NAME + ' COLLATE Latin1_General_CI_AS AS ' + A.COLUMN_NAME + ','  
	ELSE A.COLUMN_NAME + ' , '
	END, *


 FROM INFORMATION_SCHEMA.COLUMNS A LEFT JOIN INFORMATION_SCHEMA.KEY_COLUMN_USAGE B
  ON A.TABLE_NAME = B.TABLE_NAME AND A.COLUMN_NAME = B.COLUMN_NAME
 
  WHERE A.table_name = 'ALTERADO';
