DECLARE @Data_de date, @Data_ate date, @COD CHAR(15), @MARNOM CHAR(60), @DESCRI CHAR(60)
set @Data_de = '20150101'
set @Data_ate = '20150130'
SET @COD = ''
SET @MARNOM =''
SET @DESCRI =''


WITH ENT AS (SELECT PROCOD,
							AVG(dbo.NFETOTITE(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE)/(NFEQTD*NFEQTDEMB)) AS VALORMD        
							FROM TBS0591 A (NOLOCK) RIGHT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD 
							AND A.NFECOD = B.NFECOD 
							AND A.NFENUM = B.NFENUM 
							AND A.NFETIP = B.NFETIP 
							AND A.SERCOD = B.SERCOD 
							AND A.SEREMPCOD = B.SEREMPCOD
			
							WHERE (B.NFECAN = 'N' OR B.NFECAN = '')
							AND (B.NFEDATENT BETWEEN getdate() - CONVERT(DATETIME,91) and getdate()) 
							AND B.NFENOM NOT LIKE('%TANBY%')
							AND B.NFETIP = 'N'
							GROUP BY PROCOD ) ,
							
PRE AS (SELECT TDPPROCOD , TDPCUSBAS AS PRECUS , 
		CASE WHEN TDPVALPROI<=GETDATE() AND TDPVALPROF >= GETDATE() AND TDPPROLOJ = 'S' THEN TDPPREPRO1 ELSE TDPPRELOJ1 END AS PREVEN
							FROM TBS031 nolock ),
					

SAL_EST AS (SELECT PROCOD,ESTQTDATU-ESTQTDRES AS QTD,ESTQTDCMP 
							FROM TBS032 NOLOCK 
							WHERE ESTLOC = 1
							AND (ESTQTDATU <> 0 or ESTQTDRES <> 0 or ESTQTDPEN <> 0 or ESTQTDCMP <> 0)), 
						

SAL_LOJ AS  (SELECT PROCOD,ESTQTDATU-ESTQTDRES AS QTD 
							FROM TBS032 NOLOCK 
							WHERE ESTLOC = 2
							AND (ESTQTDATU <> 0 or ESTQTDRES <> 0 or ESTQTDPEN <> 0 or ESTQTDCMP <> 0)), 
							

VND_LOJA AS (SELECT M2_PROCOD, SUM(M2_QTD) AS VENDAS, avg(M2_VALUNI-M2_ABT) AS VALMED,SUM(M2_VALTOT)-SUM(M2_ABT) AS VALOR
							FROM MSL002 (NOLOCK)
							WHERE MSL002.M2_DAT between @Data_de and @Data_ate 
							and MSL002.M2_TIPREG='01' 
							and MSL002.M2_REGCAN='F'
							and M2_QTD > 0
							GROUP BY M2_PROCOD), 
					

VND_INT AS(SELECT A.PROCOD, SUM(A.NFSQTD*NFSQTDEMB)AS VENDAS,AVG(NFSPRE/NFSQTDEMB) AS VALMED,
			SUM(dbo.NFSTOTITE(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE)) AS VALOR
							FROM TBS0671 A (NOLOCK) JOIN TBS067 B ON 
							A.NFSEMPCOD = B.NFSEMPCOD and 
							A.NFSNUM = B.NFSNUM and 
							A.SNEEMPCOD = B.SNEEMPCOD and 
							A.SNESER = B.SNESER

							WHERE NFSTIP = 'N' and NFSCAN = 'N' and NFSDEV='N'
							and NFSCLINOM not Like('%BEST BAG%') and NFSCLINOM not Like('%BEST OFFICE%') 
							and NFSCLINOM not Like('%MISASPEL%') and NFSCLINOM not Like('%PAPELYNA%') 
							and NFSCLINOM not Like('%TANBY%') and
							NFSDATEMI between @Data_de and @Data_ate 
							AND NFSTIP = 'N' and NFSCAN = 'N'
							AND A.NFSQTD*NFSQTDEMB > 0
							GROUP BY A.PROCOD),

PDC AS ( SELECT PROCOD AS COD,dbo.PDCTOTITE(TBS0451.PDCEMPCOD,TBS0451.PDCNUM,PDCITE) as VALOR_ITEM,
		(dbo.PDCTOTITE(TBS0451.PDCEMPCOD,TBS0451.PDCNUM,PDCITE) / PDCQTD) * PDCQTDENT AS VALOR_ENT,
		(dbo.PDCTOTITE(TBS0451.PDCEMPCOD,TBS0451.PDCNUM,PDCITE) / PDCQTD) * PDCQTDRES AS VALOR_RES, 
		CASE WHEN (PDCQTD - PDCQTDENT - PDCQTDRES ) > 0 THEN dbo.PDCTOTITE(TBS0451.PDCEMPCOD,TBS0451.PDCNUM,PDCITE) ELSE 0 END AS VALOR_ABERTO
		FROM TBS0451(NOLOCK)FULL OUTER JOIN TBS045 (NOLOCK)ON TBS0451.PDCNUM = TBS045.PDCNUM
		WHERE PDCDATCAD between @Data_de and @Data_ate )
		

SELECT A.PROCOD,
A.PRODES AS DES,
CASE WHEN len(A.MARCOD) = 4 
	THEN rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) 
	ELSE RIGHT(('00' + ltrim(str(A.MARCOD))), 3) + ' - ' + rtrim(A.MARNOM) END AS MARCA,
A.PROUM1 AS  UNI1, 
ISNULL(CASE WHEN ENT.VALORMD > 0 THEN CASE WHEN ENT.VALORMD > (PRE.PRECUS*2) THEN PRE.PRECUS ELSE ENT.VALORMD END  ELSE PRE.PRECUS END,0) AS VALMEDENTR,
ISNULL(PRE.PRECUS,0) AS VALCUSTATU,
ISNULL(VND_LOJA.VALMED,VND_INT.VALMED) AS VALMEDVEN,
ISNULL(PRE.PREVEN,0) AS VALVENATU,
ISNULL(SAL_LOJ.QTD,0) AS SALLOJA,
ISNULL(SAL_LOJ.QTD * CASE WHEN ENT.VALORMD > 0 THEN CASE WHEN ENT.VALORMD > (PRE.PRECUS*2) THEN PRE.PRECUS ELSE ENT.VALORMD END  ELSE PRE.PRECUS END,0) AS CUS_SALOJA,
ISNULL(VND_LOJA.VENDAS,0) AS VENDASLOJ,
ISNULL(VND_LOJA.VALOR,0) AS VALORLOJA,
ISNULL(VND_LOJA.VENDAS * CASE WHEN ENT.VALORMD > 0 THEN CASE WHEN ENT.VALORMD > (PRE.PRECUS*2) THEN PRE.PRECUS ELSE ENT.VALORMD END  ELSE PRE.PRECUS END,0) AS CUSTOLOJA,
ISNULL(SAL_EST.QTD,0) AS SALEST,
ISNULL(SAL_EST.QTD * CASE WHEN ENT.VALORMD > 0 THEN CASE WHEN ENT.VALORMD > (PRE.PRECUS*2) THEN PRE.PRECUS ELSE ENT.VALORMD END  ELSE PRE.PRECUS END,0) AS CUS_SALEST,
ISNULL(VND_INT.VENDAS,0) AS VENDASINT,
ISNULL(VND_INT.VALOR,0) AS VALORINT,
ISNULL(VND_INT.VENDAS * CASE WHEN ENT.VALORMD > 0 THEN CASE WHEN ENT.VALORMD > (PRE.PRECUS*2) THEN PRE.PRECUS ELSE ENT.VALORMD END  ELSE PRE.PRECUS END,0) AS CUSTOINT,
ISNULL(VALOR_ITEM,0) AS VAL_CMP,
ISNULL(VALOR_ENT,0) AS VAL_ENTR,
ISNULL(VALOR_RES,0 ) AS VAL_RES,
ISNULL(VALOR_ABERTO,0) AS VAL_ABER

FROM TBS010 A FULL OUTER JOIN ENT ON ENT.PROCOD = A.PROCOD
				FULL OUTER JOIN VND_LOJA ON VND_LOJA.M2_PROCOD = A.PROCOD
				FULL OUTER JOIN VND_INT ON VND_INT.PROCOD = A.PROCOD
				FULL OUTER JOIN SAL_EST ON SAL_EST.PROCOD = A.PROCOD
				FULL OUTER JOIN SAL_LOJ ON SAL_LOJ.PROCOD = A.PROCOD 
				FULL OUTER JOIN PRE ON PRE.TDPPROCOD = A.PROCOD
				FULL OUTER JOIN PDC ON PDC.COD = A.PROCOD
				
WHERE (SAL_EST.QTD <> 0 OR SAL_EST.QTD <> 0 OR VND_LOJA.VENDAS <> 0 OR VND_INT.VENDAS <> 0 OR VALOR_ITEM <> 0 OR VALOR_ENT <> 0 OR VALOR_RES <> 0 OR VALOR_ABERTO <> 0)
AND A.PROCOD LIKE(CASE WHEN @COD = '' THEN '% %' ELSE RTRIM(@COD)+'%' END)
AND A.MARNOM LIKE (CASE WHEN @MARNOM = '' THEN '% %' ELSE UPPER(RTRIM(@MARNOM))+'%' END)
AND A.PRODES LIKE(CASE WHEN @DESCRI = '' THEN '% %' ELSE UPPER(RTRIM(@DESCRI))+'%' END )
				
GROUP BY A.PROCOD, A.MARCOD, A.MARNOM, ENT.VALORMD,PRE.PRECUS,
			VND_LOJA.VENDAS,VND_INT.VENDAS,VND_LOJA.VALMED,VND_INT.VALMED,
			SAL_EST.QTD,SAL_LOJ.QTD,VND_INT.VALOR,VND_LOJA.VALOR,PRE.PREVEN,
			A.PROUM1, A.PRODES,VALOR_ITEM,VALOR_ENT,VALOR_RES,VALOR_ABERTO

DECLARE @Data_de date, @Data_ate date, @COD CHAR(15), @MARNOM CHAR(60), @DESCRI CHAR(60)
set @Data_de = '20151001'
set @Data_ate = '20160101'
SET @COD = '0040001'
SET @MARNOM =''
SET @DESCRI =''


SELECT A.PROCOD,
A.PRODES AS DES,
CASE WHEN len(A.MARCOD) = 4 
	THEN rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) 
	ELSE RIGHT(('00' + ltrim(str(A.MARCOD))), 3) + ' - ' + rtrim(A.MARNOM) END AS MARCA,
A.PROUM1 AS  UNI1, 
ISNULL(CASE WHEN ENT.VALORMD > 0 THEN CASE WHEN ENT.VALORMD > (PRE.PRECUS*2) THEN PRE.PRECUS ELSE ENT.VALORMD END  ELSE PRE.PRECUS END,0) AS VALMEDENTR,
ISNULL(PRE.PRECUS,0) AS VALCUSTATU,
ISNULL(VND_LOJA.VALMED,VND_INT.VALMED) AS VALMEDVEN,
ISNULL(PRE.PREVEN,0) AS VALVENATU,
ISNULL(SAL_LOJ.QTD,0) AS SALLOJA,
ISNULL(SAL_LOJ.QTD * CASE WHEN ENT.VALORMD > 0 THEN CASE WHEN ENT.VALORMD > (PRE.PRECUS*2) THEN PRE.PRECUS ELSE ENT.VALORMD END  ELSE PRE.PRECUS END,0) AS CUS_SALOJA,
ISNULL(VND_LOJA.VENDAS,0) AS VENDASLOJ,
ISNULL(VND_LOJA.VALOR,0) AS VALORLOJA,
ISNULL(VND_LOJA.VENDAS * CASE WHEN ENT.VALORMD > 0 THEN CASE WHEN ENT.VALORMD > (PRE.PRECUS*2) THEN PRE.PRECUS ELSE ENT.VALORMD END  ELSE PRE.PRECUS END,0) AS CUSTOLOJA,
ISNULL(SAL_EST.QTD,0) AS SALEST,
ISNULL(SAL_EST.QTD * CASE WHEN ENT.VALORMD > 0 THEN CASE WHEN ENT.VALORMD > (PRE.PRECUS*2) THEN PRE.PRECUS ELSE ENT.VALORMD END  ELSE PRE.PRECUS END,0) AS CUS_SALEST,
ISNULL(VND_INT.VENDAS,0) AS VENDASINT,
ISNULL(VND_INT.VALOR,0) AS VALORINT,
ISNULL(VND_INT.VENDAS * CASE WHEN ENT.VALORMD > 0 THEN CASE WHEN ENT.VALORMD > (PRE.PRECUS*2) THEN PRE.PRECUS ELSE ENT.VALORMD END  ELSE PRE.PRECUS END,0) AS CUSTOINT,
ISNULL(VALOR_ITEM,0) AS VAL_CMP,
ISNULL(VALOR_ENT,0) AS VAL_ENTR,
ISNULL(VALOR_RES,0 ) AS VAL_RES,
ISNULL(VALOR_ABERTO,0) AS VAL_ABER

FROM TBS010 A FULL OUTER JOIN (SELECT D.PROCOD ,
								convert(int,subString(convert(char(6),NFEDATENT,112),1,4)) AS ANO,
								convert(int,subString(convert(char(6),NFEDATENT,112),5,2))AS MES ,
										
										
								CASE WHEN SUM(PDCQTDENT*D.PDCQTDEMB) - isnull(SUM(C.NFENFDQTD*A.NFEQTDEMB),0)  > 0 THEN 
										
								round(AVG(convert(decimal(10,4),(dbo.PDCTOTITE(D.PDCEMPCOD,D.PDCNUM,PDCITE) / PDCQTD )/ PDCQTDEMB)),2)
								ELSE 0 END AS VALORMD
								
								FROM TBS0591 A (NOLOCK) LEFT JOIN TBS0451 D(nolock) ON D.PDCNUM = A.NFEPDCNUM AND A.PROCOD = D.PROCOD 
														LEFT JOIN  TBS059 B(nolock)  on A.SERCOD=B.SERCOD and A.NFETIP=B.NFETIP and A.NFECOD=B.NFECOD and A.NFENUM=B.NFENUM
														LEFT JOIN  TBS0597 C(NOLOCK) ON A.SERCOD=C.SERCOD and A.NFETIP=C.NFETIP AND A.NFECOD=C.NFECOD AND A.NFEITE=C.NFEITE
																									AND A.NFENUM = C.NFENUM
														
														
										
								WHERE 
								A.NFETIP='N' and NFEDATENT between '20150101' and '20151231' and
								NFENOM not like('BEST BAG%') and NFENOM not like('BEST OFFICE%') and 
								NFENOM not like('MISASPEL%') and NFENOM not like('%PAPELYNA%') and NFENOM not like('TANBY%')
								AND A.SERCOD NOT IN('CAN') 
								AND D.PDCNUM > 0 --AND PROCOD IS NULL
								
								GROUP BY convert(char(6),NFEDATENT,112), D.PROCOD
								 )  AS ENT ON ENT.PROCOD = A.PROCOD
							
							
				FULL OUTER JOIN (SELECT M2_PROCOD, SUM(M2_QTD) AS VENDAS, avg(M2_VALUNI-M2_ABT) AS VALMED,SUM(M2_VALTOT)-SUM(M2_ABT) AS VALOR
								FROM MSL002 (NOLOCK)
								WHERE MSL002.M2_DAT between @Data_de and @Data_ate
								and MSL002.M2_TIPREG='01' 
								and MSL002.M2_REGCAN='F'
								and M2_QTD > 0
								GROUP BY M2_PROCOD) AS VND_LOJA ON VND_LOJA.M2_PROCOD = A.PROCOD
								
				FULL OUTER JOIN (SELECT A.PROCOD, SUM(A.NFSQTD*NFSQTDEMB)AS VENDAS,AVG(NFSPRE/NFSQTDEMB) AS VALMED,
								SUM(dbo.NFSTOTITE(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE)) AS VALOR
								FROM TBS0671 A (NOLOCK) JOIN TBS067 B ON 
								A.NFSEMPCOD = B.NFSEMPCOD and 
								A.NFSNUM = B.NFSNUM and 
								A.SNEEMPCOD = B.SNEEMPCOD and 
								A.SNESER = B.SNESER
	
								WHERE NFSTIP = 'N' and NFSCAN = 'N' and NFSDEV='N'
								and NFSCLINOM not Like('%BEST BAG%') and NFSCLINOM not Like('%BEST OFFICE%') 
								and NFSCLINOM not Like('%MISASPEL%') and NFSCLINOM not Like('%PAPELYNA%') 
								and NFSCLINOM not Like('%TANBY%') and
								NFSDATEMI between @Data_de and @Data_ate
								AND NFSTIP = 'N' and NFSCAN = 'N'
								AND A.NFSQTD*NFSQTDEMB > 0
								GROUP BY A.PROCOD) AS VND_INT ON VND_INT.PROCOD = A.PROCOD
				
				FULL OUTER JOIN (SELECT PROCOD,ESTQTDATU-ESTQTDRES AS QTD,ESTQTDCMP 
								FROM TBS032 NOLOCK 
								WHERE ESTLOC = 1
								AND (ESTQTDATU <> 0 or ESTQTDRES <> 0 or ESTQTDPEN <> 0 or ESTQTDCMP <> 0)) AS SAL_EST ON SAL_EST.PROCOD = A.PROCOD
								
				FULL OUTER JOIN (SELECT PROCOD,ESTQTDATU-ESTQTDRES AS QTD 
								FROM TBS032 NOLOCK 
								WHERE ESTLOC = 2
								AND (ESTQTDATU <> 0 or ESTQTDRES <> 0 or ESTQTDPEN <> 0 or ESTQTDCMP <> 0)) AS SAL_LOJ ON SAL_LOJ.PROCOD = A.PROCOD 
				
				FULL OUTER JOIN (SELECT TDPPROCOD , TDPCUSBAS AS PRECUS , 
								CASE WHEN TDPVALPROI<=GETDATE() AND TDPVALPROF >= GETDATE() AND TDPPROLOJ = 'S' THEN TDPPREPRO1 ELSE TDPPRELOJ1 END AS PREVEN
								FROM TBS031 nolock ) AS PRE ON PRE.TDPPROCOD = A.PROCOD
								
				FULL OUTER JOIN ( SELECT PROCOD AS COD,dbo.PDCTOTITE(TBS0451.PDCEMPCOD,TBS0451.PDCNUM,PDCITE) as VALOR_ITEM,
								(dbo.PDCTOTITE(TBS0451.PDCEMPCOD,TBS0451.PDCNUM,PDCITE) / PDCQTD) * PDCQTDENT AS VALOR_ENT,
								(dbo.PDCTOTITE(TBS0451.PDCEMPCOD,TBS0451.PDCNUM,PDCITE) / PDCQTD) * PDCQTDRES AS VALOR_RES, 
								CASE WHEN (PDCQTD - PDCQTDENT - PDCQTDRES ) > 0 THEN dbo.PDCTOTITE(TBS0451.PDCEMPCOD,TBS0451.PDCNUM,PDCITE) ELSE 0 END AS VALOR_ABERTO
								FROM TBS0451(NOLOCK)FULL OUTER JOIN TBS045 (NOLOCK)ON TBS0451.PDCNUM = TBS045.PDCNUM
								WHERE PDCDATCAD between @Data_de and @Data_ate ) AS PDC ON PDC.COD = A.PROCOD
				
WHERE (SAL_EST.QTD <> 0 OR SAL_EST.QTD <> 0 OR VND_LOJA.VENDAS <> 0 OR VND_INT.VENDAS <> 0 OR VALOR_ITEM <> 0 OR VALOR_ENT <> 0 OR VALOR_RES <> 0 OR VALOR_ABERTO <> 0)

AND A.PROCOD LIKE(CASE WHEN @COD = '' THEN '% %' ELSE RTRIM(@COD)+'%' END)
AND A.MARNOM LIKE (CASE WHEN @MARNOM = '' THEN '% %' ELSE UPPER(RTRIM(@MARNOM))+'%' END)
AND A.PRODES LIKE(CASE WHEN @DESCRI = '' THEN '% %' ELSE UPPER(RTRIM(@DESCRI))+'%' END )
				
GROUP BY A.PROCOD, A.MARCOD, A.MARNOM, ENT.VALORMD,PRE.PRECUS,
			VND_LOJA.VENDAS,VND_INT.VENDAS,VND_LOJA.VALMED,VND_INT.VALMED,
			SAL_EST.QTD,SAL_LOJ.QTD,VND_INT.VALOR,VND_LOJA.VALOR,PRE.PREVEN,
			A.PROUM1, A.PRODES,VALOR_ITEM,VALOR_ENT,VALOR_RES,VALOR_ABERTO
