-- CLIENTES DA TANBY MATRIZ

IF OBJECT_ID('tempDB.dbo.#TB2ND') IS NOT NULL 
DROP TABLE #TB2ND

SELECT
'ND' AS EMP, 
CLITIPPES,
CLICOD,
CLINOM,
CLINOMFAN,
REPLACE(REPLACE(REPLACE(REPLACE(RTRIM(LTRIM(CLICGC)),' ',''),'/',''),'.',''),'-','')  AS CLICGC,
RTRIM(LTRIM(CLICPF))  AS CLICPF,
RTRIM(LTRIM(CLIIES))  AS CLIIES,
CLIDATCAD,
CLIUCPDAT,
CLIEMAIL,
VENCOD,
CASE WHEN VENCOD <> 0 
	THEN ISNULL((SELECT VENNOM FROM TBS004 B (NOLOCK) WHERE A.VENCOD = B.VENCOD),'')
	ELSE 'SEM VENDEDOR'
END VENNOM , 
CLICLA

INTO #TB2ND
FROM TBS002 A (NOLOCK)

-- INDICE juridico em comomum 

SELECT 
EMP,
CLITIPPES,
CLICOD,
CLINOM,
CLICGC,
CLICPF as CPF,
VENCOD, 
VENNOM

FROM #CLI 

WHERE 
CLICGC <> '' AND  
CLICGC NOT LIKE('00000000000%') AND 
CLICGC <>'0' AND 
CLICGC <> '0000000000' AND 
CLICGC IN (SELECT CLICGC FROM #CLI GROUP BY CLICGC HAVING COUNT(*) > 1)

UNION 

SELECT 
EMP,
CLITIPPES,
CLICOD,
CLINOM,
CLICGC AS CNPJ,
CLICPF,
VENCOD, 
VENNOM

FROM #CLI 

WHERE 
CLICPF <> '' AND  
CLICPF NOT LIKE('00000000000%') AND 
CLICPF <>'0' AND 
CLICPF <> '0000000000' AND 
CLICPF IN (SELECT CLICPF FROM #CLI GROUP BY CLICPF HAVING COUNT(*) > 1)
