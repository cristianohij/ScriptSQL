-- observações 
-- é necessario pelo menos filtrar um dos campos de filtro abaixo
-- a verificação dos clientes na outras unidades, ocorre atraves do CNPJ e CFP, ou seja, se o cliente estiver cadastrado na outra empresa porem sem o CNPJ, o select irá retornar vazio para essa empresa, porem existe o cliente, essa informação não da para trazer

-- FILTROS

DECLARE  @CODVENDEDOR VARCHAR(3), @CNPJ VARCHAR(16), @CPF VARCHAR(16)

SET @CODVENDEDOR = '49'
SET @CNPJ = ''
SET @CPF = ''

-----------------------------------------------------------------------------------------------------------------------------------------

-- CLIENTES DA TANBY MATRIZ

IF OBJECT_ID('tempDB.dbo.#TB2ND') IS NOT NULL 
DROP TABLE #TB2ND

SELECT
'ND' AS EMP, 
A.CLITIPPES,
A.VENCOD,
CASE WHEN A.VENCOD <> 0 
	THEN ISNULL((SELECT RTRIM(LTRIM(VENNOM)) FROM TBS004 B (NOLOCK) WHERE A.VENCOD = B.VENCOD),'')
	ELSE 'SEM VENDEDOR'
END VENNOM , 
A.CLICOD,
RTRIM(LTRIM(A.CLINOM)) AS CLINOM,
RTRIM(LTRIM(A.CLINOMFAN)) AS CLINOMFAN,
REPLACE(REPLACE(REPLACE(REPLACE(RTRIM(LTRIM(A.CLICGC)),' ',''),'/',''),'.',''),'-','')  AS CLICGC,
RTRIM(LTRIM(A.CLICPF))  AS CLICPF,
RTRIM(LTRIM(A.CLIIES))  AS CLIIES,
CONVERT(CHAR(10),A.CLIDATCAD,103) AS CLIDATCAD,
CASE WHEN CONVERT(CHAR(10),A.CLIUCPDAT,103) = '01/01/1753' OR CONVERT(CHAR(10),A.CLIUCPDAT,103) = '01/01/1990'
	THEN ''
	ELSE CONVERT(CHAR(10),A.CLIUCPDAT,103)
END AS CLIUCPDAT,
RTRIM(LTRIM(A.CLIEMAIL)) AS CLIEMAIL,
A.CLICLA

INTO #TB2ND
FROM TBS002 A (NOLOCK) 

WHERE 
RTRIM(LTRIM(STR(VENCOD))) = CASE WHEN @CODVENDEDOR = '' THEN VENCOD ELSE @CODVENDEDOR END AND 
REPLACE(REPLACE(REPLACE(REPLACE(RTRIM(LTRIM(CLICGC)),' ',''),'/',''),'.',''),'-','') = CASE WHEN @CNPJ = '' 
																					   		THEN REPLACE(REPLACE(REPLACE(REPLACE(RTRIM(LTRIM(CLICGC)),' ',''),'/',''),'.',''),'-','') 
																							ELSE @CNPJ 
																						END AND 
RTRIM(LTRIM(CLICPF)) = CASE WHEN @CPF = '' THEN RTRIM(LTRIM(CLICPF)) ELSE @CPF END
