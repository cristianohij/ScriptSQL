-- CADASTRO DE CLIENTES NOVOS

DECLARE @DATADE DATETIME, @DATEATE DATETIME , @VENNOM VARCHAR(60) -- , @GVECOD INT

SET @DATADE = '20171018'
SET @DATEATE = '20171020'
SET @VENNOM = ''
-- SET @GVECOD = 0

SELECT 
A.CLIDATCAD,
CASE CLITIPPES
	WHEN 'J' THEN 'JURIDICA'
	WHEN 'F' THEN 'FISICA'
	ELSE ''
END AS CLITIPPES,
CLICOD, 
RTRIM(LTRIM(CLINOM)) AS CLINOM,
CLICGC, 
CLICPF,
A.CLICLA,
A.CLILIC,
CASE A.CLISIT
	WHEN 'A' THEN 'ATIVO'
	WHEN 'I' THEN 'INATIVO'
	ELSE ''
END CLISIT,
CASE WHEN A.CLIBLQTRN = 'S' THEN 'SIM' ELSE 'NAO' END AS CLIBLQTRN, -- CAMPO CLIBLQTRN SIGNIFICA SOMENTE A VISTA ? 
A.CLIUSUALT,
ISNULL(RTRIM(LTRIM(STR(A.VENCOD))) + ' - ' + RTRIM(LTRIM(VENNOM)), 'SEM VENDEDOR') AS VENNOM,
ISNULL(RTRIM(LTRIM(STR(B.GVECOD))) + ' - ' + RTRIM(LTRIM(C.GVEDES)), '0 - SEM GRUPO') AS GVDES

FROM 
TBS002 A (NOLOCK)
LEFT JOIN 
TBS004 B (NOLOCK) ON A.VENCOD = B.VENCOD AND A.VENEMPCOD = B.VENEMPCOD
LEFT JOIN 
TBS091 C (NOLOCK) ON B.GVECOD = C.GVECOD AND B.GVEEMPCOD = C.GVEEMPCOD

WHERE
A.CLIDATCAD BETWEEN @DATADE AND @DATEATE AND 
ISNULL(RTRIM(LTRIM(STR(A.VENCOD))) + ' - ' + RTRIM(LTRIM(VENNOM)), '0 - SEM VENDEDOR') 
LIKE (CASE WHEN @VENNOM = '' 
		THEN ISNULL(RTRIM(LTRIM(STR(A.VENCOD))) + ' - ' + RTRIM(LTRIM(VENNOM)), 'SEM VENDEDOR')
		ELSE upper(@VENNOM)
	END) /*AND 
ISNULL(B.GVECOD,0) IN (@GVECOD)		*/ -- uso no reports

ORDER BY 
CLICOD
