declare @data date 
set @data = '20170701'


SELECT CONVERT(DATE,LMEDATHOR) AS LMEDATMOV,* 
INTO #TES 
FROM TBS051 (NOLOCK) 
WHERE LMEDATHOR < @data --AND LMEDATHOR < '20150102'
AND LMEINFALT = 'E'  -- SIGNIFICA ESTOQUE
ORDER BY PROCOD, LMEREG DESC, LMEDATHOR DESC

SELECT 
ESTLOC,
PROCOD,
ESTQTDATU,
SINQTD,
ISNULL((SELECT 
	 	top 1 LMEQTDSAL 
	 	FROM #TES C
	 	WHERE LMEINFALT = 'E' 
							  
		AND A.PROCOD = C.PROCOD 
	 	AND C.LMELOCEST = A.ESTLOC
		 
	 	ORDER BY 
		LMEREG DESC ,PROCOD, LMEDATHOR DESC),0) AS TBS051,
CASE WHEN ESTQTDATU <> SINQTD
	THEN 'ERRADO'
	ELSE 'CERTO'
END AS COMPARAR

FROM SALDODIARIO A (NOLOCK)
LEFT JOIN TBS124 B (NOLOCK) ON A.PROCOD = B.SINPROCOD AND A.ESTLOC = B.LESCOD

WHERE 
SINDAT =@data AND 
ESTDATSAL = dateadd(D,-1,@data)

-- COMPARA DIA POR DIA COM A TBS124

DECLARE @DATADE DATETIME, @DATAATE DATETIME , @ESTLOC INT , @PROCOD CHAR(15)
SET @DATADE = '20170306'
SET @DATAATE = '20170718'
SET @ESTLOC = 1
SET @PROCOD ='3261301'

-----------------------------------------------------------------------

if object_id('TempDB.dbo.#SAIDA') is not null
   drop table #SAIDA

SELECT 
PROCOD, 
SUM(LMEQTDMOV) AS QTD , 
CONVERT(DATE,LMEDATHOR) AS DATA

INTO #SAIDA
FROM TBS051 NOLOCK 

WHERE 
PROCOD = @PROCOD AND 
CONVERT(DATE,LMEDATHOR) BETWEEN @DATADE AND @DATAATE AND
LMEACA = 'S' AND
LMEINFALT = 'E' AND
LMELOCEST = @ESTLOC

GROUP BY 
PROCOD , CONVERT(DATE,LMEDATHOR) 

-------------------------------------------------------

if object_id('TempDB.dbo.#INICIO') is not null
   drop table #INICIO
   
SELECT 
TOP 1 PROCOD, 
LMEQTDATU AS QTD , 
CONVERT(DATE,LMEDATHOR) AS DATA 

INTO #INICIO
FROM TBS051 A (NOLOCK) 

WHERE 
PROCOD = @PROCOD AND 
CONVERT(DATE,LMEDATHOR) BETWEEN @DATADE AND @DATAATE AND
LMEINFALT = 'E' AND
LMELOCEST = @ESTLOC

ORDER BY 
LMEREG

-------------------------------

if object_id('TempDB.dbo.#ENTRADA') is not null
   drop table #ENTRADA

SELECT PROCOD, 
SUM(LMEQTDMOV) AS QTD , 
CONVERT(DATE,LMEDATHOR) AS DATA

INTO #ENTRADA
FROM TBS051 NOLOCK 

WHERE 
PROCOD = @PROCOD AND 
CONVERT(DATE,LMEDATHOR) BETWEEN @DATADE AND @DATAATE AND
LMEACA = 'E' AND
LMEINFALT = 'E' AND
LMELOCEST = @ESTLOC

GROUP BY 
PROCOD , CONVERT(DATE,LMEDATHOR) 

-----------------------------------------------------------------------

if object_id('TempDB.dbo.#KARDEX') is not null
   drop table #KARDEX

SELECT 
IDENTITY(INT,1,1) AS ORDEM, 
ISNULL(ISNULL(A.DATA,B.DATA),C.DATA) AS DATA,
ISNULL(ISNULL(A.PROCOD,B.PROCOD),C.PROCOD) AS PROCOD,
ISNULL(A.QTD,0) AS INICIO,
ISNULL(C.QTD,0) AS ENTRADA,
ISNULL(B.QTD,0) AS SAIDA

INTO #KARDEX
FROM #INICIO A 
FULL OUTER JOIN #SAIDA B ON A.PROCOD = B.PROCOD AND A.DATA = B.DATA
FULL OUTER JOIN #ENTRADA C ON ISNULL(A.PROCOD,B.PROCOD) = C.PROCOD AND ISNULL(A.DATA, B.DATA) = C.DATA

ORDER BY 
ISNULL(ISNULL(A.DATA,B.DATA),C.DATA)


DECLARE @i INT

SET @i = 2

WHILE @i <= (SELECT MAX(ORDEM) FROM #KARDEX)

	BEGIN 
	
	UPDATE #KARDEX SET
	INICIO = (SELECT INICIO FROM #KARDEX WHERE ORDEM = @i - 1) + (SELECT ENTRADA FROM #KARDEX WHERE ORDEM = @i - 1)  - (SELECT SAIDA FROM #KARDEX WHERE ORDEM = @i - 1)
	WHERE ORDEM = @i 
		
	SET @i = @i + 1

END

if object_id('TempDB.dbo.#KES') is not null
   drop table #KES

SELECT 
CONVERT(DATE,KESDAT) AS KESDAT, 
LESCOD, 
KESPROCOD, 
KESNFENT + KESENTDEV + KESCANNFSAI + KESCANCUPFIS + KESCANNFDEV + KESMOVENT + KESSALENT AS ENTRADA,
KESNFSAI + KESCUPFIS + KESNFDEVSAI + KESMOVSAI + KESSALSAI AS SAIDA, 
KESNFSAI , KESCUPFIS , KESNFDEVSAI , KESMOVSAI , KESSALSAI,  
KESNFENT , KESENTDEV , KESCANNFSAI , KESCANCUPFIS , KESCANNFDEV , KESMOVENT , KESSALENT

INTO #KES
FROM TBS125 NOLOCK 

WHERE 
KESDAT BETWEEN @DATADE AND @DATAATE AND 
LESCOD= @ESTLOC AND 
KESPROCOD = @PROCOD

----------------------------------------------------------------------------------------

SINPROCOD, 
SINQTD

SINDAT BETWEEN @DATADE AND @DATAATE AND

SINPROCOD = @PROCOD

SELECT A.* , B.*
FROM #KARDEX A
FULL OUTER JOIN #KES B ON A.PROCOD = B.KESPROCOD AND A.DATA = B.KESDAT 
WHERE
A.ENTRADA <> B.ENTRADA
