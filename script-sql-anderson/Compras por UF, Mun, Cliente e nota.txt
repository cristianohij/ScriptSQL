-- contabilizar COMPRAS

if object_id('TempDB.dbo.#NFE') is not null
   drop table #NFE

SELECT
B.NFEESTORI AS UF,
C.MUNCOD,
(SELECT MUNNOM FROM TBS003 E (NOLOCK) WHERE E.MUNCOD = C.MUNCOD) AS MUNNOM,
B.NFECOD as NFSCLICOD,
B.NFENOM AS NFSCLINOM,
B.NFENUM AS NFSNUM,
A.PROCOD ,
A.NFEDES AS NFSPRODES,
D.PROUM1,
sum(ROUND(NFETOTOPEITE,2)) / sum(A.NFEQTD * A.NFEQTDEMB) AS NFSPREMED,
sum(A.NFEQTD * A.NFEQTDEMB) AS NFSQTDVEN,
sum(ROUND(NFETOTOPEITE,2)) AS NFSTOTITEST

INTO #NFE
FROM TBS0591 A (NOLOCK)
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND 
A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND  A.SEREMPCOD = B.SEREMPCOD
LEFT JOIN #TBS006 C (NOLOCK) ON B.NFECOD = C.FORCOD
LEFT JOIN #TBS010 D (NOLOCK) ON A.PROCOD = D.PROCOD
		
WHERE
B.NFEDATEFE BETWEEN @data1de and @data1ate and 
B.NFETIP NOT IN ('T','D') AND 
B.NFECAN <> 'S' AND 
NFEDATEFE <> '' AND 
A.NFECFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E') AND 
B.NFECOD NOT IN (SELECT CLICOD FROM #GRUPOCLI)

GROUP BY 
B.NFEESTORI ,
C.MUNCOD,
B.NFECOD,
B.NFENOM,
B.NFENUM ,
A.PROCOD,
A.NFEDES,
D.PROUM1

SELECT * FROM #NFE
