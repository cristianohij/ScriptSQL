DECLARE @DATACAD1 DATE, @DATACAD2 DATE , @DATUAL1 DATE , @DATUAL2 DATE, @DATARES1 DATE, @DATARES2 DATE,
@PDCNUM INT, @COD char(15) , @DESCR CHAR(60), @FORNE CHAR(60), @CODFOR INT, @USUCAD CHAR(60), @USURES CHAR(60), @COMPRADOR CHAR(60)


SET @DATACAD1 = ''
SET @DATACAD2 = ''
SET @DATARES1 = '' 
SET @DATARES2 = ''
SET @DATUAL1 = ''
SET @DATUAL2 = ''
SET @COD = '1640054'
SET @CODFOR = ''
SET @FORNE = ''
SET @DESCR = ''
SET @COMPRADOR = ''
SET @PDCNUM = ''


SELECT TBS045.COMCOD AS COD_COMPRADOR, 
TBS046.COMNOM AS COMPRADOR, 
TBS045.PDCNUM AS PD_COMPRAS, 
TBS006.FORCOD AS COD_FOR, 
FORNOM AS FORNECEDOR,
PDCDATCAD AS DATA_CADASTRO,
PDCUSUCAD AS USU_CADASTROU,
PDCUSURES AS USU_RESIDUO,
PDCDATRES AS DATA_RESIDUO,
PDCUSUALT AS USU_ATUALIZOU,
PDCDATALT AS DATA_ATUALIZOU,
PROCOD AS CODIGO,
PDCDES AS DESCRIÇÃO,
RTRIM(PDCUNI)+''+ CASE WHEN PDCQTDEMB > 1 THEN +' c/'+ RTRIM(CONVERT(DECIMAL,PDCQTDEMB,0)) ELSE '' END AS UNI_MED,
PDCQTD AS QTD,
PDCQTDENT AS QTD_ENTRADA,
PDCQTDRES AS QTD_RESIDUO,
CASE WHEN (PDCQTD - PDCQTDENT - PDCQTDRES ) > 0 THEN PDCQTD ELSE 0 END AS QTD_ABERTO,
dbo.PDCTOTITE(TBS0451.PDCEMPCOD,TBS0451.PDCNUM,PDCITE) as VALOR_ITEM,
(dbo.PDCTOTITE(TBS0451.PDCEMPCOD,TBS0451.PDCNUM,PDCITE) / PDCQTD) * PDCQTDENT AS VALOR_ENT,
(dbo.PDCTOTITE(TBS0451.PDCEMPCOD,TBS0451.PDCNUM,PDCITE) / PDCQTD) * PDCQTDRES AS VALOR_RES, 
CASE WHEN (PDCQTD - PDCQTDENT - PDCQTDRES ) > 0 THEN dbo.PDCTOTITE(TBS0451.PDCEMPCOD,TBS0451.PDCNUM,PDCITE) ELSE 0 END AS VALOR_ABERTO,
PDCOBS AS OBS,
PDCDATPEN AS PREV_ENT

FROM TBS0451(NOLOCK)FULL OUTER JOIN TBS045 (NOLOCK)ON TBS0451.PDCNUM = TBS045.PDCNUM
				FULL OUTER JOIN TBS046 (NOLOCK) ON TBS045.COMCOD = TBS046.COMCOD
				FULL OUTER JOIN TBS006 (NOLOCK) ON TBS045.FORCOD = TBS006.FORCOD AND TBS045.FOREMPCOD = TBS006.FOREMPCOD

WHERE 
PDCDATCAD BETWEEN CASE WHEN @DATACAD1 = '' THEN '17530101' ELSE @DATACAD1 END AND CASE WHEN @DATACAD2 = '' THEN '20301231' ELSE @DATACAD2 END
AND PDCDATRES BETWEEN CASE WHEN @DATARES1 = '' THEN '17530101' ELSE @DATARES1 END AND CASE WHEN @DATARES2 = '' THEN '20301231' ELSE @DATARES2 END
AND PDCDATALT BETWEEN CASE WHEN @DATUAL1 = '' THEN '17530101' ELSE @DATUAL1 END AND CASE WHEN @DATUAL2 = '' THEN '20301231' ELSE @DATUAL2 END
AND (TBS046.COMNOM LIKE(CASE WHEN @COMPRADOR = '' THEN TBS046.COMNOM -- FUNCIONA NO SQL , ASSIM '% %'- FUNCIONA NO RS 
ELSE RTRIM(UPPER(@COMPRADOR))+'%' END) OR TBS046.COMNOM IS NULL)
AND PROCOD LIKE(CASE WHEN @COD = 0 THEN '% %' ELSE RTRIM(@COD)+'%' END)
AND PDCDES LIKE(CASE WHEN @DESCR = '' THEN '% %' ELSE RTRIM(UPPER(@DESCR))+'%' END)
AND FORNOM LIKE(CASE WHEN @FORNE = '' THEN '% %' ELSE RTRIM(UPPER(@FORNE))+'%' END)
AND TBS045.PDCNUM in(CASE WHEN @PDCNUM = '' THEN TBS045.PDCNUM ELSE @PDCNUM END)
AND TBS006.FORCOD in(CASE WHEN @CODFOR = '' THEN TBS006.FORCOD ELSE @CODFOR END)
