-- SOMA VALOR

-- SOMA QTD DEVOLUÇÃO
ISNULL(E.QTD1,0) + ISNULL(E.QTD2,0) + ISNULL(E.QTD3,0) AS DEVQTDTOT,

-- SOMA DEVOLUÇÕES
ISNULL(E.VAL1,0) + ISNULL(E.VAL2,0) + ISNULL(E.VAL3,0) AS DEVVALTOT,

-- dias uteis 
dbo.FCNDIASUTEIS (@data1de, @data1ate) - 
(	SELECT count(*) 
	FROM #FER 
	WHERE 
	(Sg_UF = 'SJ' or Tp_Feriado = 1 ) and 
	(Nr_Ano between substring(convert(char(10),@data1de,103),7,4) and substring(convert(char(10),@data1ate,103),7,4) OR Nr_Ano between 0 and 0) and 
	 Nr_Mes between substring(convert(char(10),@data1de,103),4,2) and substring(convert(char(10),@data1ate,103),4,2) and 
	 Nr_Dia between substring(convert(char(10),@data1de,103),1,2) and substring(convert(char(10),@data1ate,103),1,2)) AS DIA1,

dbo.FCNDIASUTEIS (@data2de, @data2ate) - 
(	SELECT count(*) 
	FROM #FER
	WHERE 
	(Sg_UF = 'SJ' or Tp_Feriado = 1 ) AND
	(Nr_Ano between substring(convert(char(10),@data2de,103),7,4) and substring(convert(char(10),@data2ate,103),7,4) OR Nr_Ano between 0 and 0) AND
	 Nr_Mes between substring(convert(char(10),@data2de,103),4,2) and substring(convert(char(10),@data2ate,103),4,2) AND
	 Nr_Dia between substring(convert(char(10),@data2de,103),1,2) and substring(convert(char(10),@data2ate,103),1,2)) AS DIA2,

dbo.FCNDIASUTEIS (@data3de, @data3ate) - 
(	SELECT count(*) 
	FROM #FER
	WHERE 
	(Sg_UF = 'SJ' or Tp_Feriado = 1 ) AND
	(Nr_Ano between substring(convert(char(10),@data3de,103),7,4) and substring(convert(char(10),@data3ate,103),7,4) OR Nr_Ano between 0 and 0) AND
	 Nr_Mes between substring(convert(char(10),@data3de,103),4,2) and substring(convert(char(10),@data3ate,103),4,2) AND
	 Nr_Dia between substring(convert(char(10),@data3de,103),1,2) and substring(convert(char(10),@data3ate,103),1,2)) AS DIA3,
	 
SUM(ISNULL(D.VAL1,0) + ISNULL(D.VAL2,0) + ISNULL(D.VAL3,0) + 
	ISNULL(C.VAL1,0) + ISNULL(C.VAL2,0) + ISNULL(C.VAL3,0)) OVER (PARTITION BY 1) AS VALOR,
	
	
CASE WHEN 	SUM(ISNULL(D.VAL1,0) + ISNULL(D.VAL2,0) + ISNULL(D.VAL3,0) + 
			ISNULL(C.VAL1,0) + ISNULL(C.VAL2,0) + ISNULL(C.VAL3,0)) OVER (PARTITION BY 1) <= 0
	THEN 0
	ELSE (ISNULL(D.VAL1,0) + ISNULL(D.VAL2,0) + ISNULL(D.VAL3,0) + 
			ISNULL(C.VAL1,0) + ISNULL(C.VAL2,0) + ISNULL(C.VAL3,0)) / 
			
			SUM(ISNULL(D.VAL1,0) + ISNULL(D.VAL2,0) + ISNULL(D.VAL3,0) + 
			ISNULL(C.VAL1,0) + ISNULL(C.VAL2,0) + ISNULL(C.VAL3,0)) OVER (PARTITION BY 1)
END * 100 AS PORC

INTO #ABC
FROM 
TBS010 A (NOLOCK)
LEFT JOIN #SALDO B (NOLOCK) ON A.PROCOD = B.PROCOD
LEFT JOIN #CORP  C (NOLOCK) ON A.PROCOD = C.PROCOD
LEFT JOIN #LOJA  D (NOLOCK) ON A.PROCOD = D.PROCOD
LEFT JOIN #DEV   E (NOLOCK) ON A.PROCOD = E.PROCOD

WHERE 
(ISNULL(D.QTD1,0) + ISNULL(D.QTD2,0) + ISNULL(D.QTD3,0) +  -- AQUI ESTOU SOMANDO TUDO PORQUE QUERO VER QUALQUER QUANTIDADE
ISNULL(C.QTD1,0) + ISNULL(C.QTD2,0) + ISNULL(C.QTD3,0) +  -- PORQUE DEPENDENDO DA CONTA PODERIA DAR ZERO, SE EU SUBTRAIR AS DEVOLUÇÕES
ISNULL(E.QTD1,0) + ISNULL(E.QTD2,0) + ISNULL(E.QTD3,0) > 0 OR
ISNULL(EST1,0) <> 0 OR ISNULL(EST2,0) <> 0 OR ISNULL(ESTQTDRES,0) <> 0 OR ISNULL(ESTQTDPEN,0) <> 0 OR ISNULL(ESTQTDCMP,0) <> 0) AND -- SALDOS # DE ZERO 
A.MARCOD =(CASE WHEN @marcod = '' THEN A.MARCOD ELSE Upper(RTRIM(@marcod)) END) AND 
A.MARNOM LIKE(CASE WHEN @marcanom = '' THEN '% %' ELSE Upper(RTRIM(@marcanom))+'%' END) AND 
A.PRODES LIKE (CASE WHEN @PRODES = '' THEN '% %' ELSE UPPER(RTRIM(@PRODES))+'%' END) 

ORDER BY 
ISNULL(D.VAL1,0) + ISNULL(D.VAL2,0) + ISNULL(D.VAL3,0) + 
ISNULL(C.VAL1,0) + ISNULL(C.VAL2,0) + ISNULL(C.VAL3,0) DESC

-- SOMA AS QTDS VENDIDAS / PELA QUANTIDADE DE DIAS UTEIS, CASO >0 , SOMA SALDO TOTAL + COMPRAS + SALDO CD - PENDENTE / (VEN_UTEIS) 

CASE WHEN -- SABER QUANTIDADE DIARIA DAS VENDAS
		VENQTDTOT / ((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))> 0 
	THEN 
		str(round((ESTQTDDIS + ESTQTDCMP - ESTQTDPEN)/ -- QUANTOS DIAS AINDA TENHO EM ESTOQUE
	(VENQTDTOT / ((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0),10,0)
	ELSE 0 
END AS 'TEMPO RESTANTE',


-- USUARIO ESCOLHE A QUANTIDADE DE DIAS UTEIS PARA REPOSIÇÃO 
-- @DIAS - (SOMA AS QTDS VENDIDAS / PELA QUANTIDADE DE DIAS UTEIS, CASO >0 , 
-- SOMA SALDO TOTAL + COMPRAS + SALDO CD - PENDENTE / (QTDS VENDAS / PELA QUANTIDADE DE DIAS UTEIS)) * VEN_UTEIS CASO <=0.9 É 0 (ZERO)
-- ELSE @DIAS - (SOMA AS QTDS VENDIDAS / PELA QUANTIDADE DE DIAS UTEIS, CASO >0 , 
-- SOMA SALDO TOTAL + COMPRAS + SALDO CD - PENDENTE / (QTDS VENDAS / PELA QUANTIDADE DE DIAS UTEIS)) (

				(VENQTDTOT /

			ELSE 
				0

			((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0)))) <=0.9 
	THEN 0 
	ELSE 
		((@DIAS - 
		(CASE WHEN -- SABER QUANTIDADE DIARIA DAS VENDAS
				VENQTDTOT / 
				((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))> 0

				(VENQTDTOT / 
				((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0),10,0)
			ELSE 
				0

			((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))))

dbo.FCNDIASUTEIS (@data3de, @data3ate) - 
(	SELECT count(*) 
	FROM #FER
	WHERE 
	(Sg_UF = 'SJ' or Tp_Feriado = 1 ) AND
	(Nr_Ano between substring(convert(char(10),@data3de,103),7,4) and substring(convert(char(10),@data3ate,103),7,4) OR Nr_Ano between 0 and 0) AND

	FROM #FER
	WHERE 
	(Sg_UF = 'SJ' or Tp_Feriado = 1 ) AND
	(Nr_Ano between substring(convert(char(10),@data3de,103),7,4) and substring(convert(char(10),@data3ate,103),7,4) OR Nr_Ano between 0 and 0) AND

-- CRIANDO TABELA TEMP COM OS VALORES REAIS DE DESCONTO

if object_id('TempDB.dbo.#TMP') is not null
   drop table #TMP
   
	select
	M2_DAT,
	M2_PROCOD,
	M2_QTD,
	M2_VALBRU - M2_ABT + M2_ACRCUP - 
	CASE WHEN  A.DESCONTO <> ISNULL(B.DESCONTO,0) AND M2_ITE = (SELECT TOP 1 M2_ITE 
								FROM #MSLAUX E 
																
								WHERE 
								A.M2_COO = E.M2_COO and A.M2_ECF = E.M2_ECF AND A.M2_CUP = E.M2_CUP 
																
								ORDER BY 
								M2_DESCUP DESC)
																
		THEN
			CASE WHEN A.DESCONTO > ISNULL(B.DESCONTO,0) 
				THEN ROUND(M2_DESCUP,2) - (A.DESCONTO - ISNULL(B.DESCONTO,0))
				ELSE ROUND(M2_DESCUP,2) + (ISNULL(B.DESCONTO,0) - A.DESCONTO)
			END
		ELSE ROUND(M2_DESCUP,2)
	END AS M2_LIQUIDO
	
	INTO #TMP	
	FROM (
		SELECT
		M2_DAT, 
		M2_ITE,M2_PROCOD,
		M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , 
		M2_DESCUP ,
		M2_ACRCUP,
		M2_ABT ,
		M2_VALBRU,
		M2_QTD,
		SUM(ISNULL(ROUND(M2_DESCUP,2),0)) OVER (PARTITION BY M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA ) AS DESCONTO
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 1 ) AS A 
			
		LEFT JOIN (
		SELECT 
		M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , M2_VALBRU AS DESCONTO 
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 2 ) AS B ON A.M2_LOJ = B.M2_LOJ AND A.M2_CUP = B.M2_CUP AND A.M2_COO= B.M2_COO AND 
														A.M2_ECF = B.M2_ECF AND A.M2_CXA = B.M2_CXA

if object_id('TempDB.dbo.#ABC') is not null
   drop table #ABC

SELECT 
Identity(INT,1,1) as ordem,
UNIDADE,
A.PROSTATUS,
A.PROLOCFIS,
A.PROCOD,
A.PRODES,
MARCA,
A.PROUM1,

ISNULL(EST1,0) AS ESTOQUE,
ISNULL(EST2,0) AS LOJA,
ISNULL(EST1,0)+ISNULL(EST2,0) AS ESTQTDDIS,
ISNULL(ESTQTDRES,0) AS ESTQTDRES,
ISNULL(ESTQTDPEN,0) AS ESTQTDPEN,
ISNULL(ESTQTDCMP,0) AS ESTQTDCMP,

-- 1° PERIODO QTD
ISNULL(D.QTD1,0) AS LOJAQTD1,
ISNULL(C.QTD1,0) AS CORPQTD1,
ISNULL(E.QTD1,0) AS DEVQTD1,

-- 1° PERIODO VALOR
ISNULL(D.VAL1,0) AS LOJAVAL1,
ISNULL(C.VAL1,0) AS CORPVAL1,
ISNULL(E.VAL1,0) AS DEVVAL1,

-- 2° PERIODO QTD
ISNULL(D.QTD2,0) AS LOJAQTD2,
ISNULL(C.QTD2,0) AS CORPQTD2,
ISNULL(E.QTD2,0) AS DEVQTD2,

-- 2° PERIODO VALOR
ISNULL(D.VAL2,0) AS LOJAVAL2,
ISNULL(C.VAL2,0) AS CORPVAL2,
ISNULL(E.VAL2,0) AS DEVVAL2,

-- 3° PERIODO QTD
ISNULL(D.QTD3,0) AS LOJAQTD3,
ISNULL(C.QTD3,0) AS CORPQTD3,
ISNULL(E.QTD3,0) AS DEVQTD3,

-- 3° PERIODO VALOR
ISNULL(D.VAL3,0) AS LOJAVAL3,
ISNULL(C.VAL3,0) AS CORPVAL3,
ISNULL(E.VAL3,0) AS DEVVAL3,

-- SOMA QTD
ISNULL(D.QTD1,0) + ISNULL(D.QTD2,0) + ISNULL(D.QTD3,0) + 
ISNULL(C.QTD1,0) + ISNULL(C.QTD2,0) + ISNULL(C.QTD3,0) AS VENQTDTOT ,

-- SOMA VALOR 
ISNULL(D.VAL1,0) + ISNULL(D.VAL2,0) + ISNULL(D.VAL3,0) + 
ISNULL(C.VAL1,0) + ISNULL(C.VAL2,0) + ISNULL(C.VAL3,0)  AS VENVALTOT ,

-- SOMA QTD DEVOLUÇÃO
ISNULL(E.QTD1,0) + ISNULL(E.QTD2,0) + ISNULL(E.QTD3,0) AS DEVQTDTOT,

-- SOMA DEVOLUÇÕES
ISNULL(E.VAL1,0) + ISNULL(E.VAL2,0) + ISNULL(E.VAL3,0) AS DEVVALTOT,
	 
SUM(ISNULL(D.VAL1,0) + ISNULL(D.VAL2,0) + ISNULL(D.VAL3,0) + 
	ISNULL(C.VAL1,0) + ISNULL(C.VAL2,0) + ISNULL(C.VAL3,0)) OVER (PARTITION BY 1) AS VALOR,
	
	
CASE WHEN 	SUM(ISNULL(D.VAL1,0) + ISNULL(D.VAL2,0) + ISNULL(D.VAL3,0) + 
			ISNULL(C.VAL1,0) + ISNULL(C.VAL2,0) + ISNULL(C.VAL3,0)) OVER (PARTITION BY 1) <= 0
	THEN 0
	ELSE (ISNULL(D.VAL1,0) + ISNULL(D.VAL2,0) + ISNULL(D.VAL3,0) + 
			ISNULL(C.VAL1,0) + ISNULL(C.VAL2,0) + ISNULL(C.VAL3,0)) / 
			
			SUM(ISNULL(D.VAL1,0) + ISNULL(D.VAL2,0) + ISNULL(D.VAL3,0) + 
			ISNULL(C.VAL1,0) + ISNULL(C.VAL2,0) + ISNULL(C.VAL3,0)) OVER (PARTITION BY 1)
END * 100 AS PORC,

@DIA1 AS DIA1,
@DIA2 AS DIA2,
@DIA3 AS DIA3

INTO #ABC
FROM 
#TBS010 A (NOLOCK)
LEFT JOIN #SALDO B (NOLOCK) ON A.PROCOD = B.PROCOD
LEFT JOIN #CORP  C (NOLOCK) ON A.PROCOD = C.PROCOD
LEFT JOIN #LOJA  D (NOLOCK) ON CONVERT(INT,A.PROCOD) = D.PROCOD
LEFT JOIN #DEV   E (NOLOCK) ON A.PROCOD = E.PROCOD

WHERE 
(ISNULL(D.QTD1,0) + ISNULL(D.QTD2,0) + ISNULL(D.QTD3,0) +  -- AQUI ESTOU SOMANDO TUDO PORQUE QUERO VER QUALQUER QUANTIDADE
ISNULL(C.QTD1,0) + ISNULL(C.QTD2,0) + ISNULL(C.QTD3,0) +  -- PORQUE DEPENDENDO DA CONTA PODERIA DAR ZERO, SE EU SUBTRAIR AS DEVOLUÇÕES
ISNULL(E.QTD1,0) + ISNULL(E.QTD2,0) + ISNULL(E.QTD3,0) > 0 OR
ISNULL(EST1,0) <> 0 OR ISNULL(EST2,0) <> 0 OR ISNULL(ESTQTDRES,0) <> 0 OR ISNULL(ESTQTDPEN,0) <> 0 OR ISNULL(ESTQTDCMP,0) <> 0) -- SALDOS # DE ZERO 

ORDER BY 
ISNULL(D.VAL1,0) + ISNULL(D.VAL2,0) + ISNULL(D.VAL3,0) + 
ISNULL(C.VAL1,0) + ISNULL(C.VAL2,0) + ISNULL(C.VAL3,0) DESC

	THEN

	(VENQTDTOT / ((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0),10,0)
	ELSE 0

-- @DIAS - (SOMA AS QTDS VENDIDAS / PELA QUANTIDADE DE DIAS UTEIS, CASO >0 ,

-- ELSE @DIAS - (SOMA AS QTDS VENDIDAS / PELA QUANTIDADE DE DIAS UTEIS, CASO >0 , 
-- SOMA SALDO TOTAL + COMPRAS + SALDO CD - PENDENTE / (QTDS VENDAS / PELA QUANTIDADE DE DIAS UTEIS)) (

CASE WHEN 
	((@DIAS - 
		(CASE WHEN -- SABER QUANTIDADE DIARIA DAS VENDAS
				VENQTDTOT / 
				((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))> 0 
			THEN 
				str(round((ESTQTDDIS + ESTQTDCMP - ESTQTDPEN)/ -- QUANTOS DIAS AINDA TENHO EM ESTOQUE
				(VENQTDTOT / 
				((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0),10,0)
			ELSE 
				0 
		END)) * (VENQTDTOT / 
			((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0)))) <=0.9 
	THEN 0 
	ELSE 
		((@DIAS - 
		(CASE WHEN -- SABER QUANTIDADE DIARIA DAS VENDAS
				VENQTDTOT / 
				((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))> 0 
			THEN 
				str(round((ESTQTDDIS + ESTQTDCMP - ESTQTDPEN)/ -- QUANTOS DIAS AINDA TENHO EM ESTOQUE
				(VENQTDTOT /

			ELSE 
				0 
		END)) * (VENQTDTOT / 
			((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))))
END AS 'QTD PARA REPOR' ,
