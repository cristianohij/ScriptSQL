DECLARE  @NFENUM INT, @FORCOD INT , @FORDES CHAR (60), @PROCODDE CHAR(15) , @PROCODATE CHAR(15) ,@MARCOD INT , @MARNOM CHAR(60), @PRODES CHAR(60), @ESTLOC INT, 
@SALDO CHAR (1) --, @ZERO CHAR (1)

SET @NFENUM = 0
SET @FORCOD = 0
SET @FORDES = ''
SET @PROCODDE =''
SET @PROCODATE =''
SET @MARCOD = 0
SET @MARNOM = ''
SET @PRODES = ''
SET @ESTLOC = 1
SET @SALDO = 'N'
--SET @ZERO = 'N'

IF OBJECT_ID ('TempDB.dbo.#BARRAS') is not null
	drop table #BARRAS

SELECT
		Identity(INT,1,1) AS RANK,		
		A.PROCOD,
		B.NFENUM AS NFENUM,
		B.NFECHAACE AS NFECHAACE,
		CAST(CAST(A.NFECODBARXML as NUMERIC(16,0)) AS VARCHAR(20)) AS NFECODBARXML,
		isnull(A.NFEFATOR,0) AS EMBXML,
		ltrim(rtrim(str(B.NFECOD))) + ' - ' + rtrim(B.NFENOM) AS NFECOD,
		E.FORCGC AS FORCGC, 
		B.NFEDATEFE+B.NFEHOREFE AS NFEDATEFE
		
		into #BARRAS
		FROM
		TBS0591 A (NOLOCK) 
		LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND 
		A.NFENUM = B.NFENUM AND A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND A.SEREMPCOD = B.SEREMPCOD
		LEFT JOIN TBS006 E (NOLOCK) ON B.NFECOD = E.FORCOD 
		
		WHERE 
		B.NFECOD = CASE WHEN @FORCOD = 0 THEN B.NFECOD ELSE @FORCOD END AND
		B.NFENOM LIKE CASE WHEN @FORDES = '' THEN B.NFENOM ELSE(RTRIM(LTRIM(upper(@FORDES)))+'%') END AND
		E.FORCGC NOT IN ('05118717000237','05118717000156','44125185000136','52080207000117','09135487000194') AND
		E.FORCGC NOT LIKE('65069593000%') AND  
		B.NFETIPENT = 'XML' AND
		B.NFETIP = 'N' AND 
		B.NFECAN = 'N' AND
		A.NFECODBARXML <> '' 
		
		ORDER BY B.NFEDATEFE+B.NFEHOREFE DESC
