DECLARE  @NFENUM INT, @FORCOD INT , @FORDES CHAR (60), @PROCODDE CHAR(15) , @PROCODATE CHAR(15) ,@MARCOD INT , @MARNOM CHAR(60), @PRODES CHAR(60)
SET @NFENUM = 0
SET @FORCOD = 0
SET @FORDES = ''
SET @PROCODDE =''
SET @PROCODATE =''
SET @MARCOD = 0
SET @MARNOM = ''
SET @PRODES =''


--- DIVERGENCIA DE NCM
IF OBJECT_ID ('TempDB.dbo.#TBS059') is not null
drop table #TBS059

SELECT
A.PROCOD,
B.NFENUM AS NFENUM,
B.NFECHAACE AS NFECHAACE,
A.NFENCMXML AS NFENCMXML,
B.NFECOD AS NFECOD, 
B.NFENOM AS NFENOM,
E.FORCGC AS FORCGC, 
B.NFEDATEFE,
NFECESTXML

INTO #TBS059
FROM
TBS0591 A (NOLOCK) 
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND 
A.NFENUM = B.NFENUM AND A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND A.SEREMPCOD = B.SEREMPCOD
LEFT JOIN TBS006 E (NOLOCK) ON B.NFECOD = E.FORCOD 

WHERE 
B.NFECOD = CASE WHEN @FORCOD = 0 THEN B.NFECOD ELSE @FORCOD END AND
B.NFENOM LIKE CASE WHEN @FORDES = '' THEN B.NFENOM ELSE(RTRIM(LTRIM(upper(@FORDES)))+'%') END AND
E.FORCGC NOT IN ('05118717000237','05118717000156','44125185000136','52080207000117','09135487000194') AND
E.FORCGC NOT LIKE('65069593000%') AND  
B.NFETIPENT = 'XML' AND
B.NFETIP = 'N' AND 
B.NFECAN = 'N' AND
(A.NFENCMXML <> '' OR CONVERT(FLOAT,NFECESTXML) > 0)

ORDER BY NFEDATEFE


IF OBJECT_ID ('TempDB.dbo.#TBS010') is not null
drop table #TBS010

SELECT
MARNOM,
MARCOD AS MAR,
PROSTATUS,
case when len(D.MARCOD) = 4 
	then rtrim(D.MARCOD) + ' - ' + rtrim(D.MARNOM) 
	else right(('00' + ltrim(str(D.MARCOD))),3) + ' - ' + rtrim(D.MARNOM)
end collate Latin1_General_BIN as MARCOD,
D.PROCOD AS PROCOD, 
PRODES AS PRODES,
PROCLAFIS AS PROCLAFIS,
PROCEST AS PROCEST

INTO #TBS010
FROM
TBS010 D (NOLOCK) 

WHERE 
D.PROCOD IN (SELECT PROCOD FROM #TBS059) 

;WITH 

T10 AS (SELECT
		MARNOM,
		PROSTATUS,
		MARCOD,
		D.PROCOD AS PROCOD, 
		PRODES AS PRODES,
		PROCLAFIS AS PROCLAFIS,
		PROCEST AS PROCEST
		
		FROM
		#TBS010 D
		
		WHERE 
		D.PROCOD BETWEEN @PROCODDE AND CASE WHEN @PROCODATE ='' THEN 'Z' ELSE @PROCODATE END AND 
		D.PROCOD IN (SELECT PROCOD FROM #TBS059) AND
		MAR = CASE WHEN @MARCOD = 0 THEN D.MAR ELSE @MARCOD END AND 
		MARNOM LIKE (RTRIM(LTRIM(upper(@MARNOM)))+'%') AND 
		PRODES LIKE (RTRIM(LTRIM(upper(@PRODES)))+'%') 
		),

NCM AS (SELECT
		PROCOD,
		NFENUM,
		NFECHAACE,
		NFENCMXML,
		NFECOD, 
		NFENOM,
		FORCGC, 
		NFEDATEFE
		
		FROM #TBS059
		
		WHERE 
		NFENCMXML <> ''),
		
CEST AS (SELECT
		PROCOD,
		NFENUM,
		NFECHAACE,
		NFECESTXML,
		NFECOD, 
		NFENOM,
		FORCGC, 
		NFEDATEFE
		
		FROM #TBS059
		
		WHERE 
		NFECESTXML <> '')

SELECT 
PROSTATUS,
MARCOD,
A.PROCOD, 
PRODES,
PROCLAFIS,
ISNULL((SELECT TOP 1 NFENCMXML FROM NCM B WHERE A.PROCOD COLLATE Latin1_General_BIN= B.PROCOD ORDER BY NFEDATEFE DESC),'') AS NCMXML,
ISNULL((SELECT TOP 1 NFECOD FROM NCM B WHERE A.PROCOD COLLATE Latin1_General_BIN= B.PROCOD ORDER BY NFEDATEFE DESC),0) AS NCMNFECOD,
ISNULL((SELECT TOP 1 NFENOM FROM NCM B WHERE A.PROCOD COLLATE Latin1_General_BIN= B.PROCOD ORDER BY NFEDATEFE DESC),'') AS NCMNFENOM,
ISNULL((SELECT TOP 1 NFENUM FROM NCM B WHERE A.PROCOD COLLATE Latin1_General_BIN= B.PROCOD ORDER BY NFEDATEFE DESC),0) AS NCMNFENUM,
ISNULL((SELECT TOP 1 NFECHAACE FROM NCM B WHERE A.PROCOD COLLATE Latin1_General_BIN= B.PROCOD ORDER BY NFEDATEFE DESC),'') AS NCMNFECHAACE,

PROCEST,
ISNULL((SELECT TOP 1 NFECESTXML FROM CEST B WHERE A.PROCOD COLLATE Latin1_General_BIN= B.PROCOD ORDER BY NFEDATEFE DESC),'') AS CESTXML,
ISNULL((SELECT TOP 1 NFECOD FROM CEST B WHERE A.PROCOD COLLATE Latin1_General_BIN= B.PROCOD ORDER BY NFEDATEFE DESC),0) AS CESTNFECOD,
ISNULL((SELECT TOP 1 NFENOM FROM CEST B WHERE A.PROCOD COLLATE Latin1_General_BIN= B.PROCOD ORDER BY NFEDATEFE DESC),'') AS CESTNFENOM,
ISNULL((SELECT TOP 1 NFENUM FROM CEST B WHERE A.PROCOD COLLATE Latin1_General_BIN= B.PROCOD ORDER BY NFEDATEFE DESC),0) AS CESTNFENUM,
ISNULL((SELECT TOP 1 NFECHAACE FROM CEST B WHERE A.PROCOD COLLATE Latin1_General_BIN= B.PROCOD ORDER BY NFEDATEFE DESC),'') AS CESTNFECHAACE

FROM 
T10 A
WHERE 
(A.PROCLAFIS  COLLATE Latin1_General_BIN <> (SELECT TOP 1 NFENCMXML FROM NCM B WHERE A.PROCOD COLLATE Latin1_General_BIN= B.PROCOD ORDER BY NFEDATEFE DESC) OR 
A.PROCEST COLLATE Latin1_General_BIN <> (SELECT TOP 1 NFECESTXML FROM CEST B WHERE A.PROCOD COLLATE Latin1_General_BIN= B.PROCOD ORDER BY NFEDATEFE DESC)) --AND
--MARNOM LIKE ( CASE WHEN @MARNOM = '' THEN MARNOM ELSE RTRIM(LTRIM(upper(@MARNOM)))+'%' END ) AND 
--PRODES LIKE (RTRIM(LTRIM(upper(@PRODES)))+'%')
