   drop table #VENDAS
   
SELECT M2_PROCOD,Ceiling(SUM(M2_QTD)/3/4.35) as SEMANAL 
INTO #VENDAS FROM MSL002 (NOLOCK)
WHERE MSL002.M2_DAT between getdate() - CONVERT(DATETIME,91) and getdate()
and MSL002.M2_TIPREG='01' 
and MSL002.M2_REGCAN='F'
GROUP BY M2_PROCOD



SELECT  rank() OVER (ORDER BY A.PROCOD) AS [RANK],
A.PROSTATUS AS STS, 
RTRIM(LTRIM(A.PROCOD)) AS CÓDIGO, 
RTRIM(LTRIM(A.PRODES)) AS DESCRIÇÃO, 
RTRIM(LTRIM(A.MARNOM)) AS MARCA, 
CASE WHEN B.TDPVALPROI<=GETDATE() AND B.TDPVALPROF >= GETDATE() AND B.TDPPROLOJ = 'S' THEN B.TDPPREPRO1 ELSE B.TDPPRELOJ1 END AS PRECO1,
CASE WHEN PROUM2QTD > 0 
		THEN CASE WHEN B.TDPVALPROI<=GETDATE() AND B.TDPVALPROF >= GETDATE() AND B.TDPPROLOJ = 'S'
				THEN B.TDPPREPRO2*PROUM2QTD 
				ELSE B.TDPPRELOJ2*PROUM2QTD END
		ELSE 0 END AS PRECO2,
'*'+RTRIM(A.PROCOD)+'*'  AS CODBAR1,
CASE WHEN PROUM1QTD = 1 THEN RTRIM(PROUM1) ELSE
	CASE WHEN PROUM1QTD > 1 THEN rtrim(PROUM1) + ' C/' + rtrim(CAST(PROUM1QTD AS DECIMAL(10,0))) +''+ RTRIM(PROUMV) ELSE '' END END  as UN1,
CASE WHEN PROUM2QTD > 0 THEN rtrim(PROUM2) + ' C/' + rtrim(CAST(PROUM2QTD AS DECIMAL(10,0)))+''+ RTRIM(PROUM1) ELSE '' END  as UN2,
ISNULL(VND_GZ.SEMANAL,1) AS PP,
A.PROUM1 AS UNpp

FROM  TBS010 A(NOLOCK) JOIN TBS031 B (NOLOCK) ON A.PROCOD = B.TDPPROCOD

FULL OUTER JOIN (SELECT M2_PROCOD,SEMANAL FROM #VENDAS (NOLOCK)) AS VND_GZ ON A.PROCOD = VND_GZ.M2_PROCOD AND B.TDPPROCOD = VND_GZ.M2_PROCOD

FULL OUTER JOIN TBS032 C (nolock) ON A.PROCOD = C.PROCOD

WHERE A.PROCOD IN ((SELECT PROCOD FROM T ))
AND A.MARNOM LIKE(CASE WHEN @MARNON = '' THEN '% %' ELSE '%'+UPPER(RTRIM(@MARNON))+'%' END)
AND A.PROSTATUS IN( CASE WHEN @STA = 'T' THEN A.PROSTATUS ELSE @STA END )
AND TDPDATATU BETWEEN CASE WHEN @ATL = 'N' THEN '17530101' ELSE @dataDE END AND CASE WHEN @dataAte='' THEN GETDATE() ELSE @dataAte END
AND (C.ESTQTDATU-C.ESTQTDRES >= case when @somenteComSaldo='N' then -9999 else 0.001 end AND C.ESTLOC = 2)
					
ORDER BY [RANK]

DROP TABLE T
