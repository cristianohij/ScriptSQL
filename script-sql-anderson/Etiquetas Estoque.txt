-- TABELA DE FILTROS NA TBS010

if object_id('TempDB.dbo.#TBS010') is not null
		drop table #TBS010

SELECT  
-- rank() OVER (ORDER BY PROLOCFIS,A.PROCOD) AS [RANK],
A.PROSTATUS AS STS, 
RTRIM(LTRIM(A.PROCOD)) AS CÓDIGO, 
RTRIM(LTRIM(A.PRODES))  AS DESCRIÇÃO, 
RTRIM(LTRIM(MARNOM)) AS MARCA, 
RTRIM(LTRIM(PROLOCFIS)) AS LOC, 
RTRIM(LTRIM(PROLOCFIS2)) AS LOC2,
'*'+RTRIM(A.PROCOD)+'*'  AS CODBAR1,
CASE WHEN PROUM2QTD > 0 
	THEN '*'+RTRIM(LTRIM(A.PROCOD))+'-2*' 
	ELSE '' 
END AS  CODBAR2,
CASE WHEN PROUM1QTD = 1 
	THEN RTRIM(PROUM1) 
	ELSE
		CASE WHEN PROUM1QTD > 1 
			THEN rtrim(PROUM1) + ' C/ ' + rtrim(CAST(PROUM1QTD AS DECIMAL(10,0))) +' '+ RTRIM(PROUMV) 
			ELSE '' 
		END 
END  as UN1,
CASE WHEN PROUM2QTD > 0 
	THEN rtrim(PROUM2) + ' C/ ' + rtrim(CAST(PROUM2QTD AS DECIMAL(10,0)))+' '+ RTRIM(PROUM1) 
	ELSE '' 
END AS UN2

INTO #TBS010
FROM  
TBS010 A (NOLOCK)

WHERE 
A.MARCOD IN (CASE WHEN @MARCOD = 0 THEN A.MARCOD ELSE @MARCOD END) AND
A.PROCOD COLLATE DATABASE_DEFAULT IN ((SELECT PROCOD FROM #T )) AND
PROLOCFIS LIKE(CASE WHEN @Locger = '' THEN PROLOCFIS ELSE rtrim(upper((@Locger)))+'%' END)
