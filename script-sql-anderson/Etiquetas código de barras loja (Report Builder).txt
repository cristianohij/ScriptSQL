IF OBJECT_ID ('TempDB.dbo.#tbs') is not null
drop table #tbs


SELECT  
rank() OVER (ORDER BY A.PROCOD) AS [RANKS],
A.PROSTATUS AS STS, 
RTRIM(LTRIM(A.PROCOD)) AS CÓDIGO, 
RTRIM(LTRIM(A.PRODES)) AS DESCRIÇÃO, 
RTRIM(LTRIM(A.MARNOM)) AS MARCA, 

'*'+RTRIM(A.PROCOD)+'*'  AS CODBAR1,
CASE WHEN PROUM2QTD > 0 
		THEN '*'+RTRIM(LTRIM(A.PROCOD))+'-2*' ELSE '' END AS  CODBAR2,


convert(decimal (12,2),
CASE WHEN B.TDPVALPROI<=GETDATE() AND B.TDPVALPROF >= GETDATE() AND B.TDPPROLOJ = 'S' 
	THEN ROUND(B.TDPPREPRO1,2) 
	ELSE ROUND(B.TDPPRELOJ1,2) 
END) AS PRECO1,

convert(decimal (12,2),
CASE WHEN PROUM2QTD > 0 
	THEN 
		CASE WHEN B.TDPVALPROI<=GETDATE() AND B.TDPVALPROF >= GETDATE() AND B.TDPPROLOJ = 'S'
			THEN ROUND(B.TDPPREPRO2,2)*round(PROUM2QTD,2)
			ELSE ROUND(B.TDPPRELOJ2,2)*round(PROUM2QTD,2) 
		END
	ELSE 0 
END) AS PRECO2,
		

CASE WHEN PROUM1QTD = 1
	THEN RTRIM((SELECT UNIDES FROM TBS011 D (NOLOCK) WHERE A.PROUM1 = D.UNICOD))   
	ELSE
		CASE WHEN PROUM1QTD > 1 
		THEN rtrim(PROUM1) + ' C/' + rtrim(CAST(PROUM1QTD AS DECIMAL(10,0))) +''+ RTRIM(PROUMV) 
		ELSE '' 
	END 
END  as UN1,
'R$' sifra,

CASE WHEN PROUM2QTD > 0 and TDPPRELOJ2 > 0
	THEN rtrim(PROUM2) + ' C/' + rtrim(CAST(PROUM2QTD AS DECIMAL(10,0)))+''+ RTRIM(PROUM1) 
	ELSE '' 
END  as UN2,

ISNULL(CONVERT(INT,ROUND(C.QUANT,0)),0) AS PP,

A.PROUM1 AS UNpp

into #tbs
FROM TBS010 A (NOLOCK) 
LEFT JOIN TBS031 B (NOLOCK) ON A.PROCOD = B.TDPPROCOD
LEFT JOIN #MSL002 C (NOLOCK) ON CONVERT(NUMERIC,A.PROCOD) = C.PROCOD
LEFT JOIN TBS032 D (nolock) ON A.PROCOD = D.PROCOD

WHERE 
A.PROCOD COLLATE Latin1_General_BIN IN ((SELECT PROCOD FROM #T ))-- AND 
--A.MARNOM LIKE(CASE WHEN @MARNON = '' THEN '% %' ELSE '%'+UPPER(RTRIM(@MARNON))+'%' END) AND 
--A.PROSTATUS IN( CASE WHEN @STA = 'T' THEN A.PROSTATUS ELSE @STA END ) AND 
--convert(DATE,TDPDATATU) BETWEEN CASE WHEN @ATL = 'N' THEN '17530101' ELSE @dataDE END AND CASE WHEN @dataAte='' THEN GETDATE()+1 ELSE @dataAte END
--and convert(nvarchar(8),TDPDATATU,108) BETWEEN CASE WHEN @ATL = 'N' THEN '00:00:00' ELSE @horaDe END AND CASE WHEN @horaAte='' THEN '23:59:59' ELSE @horaAte END AND
--(D.ESTQTDATU-D.ESTQTDRES > case when @somenteComSaldo='N' then -999999 else 0 end 
AND D.ESTLOC = 2 --AND
--A.MARCOD = (CASE WHEN @MARCOD = 0 THEN A.MARCOD ELSE RTRIM(LTRIM(@MARCOD))END) AND 
--A.PRODES LIKE (CASE WHEN @PRODES = '' THEN '% %' ELSE UPPER(RTRIM(@PRODES))+'%' END) 
					
ORDER BY [RANKS]
