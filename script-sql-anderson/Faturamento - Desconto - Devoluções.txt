-- DESCONTO CONCEDIDO POR VENDEDOR 

declare @dateDe datetime, @dateAte datetime , @nomecli char (50), @codcli smallint,
@codven smallint, @nomven VARchar(60), @Coluna int, @DECR CHAR (1)

set @dateDe = '20170501'
set @dateAte = '20170531'
--set @codcli = ''
--set @nomecli = ''
set @codven = 114
set @nomven = 'ama'
set @Coluna = 2 
SET @DECR = 'S'

;with 
-- TABCLI as (select CLICOD from TBS002 where VENCOD IN (CASE WHEN @codven = 0 THEN VENCOD ELSE @codven END)), 
TABCLI as (
		SELECT CLICOD
		FROM TBS002 (NOLOCK)
		WHERE CLICGC LIKE ('65069593000%') OR CLICGC LIKE('05118717000%') OR CLICGC LIKE('44125185000%') OR CLICGC LIKE('52080207000%')) ,

--select CLICOD, CLINOM from TBS002 where VENCOD = '32'

TABVEN as (
		SELECT 
		VENCOD,
		-- ISNULL((SELECT VENCOD FROM TBS002 E (NOLOCK) WHERE B.NFSCLICOD = E.CLICOD),0) AS VENCODCLI,
		ISNULL((SELECT RTRIM(LTRIM(VENNOM)) FROM TBS004 D (NOLOCK) WHERE A.VENCOD = D.VENCOD),'SEM VENDEDOR') AS VENNOM,
		CONVERT(CHAR(12),NFSDATEMI,103) AS DATA,
		CONVERT(CHAR(12),NFSDATEMI,111) AS DATA2,
		A.SNESER,
		NFSCLICOD,
		NFSCLINOM,
		NFSNUM,
		NFSTIP,
		CASE WHEN NFSTIP = 'C'
			THEN ROUND(NFSCOMTOT,2)
			ELSE ROUND(dbo.NFSTOTLIQ(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER),2)
		END AS NFSTOTLIQ, 
		ROUND(dbo.NFSVDDTOT(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER),2) AS NFSVDDTOT, 
		ROUND(dbo.NFSTOTDEVMEN(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER),2) AS NFSTOTDEV		
				
		FROM 
		TBS067 A (NOLOCK)
		LEFT JOIN TBS080 D (NOLOCK) ON A.NFSNUM = D.ENFNUM AND A.SNEEMPCOD = D.SNEEMPCOD AND A.SNESER = D.SNESER 
			  
		WHERE 
		NFSDATEMI BETWEEN @dateDe AND @dateAte AND
		D.ENFSIT = 6 AND
		D.ENFTIPDOC = 1 AND
		NFSFINNFE NOT IN (4) AND 
		VENCOD in (@codven) AND 
		NFSCLICOD NOT IN (SELECT CLICOD FROM TABCLI)
		 )

SELECT * FROM TABVEN 

WHERE 
VENNOM LIKE (CASE WHEN @nomven = '' THEN VENNOM ELSE RTRIM(LTRIM(UPPER(@nomven)))+'%' END)
