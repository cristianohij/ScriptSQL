;WITH ENTRS AS (

	SELECT 
	D.PROCOD AS COD,
	convert(int,subString(convert(char(6),NFEDATEFE,112),1,4)) AS ANO,
	convert(int,subString(convert(char(6),NFEDATEFE,112),5,2)) AS MES ,

		THEN round(AVG(convert(decimal(10,4),(dbo.PDCTOTITE(D.PDCEMPCOD,D.PDCNUM,PDCITE) / PDCQTD )/ PDCQTDEMB)),2)
		ELSE 0 
	END AS VALORMD
	/*SUM(PDCQTDENT) AS QTDPDC,

--	D.PDCNUM AS PEDCOMPRAS,

	-- INTO CUSTOENTRADA2015 --USEI ESSE
	FROM #TBS0591 A (NOLOCK) 
	LEFT JOIN TBS0451 D (nolock) ON A.PROCOD = D.PROCOD AND A.NFEPEDNUM = D.PDCNUM
	LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND A.NFETIP = B.NFETIP AND 
	A.SERCOD = B.SERCOD AND A.SEREMPCOD = B.SEREMPCOD
	LEFT JOIN TBS0597 C (NOLOCK) ON A.NFEEMPCOD = C.NFEEMPCOD AND A.NFECOD = C.NFECOD AND A.NFENUM = C.NFENUM AND A.NFETIP = C.NFETIP AND 
	A.SERCOD = C.SERCOD AND A.SEREMPCOD = C.SEREMPCOD
	
	WHERE 
	A.NFETIP = 'N' and 
	NFEDATEFE between '20160101' and '20161231' and

	NFENOM not like('TANBY%') AND 
	B.NFECAN = 'N'  AND 
	D.PDCNUM > 0 -- AND PROCOD IS NULL
	
	GROUP BY 
	convert(char(6),NFEDATEFE,112), D.PROCOD  --, PDCNUM, A.NFENUM, C.NFENFDQTD
	
	)--,PDCNUM)
