WITH 
VENDASGZ AS (select PROCOD,PRODES,MARCOD,
SUM(M2_QTD) AS VENDAS_GZ,
SUM(M2_VALTOT)-SUM(M2_ABT) AS VALOR_VENDAS_GZ
from MSL002 A join TBS010 B ON A.M2_PROCOD = B.PROCOD

where M2_DAT between '20150601'and '20150831' and
M2_PROCOD LIKE ('%') AND 
M2_TIPREG = '01' AND 
M2_REGCAN = 'F'

GROUP BY M2_PROCOD,PRODES,PROCOD,MARCOD),

VENDASINTEGROS AS (SELECT D.PROCOD,D.PRODES,
SUM(A.NFSQTD*A.NFSQTDEMB) AS VENDAS_INT,
SUM(A.NFSQTD*(NFSPRE-((NFSPRE/100)*NFSPDDITE))) AS VALOR_VENDAS_INT 
FROM TBS010 D JOIN TBS0671 A ON
	A.PROCOD = D.PROCOD
	JOIN TBS067 B ON 
	A.NFSEMPCOD = B.NFSEMPCOD and 
	A.NFSNUM = B.NFSNUM and 
	A.SNEEMPCOD = B.SNEEMPCOD and 
	A.SNESER = B.SNESER
	join TBS042 C on 
	C.TESCOD = A.TESCOD


WHERE A.PROCOD LIKE ('%')AND 
NFSDATEMI between '20150601'and '20150831' and
NFSTIP = 'N' and NFSCAN = 'N' and TESCNTVEN = 'S'


GROUP BY D.PROCOD,D.PRODES)

SELECT rtrim(ltrim(A.PROCOD)) AS CÓD,
A.PROLOCFIS AS LOC,
PROSTATUS AS STATUS,
A.PRODES AS DESCRIÇÃO,
PROUM1 AS UN1,
replace(str(isnull(SaldoEstND.SDISEST,0),10,4),'.',',') as 'Saldo Disponivel Estoque',
replace(str(isnull(SDISLOJA,0),10,4),'.',',') as 'Saldo Disponivel Loja',
' ' as 'Saldo Loja+Est',
rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) as 'MARCA',
replace(str(isnull(B.VENDAS_GZ,0),10,4),'.',',')as 'Vendas GZ',
replace(str(isnull(C.VENDAS_INT,0),10,4),'.',',') as 'Vendas INT',
' ' as 'Vendas GZ+INT',
replace(str(isnull(F.TDPPRELOJ1,0),10,4),'.',',') as 'PRECO LOJA',
replace(str(isnull(F.TDPCUSBAS,0),10,4),'.',',') as 'PRECO CUSTO',
isnull(rtrim(D.GRUCOD),0) + ' - ' + isnull(RTRIM(D.GRUDES),0) as 'Grupo',
isnull(rtrim(E.SUBGRUCOD),0) + ' - '+ isnull(rtrim(E.SUBGRUDES),0) as 'Sub Grupo',
replace(str(isnull(rtrim(B.VALOR_VENDAS_GZ),0),10,2),'.',',') as 'Valor GZ',
replace(str(isnull(rtrim(C.VALOR_VENDAS_INT),0),10,2),'.',',') as 'Valor Int',
' ' as 'Valor Total'
FROM TBS010 A

Full Outer Join VENDASGZ B ON A.PROCOD = B.PROCOD 
Full Outer Join VENDASINTEGROS C ON A.PROCOD = C.PROCOD

Full Outer Join TBS012 D ON
	A.GRUCOD = D.GRUCOD AND 
	A.GRUEMPCOD = D.GRUEMPCOD
	
Full Outer Join TBS0121 E ON
	A.GRUCOD = E.GRUCOD AND
	A.GRUEMPCOD = E.GRUEMPCOD AND
	A.SUBGRUCOD = E.SUBGRUCOD
	
Full Outer Join TBS031 F ON
	A.PROCOD = F.TDPPROCOD
	
	Full Outer Join(select A.PROCOD as PRO, sum(A.ESTQTDATU-A.ESTQTDRES) as SDISLOJA, A.ESTQTDRES as SRESLOJA
from TBS032 A join TBS010 B on A.PROCOD = B.PROCOD
where A.ESTLOC in (2) and (A.ESTQTDATU <> 0 or A.ESTQTDRES <> 0 or A.ESTQTDPEN <> 0 or A.ESTQTDCMP <> 0)
group by A.PROCOD, A.ESTQTDRES) as SaldoLoja on A.PROCOD = SaldoLoja.PRO

	Full Outer Join (select A.PROCOD as PRO, sum(A.ESTQTDATU-A.ESTQTDRES) as SDISEST
from TBS032 A join TBS010 B on A.PROCOD = B.PROCOD
where A.ESTLOC in (1) and (A.ESTQTDATU <> 0 or A.ESTQTDRES <> 0 or A.ESTQTDPEN <> 0 or A.ESTQTDCMP <> 0)
group by A.PROCOD) as SaldoEstND on A.PROCOD = SaldoEstND.PRO

WHERE A.PROCOD LIKE ('%') 

GROUP BY A.PROLOCFIS, PROSTATUS,A.PROCOD, A.PRODES,A.MARCOD,A.MARNOM,PROUM1,B.VENDAS_GZ,C.VENDAS_INT,
D.GRUCOD,D.GRUDES,E.SUBGRUCOD,E.SUBGRUDES,F.TDPPRELOJ1,SaldoEstND.SDISEST,SDISLOJA,F.TDPCUSBAS,B.VALOR_VENDAS_GZ,
C.VALOR_VENDAS_INT


order by A.PROCOD
