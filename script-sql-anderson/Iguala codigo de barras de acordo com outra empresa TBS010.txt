SELECT 
A.PROCOD AS PROCOD,
--ISNULL(ISNULL(ISNULL(B.PRODES, C.PRODES), D.PRODES), E.PRODES) AS PRODES,
--B.PRODES AS DESND,
C.PRODES AS DESPP,
D.PRODES AS DESMI,
E.PRODES AS DESBB,

--B.PROCODBAR1 AS ND,
C.PROCODBAR1 AS PP,
D.PROCODBAR1 AS MI,
E.PROCODBAR1 AS BB,

CASE WHEN D.PROCODBAR1 <> '' AND E.PROCODBAR1 = '' 
	THEN 1
	ELSE 
		CASE WHEN D.PROCODBAR1 = '' AND E.PROCODBAR1 <> ''
			THEN 2
			ELSE 
				CASE WHEN  D.PROCODBAR1 = E.PROCODBAR1
					THEN 3
					ELSE 0
				END 
		END
END AS SIT,

CASE WHEN 
		CASE WHEN D.PROCODBAR1 <> '' AND E.PROCODBAR1 = '' 
			THEN 1
			ELSE 
				CASE WHEN D.PROCODBAR1 = '' AND E.PROCODBAR1 <> ''
					THEN 2
					ELSE 
						CASE WHEN  D.PROCODBAR1 = E.PROCODBAR1
							THEN 3
							ELSE 0
						END 
				END
		END = 1 AND SOUNDEX(RTRIM(LTRIM(C.PRODES))) = SOUNDEX(RTRIM(LTRIM(D.PRODES)))
	THEN 'OK'
	ELSE
		CASE WHEN 
				CASE WHEN D.PROCODBAR1 <> '' AND E.PROCODBAR1 = '' 
					THEN 1
					ELSE 
						CASE WHEN D.PROCODBAR1 = '' AND E.PROCODBAR1 <> ''
							THEN 2
							ELSE 
								CASE WHEN  D.PROCODBAR1 = E.PROCODBAR1
									THEN 3
									ELSE 0
								END 
						END
				END = 2 AND SOUNDEX(RTRIM(LTRIM(C.PRODES))) = SOUNDEX(RTRIM(LTRIM(E.PRODES)))
			THEN 'OK'
			ELSE
				CASE WHEN 
						CASE WHEN D.PROCODBAR1 <> '' AND E.PROCODBAR1 = '' 
							THEN 1
							ELSE 
								CASE WHEN D.PROCODBAR1 = '' AND E.PROCODBAR1 <> ''
									THEN 2
									ELSE 
										CASE WHEN  D.PROCODBAR1 = E.PROCODBAR1
											THEN 3
											ELSE 0
										END 
								END
						END = 3 AND SOUNDEX(RTRIM(LTRIM(C.PRODES))) = SOUNDEX(RTRIM(LTRIM(E.PRODES)))
					THEN 'OK'
					ELSE 'NAO'
				END 
		END 
END AS SIT2	

INTO #TBS010
FROM #IND A
-- LEFT JOIN #ND B ON A.PROCOD = B.PROCOD 
LEFT JOIN #PP C ON A.PROCOD = C.PROCOD 
LEFT JOIN #MI D ON A.PROCOD = D.PROCOD 
LEFT JOIN #BB E ON A.PROCOD = E.PROCOD 

WHERE 
D.PROCODBAR1 <>'' or E.PROCODBAR1 <>''


SELECT 
*, 
CASE WHEN SIT = 1 AND SIT2 = 'OK'
	THEN MI
	ELSE
		CASE WHEN SIT = 2 AND SIT2 = 'OK'
			THEN BB
			ELSE 
				CASE WHEN SIT = 3 AND SIT2 = 'OK'
					THEN BB
					ELSE ''
				END
		END
	END
AS BARRAS

INTO #FIM
FROM #TBS010

WHERE 
SIT2 = 'OK' 


SELECT 
*

FROM #FIM 

WHERE 
CONVERT(DECIMAL(20,0),BARRAS) <> 0

'PROCODBAR1' 			AS LOGATT,
'' 						AS LOGVALANT,

'SQL'					AS LOGROTDES
