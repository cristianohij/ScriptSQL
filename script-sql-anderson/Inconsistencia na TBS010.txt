SELECT 
A.PROSTATUS AS STA ,
RTRIM(A.PROCOD) AS CODIGO, 
RTRIM(A.PRODES) AS DESCRIÇÃO, 
RTRIM(A.MARCOD) AS 'COD.MAR', 
RTRIM(MARNOM) AS 'NOM.MAR',
RTrim(ltrim(PROUM1)) AS UN1, 
REPLACE(A.PROUM1QTD,'.',',')AS QTD1,
RTRIM(LTRIM(PROUM2)) AS UN2, 
REPLACE(A.PROUM2QTD,'.',',') AS QTD2,
RTRIM(LTRIM(PROUM3)) AS UN3, 
REPLACE(A.PROUM3QTD,'.',',') AS QTD3,
RTRIM(LTRIM(PROUM4)) AS UN4 ,
REPLACE(A.PROUM4QTD,'.',',') AS QTD4,
RTRIM(LTRIM(PROCODBAR1)) AS CODBAR1,
RTRIM(LTRIM(PROCODBAR2)) AS CODBAR2,
RTRIM(LTRIM(PROCODBAR3)) AS CODBAR3,
RTRIM(LTRIM(PROCODBAR4)) AS CODBAR4,
REPLACE(SUM(ESTQTDATU),'.',',') AS 'SALDO GERAL'

FROM TBS010 A (NOLOCK) 
LEFT JOIN TBS032 B (NOLOCK) ON A.PROCOD = B.PROCOD
WHERE PROUM1 = '' AND PROUM1QTD > 0 
-- PROUM2QTD = 1 
-- BARRAS IGUAIS
-- PROUM2 = '' 
-- PROUM3 = ''
-- PROUM4 = ''
-- 

GROUP BY 
A.PROSTATUS,
A.PROCOD,
A.PRODES,
A.MARCOD,
MARNOM,
PROUM1,
A.PROUM1QTD,
PROUM2,
A.PROUM2QTD,
PROUM3 ,
A.PROUM3QTD,
PROUM4 ,
A.PROUM4QTD,
PROCODBAR1,
PROCODBAR2,
PROCODBAR3 ,
PROCODBAR4
