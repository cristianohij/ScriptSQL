SELECT 
	'CD' as 'UNIDADE',
	rtrim(ltrim(A.PROCOD)) collate Latin1_General_BIN as 'CODIGO',
	A.PRODES collate Latin1_General_BIN as 'DESCRICAO',							
	case when len(A.MARCOD) = 4 then rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) else right(('00' + ltrim(str(A.MARCOD))),3) + ' - ' + rtrim(A.MARNOM) end collate Latin1_General_BIN as 'MARCA',
	A.PROUM1 collate Latin1_General_BIN as 'UNID. MEDIDA',
	ISNULL(B.LOJA,0) AS 'LOJA',						
	ISNULL(B.EST,0) AS 'DISPONIVEL',
	ISNULL(B.RES,0) AS 'RESERVADO',
	ISNULL(B.PEN,0) AS 'PENDENTE',
	ISNULL(C.QTD,0) AS 'COMPRAS',
	C.DAT AS 'PRE_ENT',
	ISNULL(B.CMP,0) AS 'COMPRASTT'
	
	from (SELECT PROCOD ,
		  ISNULL((SELECT ESTQTDATU-ESTQTDRES AS LOJA FROM TBS032CD B (NOLOCK) WHERE ESTLOC = 2 AND B.PROCOD = C.PROCOD),0) AS LOJA , 
		  ISNULL((SELECT ESTQTDATU-ESTQTDRES AS EST FROM TBS032CD B (NOLOCK) WHERE ESTLOC = 1 AND B.PROCOD = C.PROCOD),0) AS EST,
		  SUM(ESTQTDRES) AS RES , SUM(ESTQTDPEN) AS PEN, SUM(ESTQTDCMP) AS CMP,DATULTALT
		  FROM TBS032CD C (NOLOCK) 
		  GROUP BY PROCOD,DATULTALT) B
		  
		  LEFT JOIN COMPRASCD C (NOLOCK) ON B.PROCOD COLLATE Latin1_General_CI_AS = C.PROCOD
		  
		  LEFT JOIN TBS010 A (NOLOCK) ON B.PROCOD COLLATE Latin1_General_CI_AS = A.PROCOD		
		  
	
	where 
	(A.PROCOD LIKE(CASE WHEN @COD = '' THEN '% %' ELSE Upper(RTRIM(@COD))+'%' END))  
	AND(A.PRODES LIKE(CASE WHEN @DESCRI = '' THEN '% %' ELSE Upper(RTRIM(@DESCRI))+'%' END))
