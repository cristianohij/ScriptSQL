WITH 
ND as  (SELECT 
	A.PROCOD collate Latin1_General_CI_AS AS COD,
	'ND' collate Latin1_General_CI_AS  as 'Unidade',
	ISNULL(SUM(A.NFEQTD*A.NFEQTDEMB),0)  AS QNT_ND,
	ISNULL(SUM(A.NFEQTD*(NFEPRE-((NFEPRE/100)*NFEPDDITE))),0)  AS VALOR_ND
	
	FROM nd.SIBD.dbo.TBS010 C 
	JOIN nd.SIBD.dbo.TBS0591 A ON
	C.PROCOD = A.PROCOD
	join nd.SIBD.dbo.TBS059 B  on 
	A.NFEEMPCOD = B.NFEEMPCOD AND
	A.NFECOD = B.NFECOD AND
	A.NFENUM = B.NFENUM AND
	A.NFETIP = B.NFETIP AND
	A.SERCOD = B.SERCOD AND
	A.SEREMPCOD = B.SEREMPCOD 

	WHERE B.NFETIP = 'N' AND B.NFECAN = 'N' AND
	A.PROCOD LIKE ('%')
	AND NFEDATENT  between '20150501' and '20150731'
	GROUP BY A.PROCOD,C.PROUM1 ),

CD as ( SELECT 
	A.PROCOD collate Latin1_General_CI_AS AS COD,
	'CD' collate Latin1_General_CI_AS as 'Unidade',
	ISNULL(SUM(A.NFEQTD*A.NFEQTDEMB),0)  AS QNT_CD,
	ISNULL(SUM(A.NFEQTD*(NFEPRE-((NFEPRE/100)*NFEPDDITE))),0)  AS VALOR_CD 
	
	FROM cd.SIBD.dbo.TBS010 C 
	JOIN cd.SIBD.dbo.TBS0591 A ON
	C.PROCOD = A.PROCOD
	join cd.SIBD.dbo.TBS059 B  on
	A.NFEEMPCOD = B.NFEEMPCOD AND
	A.NFECOD = B.NFECOD AND
	A.NFENUM = B.NFENUM AND
	A.NFETIP = B.NFETIP AND
	A.SERCOD = B.SERCOD AND
	A.SEREMPCOD = B.SEREMPCOD
	
	WHERE B.NFETIP = 'N' AND B.NFECAN = 'N' AND
	A.PROCOD LIKE ('%')
	AND NFEDATENT  between '20150501' and '20150731'
	GROUP BY A.PROCOD,C.PROUM1),

TTÉ AS (SELECT 
	A.PROCOD collate Latin1_General_CI_AS AS COD,
	'TTÉ' collate Latin1_General_CI_AS as 'Unidade',
	ISNULL(SUM(A.NFEQTD*A.NFEQTDEMB),0)  AS QNT_TTE,
	ISNULL(SUM(A.NFEQTD*(NFEPRE-((NFEPRE/100)*NFEPDDITE))),0)  AS VALOR_TTE 
	
	
	FROM tt.SIBD.dbo.TBS010 C 
	JOIN tt.SIBD.dbo.TBS0591 A ON
	C.PROCOD = A.PROCOD
	join tt.SIBD.dbo.TBS059 B on
	A.NFEEMPCOD = B.NFEEMPCOD AND
	A.NFECOD = B.NFECOD AND
	A.NFENUM = B.NFENUM AND
	A.NFETIP = B.NFETIP AND
	A.SERCOD = B.SERCOD AND
	A.SEREMPCOD = B.SEREMPCOD
  
	WHERE B.NFETIP = 'N' AND B.NFECAN = 'N' AND
	A.PROCOD LIKE ('%')
	AND NFEDATENT  between '20150501' and '20150731'
	GROUP BY A.PROCOD,C.PROUM1)
	
SELECT 
	A.PROCOD AS CÓD,
	replace(str(isnull(SUM(B.QNT_ND+C.QNT_CD+D.QNT_TTE),0),10,4),'.',',') collate Latin1_General_CI_AS AS QNT_COMPRADA,
	replace(str(isnull(SUM(B.VALOR_ND+C.VALOR_CD+D.VALOR_TTE),0),10,4),'.',',') collate Latin1_General_CI_AS AS VALOR_INVES
	
	FROM TBS010 A 
	FULL OUTER JOIN ND B ON A.PROCOD = B.COD
	FULL OUTER JOIN CD C ON A.PROCOD = C.COD
	FULL OUTER JOIN TTÉ D ON A.PROCOD = D.COD
	
GROUP BY A.PROCOD
	
ORDER BY A.PROCOD
