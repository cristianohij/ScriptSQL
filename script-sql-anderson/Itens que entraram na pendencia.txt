-- TABELA FINAL DE SALDO E ENTRADA 

if object_id('TempDB.dbo.#SALDO') is not null
   drop table #SALDO

SELECT 
ISNULL(ISNULL(ISNULL(ISNULL(A.DATA,B.DATA),C.DATA),D.DATA),E.DATA) AS DATA,
ISNULL(ISNULL(ISNULL(ISNULL(A.PROCOD,B.PROCOD),C.PROCOD),D.PROCOD),E.PROCOD) AS PROCOD,
ISNULL(B.QUANT,0) + ISNULL(D.QUANT,0) AS ENT,
ISNULL(A.ESTQTDDIS,0) AS SALDOCD,
--ISNULL(A.ESTQTDCMP,0) AS CMPCD,
--ISNULL(A.ESTQTDPEN,0) AS PENCD,
--ISNULL(A.ESTQTDRES,0) AS RESCD,
--ISNULL(D.QUANT,0) AS ENTND,
ISNULL(C.ESTQTDDIS,0) AS SALDOND,
ISNULL(E.ESTQTDDIS,0) AS SALDLOJ,
ISNULL(C.ESTQTDCMP,0) + ISNULL(E.ESTQTDCMP,0) AS CMP,
--ISNULL(E.ESTQTDCMP,0) AS CMPLOJ,
ISNULL(C.ESTQTDPEN,0) AS PENND,
ISNULL(C.ESTQTDRES,0) AS RESND

INTO #SALDO
FROM 
#SALCD A
FULL OUTER JOIN #TBS059CD B ON A.PROCOD = B.PROCOD AND A.DATA = B.DATA
FULL OUTER JOIN #SALND C	ON ISNULL(A.PROCOD,B.PROCOD) COLLATE DATABASE_DEFAULT = C.PROCOD AND ISNULL(A.DATA,B.DATA) = C.DATA
FULL OUTER JOIN #TBS059ND D	ON ISNULL(ISNULL(A.PROCOD,B.PROCOD),C.PROCOD) COLLATE DATABASE_DEFAULT = D.PROCOD AND ISNULL(ISNULL(A.DATA,B.DATA),C.DATA) = D.DATA
FULL OUTER JOIN #SALNDLOJ E	ON ISNULL(ISNULL(ISNULL(A.PROCOD,B.PROCOD),C.PROCOD),D.PROCOD) COLLATE DATABASE_DEFAULT = E.PROCOD AND 
							   ISNULL(ISNULL(ISNULL(A.DATA,B.DATA),C.DATA),D.DATA) = CONVERT(DATE,E.DATA)

-- TABELA FINAL DE QUANTIDADES

if object_id('TempDB.dbo.#QTD') is not null
	drop table #QTD  

SELECT 
ISNULL(ISNULL(ISNULL(ISNULL(ISNULL(ISNULL(ISNULL(A.DATA,B.DATA),C.DATA),D.DATA),E.DATA),F.DATA),G.DATA),H.DATA) AS DATA,
ISNULL(ISNULL(ISNULL(ISNULL(ISNULL(ISNULL(ISNULL(A.PROCOD,B.PROCOD),C.PROCOD),D.PROCOD),E.PROCOD),F.PROCOD),G.PROCOD),H.PROCOD) AS PROCOD,
ISNULL(A.QUANT,0) 	AS NFSQUANT,
ISNULL(B.QUANT,0) 	AS LOJQUANT,
ISNULL(C.QUANT,0) 	AS PENQUANT,

ISNULL(E.QUANT,0) 	AS PENESTQUANT,
ISNULL(F.QTD,0) 	AS PENESTQTD,

ISNULL(H.QTD,0) 	AS ORCQTD

INTO #QTD
FROM 
#NFS A
FULL OUTER JOIN #MSL B 		ON A.PROCOD = B.PROCOD AND A.DATA = B.DATA
FULL OUTER JOIN #PEN C		ON ISNULL(A.PROCOD,B.PROCOD) COLLATE DATABASE_DEFAULT = C.PROCOD AND ISNULL(A.DATA,B.DATA) = C.DATA
FULL OUTER JOIN #PENQTD D	ON ISNULL(ISNULL(A.PROCOD,B.PROCOD),C.PROCOD) COLLATE DATABASE_DEFAULT = D.PROCOD AND ISNULL(ISNULL(A.DATA,B.DATA),C.DATA) = D.DATA
FULL OUTER JOIN #PENEST E	ON ISNULL(ISNULL(ISNULL(A.PROCOD,B.PROCOD),C.PROCOD),D.PROCOD) COLLATE DATABASE_DEFAULT = E.PROCOD AND 
							   ISNULL(ISNULL(ISNULL(A.DATA,B.DATA),C.DATA),D.DATA) = E.DATA
FULL OUTER JOIN #PENESTQTD F ON ISNULL(ISNULL(ISNULL(ISNULL(A.PROCOD,B.PROCOD),C.PROCOD),D.PROCOD),E.PROCOD) COLLATE DATABASE_DEFAULT = F.PROCOD AND 
							   ISNULL(ISNULL(ISNULL(ISNULL(A.DATA,B.DATA),C.DATA),D.DATA),E.DATA) = F.DATA
FULL OUTER JOIN #ORC G 		ON ISNULL(ISNULL(ISNULL(ISNULL(ISNULL(A.PROCOD,B.PROCOD),C.PROCOD),D.PROCOD),E.PROCOD),F.PROCOD) COLLATE DATABASE_DEFAULT = G.PROCOD AND 
							   ISNULL(ISNULL(ISNULL(ISNULL(ISNULL(A.DATA,B.DATA),C.DATA),D.DATA),E.DATA),F.DATA) = G.DATA
FULL OUTER JOIN #ORCQTD H ON ISNULL(ISNULL(ISNULL(ISNULL(ISNULL(ISNULL(A.PROCOD,B.PROCOD),C.PROCOD),D.PROCOD),E.PROCOD),F.PROCOD),G.PROCOD) COLLATE DATABASE_DEFAULT = 
								H.PROCOD AND 
							   ISNULL(ISNULL(ISNULL(ISNULL(ISNULL(ISNULL(A.DATA,B.DATA),C.DATA),D.DATA),E.DATA),F.DATA),G.DATA) = H.DATA

-----------------------------------------------------------------------------------------------------------------------------------------------------------------------

-- TABELA FINAL 

SELECT 
ROW_NUMBER ( ) OVER (ORDER BY ISNULL(A.PROCOD,B.PROCOD),ISNULL(A.DATA,B.DATA)) AS ORDEM,
CASE WHEN ROW_NUMBER ( ) OVER (ORDER BY ISNULL(A.PROCOD,B.PROCOD),ISNULL(A.DATA,B.DATA)) BETWEEN 1 AND 65000
	THEN 1 
	ELSE 
		CASE WHEN ROW_NUMBER ( ) OVER (ORDER BY ISNULL(A.PROCOD,B.PROCOD),ISNULL(A.DATA,B.DATA)) BETWEEN 65001 AND 130001
			THEN 2
			ELSE 
				CASE WHEN ROW_NUMBER ( ) OVER (ORDER BY ISNULL(A.PROCOD,B.PROCOD),ISNULL(A.DATA,B.DATA)) BETWEEN 130002 AND 195002
					THEN 3
					ELSE 
						CASE WHEN ROW_NUMBER ( ) OVER (ORDER BY ISNULL(A.PROCOD,B.PROCOD),ISNULL(A.DATA,B.DATA)) BETWEEN 195003 AND 260003
							THEN 4
							ELSE
								CASE WHEN ROW_NUMBER ( ) OVER (ORDER BY ISNULL(A.PROCOD,B.PROCOD),ISNULL(A.DATA,B.DATA)) BETWEEN 260004 AND 325004
									THEN 5
									ELSE
										CASE WHEN ROW_NUMBER ( ) OVER (ORDER BY ISNULL(A.PROCOD,B.PROCOD),ISNULL(A.DATA,B.DATA)) BETWEEN 325005 AND 390005
											THEN 6
											ELSE 7
										END
								END
						END
				END
		END
END AS RANK,
ISNULL(A.DATA,B.DATA) AS DATA,
ISNULL(A.PROCOD,B.PROCOD) AS PROCOD,
ISNULL(C.PRODES,'PRODUTO SEM CADASTRO') as PRODES,
ISNULL(
CASE WHEN len(C.MARCOD) = 4
	THEN rtrim(C.MARCOD) + ' - ' + rtrim(C.MARNOM) 
	else right(('00' + ltrim(str(C.MARCOD))),3) + ' - ' + rtrim(C.MARNOM) 
end, '000 - SEM MARCA') as MARCOD,
ISNULL(ENT,0) AS ENT ,
ISNULL(SALDOCD,0) AS SALDOCD,
ISNULL(SALDOND,0) AS SALDOND,
ISNULL(SALDLOJ,0) AS SALDLOJ,
ISNULL(CMP,0)  AS CMP,
ISNULL(PENND,0) AS PENND,
ISNULL(RESND,0) AS RESND,
ISNULL(NFSQUANT,0) 	AS NFSQUANT,
ISNULL(LOJQUANT,0) 	AS LOJQUANT,
ISNULL(PENQUANT,0) 	AS PENQUANT,
ISNULL(PENQTD,0) 	AS PENQTD,
ISNULL(PENESTQUANT,0) AS PENESTQUANT,
ISNULL(PENESTQTD,0) AS PENESTQTD,
ISNULL(ORCQUANT,0)	AS ORCQUANT,
ISNULL(ORCQTD,0) 	AS ORCQTD,
SUM(ISNULL(PENQTD,0) + ISNULL(ORCQTD,0)) OVER (PARTITION BY ISNULL(A.PROCOD,B.PROCOD)) AS VALOR

SELECT 
ANO,
MES,
CASE SUBSTRING(MES,1,2) 
	WHEN '01' THEN 'JAN'
	WHEN '02' THEN 'FEV'
	WHEN '03' THEN 'MAR'
	WHEN '04' THEN 'ABR'
	WHEN '05' THEN 'MAI'
	WHEN '06' THEN 'JUN'
	WHEN '07' THEN 'JUL'
	WHEN '08' THEN 'AGO'
	WHEN '09' THEN 'SET'
	WHEN '10' THEN 'OUT'
	WHEN '11' THEN 'NOV'
	WHEN '12' THEN 'DEZ'
END + '/' + LTRIM(STR(ANO)) AS MES1,
X,
A.PROCOD,
rtrim(PRODES) AS PRODES,

	then rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) 
	else right(('00' + ltrim(str(A.MARCOD))),3) + ' - ' + rtrim(A.MARNOM) 
end as MARNOM
