	-- QUANTIDADE DE SAIDA CONTANDO TUDO ATÉ AS NFSCAN -- 1° -- MES ANTERIOR AO MES ATUAL

ISNULL((SELECT SUM(D.NFSQTD*D.NFSQTDEMB) AS QTDSAI

	   D.NFSNUM = E.NFSNUM and 
	   D.SNEEMPCOD = E.SNEEMPCOD and 
	   D.SNESER = E.SNESER
	   
	   WHERE 
	   NFSDATEMI between '20150101' and '20151231'
	   
	   AND NFSTIP ='N' AND NFSMOVEST = 'S'
	   AND LESCOD = 2 
	   AND E.NFSDATEMI = B.NFSDATEMI
	   AND D.PROCOD = C.PROCOD
	   
	   GROUP BY PROCOD,NFSDATEMI),0) AS QTDSAILJ
	   
INTO #SAIDA
FROM TBS0671 C (nolock) JOIN TBS067 B (nolock)  ON 
C.NFSEMPCOD = B.NFSEMPCOD and 
C.NFSNUM = B.NFSNUM and 
C.SNEEMPCOD = B.SNEEMPCOD and 
C.SNESER = B.SNESER

WHERE NFSDATEMI between '20150101' and '20151231' AND NFSTIP ='N' AND NFSMOVEST = 'S'
	   
GROUP BY PROCOD,NFSDATEMI

ORDER BY NFSDATEMI

--COLOCAR SALDO KARDEX

SELECT 
rank() OVER (ORDER BY A.COD,A.DAT) AS [RANK],
A.COD,
isnull(SALDOINICIAL,0) as SALDOINICIAL ,
ISNULL(SALDOINILOJ,0) AS SALDOINILOJ ,

-- SAIDAS
ISNULL(SAIFOR,0) + ISNULL(QTDMNUSAI,0) AS INVSAI, -- SAIDA FORÇADA
ISNULL(SAIFORLJ,0) + ISNULL(QTDMNUSAILJ,0) AS INVSAILJ, -- SAIDA FORÇADA LOJA

ISNULL(QTDSAI,0) AS SAIDANF, -- SAIDA POR NF
ISNULL(QTDSAILJ,0) AS SAIDANFLJ, -- SAIDA POR NF LOJA 

ISNULL(QTDSAILOJ,0) AS SAIDACUP, -- SAIDA POR CUPOM FISCAL 

ISNULL(QTDDEV,0) AS SAIDADEV, -- SAIDA DE DEVOLUÇÃO
ISNULL(QTDDEVLJ,0) AS SAIDADEVLJ, -- SAIDA DE DEVOLUÇÃO LOJA

ISNULL(TRLOJASAI,0) AS TRLOJASAI , -- SAIDA POR TRANSFERENCIA DA LOJA

ISNULL(QTDSAI,0) + ISNULL(QTDSAILOJ,0) + ISNULL(QTDDEV,0) +ISNULL(SAIFOR,0) + ISNULL(QTDMNUSAI,0) as SAIDAS, -- SOMA DAS SAIDAS

ISNULL(SAIFORLJ,0)+ISNULL(QTDSAILJ,0)+ISNULL(QTDSAILOJ,0)+ISNULL(QTDDEVLJ,0) + ISNULL(TRLOJASAI,0) +ISNULL(QTDMNUSAILJ,0) AS SAIDASLJ , -- SOMA DAS SAIDAS LOJA + TRANSF SAI LOJA

A.DAT,

-- ENTRADAS
ISNULL(QTDENT,0) AS ENTRDNFE,  -- ENTRDA DE NOTAS
ISNULL(QTDENTLJ,0) AS ENTRDNFELJ, -- ENTRDA DE NOTAS DA LOJA 

ISNULL(ENTFOR,0) + ISNULL(QTDRESMAN,0) + ISNULL(QTDMNUENT,0) AS INVENT, -- ENTRADA FORÇADA + ENTRADA POR RESERVA MANUAL
ISNULL(ENTFORLJ,0)+ ISNULL(QTDMNUENTLJ,0) AS INVENTLJ , -- ENTRADA FORÇADA LOJA

ISNULL(QTDENTDEV,0) AS ENTDEV, -- ENTRADA POR DEVOLUÇÃO
ISNULL(QTDENTDEVLJ,0) AS ENTDEVLJ, -- ENTRADA POR DEVOLUÇÃO LOJA

ISNULL(QTDSAICAN,0) AS ENTSAICAN, -- ENTRADA POR NOTAS CANCELADAS

SUM(ESTQTDATU) AS ATU,--soma de todos os estoques, nao tem condiçao no where 
ISNULL(QTDSAI,0) AS SAI,
ISNULL(QTDSAILOJ,0) AS SAILOJ, 
ISNULL(QTDENT,0) AS ENT, 
ISNULL(MVIEST,0) AS MVEST, 
ISNULL(MVILOJA,0) AS MVLOJ, 
ISNULL(QTDDEV,0) AS DEV,
ISNULL(MVI4,0) AS MVI4, 
ISNULL(MVI5,0) AS MVI5, 
ISNULL(MVI6,0) AS MVI6, 
ISNULL(MVI7,0) AS MVI7,
ISNULL(MVI8,0) AS MVI8,

SUM(ESTQTDATU) + ISNULL(QTDSAI,0) + ISNULL(QTDSAILOJ,0) - ISNULL(QTDENT,0) - ISNULL(MVIEST,0) - ISNULL(MVILOJA,0) + ISNULL(QTDDEV,0) AS SALDO
--INTO #SALDOEST2015
FROM TBS032 T (NOLOCK) FULL OUTER JOIN ENTR E ON T.PROCOD = E.COD
		 	  FULL OUTER JOIN SAIDA S ON T.PROCOD = S.COD
			  FULL OUTER JOIN MVI M ON T.PROCOD = M.COD
			  FULL OUTER JOIN DEV D ON T.PROCOD = D.COD
			  FULL OUTER JOIN SAILOJ L ON T.PROCOD = L.COD
			  

WHERE --((ESTQTDATU) <> 0 OR ISNULL(QTDSAI,0) <> 0 OR ISNULL(QTDENT,0) <> 0 OR ISNULL(MVIEST,0)<> 0 OR ISNULL(MVILOJA,0)<>0 OR ISNULL(QTDDEV,0) <> 0) --PROCOD = '4250435'
(ESTQTDATU) + ISNULL(QTDSAI,0) - ISNULL(QTDENT,0) - ISNULL(MVIEST,0) - ISNULL(MVILOJA,0) + ISNULL(QTDDEV,0) <> 0
GROUP BY PROCOD,QTDSAI,QTDSAILOJ,QTDENT,MVILOJA,MVIEST,QTDDEV,MVI4,MVI5,MVI6, MVI7, MVI8
