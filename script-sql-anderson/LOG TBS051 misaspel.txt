SELECT A.PROCOD AS 'CÓDIGO',
B.PRODES AS 'DESCRIÇÃO' ,
case when len(B.MARCOD) = 4 then 
	rtrim(B.MARCOD) + ' - ' + rtrim(B.MARNOM) 
	else right(('00' + ltrim(str(B.MARCOD))),3) + ' - ' + rtrim(B.MARNOM) end as 'MARCA',
PROUM1 AS 'UNI1',
REPLACE(ESTQTDATU,'.',',') AS ATUAL,
REPLACE(ESTQTDRES,'.',',') AS RESERVADO,
REPLACE((ESTQTDATU - ESTQTDRES),'.',',') AS DISPONIVEL,
REPLACE(ESTQTDPEN,'.',',') AS PENDENTE,
REPLACE(ESTQTDCMP,'.',',') AS COMPRAS

FROM 
TBS032 A (NOLOCK) 
LEFT JOIN TBS010 B (NOLOCK) ON A.PROCOD = B.PROCOD 
WHERE ESTLOC = 2 AND (ESTQTDATU > 0 or ESTQTDPEN <> 0 OR ESTQTDCMP <> 0 OR ESTQTDRES <> 0)

ORDER BY A.PROCOD
