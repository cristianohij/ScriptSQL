DECLARE @Loc CHAR(17), @Sta char(1), @Marnom CHAR(30),@somenteComSaldo CHAR (1), @Cod CHAR(15), @Descr CHAR(60), @Opcao INT, @ESTLOC INT
SET @Loc = ''
set @Sta ='T'
SET @Marnom = 'acp'
SET @somenteComSaldo ='N'
SET @Cod = ''
SET @Descr = ''
SET @Opcao = 3
SET @ESTLOC = 1


IF @Opcao = 1 -- TODAS AS LOCALIZAÇOES 
SELECT 
SUBSTRING(RTRIM(LTRIM(PROLOCFIS)),1,2) AS RUA,
ROW_NUMBER() OVER(ORDER BY PROLOCFIS,A.PROCOD ) AS RANK,
'ND' AS EMPRESA,
A.PROSTATUS AS 'ST', 
RTRIM(LTRIM(A.PROCOD)) AS 'COD', 
RTRIM(A.PRODES) AS DESCRIÇÃO, 
CASE WHEN len(A.MARCOD) = 4 
		THEN rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) 
		ELSE RIGHT(('00' + ltrim(str(A.MARCOD))), 3) + ' - ' + rtrim(A.MARNOM) END AS 'MARCA', 
A.PROLOCFIS AS 'LOC 1',
CASE WHEN PROUM1QTD > 1 THEN RTRIM(LTRIM(PROUM1))+' '+ 
	RTRIM(CONVERT(DECIMAL,PROUM1QTD,0))+ '' + CASE WHEN PROUMV = '' THEN PROUM1 ELSE PROUMV END ELSE 
	RTRIM(CONVERT(DECIMAL,PROUM1QTD,0))+ '' + CASE WHEN PROUMV = '' THEN PROUM1 ELSE PROUMV END END AS 'UN1',
'' AS 'QTD'

FROM TBS010 AS A (nolock) 
INNER JOIN TBS032 AS B (nolock) ON A.PROCOD = B.PROCOD
				 
WHERE
(A.PROLOCFIS LIKE(CASE WHEN @Loc = '' THEN '% %' ELSE RTRIM(UPPER(@Loc)) + '%'END) ) AND 
(B.ESTLOC = @ESTLOC) AND 
A.PROSTATUS IN(CASE WHEN @Sta = 'T' THEN A.PROSTATUS ELSE UPPER(@Sta) END )AND 
A.MARNOM LIKE (CASE WHEN @Marnom = '' THEN '% %' ELSE RTRIM(UPPER(@Marnom))+ '%' END ) AND 
(B.ESTQTDATU-B.ESTQTDRES > case when @somenteComSaldo='N' then -999999 else 0 end )AND 
A.PROCOD LIKE (CASE WHEN @Cod = '' THEN '% %' ELSE RTRIM(@Cod)+'%' END)AND 
A.PRODES LIKE (CASE WHEN @Descr = '' THEN '% %' ELSE RTRIM(UPPER(@Descr)) + '%' END)


ORDER BY 'LOC 1'
