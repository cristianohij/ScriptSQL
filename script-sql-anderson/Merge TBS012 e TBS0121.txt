if object_id('TempDB.dbo.#TB12') is not null 
	drop table #TB12

SELECT TB12.* INTO #TB12 FROM OPENROWSET('SQLNCLI','192.168.1.205';'si';'123',
'SELECT * FROM SIBD.dbo.TBS012 NOLOCK') AS TB12

BEGIN TRAN 
MERGE TBS012 A
USING #TB12  B
ON A.GRUEMPCOD = B.GRUEMPCOD AND A.GRUCOD = B.GRUCOD 

WHEN MATCHED AND (A.GRUDES COLLATE Latin1_General_CI_AS <> B.GRUDES OR A.SUBGRUUIT <> B.SUBGRUUIT)
THEN UPDATE SET 
A.GRUDES = B.GRUDES ,
A.SUBGRUUIT = B.SUBGRUUIT

WHEN NOT MATCHED 
THEN INSERT (
GRUEMPCOD,
GRUCOD,
GRUDES,
GRUDATCAD,
SUBGRUUIT)

VALUES (
B.GRUEMPCOD,
B.GRUCOD,
B.GRUDES,
GETDATE(),
B.SUBGRUUIT)

OUTPUT 
INSERTED.*, $action, DELETED.* ;
