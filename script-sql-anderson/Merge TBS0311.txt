create procedure sp_MGTBS0311
AS 
BEGIN

SELECT Scr.* INTO #T311
FROM OPENROWSET('SQLNCLI','192.168.1.205';'si';'123',
     'SELECT * FROM SIBD.dbo.TBS0311 WHERE TDPHISDAT > GETDATE ()-5 ORDER BY TDPPROCOD') AS Scr

MERGE TBS0311 A  -- DESTINO
USING #T311 B (NOLOCK) -- ORIGEM
ON A.TDPPROCOD collate Latin1_General_BIN = B.TDPPROCOD and
A.TDPHISREG = B.TDPHISREG AND
A.TDPEMPCOD = B.TDPEMPCOD

WHEN NOT
MATCHED 

THEN 
INSERT (
TDPEMPCOD,
TDPPROCOD,
TDPHISREG,
TDPHISDAT,
TDPHISCPO,
TDPHISTIP,
TDPHISVAL,
TDPHISVALPRO,
TDPHISDATPROI,
TDPHISDATPROF,
TDPHISUSU,
TDPHISMOD)

VALUES (
B.TDPEMPCOD,
B.TDPPROCOD,
B.TDPHISREG,
B.TDPHISDAT,
B.TDPHISCPO,
B.TDPHISTIP,
B.TDPHISVAL,
B.TDPHISVALPRO,
B.TDPHISDATPROI,
B.TDPHISDATPROF,
B.TDPHISUSU,
B.TDPHISMOD)

OUTPUT 
HOST_NAME(),
GETDATE(),
UPPER($action),
INSERTED.*,
DELETED.*
INTO LOGTBS0311;

END
