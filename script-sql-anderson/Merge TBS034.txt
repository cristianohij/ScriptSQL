SELECT * into #TBS034 FROM cd.SIBD.dbo.TBS034

SELECT * FROM #TBS034

BEGIN TRAN 

MERGE TBS034 A
USING #TBS034  B
ON A.LESEMPCOD = B.LESEMPCOD AND A.LESCOD = B.LESCOD 

WHEN MATCHED AND (A.LESDES COLLATE Latin1_General_CI_AS <> B.LESDES )

THEN UPDATE SET 
A.LESDES = B.LESDES

WHEN NOT MATCHED 

THEN INSERT (
LESEMPCOD,
LESCOD,
LESDES,
LESDATCAD)

VALUES (
B.LESEMPCOD,
B.LESCOD,
B.LESDES,
GETDATE())

OUTPUT 
INSERTED.*, $action, DELETED.* ;
