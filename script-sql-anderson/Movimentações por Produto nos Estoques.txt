-- FILTROS

DECLARE @PROCOD CHAR(15), @DATADE DATETIME, @DATEATE DATETIME, @LMEACA CHAR(1), @LMEINFALT CHAR(1), @LMEDOC VARCHAR(10)
SET @PROCOD = '0'
SET @DATADE ='20170901'
SET @DATEATE='20170904'
SET @LMEACA= 'E'
SET @LMEINFALT = 'C'
SET @LMEDOC = '48061' 

SELECT 
LMEREG,
LMEDATHOR,
LMEDOC,
PROCOD, 
CASE WHEN LMEACA = 'E'
	THEN 'ENTROU'
	ELSE 'SAIU'
END AS LMEACA,
CASE LMEINFALT 
	WHEN 'C' THEN 'COMPRAS'
	WHEN 'P' THEN 'PENDENCIA'
	WHEN 'E' THEN 'ESTOQUE'
	WHEN 'R' THEN 'RESERVA'
END AS LMEINFALT,	
LTRIM(RTRIM(STR(LMELOCEST))) + '-' + B.LESDES AS LMELOCEST, 
LMEUNI,
LMEQTDMOV,
CASE LMEINFALT 
	WHEN 'C' THEN LMEQTDCMP
	WHEN 'P' THEN LMEQTDPEN
	WHEN 'E' THEN LMEQTDATU
	WHEN 'R' THEN LMEQTDRES
END AS LMEQTD,
LMEQTDSAL, 
LMEUSU,
CASE WHEN C.LMEDESROT IS NULL 
	THEN 'NÃO CATALOGADO'
	ELSE
		CASE WHEN C.LMEDESROT = '' AND A.LMEROT = 'SQL' 
			THEN rtrim(A.LMEDESROT) COLLATE DATABASE_DEFAULT
			ELSE rtrim(C.LMEDESROT) COLLATE DATABASE_DEFAULT
		END
END AS LMEDESROT

FROM TBS051 A (NOLOCK) 
LEFT JOIN TBS034 B (NOLOCK) ON A.LMELOCEST = B.LESCOD
LEFT JOIN #ROTMOV C ON A.LMEROT COLLATE DATABASE_DEFAULT = C.LMEROT 

WHERE 
CONVERT(DATE,LMEDATHOR) BETWEEN @DATADE AND @DATEATE AND 
PROCOD = CASE WHEN @PROCOD = '0' THEN PROCOD ELSE @PROCOD END AND
LMEACA IN (@LMEACA) AND
LMEINFALT IN (@LMEINFALT) AND
--LMELOCEST IN (@LMELOCEST) AND 
LMEDOC = CASE WHEN @LMEDOC = '' THEN LMEDOC ELSE RTRIM(LTRIM(@LMEDOC)) END
