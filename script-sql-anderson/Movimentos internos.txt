SELECT PROCOD AS COD,

		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM TBS0371 P JOIN TBS037 Q ON P.MVIDOC = Q.MVIDOC
		WHERE Q.MVIDATEFE BETWEEN '20160112' AND GETDATE() AND TMVCOD <= 499 AND MVILOCDES = 1 AND Z.PROCOD = P.PROCOD),0)-

		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM TBS0371 R JOIN TBS037 S ON R.MVIDOC = S.MVIDOC
		WHERE S.MVIDATEFE BETWEEN '20160112' AND GETDATE() AND TMVCOD >= 500 AND MVILOCORI = 1 AND Z.PROCOD = R.PROCOD),0) AS MVIEST ,-- MOVIMENTOS DO ESTOQUE
		
		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM TBS0371 T JOIN TBS037 U ON T.MVIDOC = U.MVIDOC
		WHERE U.MVIDATEFE BETWEEN '20160112' AND GETDATE() AND TMVCOD <= 499 AND MVILOCDES = 2 AND Z.PROCOD = T.PROCOD),0)- 

		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM TBS0371 V JOIN TBS037 X ON V.MVIDOC = X.MVIDOC
		WHERE X.MVIDATEFE BETWEEN '20160112' AND GETDATE() AND TMVCOD >= 500 AND MVILOCORI = 2 AND Z.PROCOD = V.PROCOD),0)AS MVILOJA -- MOVIMENTOS DA LOJA
		
		FROM TBS010 Z
