SELECT
				convert(int,subString(convert(char(6),NFEDATENT,112),1,4)) AS ANO,
       				convert(int,subString(convert(char(6),NFEDATENT,112),5,2))AS MES ,
       				PROCOD AS COD,
					NFEDES AS DESCR,
					NFEPREXML AS PREXM,
					NFEQTDXML AS QTDXML,
					NFEPRE AS PRE,
					NFEQTD AS QTD,
					A.NFENUM AS NF,
					NFECHAACE AS CHAVE,
					NFEVALICMSST AS CST,
					dbo.NFEVALIPI(A.NFEEMPCOD,A.NFETIP,A.NFENUM,A.NFECOD,A.SEREMPCOD,A.SERCOD,A.NFEITE)  AS IPI,
					
					(SELECT COUNT(F.NFEITE)  
					from TBS0591 F(nolock)  
					FULL OUTER JOIN  TBS059 B(nolock)  on 
					F.SERCOD=B.SERCOD and F.NFETIP=B.NFETIP and 
					F.NFECOD=B.NFECOD and F.NFENUM=B.NFENUM 
					WHERE F.NFENUM = A.NFENUM AND
					F.NFETIP='T' and NFEDATENT between '20151201' and GETDATE() and 
					F.SERCOD NOT IN('CAN')
					GROUP BY F.NFENUM )    as TTITEM,
					
					A.NFEITE AS NUMITEM,
					A.NFEITEXML AS ITEMXML

from TBS0591 A(nolock)  FULL OUTER JOIN  TBS059 B(nolock)  on A.SERCOD=B.SERCOD and A.NFETIP=B.NFETIP and A.NFECOD=B.NFECOD and A.NFENUM=B.NFENUM
	  					FULL OUTER JOIN  TBS0597 C(NOLOCK) ON A.SERCOD=C.SERCOD and A.NFETIP=C.NFETIP and 
	  					A.NFECOD=C.NFECOD AND A.NFEITE=C.NFEITE AND A.NFENUM=C.NFENUM 
	   
	  
where A.NFETIP='T' and NFEDATENT between '20151201' and GETDATE() and
       /*NFENOM not like('BEST BAG%') and NFENOM not like('BEST OFFICE%') and 
	   NFENOM not like('MISASPEL%') and NFENOM not like('%PAPELYNA%') and NFENOM not like('TANBY%')*/
	   A.SERCOD NOT IN('CAN') AND A.NFEITE >  (SELECT COUNT(F.NFEITE)  
					from TBS0591 F(nolock)  
					FULL OUTER JOIN  TBS059 B(nolock)  on 
					F.SERCOD=B.SERCOD and F.NFETIP=B.NFETIP and 
					F.NFECOD=B.NFECOD and F.NFENUM=B.NFENUM 
					WHERE F.NFENUM = A.NFENUM AND
					F.NFETIP='T' and NFEDATENT between '20151201' and GETDATE() and 
					F.SERCOD NOT IN('CAN')
					GROUP BY F.NFENUM )
	   --NFEPREXML*NFEQTDXML <> NFEPRE*NFEQTD 
	   --A.NFENUM ='284458' AND 
	   --A.NFEITE <> A.NFEITEXML
	   
	   
group by
	convert(char(6),NFEDATENT,112),PROCOD,NFEPREXML,NFEQTDXML,NFEPRE,NFEQTD,NFEVALICMSST,A.NFEITE,
	A.NFENUM,NFECHAACE,NFEDES,A.NFEEMPCOD,A.NFETIP,A.NFENUM,A.NFECOD,A.SEREMPCOD,A.SERCOD,A.NFEITE,A.NFEITEXML
	
	ORDER BY A.NFENUM,A.NFEITE
