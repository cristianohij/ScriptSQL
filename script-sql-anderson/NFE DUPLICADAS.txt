
SELECT  NFENUM AS NF ,
NFECOD AS CODFOR, 
NFENOM AS FORNECEDOR,
NFEDATEMI AS EMISAO,
NFEDATENT AS ENTRADA, 
NFECAN AS CANCELADA ,
NFECHAACE AS CHAVE

FROM TBS059 (NOLOCK) 

WHERE NFENUM IN (SELECT NFENUM

 					FROM TBS059 B (nolock)  
  
  					--WHERE  NFECHAACE <> ''

					GROUP BY NFECOD,NFECONDPAG,NFECTREFE,NFEDADVAL,NFEDATEMI,NFEDATENT,NFEEMPCOD,NFEEMPFC,NFEEMPNFS,NFEESTORI,NFEEVEULTCOD,
					NFEGERPDC,NFEIMPXML,NFELIBDEV,NFEMOD,NFEMODFRE,NFENFRREGULT,NFENOM,NFENOSFOR,NFENUM,NFENUMCOL,NFEPARNF,NFESERDOC,NFESHELLBY,NFESUBTIP,
					NFETIPFRE,NFETOTBAS,NFETOTICMS,NFETOTIPI,NFEULTITE,NFEVALDES,NFEVALFRE,NFEVALRETCOFINS,NFEVALRETCSLL,NFEVALRETPIS,
					NFEVALSEG,NFEVALSUB,NFEVDDTOT,SEREMPCOD,NFECHAACE
					
					HAVING COUNT(NFECHAACE) = 2)
					

ORDER BY NFENUM
