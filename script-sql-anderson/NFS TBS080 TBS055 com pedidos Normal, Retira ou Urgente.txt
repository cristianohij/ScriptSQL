
DECLARE @DATADE DATETIME , @DATAATE DATETIME

SET @DATADE ='20171101'
SET @DATAATE = '20171105'

if object_id('TempDB.dbo.#TBS080') is not null 
drop table #TBS080

SELECT 
D.ENFDATEMI,
D.SNESER,
D.ENFNUM,
D.ENFCODDES,
D.ENFDESREM,
D.ENFESTDES,
ISNULL(A.NFSPDVNUM,0) AS PDVNUM

INTO #TBS080
FROM 
TBS080 D (NOLOCK)
LEFT JOIN
TBS0672 A (NOLOCK) ON A.NFSNUM = D.ENFNUM AND A.SNEEMPCOD = D.SNEEMPCOD AND A.SNESER = D.SNESER 

WHERE
-- ENFCNPJCPF = @CNPJ AND 
ENFSIT = 6 AND 
ENFFINEMI <> 4 AND 
ENFTIPDOC = 1 AND 
ENFDATEMI BETWEEN @DATADE AND @DATAATE

SELECT 
ROW_NUMBER() OVER (Partition by ENFDATEMI,ENFNUM order by ENFDATEMI,ENFNUM) AS RANK,
ENFDATEMI,
SNESER,
ENFNUM,
ENFCODDES,
ENFDESREM,
ENFESTDES,
CASE B.TRNCOD
	WHEN 1 THEN 'NORMAL'
	WHEN 2 THEN 'RETIRA'
	WHEN 3 THEN 'URGENTE'
	ELSE ''
END AS TRNCOD

FROM #TBS080 A 
LEFT JOIN TBS055 B (NOLOCK) ON A.PDVNUM = B.PDVNUM

GROUP BY 
ENFDATEMI,
SNESER,
ENFNUM,
ENFCODDES,
ENFDESREM,
ENFESTDES,
TRNCOD

ORDER BY 
ENFDATEMI, ENFNUM
