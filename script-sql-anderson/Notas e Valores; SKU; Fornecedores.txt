DECLARE @dataDe datetime, @dataAte datetime, @CNPJ VARCHAR(30)
SET @dataDe = '20170601'
SET @dataAte = '20170930'

-- SUM(dbo.NFSTOTPRO(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE)) AS NFSTOTPRO,  					-- TOTAL DOS PRODUTOS
if object_id('TempDB.dbo.#NFS') is not null
BEGIN
   drop table #NFS
END

-- CNPJ DA TANBY|DATA NF|NF|VALOR NF|BASE ICMS|VALOR ICMS

SELECT 
'SAIDA' AS NOME,
A.NFSNUM 		AS NF,
SUM(ROUND(dbo.NFSTOTITEST(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) 	AS NFSTOTITEST, -- VALOR NF
(SELECT MUNNOM FROM TBS003 D WHERE C.MUNCOD = D.MUNCOD ) AS CIDADE

INTO #NFS
FROM 
TBS0671 A (NOLOCK)
LEFT JOIN TBS067 B (NOLOCK) ON A.NFSEMPCOD = B.NFSEMPCOD AND A.NFSNUM = B.NFSNUM AND A.SNEEMPCOD = B.SNEEMPCOD AND A.SNESER = B.SNESER
LEFT JOIN TBS080 D (NOLOCK) ON A.NFSNUM = D.ENFNUM AND A.SNEEMPCOD = D.SNEEMPCOD AND A.SNESER = D.SNESER 
LEFT JOIN TBS002 C (NOLOCK) ON B.NFSCLICOD = C.CLICOD

WHERE
-- SUBSTRING(CONVERT(VARCHAR(12),B.NFSDATEMI,111),1,7) = @anoDe+'/'+@mesDe AND
B.NFSDATEMI BETWEEN @dataDe AND @dataAte AND 
-- NFSTIP IN('N','C','L') and 
D.ENFSIT = 6 AND
D.ENFTIPDOC = 1 

GROUP BY 
A.NFSNUM,C.MUNCOD -- , CONVERT(CHAR(10),B.NFSDATEMI,103), CONVERT(CHAR(10),B.NFSDATEMI,102)

----------------------------------------------------------------------\\------------------------------------------------------------------------------------------
-- CRIANDO TABELA DE DEVOLU플O

UNION 
   
-- CNPJ DA TANBY|DATA NF|NF|VALOR NF|BASE ICMS|VALOR ICMS   
   
SELECT 
'SAIDA-DEVOLUCAO' AS NOME,
A.NFDNUM 		AS NF,
SUM(round(dbo.NFDTOTITEST(A.NFDEMPCOD,A.SNEEMPCOD,A.SNESER,A.NFDNUM,A.NFDITE),2)) + -- NOVA FORMULA N홒 CONTABILIZA O IPI, POR ISSO PRECISO SOMAR COM IPI
SUM(round(NFDVALIPI,2)) AS NFDTOTITEST, 	  -- TOTAL DA NOTA
(SELECT MUNNOM FROM TBS003 D WHERE F.MUNCOD = D.MUNCOD ) AS CIDADE

FROM
TBS1172 A (NOLOCK)
LEFT JOIN TBS117 B (NOLOCK) ON A.NFDEMPCOD = B.NFDEMPCOD AND A.NFDNUM = B.NFDNUM AND A.SNEEMPCOD = B.SNEEMPCOD AND A.SNESER = B.SNESER
LEFT JOIN TBS080 D (NOLOCK) ON A.NFDNUM = D.ENFNUM AND A.SNEEMPCOD = D.SNEEMPCOD AND A.SNESER = D.SNESER
LEFT JOIN TBS006 F (NOLOCK) ON B.FORCOD = F.FORCOD

WHERE
--SUBSTRING(CONVERT(VARCHAR(12),B.NFDDATEMI,111),1,7) = @anoDe+'/'+@mesDe AND
B.NFDDATEMI BETWEEN @dataDe AND @dataAte AND 
D.ENFSIT = 6 AND
D.ENFFINEMI = 4 AND
D.ENFTIPDOC = 1 

GROUP BY 
A.NFDNUM,F.MUNCOD -- A.NFDCFOP, A.NFDNUM, CONVERT(CHAR(10),B.NFDDATEMI,103), CONVERT(CHAR(10),B.NFDDATEMI,102)

-----------------------------------------------------------------------------------------------------------------------------------------------------------------

-- APURA플O DE ICMS DE ENTRADA 
UNION 

SELECT
CASE WHEN B.NFETIP = 'D'
	THEN 'ENTRADA-DEVOLUCAO'
	ELSE'ENTRADA' 
END AS NOME,
B.NFENUM AS NF,
sum(ROUND(NFETOTOPEITE,2)) AS NFETOTOPEITE,
CASE WHEN B.NFETIP = 'D'
	 THEN (SELECT MUNNOM FROM TBS003 G WHERE C.MUNCOD = G.MUNCOD ) 
	 ELSE (SELECT MUNNOM FROM TBS003 G WHERE F.MUNCOD = G.MUNCOD ) 
END AS CIDADE

-- INTO #NFE
FROM TBS0591 A (NOLOCK)
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND 
A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND  A.SEREMPCOD = B.SEREMPCOD
LEFT JOIN TBS080 D (NOLOCK) ON A.NFENUM = D.ENFNUM AND A.NFECOD = D.ENFCODDES
LEFT JOIN TBS002 C ON B.NFECOD = C.CLICOD
LEFT JOIN TBS006 F ON B.NFECOD = F.FORCOD

WHERE 
((D.ENFSIT = 6 AND D.ENFFINEMI = 4 AND D.ENFTIPDOC = 0) OR ((NFENOSFOR = 'N' OR NFENOSFOR = '') AND (B.NFECAN = 'N' OR B.NFECAN = '') AND NFEDATEFE <> '' )) AND
B.NFEDATEFE BETWEEN @dataDe AND @dataAte --AND 
--A.NFECFOP COLLATE DATABASE_DEFAULT NOT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E' and PAGAR = 'S')
-- SUBSTRING(CONVERT(VARCHAR(12),B.NFEDATEFE,111),1,7) = @anoDe+'/'+@mesDe AND

GROUP BY 
B.NFENUM, F.MUNCOD , C.MUNCOD, B.NFETIP



SELECT 
NOME AS OPERA플O,
NF AS 'N� NOTA',
REPLACE(NFSTOTITEST,'.',',') AS 'TOTAL NOTA',
CIDADE
FROM #NFS
