DECLARE @DATADE DATETIME, @DATAATE DATETIME, @ENFESTDES VARCHAR(30), @ENFCNPJCPF VARCHAR(15) , @ENFSIT INT, @ENFNUM INT, 
@ENFTIPDOC INT, @ENFFINEMI INT, @ENFCHAACE VARCHAR(60), @ENFDESREM char(60)	

SET @DATADE = '20171101'
SET @DATAATE = '20171120'
SET @ENFESTDES = 'SP'
SET @ENFCNPJCPF = '' 
SET @ENFSIT = 6 
SET @ENFNUM = '' 
SET @ENFTIPDOC = 1 
SET @ENFFINEMI = 1 
SET @ENFCHAACE = ''
SET @ENFDESREM = '' 


SELECT 
CASE ENFSIT
	WHEN 1 THEN 'Digitação'
	WHEN 2 THEN 'Dados Validos'
	WHEN 3 THEN 'Dados Invalidos'
	WHEN 4 THEN 'XML Gerado'
	WHEN 5 THEN 'Assinada'
	WHEN 6 THEN 'Autorizada'
	WHEN 7 THEN 'Cancelada'
	WHEN 8 THEN 'Denegada'
	WHEN 9 THEN 'Processamento'
	WHEN 10 THEN 'Rejeitada'
	WHEN 11 THEN 'Inutilizada'
	ELSE ''
END AS ENFSIT,
ENFNUM,
SNESER,
ENFDATEMI,
ENFVALTOT,
ENFDESREM,
ENFCNPJCPF,
ENFESTDES,
CASE ENFTIPDOC
	WHEN 0 THEN 'ENTRADA'
	WHEN 1 THEN 'SAIDA'
	ELSE ''
END ENFTIPDOC,
CASE ENFFINEMI
	WHEN 0 THEN 'NÃO EMITIDA'
	WHEN 1 THEN 'NORMAL'
	WHEN 2 THEN 'COMPLEMENTAR'
	WHEN 3 THEN 'AJUSTE'
	WHEN 4 THEN 'DEVOLUÇÃO'
	ELSE ''
END ENFFINEMI,
ENFCHAACE

FROM TBS080 NOLOCK
 
WHERE
ENFTIPDOC 	IN (@ENFTIPDOC) AND 
ENFDATEMI BETWEEN @DATADE AND @DATAATE AND 
ENFESTDES 	IN (@ENFESTDES) AND
ENFCNPJCPF 	= CASE WHEN @ENFCNPJCPF = '' THEN ENFCNPJCPF ELSE @ENFCNPJCPF END AND 
ENFSIT 		IN (@ENFSIT) AND 
ENFNUM		= CASE WHEN @ENFNUM = '' THEN ENFNUM ELSE @ENFNUM END AND 
ENFFINEMI	IN (@ENFFINEMI) AND 
ENFCHAACE	= CASE WHEN @ENFCHAACE = '' THEN ENFCHAACE ELSE @ENFCHAACE END AND 
ENFDESREM	LIKE( CASE WHEN @ENFDESREM = '' THEN ENFDESREM ELSE RTRIM(LTRIM(UPPER(@ENFDESREM)))+'%' END)

ORDER BY 
ENFDATEMI , ENFNUM
