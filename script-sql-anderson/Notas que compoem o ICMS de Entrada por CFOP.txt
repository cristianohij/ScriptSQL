-- BUSCAR VALORES NO CONTAS A PAGAR

if object_id('TempDB.dbo.#ENTRADA') is not null
   drop table #ENTRADA

SELECT 
CONVERT(CHAR(4),CPAREFANO) +'.'+ right(('0' + ltrim(str(CONVERT(CHAR(2),CPAREFMES)))),2) AS DATAS,
right(('0' + ltrim(str(CONVERT(CHAR(2),CPAREFMES)))),2) +'/'+ CONVERT(CHAR(4),CPAREFANO) AS NFEDATEFE,
'E' AS TIPO,
CPATIT AS NFE,
CPACFOP collate DATABASE_DEFAULT AS CFOP,
'' collate DATABASE_DEFAULT AS CST,
COUNT(*) AS QTD,
SUM(ROUND(CPAVAL,2)) AS NFETOTITE,
SUM(ROUND(CPAVAL,2)) AS NFETOTOPEITE,
SUM(ROUND(CPABASCAL,2)) AS NFEBASICMS,
ROUND(SUM(ROUND(CPABASCAL,2)) *  CPAALIICMS / 100 ,2) AS NFEVALICMS,
0 AS NFEBASICMSANT,
0 AS NFEVALICMSANT,
0 AS NFEBASICMSST,
0 AS NFEVALICMSST,
0 AS NFEBASIPI,
0 AS NFEVALIPI,
0 AS ICMSISENTO,		-- ICMS ISENTO
0 AS ICMSNAOTRI,		-- ICMS NÃO TRIBUTADO
0 AS ICMSTRIANT,		-- ICMS TRIBUTADO ANTERIORMENTE
SUM(ROUND(CPAVAL,2)) - SUM(ROUND(CPABASCAL,2)) AS ICMSOUTROS

INTO #ENTRADA
FROM 
TBS057 NOLOCK 

WHERE 
--CPAREFMES BETWEEN MONTH(@Data_de) AND  MONTH(@Data_ate) and  
--CPAREFANO BETWEEN YEAR(@Data_de) AND  YEAR(@Data_ate) and  
CONVERT(CHAR(4),CPAREFANO) +'.'+ right(('0' + ltrim(str(CONVERT(CHAR(2),CPAREFMES)))),2) 
BETWEEN SUBSTRING(CONVERT(CHAR(10),@Data_de,102),1,7) AND SUBSTRING(CONVERT(CHAR(10),@Data_ate,102),1,7) AND 
CPACFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E' AND PAGAR = 'S') and
CPACFOP COLLATE DATABASE_DEFAULT IN (SELECT NFECFOP FROM #CFOP1)

GROUP BY 
CPACFOP, 
CPAALIICMS,
CPATIT, 
CPAREFANO,
CPAREFMES

------------------------------------------------------------------------------------------------------------------------------------------

UNION

SELECT
convert(CHAR(10),B.NFEDATEFE,111) AS DATAS,
convert(CHAR(10),B.NFEDATEFE,103) AS NFEDATEFE,
B.NFETIP AS TIPO,
A.NFENUM AS NFE,
A.NFECFOP COLLATE DATABASE_DEFAULT AS CFOP,	
A.NFECST COLLATE DATABASE_DEFAULT AS CST,
count(PROCOD) AS QTD,
sum(ROUND(dbo.NFETOTITE(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE),2)) AS NFETOTITE,
sum(ROUND(NFETOTOPEITE,2)) AS NFETOTOPEITE,
CASE WHEN A.NFECFOP COLLATE DATABASE_DEFAULT NOT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E')
	THEN SUM(round(A.NFEBASICMS,2))
	ELSE 0
END AS NFEBASICMS, 
CASE WHEN A.NFECFOP COLLATE DATABASE_DEFAULT NOT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E')
	THEN SUM(round(A.NFEVALICMS,2)) 
	ELSE 0
END AS NFEVALICMS,
CASE WHEN A.NFECFOP IN ('1.403','2.403')
	THEN SUM(round(A.NFEBASICMS,2))
	ELSE 0
END AS NFEBASICMSANT,
CASE WHEN A.NFECFOP IN ('1.403','2.403')
	THEN SUM(round(A.NFEVALICMS,2))
	ELSE 0
END AS NFEVALICMSANT,
SUM(round(NFEBASICMSST,2)) AS NFEBASICMSST,
SUM(round(NFEVALICMSST,2)) AS NFEVALICMSST,
SUM(ROUND(NFEBASIPI,2)) AS  NFEBASIPI,
SUM(ROUND(NFEVALIPI,2))    AS NFEVALIPI,

CASE WHEN SUBSTRING(NFECST,2,3) IN ('30','40','103','203','300')
	THEN sum(ROUND(NFETOTOPEITE,2)) 
	ELSE 0
END AS ICMSISENTO,		-- ICMS ISENTO

CASE WHEN SUBSTRING(NFECST,2,3) IN ('41','50','400','102')
	THEN sum(ROUND(NFETOTOPEITE,2)) 
	ELSE 0
END AS ICMSNAOTRI,		-- ICMS NÃO TRIBUTADO

CASE WHEN SUBSTRING(NFECST,2,3) IN ('60','500')
	THEN sum(ROUND(NFETOTOPEITE,2)) 
	ELSE 0
END AS ICMSTRIANT,		-- ICMS TRIBUTADO ANTERIORMENTE

SUM(ROUND(NFETOTOPEITE,2)) - 								-- TOTA CONTABIL
CASE WHEN A.NFECFOP COLLATE DATABASE_DEFAULT NOT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E') -- BASE ICMS 
	THEN SUM(round(A.NFEBASICMS,2))
	ELSE 0
END - 
CASE WHEN SUBSTRING(NFECST,2,3) IN ('41','50','400','102','30','40','103','203','300') -- ISENTO E NÃO TRIBUTADO
	THEN sum(ROUND(NFETOTOPEITE,2)) 
	ELSE 0
END AS ICMSOUTROS  /* CONTABIL - BASE - ISENTO - NÃO TRIBUTADO = ICMSOUTROS */
 
FROM TBS0591 A (NOLOCK)
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND 
A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND  A.SEREMPCOD = B.SEREMPCOD
LEFT JOIN TBS080 D (NOLOCK) ON A.NFENUM = D.ENFNUM AND A.NFECOD = D.ENFCODDES
		
WHERE
B.NFEDATEFE BETWEEN @Data_de AND @Data_ate AND 
B.NFETIP IN (@NFETIP) AND 
A.NFECFOP COLLATE DATABASE_DEFAULT IN ((SELECT NFECFOP FROM #CFOP1)) AND
((D.ENFSIT = 6 AND D.ENFFINEMI = 4 AND D.ENFTIPDOC = 0) OR (NFENOSFOR <> 'S' AND B.NFECAN <> 'S' AND NFEDATEFE <> '' )) and 
A.NFECFOP COLLATE DATABASE_DEFAULT NOT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E' AND PAGAR = 'S')

GROUP BY 
A.NFECFOP, A.NFECST, SUBSTRING(NFECST,2,3), B.NFETIP, A.NFENUM, convert(CHAR(10),B.NFEDATEFE,103), convert(CHAR(10),B.NFEDATEFE,111)


SELECT
TIPO,
DATAS,
NFEDATEFE,
CFOP,
NFE,
SUM(QTD) AS QTD,
SUM(NFETOTITE)     AS NFETOTITE,
SUM(NFETOTOPEITE)  AS NFETOTOPEITE, 
SUM(NFEBASICMS)    AS NFEBASICMS,
SUM(NFEVALICMS)    AS NFEVALICMS,
SUM(NFEBASICMSANT) AS NFEBASICMSANT,
SUM(NFEVALICMSANT) AS NFEVALICMSANT,
SUM(NFEBASICMSST)  AS NFEBASICMSST,
SUM(NFEVALICMSST)  AS NFEVALICMSST,
SUM(NFEBASIPI)     AS  NFEBASIPI,
SUM(NFEVALIPI)     AS NFEVALIPI,
SUM(ICMSISENTO)    AS ICMSISENTO,
SUM(ICMSNAOTRI)    AS ICMSNAOTRI,
SUM(ICMSTRIANT)    AS ICMSTRIANT,
SUM(ICMSOUTROS)    AS NFEVALDESITE, 
SUM(ICMSOUTROS)    AS ICMSOUTROS

FROM 
#ENTRADA 
WHERE 
NFE = CASE WHEN @NFE = 0 THEN NFE ELSE @NFE END

GROUP BY 
TIPO,
DATAS,
NFEDATEFE,
CFOP,
NFE
