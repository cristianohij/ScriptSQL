SELECT 
CASE WHEN NFSFINNFE = 4 
	THEN 'DEV'
	ELSE
		CASE WHEN NFSCLINOM LIKE('TANBY%') AND UFESIG = 'SP' AND NFSFINNFE <> 4 
			THEN 'TRF'
			ELSE 
				CASE WHEN UFESIG = 'SP' AND NFSCLINOM NOT LIKE('TANBY%') AND NFSFINNFE <> 4 
					THEN 'VDE'
					ELSE 'VFE'
				END 
		END 
END AS SUBTIPO,
A.NFSCFOP AS CFOP,
B.NFSDATEMI ,
B.NFSNUM, 
-- A.NFSCST AS CST,
-- B.UFESIG AS UF,
COUNT(A.PROCOD) AS ITENS,
SUM(round(dbo.NFSTOTPRO(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE),2)) AS NFSTOTPRO,  				-- TOTAL DOS PRODUTOS
SUM(round(dbo.NFSTOTITEST(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) AS NFSTOTITEST, 		-- TOTAL DA NOTA
CASE WHEN A.NFSCFOP COLLATE DATABASE_DEFAULT NOT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S') AND B.NFSDESICMS = 'N'
	THEN SUM(round(dbo.NFSBASICMS(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) 
	ELSE 0
END AS NFSBASICMS, 		-- BASE ICMS
CASE WHEN A.NFSCFOP COLLATE DATABASE_DEFAULT NOT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S') AND B.NFSDESICMS = 'N'
	THEN SUM(round(dbo.NFSVALICMS(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) 
	ELSE 0
END AS NFSVALICMS, 																							-- VALOR ICMS SEM O CFOP 5.922 6.922
SUM(round(dbo.NFSBASICMSST(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) AS NFSBASICMSST, 	-- BASE ICMS ST
SUM(round(dbo.NFSVALICMSST(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) AS NFSVALICMSST, 	-- VALOR ICMS ST

CASE WHEN A.NFSCFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S')
	THEN SUM(round(dbo.NFSBASICMS(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2))
	ELSE 0
END AS NFSBASICMSFUT,	-- BASE ICMS PARA ENTREGA FUTURA

CASE WHEN A.NFSCFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S')
	THEN SUM(round(dbo.NFSVALICMS(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2))
	ELSE 0
END AS NFSVALICMSFUT,	-- VALOR ICMS PARA ENTREGA FUTURA

CASE WHEN SUBSTRING(NFSCST,2,2) IN ('30','40') or B.NFSDESICMS = 'S'
	THEN sum(round(dbo.NFSTOTITEST(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2))
	ELSE 0
END AS ICMSISENTO, 		-- AS VALOR ICMS ISENTO

CASE WHEN SUBSTRING(NFSCST,2,2) IN ('41','50')
	THEN SUM(round(dbo.NFSTOTITEST(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2))
	ELSE 0
END AS ICMSNAOTRI, 		-- AS VALOR ICMS NÃO TRIBUTADO

CASE WHEN SUBSTRING(NFSCST,2,3) IN ('60')
	THEN SUM(round(dbo.NFSTOTITEST(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2))
	ELSE 0
END AS ICMSTRIANT		-- ICMS TRIBUTADO ANTERIORMENTE 
-- 0 AS NFSVALIPI

INTO #NFS
FROM 
TBS0671 A (NOLOCK)
LEFT JOIN TBS067 B (NOLOCK) ON A.NFSEMPCOD = B.NFSEMPCOD AND A.NFSNUM = B.NFSNUM AND A.SNEEMPCOD = B.SNEEMPCOD AND A.SNESER = B.SNESER
LEFT JOIN TBS080 D (NOLOCK) ON A.NFSNUM = D.ENFNUM AND A.SNEEMPCOD = D.SNEEMPCOD AND A.SNESER = D.SNESER 

WHERE
B.NFSDATEMI BETWEEN @Data_de AND @Data_ate AND
--B.NFSDATEMI BETWEEN convert(date,@ANO+'/'+@MES+'/'+'01') AND @ultimoDiaMes AND-- BETWEEN @Data_de AND @Data_ate AND 
B.NFSNUM = CASE WHEN @NFSNUM = 0 THEN B.NFSNUM ELSE @NFSNUM END AND 
NFSTIP IN('N','C') and 
D.ENFSIT = 6 AND
D.ENFTIPDOC = 1 --AND
--B.NFSTIP IN (@NFSTIP) 

GROUP BY 
A.NFSCFOP, B.NFSNUM,B.NFSDATEMI , A.NFSCST ,NFSCLINOM, NFSFINNFE, UFESIG , NFSDESICMS

B.NFDNUM, 
-- A.NFDCSTCSOSN AS CST,

SUM(round(A.NFDBASICMS,2)) AS NFSBASICMS, 		-- BASE ICMS
CASE WHEN A.NFDCFOP COLLATE DATABASE_DEFAULT NOT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S')
	THEN SUM(round(A.NFDVALICMS,2))
	ELSE 0
END AS NFSVALICMS, 		-- VALOR ICMS

CASE WHEN A.NFDCFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S')
	THEN SUM(round(A.NFDVALICMS,2))
	ELSE 0
END AS NFSVALICMSFUT,	-- VALOR ICMS PARA ENTREGA FUTURA

CASE WHEN SUBSTRING(NFDCSTCSOSN,2,2) IN ('30','40')
	THEN SUM(round(dbo.NFDTOTITEST(A.NFDEMPCOD,A.SNEEMPCOD,A.SNESER,A.NFDNUM,A.NFDITE),2)) + -- NOVA FORMULA NÃO CONTABILIZA O IPI, POR ISSO PRECISO SOMAR COM IPI
         SUM(round(NFDVALIPI,2))
	ELSE 0
END AS ICMSISENTO, 						-- AS VALOR ICMS ISENTO
	
CASE WHEN SUBSTRING(NFDCSTCSOSN,2,2) IN ('41','50')
	THEN SUM(round(dbo.NFDTOTITEST(A.NFDEMPCOD,A.SNEEMPCOD,A.SNESER,A.NFDNUM,A.NFDITE),2)) + -- NOVA FORMULA NÃO CONTABILIZA O IPI, POR ISSO PRECISO SOMAR COM IPI
         SUM(round(NFDVALIPI,2))
	ELSE 0
END AS ICMSNAOTRI, 						-- AS VALOR ICMS NÃO TRIBUTADO

CASE WHEN SUBSTRING(NFDCSTCSOSN,2,3) IN ('60')
	THEN SUM(round(dbo.NFDTOTITEST(A.NFDEMPCOD,A.SNEEMPCOD,A.SNESER,A.NFDNUM,A.NFDITE),2)) + -- NOVA FORMULA NÃO CONTABILIZA O IPI, POR ISSO PRECISO SOMAR COM IPI
         SUM(round(NFDVALIPI,2))
	ELSE 0
END AS ICMSTRIANT,						-- ICMS TRIBUTADO ANTERIORMENTE
SUM(ROUND(A.NFDVALIPI,2)) AS NFDVALIPI

INTO #NFD
FROM
TBS1172 A (NOLOCK)
LEFT JOIN TBS117 B (NOLOCK) ON A.NFDEMPCOD = B.NFDEMPCOD AND A.NFDNUM = B.NFDNUM AND A.SNEEMPCOD = B.SNEEMPCOD AND A.SNESER = B.SNESER
LEFT JOIN TBS080 D (NOLOCK) ON A.NFDNUM = D.ENFNUM AND A.SNEEMPCOD = D.SNEEMPCOD AND A.SNESER = D.SNESER 

WHERE
B.NFDDATEMI BETWEEN @Data_de AND @Data_ate AND
-- B.NFDDATEMI BETWEEN convert(date,@ANO+'/'+@MES+'/'+'01') AND @ultimoDiaMes AND -- BETWEEN @Data_de AND @Data_ate AND 
D.ENFSIT = 6 AND
D.ENFFINEMI = 4 AND 
D.ENFTIPDOC = 1 --AND
--A.NFDNFETIP IN (@NFSTIP)

GROUP BY 
A.NFDCFOP,B.NFDDATEMI, B.NFDNUM,  A.NFDCSTCSOSN , B.NFDESTDES

-- CRIANDO TABELA TEMP COM OS VALORES REAIS DE DESCONTO
if object_id('TempDB.dbo.#TMP') is not null
   drop table #TMP
   
	SELECT
	ISNULL(D.NFSDATEMI,A.M2_DAT) AS NFSDATEMI,
	CASE ISNULL(D.COO,0) WHEN 0 
		THEN 'VDE'
		ELSE 
			CASE WHEN UFESIG <> 'SP' THEN 'VFE'	ELSE 'VDE' END
	END AS SUBTIPO,
	ISNULL(D.NFSNUM,0) AS NFSNUM,
	ISNULL(A.M2_ECF,0) AS ECF,
	ISNULL(A.M2_COO,0) AS COO,
	ISNULL(A.M2_CUP,0) AS CUPOM,
	CASE ISNULL(D.COO,0) WHEN 0 
		THEN '5.929C'
		ELSE '5.929'
	END AS M2_CFOP,
	CASE ISNULL(D.COO,0) WHEN 0 
		THEN 'F'
		ELSE 'L'
	END AS M2_TIP,
	A.M2_TRI,
	A.M2_CST,
	A.M2_ITE,
	A.M2_LOJ, 
	A.M2_CUP, 
	A.M2_COO, 
	A.M2_ECF, 
	A.M2_CXA ,
	M2_PROCOD,
	M2_VALBRU,
	M2_VALBRU - M2_ABT + M2_ACRCUP - 
	CASE WHEN  A.DESCONTO <> ISNULL(B.DESCONTO,0) AND M2_ITE = (SELECT TOP 1 M2_ITE 
								    FROM #MSLAUX E 
									
								    WHERE 
								    A.M2_COO = E.M2_COO and A.M2_ECF = E.M2_ECF AND A.M2_CUP = E.M2_CUP 
																
								    ORDER BY 
								    M2_DESCUP DESC)
		THEN
			CASE WHEN A.DESCONTO > ISNULL(B.DESCONTO,0) 
				THEN ROUND(M2_DESCUP,2) - (A.DESCONTO - ISNULL(B.DESCONTO,0))
				ELSE ROUND(M2_DESCUP,2) + (ISNULL(B.DESCONTO,0) - A.DESCONTO)
			END
		ELSE ROUND(M2_DESCUP,2)
	END AS M2_LIQUIDO
	
	INTO #TMP	
	FROM (
		SELECT 
		M2_DAT, M2_ITE,M2_PROCOD,M2_TRI,M2_CST,
		M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , 
		M2_DESCUP ,
		M2_ACRCUP,
		M2_ABT ,
		M2_VALBRU,
		SUM(ISNULL(ROUND(M2_DESCUP,2),0)) OVER (PARTITION BY M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA ) AS DESCONTO
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 1 ) AS A 
			
		LEFT JOIN (
		SELECT 
		M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , M2_VALBRU AS DESCONTO 
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 2 ) AS B ON A.M2_LOJ = B.M2_LOJ AND A.M2_CUP = B.M2_CUP AND A.M2_COO= B.M2_COO AND 
														A.M2_ECF = B.M2_ECF AND A.M2_CXA = B.M2_CXA  
		
	LEFT JOIN #NFSLOJA D (NOLOCK) ON A.M2_COO = D.COO AND A.M2_ECF = D.ECF

-- CRIANDO TABELA TEMP COM OS VALORES REAIS DE DESCONTO
if object_id('TempDB.dbo.#TMP') is not null
   drop table #TMP
   
	SELECT
	ISNULL(D.NFSDATEMI,A.M2_DAT) AS NFSDATEMI,
	CASE ISNULL(D.NFSNFRCHA,'') WHEN '' 
		THEN 'VDE'
		ELSE 
			CASE WHEN UFESIG <> 'SP' THEN 'VFE'	ELSE 'VDE' END
	END AS SUBTIPO,
	ISNULL(D.NFSNUM,0) AS NFSNUM,
	--ISNULL(A.M2_ECF,0) AS ECF,
	--ISNULL(A.M2_COO,0) AS COO,
	ISNULL(A.M2_CUP,0) AS CUPOM,
	CASE ISNULL(D.NFSNFRCHA,'') WHEN '' 
		THEN '5.929C'
		ELSE '5.929'
	END AS M2_CFOP,
	CASE ISNULL(D.NFSNFRCHA,'') WHEN '' 
		THEN 'F'
		ELSE 'L'
	END AS M2_TIP,
	A.M2_TRI,
	A.M2_CST,
	A.M2_ITE,
--	A.M2_LOJ, 
--	A.M2_CUP, 
--	A.M2_COO, 
--	A.M2_ECF, 
	A.M2_CXA ,
	M2_PROCOD,
	M2_VALBRU,
	M2_VALBRU - M2_ABT + M2_ACRCUP - 
	CASE WHEN  A.DESCONTO <> ISNULL(B.DESCONTO,0) AND M2_ITE = (SELECT TOP 1 M2_ITE 
								    FROM #MSLAUX E 
									
								    WHERE 
								    A.M2_LOJ = E.M2_LOJ AND A.M2_CUP = E.M2_CUP AND A.M2_CHAVE= E.M2_CHAVE AND A.M2_CXA = E.M2_CXA
																
								    ORDER BY 
								    M2_DESCUP DESC)
		THEN
			CASE WHEN A.DESCONTO > ISNULL(B.DESCONTO,0) 
				THEN ROUND(M2_DESCUP,2) - (A.DESCONTO - ISNULL(B.DESCONTO,0))
				ELSE ROUND(M2_DESCUP,2) + (ISNULL(B.DESCONTO,0) - A.DESCONTO)
			END
		ELSE ROUND(M2_DESCUP,2)
	END AS M2_LIQUIDO
	
	INTO #TMP	
	FROM (
		SELECT 
		M2_DAT, M2_ITE,M2_PROCOD,M2_TRI,M2_CST,
		M2_LOJ, M2_CUP, M2_CHAVE, M2_CXA,
		M2_DESCUP ,
		M2_ACRCUP,
		M2_ABT ,
		M2_VALBRU,
		SUM(ISNULL(ROUND(M2_DESCUP,2),0)) OVER (PARTITION BY M2_LOJ, M2_CUP, M2_CXA, M2_CHAVE ) AS DESCONTO
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 1 ) AS A 
			
		LEFT JOIN (
		SELECT 
		M2_LOJ,M2_CXA, M2_CUP, M2_CHAVE, M2_VALBRU AS DESCONTO 
					
		FROM
