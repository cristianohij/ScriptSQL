-- CRIAR A BASE PARA PROCURAR OS PRODUTOS

IF OBJECT_ID('TempDB.dbo.#BASE') IS NOT NULL 
DROP TABLE #BASE

SELECT 
NFENUM,
CASE WHEN B.NFETIP <> 'D' 
	THEN '35'+REPLACE(CONVERT(CHAR(5),B.NFEDATEMI,11),'/','')+RTRIM(LTRIM((SELECT FORCGC FROM TBS006 C (NOLOCK) WHERE C.FORCOD = B.NFECOD)))
	ELSE '35'+REPLACE(CONVERT(CHAR(5),B.NFEDATEMI,11),'/','')+RTRIM(LTRIM((SELECT CLICGC FROM TBS002 C (NOLOCK) WHERE C.CLICOD = B.NFECOD)))
END AS NFECHAACE

INTO #BASE
FROM
TBS059 B (NOLOCK)

WHERE 
B.NFECAN <> 'S' AND 
CASE WHEN B.NFETIP <> 'D' 
	THEN '35'+REPLACE(CONVERT(CHAR(5),B.NFEDATEMI,11),'/','')+RTRIM(LTRIM((SELECT FORCGC FROM TBS006 C (NOLOCK) WHERE C.FORCOD = B.NFECOD)))
	ELSE '35'+REPLACE(CONVERT(CHAR(5),B.NFEDATEMI,11),'/','')+RTRIM(LTRIM((SELECT CLICGC FROM TBS002 C (NOLOCK) WHERE C.CLICOD = B.NFECOD)))
END IN 
(SELECT DISTINCT substring(ENFCHAACE,1,20) FROM #SAIDA WHERE EMPRESA in(@EMPRESA))


-----------------------------------------------------------------------------------------------------------

-- PROCURAR NA BASE 
SELECT 
EMPRESA,
ENFNUM, 
ENFDATEMI,
SNESER, 
ENFCHAACE,
ENFVALTOT

FROM #SAIDA A
LEFT JOIN #BASE B ON substring(A.ENFCHAACE,1,20) = B.NFECHAACE AND A.ENFNUM = B.NFENUM -- AND A.PROCOD = B.PROCOD -- AND A.NFSQTD = B.NFEQTD

WHERE 
B.NFENUM IS NULL AND 
EMPRESA IN (@EMPRESA)
