DECLARE @MES CHAR(2) , @ANO CHAR(4)
SET @MES = '03'
SET @ANO = '2017'

SELECT 
CASE WHEN LAICFOP < 5000
	THEN 'ENTRADAS'
	ELSE 'SAIDAS'
END AS OP,
SUBSTRING(CONVERT(CHAR(4),LAICFOP),1,1)+'.00' AS CF,
SUBSTRING(CONVERT(CHAR(4),LAICFOP),1,1)+'.'+SUBSTRING(CONVERT(CHAR(4),LAICFOP),2,3)	AS CFOP,
LAIVALCON	AS NFSTOTITEST, 
LAIBASCAL	AS NFSBASICMS,
LAIVALIMP	AS NFSVALICMS,
LAIVALIN	AS ICMSISENAOTRI,
CASE WHEN LAIVALCON - LAIBASCAL < 0 
	THEN 0
	ELSE LAIVALCON - LAIBASCAL
END AS ICMSOUTROS

FROM TBS1281 

WHERE 
LAIANO = @ANO AND 
LAIMES = @MES

--A.M2_COO, 
--A.M2_ECF, 
--A.M2_CXA ,
M2_PROCOD,
M2_VALBRU,
M2_VALBRU - M2_ABT + M2_ACRCUP - 
CASE WHEN  A.DESCONTO <> ISNULL(B.DESCONTO,0) AND M2_ITE = (SELECT TOP 1 M2_ITE 
																FROM #MSLAUX E 
																
																WHERE 
																A.M2_COO = E.M2_COO and A.M2_ECF = E.M2_ECF AND A.M2_CUP = E.M2_CUP 
																
																ORDER BY 
																M2_DESCUP DESC)
	THEN
		CASE WHEN A.DESCONTO > ISNULL(B.DESCONTO,0) 
			THEN ROUND(M2_DESCUP,2) - (A.DESCONTO - ISNULL(B.DESCONTO,0))
			ELSE ROUND(M2_DESCUP,2) + (ISNULL(B.DESCONTO,0) - A.DESCONTO)
		END
	ELSE ROUND(M2_DESCUP,2)
END AS M2_LIQUIDO
	
INTO #TMP	
FROM (
	SELECT 
	M2_ITE,M2_PROCOD,M2_TRI,M2_CST,
	M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , 
	M2_DESCUP ,
	M2_ACRCUP,
	M2_ABT ,
	M2_VALBRU,
	SUM(ISNULL(ROUND(M2_DESCUP,2),0)) OVER (PARTITION BY M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA ) AS DESCONTO
					
	FROM 
	#MSLAUX 
					
	WHERE
	M2_STATUS = 1 ) AS A 
			
	LEFT JOIN (
	SELECT 
	M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , M2_VALBRU AS DESCONTO 
					
	FROM 
	#MSLAUX 
					
	WHERE
	M2_STATUS = 2 ) AS B ON A.M2_LOJ = B.M2_LOJ AND A.M2_CUP = B.M2_CUP AND A.M2_COO= B.M2_COO AND A.M2_ECF = B.M2_ECF AND A.M2_CXA = B.M2_CXA
