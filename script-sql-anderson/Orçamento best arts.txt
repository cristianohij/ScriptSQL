CLIEND, 
CLINUM, 
CLIBAI,

CLITEL,

select 
'BO' AS EMP,
ORCITEM,
rtrim(ORCDES) AS ORCDES,
RTRIM(MARNOM) AS MARNOM,
ORCUNI,
ORCQTD,
ORCPRE,
ORCQTD*ORCPRE AS TOTAL

from TBS0431 A (NOLOCK)
LEFT JOIN TBS010 B (NOLOCK) ON A.PROCOD = B.PROCOD

WHERE ORCNUM = 516990

union 

select 
'ND' AS EMP,
ORCITEM,
rtrim(ORCDES) AS ORCDES,
RTRIM(MARNOM) AS MARNOM,
ORCUNI,
ORCQTD,
ORCPRE,
ORCQTD*ORCPRE AS TOTAL,  
CLICOD,
CLINOM, 
CLIEND, 
CLINUM, 
CLIBAI, 
CLICEP, 
CLITEL,
C.CLICGC AS CNPJ,
C.CLIEMAIL AS EMAIL,
C.CLIIES AS INSEST, 
MUNNOM , 
C.UFESIG

from TBS0431 A (NOLOCK) 
LEFT JOIN TBS043 B (NOLOCK) ON A.ORCNUM = B.ORCNUM AND A.ORCEMPCOD = B.ORCEMPCOD 
LEFT JOIN TBS002 C (NOLOCK) ON B.ORCCLI = C.CLICOD
LEFT JOIN TBS003 D (NOLOCK) ON C.MUNCOD = D.MUNCOD 
LEFT JOIN TBS010 E (NOLOCK) ON A.PROCOD = E.PROCOD


WHERE A.ORCNUM = 516978

UNION
SELECT 
'BB' AS EMP,
ORCITEM,
rtrim(ORCDES) AS ORCDES,
RTRIM(MARNOM) AS MARNOM,
ORCUNI,
ORCQTD,
ORCPRE,
ORCQTD*ORCPRE AS TOTAL

from TBS0431 A (NOLOCK)
LEFT JOIN TBS010 B (NOLOCK) ON A.PROCOD = B.PROCOD

WHERE ORCNUM = 516990


ORDER BY ORCITEM
