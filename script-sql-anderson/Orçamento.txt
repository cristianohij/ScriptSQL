SELECT  
C.ORCDATCAD, 
RTRIM(A.ORCITEM) AS ORCITEM ,
RTRIM(A.PROCOD)  AS PROCOD,
rtrim(A.ORCDES)  AS ORCDES,
RTRIM(MARNOM)    AS MARNOM,
RTRIM(A.ORCUNI)  AS ORCUNI, 
RTRIM(A.ORCQTD)  AS ORCQTD,
ROUND(ROUND(dbo.ORCTOTITEST(A.ORCEMPCOD,A.ORCNUM,A.ORCITEM),2) / ORCQTD,2)AS ORCPRE,
ROUND(dbo.ORCTOTITEST(A.ORCEMPCOD,A.ORCNUM,A.ORCITEM),2) AS ORCTOTITEST,
RTRIM(C.ORCCLI) AS CLICOD,
RTRIM(C.ORCNOM) AS CLINOM,
RTRIM(C.ORCREQNOM) AS ORCREQNOM,
RTRIM(E.CLITEL) AS CLITEL,
RTRIM(E.CLIRAM1) AS CLIRAMAL,
(SELECT CPGDES FROM TBS008 I (NOLOCK) WHERE C.CPGCOD = I.CPGCOD) AS CPGDES,
C.ORCVALPRO, 
C.ORCPRAENT,
A.ORCNUM ,
C.VENCOD , 
RTRIM(VENNOM) AS VENNOM ,
RTRIM(D.VENEMAIL) AS VENEMAIL ,
RTRIM(D.VENRAM) AS VENRAM, 
RTRIM(CLIEND) + CASE WHEN CHARINDEX(RTRIM(LTRIM(CLINUM)),RTRIM(CLIEND)) > 0 THEN '' ELSE ' - N° ' + RTRIM(LTRIM(CLINUM)) END + 
' , ' + RTRIM(LTRIM(CLIBAI)) + ' , ' + RTRIM(LTRIM((SELECT MUNNOM FROM TBS003 G (NOLOCK) WHERE E.MUNCOD = G.MUNCOD))) + ' - ' + UFESIG AS CLIENDFAT,
ISNULL(F.CLIENDENT,'') AS CLIENDENT,
ISNULL(H.CLIENDCOB,'') AS CLIENDCOB, 
ROUND(dbo.ORCVDDITE(A.ORCEMPCOD,A.ORCNUM,A.ORCITEM),2)    AS ORCVDDITE,
ROUND(dbo.ORCFREITEVAL(A.ORCEMPCOD,A.ORCNUM,A.ORCITEM),2) AS ORCFREITEVAL,
ROUND(dbo.ORCSEGITEVAL(A.ORCEMPCOD,A.ORCNUM,A.ORCITEM),2) AS ORCSEGITEVAL,
ROUND(dbo.ORCDESITEVAL(A.ORCEMPCOD,A.ORCNUM,A.ORCITEM),2) AS ORCDESITEVAL ,
ROUND(dbo.ORCVALICMSST(A.ORCEMPCOD,A.ORCNUM,A.ORCITEM),2) AS ORCVALICMSST,
ROUND(dbo.ORCTOTPRO(A.ORCEMPCOD,A.ORCNUM,A.ORCITEM),2) AS ORCTOTPRO,
C.ORCOBS,
C.ORCREQCDC,
ISNULL((SELECT RDVDES FROM TBS097 J (NOLOCK) WHERE E.RDVCOD = J.RDVCOD),'') AS RDVDES

FROM TBS0431 A (NOLOCK)
LEFT JOIN TBS010 B  (NOLOCK) ON A.PROCOD = B.PROCOD
LEFT JOIN TBS043 C  (NOLOCK) ON A.ORCEMPCOD = C.ORCEMPCOD AND A.ORCNUM = C.ORCNUM 
LEFT JOIN TBS004 D  (NOLOCK) ON C.VENCOD = D.VENCOD
LEFT JOIN TBS002 E  (NOLOCK) ON C.ORCCLI = E.CLICOD
LEFT JOIN #ENDENT F (NOLOCK) ON C.ORCCLI = F.CLICOD 
LEFT JOIN #ENDCOB H (NOLOCK) ON C.ORCCLI = H.CLICOD 

WHERE A.ORCNUM = @ORCNUM
