declare @dataDe char(8), @dataAte char(8),@comando varchar(max)

set @dataDe = '20161101'
set @dataAte = '20161130'

select 
A.PROCOD as codigo,
A.NFSPRODES as descricao,
D.PROUM1 as UN,
NFSQTD*NFSQTDEMB as 'quantidade',
round(dbo.NFSPRELIQ(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,NFSITE)/NFSQTDEMB,2) as PrecoUnitario,
round(NFSPRECUS/NFSQTDEMB,4) as 'CustoUnitario',
round(dbo.NFSTOTITEST(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE),2) as TotalVendido,
round(dbo.NFSVALICMS(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE),2) as ICMS,
case when D.PROSTBPIS>='06' 
	then 0 
	else round(dbo.NFSTOTITEST(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE)*0.0165,2) 
end as PIS,
case when D.PROSTBCOFINS>='06' 
	then 0 
	else round(dbo.NFSTOTITEST(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE)*0.076,2) 
end as COFINS,

round(dbo.NFSTOTITEST(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE),2)- 
round(dbo.NFSVALICMS(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE),2)-
case when D.PROSTBPIS>='06' 
	then 0 
	else round(dbo.NFSTOTITEST(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE)*0.0165,2)
end -
case when D.PROSTBCOFINS>='06' 
	then 0 
	else round(dbo.NFSTOTITEST(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE)*0.076,2) 
end as VendaLiquida,

B.NFSDATEMI as emissao,
A.NFSNUM as NF,
A.SNESER as serie,
A.NFSITE as item
  
from TBS0671 A (nolock)
right join TBS067 B (nolock) on B.SNESER=A.SNESER and B.NFSNUM=A.NFSNUM
Left join TBS042 C (nolock) on C.TESCOD=A.TESCOD
Left join TBS010 D (nolock) on D.PROCOD=A.PROCOD

where 
B.NFSDATEMI between @dataDe and @dataAte and 
B.NFSTIP='N' and 
B.NFSCAN='N' and 
(C.TESCNTVEN='S' or A.NFSROPCNTVEN='S')

order by 
B.NFSDATEMI,A.NFSNUM,A.SNESER,A.NFSITE

	WHERE A.cupom = C.M2_NUMDOC AND A.caixa = C.M2_CXA AND A.coo = C.M2_COO AND

icms,

	then 
		case when B.PROSTBPIS>='06' 
			then 0 
			else round((totven)*0.0165,2) 
		end 
	else 0
end as PIS,

case when status = 1
	then 
		case when B.PROSTBCOFINS>='06' 
			then 0 
			else round((totven)*0.076,2) 
		end 
	else 0

case when status = 1 
	then round(totven,2) - round((totven * icms)/100,2) -

		end -
		case when B.PROSTBCOFINS>='06'

	else 0
