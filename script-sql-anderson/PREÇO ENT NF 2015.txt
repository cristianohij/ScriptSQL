-- USANDO ESSE PARA VERIFICAR OS VALORES MENSAIS DO CD

WITH ENTR AS (
SELECT	PROCOD AS COD,
		(SELECT PRODES FROM TBS010 P WHERE P.PROCOD = A.PROCOD) AS DESCR,
		(SELECT 
			CASE WHEN len(P.MARCOD) = 4 
			THEN rtrim(P.MARCOD) + ' - ' + rtrim(P.MARNOM) 
			ELSE RIGHT(('00' + ltrim(str(P.MARCOD))), 3) + ' - ' + rtrim(P.MARNOM)
			END
			FROM TBS010 P 
			WHERE P.PROCOD = A.PROCOD ) AS MARCA,
			
	(SELECT PROUM1 FROM TBS010 P WHERE P.PROCOD = A.PROCOD) AS UN1,	
			
	convert(int,subString(convert(char(6),NFEDATENT,112),1,4)) AS ANO,
       	convert(int,subString(convert(char(6),NFEDATENT,112),5,2))AS MES ,
		
		-- MEDIA DO VALOR TOTAL DO ITEM NA ENTRADA			
       	CASE WHEN SUM(A.NFEQTD)-ISNULL(SUM(C.NFENFDQTD/NFEQTDEMB),0) > 0 
		THEN 
   	   		round(
				avg((((case when NFEPREXML <= 0 then NFEPRE ELSE NFEPREXML END * case when NFEQTDXML <= 0 then NFEQTD ELSE NFEQTDXML END) 
								   
					+NFEVALICMSST+dbo.NFEVALIPI(A.NFEEMPCOD,A.NFETIP,A.NFENUM,A.NFECOD,A.SEREMPCOD,A.SERCOD,A.NFEITE))
								   
					/(case when NFEQTDXML <= 0 then NFEQTD ELSE NFEQTDXML END))/case when NFEQTDEMB <= 0 then 1 else NFEQTDEMB end ),2)
	   	ELSE 
			0 
		END AS VALORMD,
		
		
		-- VALOR TOTAL DO ITEM NA ENTRADA			   
		CASE WHEN SUM(A.NFEQTD)-ISNULL(SUM(C.NFENFDQTD/NFEQTDEMB),0) > 0 
		THEN 
			SUM(dbo.NFETOTITE(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE)) 
		ELSE
			0 
		END AS VALORENT
		

from TBS0591 A(nolock)  FULL OUTER JOIN  TBS059 B(nolock)  on A.SERCOD=B.SERCOD and A.NFETIP=B.NFETIP and A.NFECOD=B.NFECOD and A.NFENUM=B.NFENUM
	  					FULL OUTER JOIN  TBS0597 C(NOLOCK) ON A.SERCOD=C.SERCOD and A.NFETIP=C.NFETIP and 
	  					A.NFECOD=C.NFECOD AND A.NFEITE=C.NFEITE AND A.NFENUM=C.NFENUM 
	   
	  
where A.NFETIP='N' and NFEDATENT between '20150101' and '20151231' and
       NFENOM not like('BEST BAG%') and NFENOM not like('BEST OFFICE%') and 
	   NFENOM not like('MISASPEL%') and NFENOM not like('%PAPELYNA%') and NFENOM not like('TANBY%')
	   AND A.SERCOD NOT IN('CAN') 
 	   
group by
	convert(char(6),NFEDATENT,112),PROCOD)
	
SELECT RTRIM(LTRIM(COD)) AS COD,
	   DESCR,
	   MARCA,
	   UN1,
	   ANO,
	   MES,
	   REPLACE(STR(VALORMD,10,4),'.',',') AS VALORMD,
	   REPLACE(STR(VALORENT,10,4),'.',',') AS VALORTT
	   --INTO CUSTOENTRADA2015
	   FROM ENTR
	
	ORDER BY MES

       convert(int,subString(convert(char(6),NFEDATENT,112),5,2)),NFECHAACE,

round(avg(dbo.NFEVALICMS(TBS0591.NFEEMPCOD,TBS0591.NFETIP,TBS0591.NFENUM,TBS0591.NFECOD,TBS0591.SEREMPCOD,TBS0591.SERCOD,TBS0591.NFEITE)/NFEQTD/NFEQTDEMB),2) AS ICMS,
