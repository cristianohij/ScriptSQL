select rtrim(ltrim(PDPCOD)) as COD, 
RTRIM(LTRIM(PRODES)) AS DESCR, 
REPLACE(STR(PDPPRELOJ1,10,4),'.',',') as POLPRELOJ, 
REPLACE(STR(TDPPRELOJ1,10,4),'.',',') AS LOJ1,
REPLACE(STR(PDPPRECOR1,10,4),'.',',') AS POLPRECOR, 
REPLACE(STR(TDPPRECOR1,10,4),'.',',') AS CORP1, 
RTRIM(LTRIM(CONVERT(VARCHAR (10),PDPDATALT,3))) AS POLPREATL, 
RTRIM(LTRIM(CONVERT(VARCHAR (10),PDPDATATU,3))) AS POLPREATU
from TBS031 join TBS015 on TDPPROCOD = PDPCOD
where PDPPRELOJ1 <> TDPPRELOJ1 OR TDPPRECOR1 <> PDPPRECOR1
