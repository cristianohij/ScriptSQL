TBS010.PRODES =  ( SELECT B.PRODES FROM #TB010 B WHERE TBS010.PROCOD COLLATE Latin1_General_BIN = B.PROCOD) ,
TBS010.PROCODBAR1 = ( SELECT B.PROCODBAR1	FROM #TB010 B WHERE TBS010.PROCOD COLLATE Latin1_General_BIN = B.PROCOD) ,

INSERTED.PRODATALT , DELETED.PRODATALT AS PRODATALTANT

WHERE TBS010.PROCOD COLLATE Latin1_General_BIN IN ( SELECT PROCOD FROM #TB010 B WHERE TBS010.PROCOD COLLATE Latin1_General_BIN = B.PROCOD)
