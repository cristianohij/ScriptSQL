DECLARE @COD CHAR(15), @DESCRI CHAR(60) , @NOMEMARCA CHAR(60), @MARCOD INT
SET @COD = ''
SET @DESCRI = ''
SET @NOMEMARCA = '' 
SET @MARCOD = 0

------------------------------------------------------------------------------------------------------------------------------------------------------------------

-- TABELA DE SALDO 

if object_id('TempDB.dbo.#TBS032') is not null
   drop table #TBS032
 
SELECT 
PROCOD,
ISNULL((SELECT SUM(ESTQTDATU-ESTQTDRES) FROM TBS032 B (nolock) WHERE ESTLOC =1 AND A.PROCOD = B.PROCOD GROUP BY PROCOD),0) AS EST,
ISNULL((SELECT SUM(ESTQTDATU-ESTQTDRES) FROM TBS032 B (nolock) WHERE ESTLOC =2 AND A.PROCOD = B.PROCOD GROUP BY PROCOD),0) AS LOJA,
SUM(ESTQTDPEN) AS ESTQTDPEN,
SUM(ESTQTDCMP) AS ESTQTDCMP

INTO #TBS032
FROM TBS032 A (NOLOCK)

WHERE 
A.PROCOD = (CASE WHEN @COD = '' THEN A.PROCOD ELSE RTRIM(@COD) END) AND
ESTLOC IN (1,2) AND 
ESTQTDPEN > 0 
  
GROUP BY 
A.PROCOD

------------------------------------------------------------------------------------------------------------------------------------------------------------------

-- TABELA DOS PRODUTOS 

if object_id('TempDB.dbo.#TBS010') is not null
   drop table #TBS010

SELECT 
RTRIM(LTRIM(PROCOD)) AS PROCOD,
RTRIM(LTRIM(PRODES)) AS PRODES,
case when len(A.MARCOD) = 4 then rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) else right(('00' + ltrim(str(A.MARCOD))),3) + ' - ' + rtrim(A.MARNOM) end as MARCOD,
PROUM1 AS UN1

INTO #TBS010
FROM TBS010 A (NOLOCK)

WHERE 
A.PROCOD IN (SELECT PROCOD FROM #TBS032) AND   
A.PRODES LIKE(CASE WHEN @DESCRI = '' THEN A.PRODES ELSE Upper(RTRIM(@DESCRI))+'%' END) AND
A.MARNOM LIKE(CASE WHEN @NOMEMARCA = '' THEN A.MARNOM ELSE Upper(RTRIM(@NOMEMARCA))+'%' END) AND 
A.MARCOD = (CASE WHEN @MARCOD = 0 THEN A.MARCOD ELSE RTRIM(@MARCOD) END) 

ORDER BY 
A.PROCOD

------------------------------------------------------------------------------------------------------------------------------------------------------------------

-- TABELA FINAL 

SELECT 
A.PROCOD,
PRODES,
MARCOD,
UN1,
EST,
LOJA,
ESTQTDPEN,
ESTQTDCMP, 
ISNULL((SELECT TOP 1 DISPONIVEL FROM cdsaldoatual C WHERE A.PROCOD COLLATE DATABASE_DEFAULT = C.CODIGO ),0) AS DISCD,
ISNULL((SELECT TOP 1 DISPONIVEL FROM ttsaldoatual D WHERE A.PROCOD COLLATE DATABASE_DEFAULT = D.CODIGO ),0) AS DISTTE,
ISNULL((SELECT TOP 1 LOJA FROM ttsaldoatual E WHERE A.PROCOD COLLATE DATABASE_DEFAULT = E.CODIGO ),0) AS LOJTTE,
ISNULL((SELECT TOP 1 DISPONIVEL FROM bbsaldoatual F WHERE A.PROCOD COLLATE DATABASE_DEFAULT = F.CODIGO ),0) AS DISBB,
ISNULL((SELECT TOP 1 LOJA FROM bbsaldoatual G WHERE A.PROCOD COLLATE DATABASE_DEFAULT = G.CODIGO ),0) AS LOJBB,
ISNULL((SELECT TOP 1 DISPONIVEL FROM pysaldoatual H WHERE A.PROCOD COLLATE DATABASE_DEFAULT = H.CODIGO ),0) AS DISPY,
ISNULL((SELECT TOP 1 DISPONIVEL FROM misaldoatual I WHERE A.PROCOD COLLATE DATABASE_DEFAULT = I.CODIGO ),0) AS DISMI

FROM #TBS010 A 
LEFT JOIN #TBS032 B ON A.PROCOD = B.PROCOD
