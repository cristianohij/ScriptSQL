
select 'POLITICA' AS TABELA,
PDPCOD AS COD,
CONVERT(VARCHAR (20),A.PDPDATALT+A.PDPHORALT,13) AS ALTERAÇÃO,

CASE WHEN CONVERT(VARCHAR(12),A.PDPDATATU,103) = '01/01/1753  '
	THEN SUBSTRING(CONVERT(VARCHAR(12),A.PDPDATATU,103),12,1) 
	ELSE CONVERT(VARCHAR(20),A.PDPDATATU + A.PDPHORATU,113) END AS ATUALIZAÇÃO,

A.PDPPROLOJ AS 'PROM. LOJA?',
CASE WHEN CONVERT(VARCHAR(12),A.PDPVALPROI,103) = '01/01/1753  '
	THEN SUBSTRING(CONVERT(VARCHAR(12),A.PDPVALPROI,103),12,1) 
	ELSE CONVERT(VARCHAR(12),A.PDPVALPROI,103) END AS 'PRO. INIC. LOJA',

CASE WHEN CONVERT(VARCHAR(12),A.PDPVALPROF,103) = '01/01/1753  '
	THEN SUBSTRING(CONVERT(VARCHAR(12),A.PDPVALPROF,103),12,1) 
	ELSE CONVERT(VARCHAR(12),A.PDPVALPROF,103) END AS 'PRO. FINAL LOJA',


case when A.PDPVALPROI<=getdate() and getdate()<=A.PDPVALPROF and A.PDPPROLOJ='S' then REPLACE(A.PDPPREPRO1,'.',',') else REPLACE(A.PDPPRELOJ1,'.',',') end as 'PREÇO'

from 
TBS015 A  (NOLOCK) 
LEFT JOIN TBS031 B (NOLOCK) ON A.PDPCOD = B.TDPPROCOD
WHERE --PDPCOD = '6533434  ' AND
--(B.TDPDATATU <> A.PDPDATALT+A.PDPHORALT OR
--B.TDPVALPROF <> A.PDPVALPROF OR 
--B.TDPVALPROI <> A.PDPVALPROI)
--AND 
--A.PDPDATALT+A.PDPHORALT >  B.TDPDATATU AND 
case when A.PDPVALPROI<=getdate() and getdate()<=A.PDPVALPROF and A.PDPPROLOJ='S' then ROUND(A.PDPPREPRO1,2) else ROUND(A.PDPPRELOJ1,2) end<>
case when B.TDPVALPROI<=getdate() and getdate()<=B.TDPVALPROF and B.TDPPROLOJ='S' then ROUND(B.TDPPREPRO1,2) else ROUND(B.TDPPRELOJ1,2) end

UNION


SELECT 'LOJA' AS TABELA, 
TDPPROCOD AS CODIGO,
SUBSTRING(CONVERT(VARCHAR(20),'',103),20,1) AS ALTERAÇÃO,

CASE WHEN CONVERT(VARCHAR(12),B.TDPDATATU,103) = '01/01/1753  '
	THEN SUBSTRING(CONVERT(VARCHAR(12),B.TDPDATATU,103),12,1) 
	ELSE CONVERT(VARCHAR(20),B.TDPDATATU,113) END AS ATUALIZAÇÃO,

B.TDPPROLOJ AS 'PROM. LOJA?',
	
CASE WHEN CONVERT(VARCHAR(12),B.TDPVALPROI,103) = '01/01/1753  '
	THEN SUBSTRING(CONVERT(VARCHAR(12),B.TDPVALPROI,103),12,1) 
	ELSE CONVERT(VARCHAR(12),B.TDPVALPROI,103) END AS 'PRO. INIC. LOJA',
	
CASE WHEN CONVERT(VARCHAR(12),B.TDPVALPROF,103) = '01/01/1753  '
	THEN SUBSTRING(CONVERT(VARCHAR(12),B.TDPVALPROF,103),12,1) 
	ELSE CONVERT(VARCHAR(12),B.TDPVALPROF,103) END AS 'PRO. FINAL LOJA',
	
case when B.TDPVALPROI<=getdate() and getdate()<=B.TDPVALPROF and B.TDPPROLOJ='S' then REPLACE(B.TDPPREPRO1,'.',',') else REPLACE(B.TDPPRELOJ1,'.',',') end as 'PREÇO'

from 
TBS015 A  (NOLOCK) 
LEFT JOIN TBS031 B (NOLOCK) ON A.PDPCOD = B.TDPPROCOD
WHERE --PDPCOD = '6533434  ' AND
--(B.TDPDATATU <> A.PDPDATALT+A.PDPHORALT OR
--B.TDPVALPROF <> A.PDPVALPROF OR 
--B.TDPVALPROI <> A.PDPVALPROI)
--AND 
--A.PDPDATALT+A.PDPHORALT >  B.TDPDATATU AND 
case when A.PDPVALPROI<=getdate() and getdate()<=A.PDPVALPROF and A.PDPPROLOJ='S' then ROUND(A.PDPPREPRO1,2) else ROUND(A.PDPPRELOJ1,2) end<>
case when B.TDPVALPROI<=getdate() and getdate()<=B.TDPVALPROF and B.TDPPROLOJ='S' then ROUND(B.TDPPREPRO1,2) else ROUND(B.TDPPRELOJ1,2) end


SELECT PDPCOD, PDPDATALT+PDPHORALT , PDPDATATU+PDPHORATU FROM TBS015 (NOLOCK) WHERE PDPDATALT+PDPHORALT > PDPDATATU+PDPHORATU AND PDPCOD = '6533434        '
