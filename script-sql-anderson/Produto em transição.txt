-- NF DE SAIDA


WITH NFS_CD AS ( 

	SELECT
	A.NFSNUM  AS NFS,
	A.PROCOD   AS CÓD,
	A.NFSPRODES AS 'DESCRIÇÃO',
	(NFSQTD*NFSQTDEMB) AS QTD, 
	C.PROUM1 AS UNI1,
	NFSCLINOM  AS DESTINATÁRIO,
	CONVERT(NVARCHAR,B.NFSDATEMI,2) AS 'DATAS',
	NFSOBS  AS OBS_NF
	
FROM 
cd.SIBD.dbo.TBS010 C 
JOIN cd.SIBD.dbo.TBS0671 A ON A.PROCOD = C.PROCOD
JOIN cd.SIBD.dbo.TBS067 B ON
	
	A.NFSEMPCOD = B.NFSEMPCOD and
	A.NFSNUM = B.NFSNUM and
	A.SNEEMPCOD = B.SNEEMPCOD and
	A.SNESER = B.SNESER
	
	
	
WHERE B.NFSCLICOD IN (590,6709) AND B.NFSCAN = 'N' AND
	NFSTIP IN ('N','T')
	AND B.NFSDATEMI  between '20150824' AND '20150831'

	
GROUP BY A.NFSNUM,NFSCLINOM,NFSOBS,A.PROCOD,NFSQTD,NFSQTDEMB,C.PROUM1,A.NFSPRODES,B.NFSDATEMI ),
	
--NFE ENTRADA

NFE_ND AS (

SELECT 
	NFENUM  AS 'NFE',
	NFENOM AS 'NOME FORNECEDOR',
	CONVERT(NVARCHAR,NFEDATENT,2) AS 'DATAE',
	RTRIM(NFEOBS) AS 'OBS_NF'
	
FROM nd.SIBD.dbo.TBS059 B  
	
	
	
WHERE  NFECAN = 'N' AND NFECOD IN (669)AND
	NFETIP IN ('N','T') AND NFEDATENT  between '20150824' AND '20150831' ) 
	
SELECT A.CÓD AS CÓD,
A.DESCRIÇÃO AS DESCRIÇÃO,
A.UNI1,
REPLACE(STR(ISNULL(A.QTD,0),10,4),'.',',') AS QTD,
A.NFS AS NFTRANSIÇÃO,
ISNULL(B.NFE,0) AS NFENTR,
A.DATAS AS 'DATA SAÍDA'

FROM NFS_CD A LEFT JOIN NFE_ND B ON A.NFS = B.NFE

WHERE NFE IS NULL

GROUP BY A.CÓD, A.DESCRIÇÃO, A.NFS, A.QTD,A.DATAS,B.NFE,A.UNI1

ORDER BY A.NFS

A.PROLOCFIS collate Latin1_General_CI_AS AS LOC,

REPLACE(STR(ISNULL(NFS.QTD,0),10,4),'.',',') collate Latin1_General_CI_AS AS QTD,
NFS.NFS AS NFTRANSIÇÃO,
ISNULL(NFE.NFE,0) AS NFENTR,
NFS.DATAS collate Latin1_General_CI_AS AS 'DATA SAÍDA',
RTRIM(NFS.OBS_NF) AS 'OBS NOTA'


FROM nd.SIBD.dbo.TBS010 A JOIN (
	SELECT
	A.NFSNUM AS NFS,
	C.PROCOD collate Latin1_General_CI_AS AS CÓD,
	A.NFSPRODES  AS 'DESCRIÇÃO',
	(NFSQTD*NFSQTDEMB) AS QTD, 
	NFSCLINOM   AS DESTINATÁRIO,
	CONVERT(NVARCHAR,B.NFSDATEMI,2)  AS 'DATAS',
	NFSOBS  AS OBS_NF
	
FROM 
cd.SIBD.dbo.TBS010 C 
JOIN cd.SIBD.dbo.TBS0671 A ON A.PROCOD = C.PROCOD
JOIN cd.SIBD.dbo.TBS067 B ON
	
	A.NFSEMPCOD = B.NFSEMPCOD and
	A.NFSNUM = B.NFSNUM and
	A.SNEEMPCOD = B.SNEEMPCOD and
	A.SNESER = B.SNESER
	
	
	
WHERE B.NFSCLICOD IN (590,6709) AND B.NFSCAN = 'N' AND
	NFSTIP IN ('N','T')  AND A.SNESER='2'
	AND B.NFSDATEMI  between '20150904' AND '20150908'

	
GROUP BY A.NFSNUM,NFSCLINOM,NFSOBS,C.PROCOD,NFSQTD,NFSQTDEMB,C.PROUM1,A.NFSPRODES,B.NFSDATEMI ) AS NFS  ON A.PROCOD = NFS.CÓD




LEFT JOIN (SELECT B.NFENUM  AS 'NFE',
	A.PROCOD  collate Latin1_General_CI_AS AS 'CÓD',
	A.PROLOCFIS   AS 'LOC',  
	B.NFENOM AS 'NOME FORNECEDOR',
	CONVERT(NVARCHAR,B.NFEDATENT,2)  AS 'DATAE',
	RTRIM(B.NFEOBS)  'OBS_NF'
	
FROM nd.SIBD.dbo.TBS010 A join nd.SIBD.dbo.TBS0591 C ON A.PROCOD = C.PROCOD AND A.PROEMPCOD= C.PROEMPCOD

LEFT JOIN nd.SIBD.dbo.TBS059 B ON B.NFECOD = C.NFECOD AND
B.NFEEMPCOD = C.NFEEMPCOD AND
B.NFENUM = C.NFENUM AND
B.NFETIP = C.NFETIP AND
B.SERCOD = C.SERCOD AND
B.SEREMPCOD = C.SEREMPCOD

  
	
	
	
WHERE  B.NFECAN = 'N' AND B.NFECOD IN (669)AND
	B.NFETIP IN ('N','T') AND B.NFEDATENT  between '20150904' AND '20150908' ) AS NFE ON A.PROCOD = NFE.CÓD AND NFE.NFE = NFS.NFS
	
	WHERE NFE IS NULL
	
	
	GROUP BY A.PROCOD, A.PRODES, A.PROLOCFIS, A.PROUM1,NFS.NFS,NFE.NFE,NFS.DATAS,NFS.QTD,NFS.OBS_NF
	
	ORDER BY NFS.NFS
