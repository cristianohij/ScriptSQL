IF OBJECT_ID('TempDB.dbo.#TBS010') IS NOT NULL 
DROP TABLE #TBS010

SELECT 
A.PROSTATUS,
PROLOCFIS,
A.PROCOD,
A.PRODES,
case when len(A.MARCOD) = 4 
	then rtrim(A.MARCOD) + ' - ' + rtrim(MARNOM) 
	else right(('00' + ltrim(str(A.MARCOD))),3) + ' - ' + rtrim(MARNOM)
end collate Latin1_General_BIN as MARCOD ,
PROUM1,
REPLACE(ESTQTDATU - ESTQTDRES,'.',',') AS ESQTDDIS

INTO #TBS010
FROM TBS010 A (NOLOCK)
LEFT JOIN TBS032 B (NOLOCK) ON A.PROCOD = B.PROCOD 

WHERE 
A.PROLOCFIS <> '' AND
B.ESTLOC = 1 


IF OBJECT_ID('TempDB.dbo.#TBS045') IS NOT NULL 
DROP TABLE #TBS045

SELECT
IDENTITY (INT,1,1) AS ID, 
D.PROCOD,
E.PDCDATCAD AS PDCDAT

INTO #TBS045 
FROM 
TBS0451 D (NOLOCK) 
LEFT JOIN TBS045 E (NOLOCK) ON  D.PDCNUM = E.PDCNUM

WHERE
PDCBLQ ='N' AND 
D.PROCOD IN (SELECT DISTINCT PROCOD FROM #TBS010)

ORDER BY
PROCOD, PDCDATCAD


SELECT *, (SELECT PDCDAT FROM #TBS045 B WHERE A.PROCOD = B.PROCOD AND ID IN (SELECT MAX(ID) FROM #TBS045 GROUP BY PROCOD)) FROM #TBS010 A
