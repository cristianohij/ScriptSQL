-- CRIANDO TABELA TEMP COM OS VALORES REAIS DE DESCONTO
if object_id('TempDB.dbo.#TMP') is not null
   drop table #TMP
   
	select
	'5.929' AS M2_CFOP,	
	A.M2_TRI,
	A.M2_CST,
	A.M2_ITE,
	A.M2_LOJ, 
	A.M2_CUP, 
	A.M2_COO, 
	A.M2_ECF, 
	A.M2_CXA ,
	M2_PROCOD,
	M2_QTD,
	M2_VALBRU,
	M2_VALBRU - M2_ABT + M2_ACRCUP - 
	CASE WHEN  A.DESCONTO <> ISNULL(B.DESCONTO,0) AND M2_ITE = (SELECT TOP 1 M2_ITE 
																FROM #MSLAUX E 
																
																WHERE 
																A.M2_COO = E.M2_COO and A.M2_ECF = E.M2_ECF AND A.M2_CUP = E.M2_CUP 
																
																ORDER BY 
																M2_DESCUP DESC)
		THEN
			CASE WHEN A.DESCONTO > ISNULL(B.DESCONTO,0) 
				THEN ROUND(M2_DESCUP,2) - (A.DESCONTO - ISNULL(B.DESCONTO,0))
				ELSE ROUND(M2_DESCUP,2) + (ISNULL(B.DESCONTO,0) - A.DESCONTO)
			END
		ELSE ROUND(M2_DESCUP,2)
	END AS M2_LIQUIDO
	
	INTO #TMP	
	FROM (
		SELECT 
		M2_ITE,M2_PROCOD,M2_TRI,M2_CST, M2_QTD,
		M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , 
		M2_DESCUP ,
		M2_ACRCUP,
		M2_ABT ,
		M2_VALBRU,
		SUM(ISNULL(ROUND(M2_DESCUP,2),0)) OVER (PARTITION BY M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA ) AS DESCONTO
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 1 ) AS A 
			
		LEFT JOIN (
		SELECT 
		M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , M2_VALBRU AS DESCONTO 
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 2 ) AS B ON A.M2_LOJ = B.M2_LOJ AND A.M2_CUP = B.M2_CUP AND A.M2_COO= B.M2_COO AND 
														A.M2_ECF = B.M2_ECF AND A.M2_CXA = B.M2_CXA  

	
-------------------------------------------------------------------------------------------------------------------------------------------------

-- CRIANDO TABELA DE ITENS COM OS TRIBUTOS NECESSARIOS

if object_id('TempDB.dbo.#LIQUIDO') is not null
   drop table #LIQUIDO

		SELECT
		M2_PROCOD,
		M2_QTD,
		C.M2_CFOP,
		-- C.M2_VALBRU AS NFSTOTPRO ,					-- TOTAL DOS PRODUTOS,		
		C.M2_LIQUIDO AS NFSTOTITEST,				-- TOTAL DAS NOTAS, NO CASO DOS CUPONS NÃO TEM NOTA, MAS COLQUEI LQUIDO
		
		CASE WHEN (CONVERT(DECIMAL(10,2),SUBSTRING(C.M2_TRI,2,5)) / 100) > 0 
			THEN C.M2_LIQUIDO  						-- valor liquido
			ELSE 0 
		END AS NFSBASICMS, 							-- BASE ICMS
		
		C.M2_LIQUIDO * (CONVERT(DECIMAL(10,2),SUBSTRING(C.M2_TRI,2,5)) / 100) AS NFSVALICMS ,  -- VALOR LIQUIDO * PORCENTAGEM TRIBUTADA -- VALOR ICMS				 
		
		CASE WHEN SUBSTRING(M2_CST,2,2) IN ('30','40','41','50')
			THEN C.M2_LIQUIDO
			ELSE 0
		END AS ICMSISENAOTRI							-- AS VALOR ICMS ISENTO/NÃO TRIBUTADO
		
		INTO #LIQUIDO
		FROM #TMP C 
		LEFT JOIN TBS010 B (NOLOCK) ON C.M2_PROCOD = CONVERT(INT,B.PROCOD)
		
----------------------------------------------------------------------------------------------------------------------------------------------------------------

if object_id('TempDB.dbo.#CUP') is not null
   drop table #CUP
   
		SELECT 
		M2_PROCOD AS PROCOD,
		SUM(M2_QTD) AS QTD,
		M2_CFOP AS CFOP,
		-- SUM(NFSTOTPRO)					AS NFSTOTPRO,		-- TOTAL DOS PRODUTOS,
		SUM(NFSTOTITEST)					AS NFSTOTITEST,		-- TOTAL DAS NOTAS, NO CASO DOS CUPONS NÃO TEM NOTA, MAS COLQUEI O VALOR LIQUIDO
		SUM(NFSBASICMS) 					AS NFSBASICMS, 		-- BASE ICMS
		SUM(NFSVALICMS) 					AS NFSVALICMS,		-- VALOR ICMS
		SUM(ICMSISENAOTRI)					AS ICMSISENAOTRI,	-- AS VALOR ICMS ISENTO/NÃO TRIBUTADO
		SUM(NFSTOTITEST) - SUM(NFSBASICMS)	AS ICMSOUTROS,		-- ICMS OUTROS 
		SUM(NFSTOTITEST)					AS NFSBASPISCOF, 	-- BASE CALCULO
		ROUND(SUM(NFSTOTITEST)* 0.0165,2) 	AS PIS, 			-- VALOR PIS
		ROUND(SUM(NFSTOTITEST)* 0.076 ,2) 	AS COFINS			-- VALOR COFINS
		
		INTO #CUP
		FROM #LIQUIDO
		
		GROUP BY
		M2_CFOP,M2_PROCOD
		
---------------------------------------------------------------------------------------------------------------------------------------------------------------

-- NOTAS DE ENTRADA, CONTABILIZAR O CUSTO MÊS A MÊS 

if object_id('TempDB.dbo.#CUSTO') is not null
   drop table #CUSTO

SELECT
PROCOD,
sum(ROUND(NFETOTOPEITE,2)) / SUM(NFEQTD*NFEQTDEMB) as NFEPREMED

INTO #CUSTO

FROM
TBS0591 A (NOLOCK) 
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND 
A.NFENUM = B.NFENUM AND A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND A.SEREMPCOD = B.SEREMPCOD
		
WHERE 
NFEDATEFE between @datade - 120 and @dataate and 
A.NFETIP = 'N' and 
(NFENOSFOR = 'N' OR NFENOSFOR = '') AND 
(B.NFECAN = 'N' OR B.NFECAN = '') AND 
NFEDATEFE <> '' AND
A.NFECFOP IN ('1.102','1.403','2.102','2.202','2.403','1.252','1.202','2.411','1.411','1.653') AND
B.NFECOD NOT IN (SELECT CLICOD FROM #GRUPOCLI) 
		
GROUP BY 
PROCOD

-- TABELA TBS010

IF OBJECT_ID('TempDB.dbo.#TBS010') IS NOT NULL
DROP TABLE #TBS010

SELECT 
CASE WHEN len(A.MARCOD) = 4 
	then rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) 
	else right(('00' + ltrim(str(A.MARCOD))),3) + ' - ' + rtrim(A.MARNOM) 
end as 'MARCA',
A.PROCOD,
PRODES,
PROUM1

INTO #TBS010
FROM TBS010 A (NOLOCK)
--WHERE

--A.MARCOD =(CASE WHEN @marcod = 0 THEN A.MARCOD ELSE Upper(RTRIM(@marcod)) END) AND 
--A.MARNOM LIKE(CASE WHEN @marcanom = '' THEN '% %' ELSE Upper(RTRIM(@marcanom))+'%' END) AND 
--A.PRODES LIKE (CASE WHEN @PRODES = '' THEN '% %' ELSE UPPER(RTRIM(@PRODES))+'%' END)

-------------------------------------------------------------------------------------------------------------------------------------------------

-- TABELA FINAL

select 
A.PROCOD,
PRODES,
MARCA,
PROUM1,
ISNULL(QTD,0) AS VENQTD,
CONVERT(DECIMAL(10,4),ROUND(ISNULL(NFSTOTITEST,0)/ CASE WHEN ISNULL(QTD,0) = 0 THEN 1 ELSE ISNULL(QTD,0) END ,4)) AS VALMED,
ISNULL(NFSTOTITEST,0) AS VENVAL,
isnull(NFEPREMED,(SELECT TDPCUSREA FROM TBS031 D (NOLOCK) WHERE A.PROCOD = D.TDPPROCOD)) AS CUSTO,
isnull(NFEPREMED,(SELECT TDPCUSREA FROM TBS031 D (NOLOCK) WHERE A.PROCOD = D.TDPPROCOD)) * ISNULL(QTD,0) as CUSVAL,

ISNULL(NFSBASICMS,0) 	AS NFSBASICMS , 	-- BASE ICMS
ISNULL(NFSVALICMS,0) 	AS NFSVALICMS,		-- VALOR ICMS
ISNULL(ICMSISENAOTRI,0) AS ICMSISENAOTRI,	-- AS VALOR ICMS ISENTO/NÃO TRIBUTADO
ISNULL(ICMSOUTROS,0) 	AS ICMSOUTROS,		-- ICMS OUTROS 
ISNULL(NFSBASPISCOF,0) 	AS NFSBASPISCOF, 	-- BASE CALCULO
ISNULL(PIS,0) 			AS PIS, 			-- VALOR PIS
ISNULL(COFINS,0) 		AS COFINS,			-- VALOR COFINS






ISNULL(NFSTOTITEST,0) - (isnull(NFEPREMED,(SELECT TDPCUSREA FROM TBS031 D (NOLOCK) WHERE A.PROCOD = D.TDPPROCOD)) * ISNULL(QTD,0)) AS LUCVAL

FROM #TBS010 A (NOLOCK)
LEFT JOIN #CUP C ON CONVERT(NUMERIC,A.PROCOD) = C.PROCOD
LEFT JOIN #CUSTO E ON A.PROCOD = E.PROCOD

WHERE 
ISNULL(QTD,0) <> 0 

ORDER BY 
ISNULL(NFSTOTITEST,0) - (isnull(NFEPREMED,(SELECT TDPCUSREA FROM TBS031 D (NOLOCK) WHERE A.PROCOD = D.TDPPROCOD)) * ISNULL(QTD,0)) DESC
