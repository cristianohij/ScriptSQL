SELECT 
CONVERT(CHAR(10),B.MVIDATEFE,103) as MVIDATEFE,
--B.MVIDOC,
A.PROCOD,
(SELECT RTRIM(LTRIM(PRODES)) FROM TBS010 C (NOLOCK) WHERE A.PROCOD = C.PROCOD) AS PRODES,
(SELECT PROUM1 FROM TBS010 C (NOLOCK) WHERE A.PROCOD = C.PROCOD) AS PROUM1,
SUM(A.MVIQTDEMB * A.MVIQTDATD) AS QTD

into #EST1
FROM TBS0371 A (NOLOCK)
LEFT JOIN TBS037 B (NOLOCK) ON A.MVIDOC = B.MVIDOC AND A.MVIEMPCOD = B.MVIEMPCOD

WHERE 
B.MVIDATEFE > '20171101' AND 
B.MVILOCORI = 1 AND 
B.MVILOCDES = 2 

GROUP BY 
CONVERT(CHAR(10),B.MVIDATEFE,103),
PROCOD

-- SELECT * FROM #EST1


SELECT 
CONVERT(CHAR(10),B.MVIDATEFE,103) as MVIDATEFE,
--B.MVIDOC,
A.PROCOD,
(SELECT RTRIM(LTRIM(PRODES)) FROM TBS010 C (NOLOCK) WHERE A.PROCOD = C.PROCOD) AS PRODES,
(SELECT PROUM1 FROM TBS010 C (NOLOCK) WHERE A.PROCOD = C.PROCOD) AS PROUM1,
SUM(A.MVIQTDEMB * A.MVIQTDATD) AS QTD

into #EST2
FROM TBS0371 A (NOLOCK)
LEFT JOIN TBS037 B (NOLOCK) ON A.MVIDOC = B.MVIDOC AND A.MVIEMPCOD = B.MVIEMPCOD

WHERE 
B.MVIDATEFE > '20171101' AND 
B.MVILOCORI = 2 AND 
B.MVILOCDES = 1 

GROUP BY 
CONVERT(CHAR(10),B.MVIDATEFE,103),
PROCOD





SELECT 
'SAIU' AS ESTOQUE,
MVIDATEFE,
PROCOD,
PRODES,
PROUM1,
REPLACE(QTD,'.',',') AS QTD

INTO #FIM
FROM #EST1 A 

WHERE 
PROCOD IN (SELECT DISTINCT A.PROCOD 
			FROM #EST1 A
			INNER JOIN #EST2 B ON A.PROCOD =B.PROCOD AND A.MVIDATEFE <= B.MVIDATEFE)
			
UNION 

SELECT 
'ENTROU' AS ESTOQUE,
MVIDATEFE,
PROCOD,
PRODES,
PROUM1,
REPLACE(QTD,'.',',') AS QTD

FROM #EST2 A 

WHERE 
PROCOD IN (SELECT DISTINCT A.PROCOD 
			FROM #EST1 A
			INNER JOIN #EST2 B ON A.PROCOD =B.PROCOD AND A.MVIDATEFE <= B.MVIDATEFE)





SELECT * FROM #FIM ORDER BY PROCOD, MVIDATEFE, ESTOQUE
