declare @data1de char(8), @data1ate char(8), @comando varchar (max)

set @data1de  ='20161001'
set @data1ate ='20170331'

SET @comando = 'EXECUTE(''
select 
caixa,
ecf,
coo,
item,
cdprod,
status,
cancelado

FROM movcaixa
where
data between "'+convert(char(8), @data1de, 112)+'" and "'+convert(char(8), @data1ate, 112)+'"  and 
status in(01,10)
'') at MYSQLGZ' 

if object_id('TempDB.dbo.#MSLAUX') is not null
   drop table #MSLAUX
    
create table #MSLAUX (M2_CXA INT, M2_ECF INT, M2_CUP INT, M2_ITE INT,M2_PROCOD INT, M2_STATUS INT, M2_CAN CHAR(1))
INSERT INTO #MSLAUX
exec(@comando)

-- 05118717000156 -- BEST BAG
-- 65069593000350 -- CD
-- 52080207000117 -- Misaspel
-- 65069593000198 -- TANBY ND
-- 44125185000136 -- PAPELYNA
-- 65069593000279 -- TTE

SELECT 
'65069593000279' +';'+
RTRIM(LTRIM(STR(M2_CXA)))+';'+
RTRIM(LTRIM(STR(M2_ECF)))+';'+
RTRIM(LTRIM(STR(M2_CUP)))+';'+
RTRIM(LTRIM(STR(M2_ITE)))+';'+
CASE WHEN Len(M2_PROCOD)< 7
	THEN right('0000000'+Ltrim(M2_PROCOD),7)
	ELSE RTRIM(LTRIM(STR(M2_PROCOD)))
END +';'+
CASE WHEN M2_STATUS = 10 OR M2_CAN = 'S'
	THEN 'S'
	ELSE 'N'
END

FROM 
#MSLAUX

ORDER BY
M2_CXA,M2_ECF,M2_CUP,M2_ITE
