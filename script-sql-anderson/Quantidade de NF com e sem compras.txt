SELECT  COUNT(E.NFENUM) AS 'TOTAL' ,(SELECT COUNT(B.NFENUM) 
	
		 from TBS0591 A
		 join TBS059 B  on
		 A.NFEEMPCOD = B.NFEEMPCOD AND
		 A.NFECOD = B.NFECOD AND
		 A.NFENUM = B.NFENUM AND
		 A.NFETIP = B.NFETIP AND
		 A.SERCOD = B.SERCOD AND
		 A.SEREMPCOD = B.SEREMPCOD
	
		 WHERE B.NFECAN = 'N' AND NFEDATENT between '20150101' and '20151231'-- AND PROCOD= '0128782'
		 AND A.NFETIP='N' AND A.SERCOD NOT IN('CAN')
		 AND A.NFEPDCNUM = 0 AND NFENOM not like('BEST BAG%') and NFENOM not like('BEST OFFICE%') and 
		 NFENOM not like('MISASPEL%') and NFENOM not like('%PAPELYNA%') and NFENOM not like('TANBY%')) AS 'SEM COMPRAS',
		 
		 (SELECT COUNT(B.NFENUM) 
	
		 from TBS0591 A
		 join TBS059 B  on
		 A.NFEEMPCOD = B.NFEEMPCOD AND
		 A.NFECOD = B.NFECOD AND
		 A.NFENUM = B.NFENUM AND
		 A.NFETIP = B.NFETIP AND
		 A.SERCOD = B.SERCOD AND
		 A.SEREMPCOD = B.SEREMPCOD
	
		 WHERE B.NFECAN = 'N' AND NFEDATENT between '20150101' and '20151231'-- AND PROCOD= '0128782'
		 AND A.NFETIP='N' AND A.SERCOD NOT IN('CAN')
		 AND A.NFEPDCNUM <> '' and NFENOM not like('BEST BAG%') and NFENOM not like('BEST OFFICE%') and 
		 NFENOM not like('MISASPEL%') and NFENOM not like('%PAPELYNA%') and NFENOM not like('TANBY%')) AS 'COM COMPRAS'
		 
		 
	
		 from TBS0591 D
		 join TBS059 E  on
		 D.NFEEMPCOD = E.NFEEMPCOD AND
		 D.NFECOD = E.NFECOD AND
		 D.NFENUM = E.NFENUM AND
		 D.NFETIP = E.NFETIP AND
		 D.SERCOD = E.SERCOD AND
		 D.SEREMPCOD = E.SEREMPCOD
	
		 WHERE E.NFECAN = 'N' AND NFEDATENT between '20150101' and '20151231'
		 AND D.NFETIP='N' AND D.SERCOD NOT IN('CAN') and NFENOM not like('BEST BAG%') and NFENOM not like('BEST OFFICE%') and 
		 NFENOM not like('MISASPEL%') and NFENOM not like('%PAPELYNA%') and NFENOM not like('TANBY%') -- AND PROCOD= '0128782'
