-- CRIANDO TABELA TEMP COM OS VALORES REAIS DE DESCONTO

if object_id('TempDB.dbo.#TMP') is not null
   drop table #TMP
   
	select
	CASE ISNULL(D.COO,0) WHEN 0 
		THEN '5.929C'
		ELSE '5.929'
	END AS M2_CFOP,
	CASE ISNULL(D.COO,0) WHEN 0 
		THEN 'F'
		ELSE 'L'
	END AS M2_TIP,
	A.M2_DAT,
	A.M2_TRI,
	A.M2_CST,
	A.M2_ITE,
	A.M2_LOJ, 
	A.M2_CUP, 
	A.M2_COO, 
	A.M2_ECF, 
	A.M2_CXA ,
	M2_PROCOD,
	M2_VALBRU,
	M2_VALBRU - M2_ABT + M2_ACRCUP - 
	CASE WHEN  A.DESCONTO <> ISNULL(B.DESCONTO,0) AND M2_ITE = (SELECT TOP 1 M2_ITE 
								FROM #MSLAUX E 
																
								WHERE 
								A.M2_COO = E.M2_COO and A.M2_ECF = E.M2_ECF AND A.M2_CUP = E.M2_CUP 
																
								ORDER BY 
								M2_DESCUP DESC)
																
		THEN
			CASE WHEN A.DESCONTO > ISNULL(B.DESCONTO,0) 
				THEN ROUND(M2_DESCUP,2) - (A.DESCONTO - ISNULL(B.DESCONTO,0))
				ELSE ROUND(M2_DESCUP,2) + (ISNULL(B.DESCONTO,0) - A.DESCONTO)
			END
		ELSE ROUND(M2_DESCUP,2)
	END AS M2_LIQUIDO
	
	INTO #TMP	
	FROM (
		SELECT
		M2_DAT, 
		M2_ITE,M2_PROCOD,M2_TRI,M2_CST,
		M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , 
		M2_DESCUP ,
		M2_ACRCUP,
		M2_ABT ,
		M2_VALBRU,
		SUM(ISNULL(ROUND(M2_DESCUP,2),0)) OVER (PARTITION BY M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA ) AS DESCONTO
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 1 ) AS A 
			
		LEFT JOIN (
		SELECT 
		M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , M2_VALBRU AS DESCONTO 
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 2 ) AS B ON A.M2_LOJ = B.M2_LOJ AND A.M2_CUP = B.M2_CUP AND A.M2_COO= B.M2_COO AND 
														A.M2_ECF = B.M2_ECF AND A.M2_CXA = B.M2_CXA  
		
	INNER JOIN #NFSLOJA D (NOLOCK) ON A.M2_COO = D.COO AND A.M2_ECF = D.ECF

SELECT 
@UTEL AS UTEL,
ISNULL(ISNULL(ISNULL(ISNULL(A.DATA,C.DATA),D.DATA),E.DATA),F.DATA) as DATA,
ISNULL(ISNULL(ISNULL(ISNULL(A.NFSDATEMI,C.NFEDATEFE),D.NFSDATEMICUP),E.NFSDATEMI),F.NFEDATEFE) as NFSDATEMI,
ISNULL(ISNULL(ISNULL(ISNULL(A.DIA,C.DIA),D.DIA),E.DIA),F.DIA) AS DIA,

ISNULL(ISNULL(ISNULL(ISNULL(A.NFSCLICOD,C.NFECOD),D.NFSCLICOD),E.NFSCLICOD),F.NFECOD) AS NFSCLICOD,
ISNULL(ISNULL(ISNULL(ISNULL(A.NFSCLINOM,C.NFENOM),D.NFSCLINOM),E.NFSCLINOM),F.NFENOM) AS NFSCLINOM,

ISNULL(A.QTDNOT,0) AS QTDNOT,
ISNULL(A.NFSTOTITEST,0) AS NFSVAL,

ISNULL(E.QTDNOT,0) AS QTDNOTGRU,
ISNULL(E.NFSTOTITEST,0) AS NFSVALGRU,

ISNULL(D.QTDNOTCUP,0) AS QTDNOTCUP,
ISNULL(D.NFSTOTITESTCUP,0) AS NFSVALCUP,

ISNULL(A.QTDNOT,0) + ISNULL(D.QTDNOTCUP,0) as QTDNOTTOT,
ISNULL(A.NFSTOTITEST,0) + ISNULL(D.NFSTOTITESTCUP,0) AS VALBRU,

ISNULL(C.QTDNOTDEV,0) AS QTDNOTDEV,
ISNULL(C.VALDEV,0) AS VALDEV,

ISNULL(F.QTDNOTDEV,0) AS QTDNOTDEVGRU,
ISNULL(F.VALDEV,0) AS VALDEVGRU,

ISNULL(A.NFSTOTITEST,0) + ISNULL(D.NFSTOTITESTCUP,0) - ISNULL(C.VALDEV,0) AS VALLIQ,
ISNULL(A.NFSTOTITEST,0) + ISNULL(D.NFSTOTITESTCUP,0) - ISNULL(C.VALDEV,0) - ISNULL(E.NFSTOTITEST,0) + ISNULL(F.VALDEV,0)  AS VALLIQSEMGRU


FROM 
#NFS A 
FULL OUTER JOIN #DEV C ON A.NFSDATEMI = C.NFEDATEFE AND A.NFSCLICOD = C.NFECOD
FULL OUTER JOIN #NFSCUP D ON ISNULL(A.NFSDATEMI,C.NFEDATEFE) = D.NFSDATEMICUP AND ISNULL(A.NFSCLICOD,C.NFECOD) = D.NFSCLICOD
FULL OUTER JOIN #NFSGRU E ON ISNULL(ISNULL(A.NFSDATEMI,C.NFEDATEFE),D.NFSDATEMICUP) = E.NFSDATEMI AND ISNULL(ISNULL(A.NFSCLICOD,C.NFECOD),D.NFSCLICOD) = E.NFSCLICOD
FULL OUTER JOIN #DEVGRU F ON ISNULL(ISNULL(ISNULL(A.NFSDATEMI,C.NFEDATEFE),D.NFSDATEMICUP),E.NFSDATEMI) = F.NFEDATEFE AND 
							 ISNULL(ISNULL(ISNULL(A.NFSCLICOD,C.NFECOD),D.NFSCLICOD),E.NFSCLICOD) = F.NFECOD

WHERE 
ISNULL(ISNULL(ISNULL(ISNULL(A.DIA,C.DIA),D.DIA),E.DIA),F.DIA) IN (@DIAS) AND 
ISNULL(ISNULL(ISNULL(ISNULL(A.NFSCLICOD,C.NFECOD),D.NFSCLICOD),E.NFSCLICOD),F.NFECOD) = 
CASE WHEN @NFSCLICOD = 0 
	THEN ISNULL(ISNULL(ISNULL(ISNULL(A.NFSCLICOD,C.NFECOD),D.NFSCLICOD),E.NFSCLICOD),F.NFECOD)
	ELSE @NFSCLICOD
END  AND

ISNULL(ISNULL(ISNULL(ISNULL(A.NFSCLINOM,C.NFENOM),D.NFSCLINOM),E.NFSCLINOM),F.NFENOM) LIKE (
CASE WHEN @NFSCLINOM = '' 
	THEN ISNULL(ISNULL(ISNULL(ISNULL(A.NFSCLINOM,C.NFENOM),D.NFSCLINOM),E.NFSCLINOM),F.NFENOM)
	ELSE RTRIM(UPPER(@NFSCLINOM))+'%'
END )

ORDER BY 
ISNULL(ISNULL(ISNULL(ISNULL(A.NFSDATEMI,C.NFEDATEFE),D.NFSDATEMICUP),E.NFSDATEMI),F.NFEDATEFE)

																			  ELSE RTRIM(UPPER(@NFSCLINOM))+'%'

;with 
NFS as (
		SELECT
		CASE DATEPART(w,NFSDATEMI) 
		WHEN 1 THEN 'Dom'
		WHEN 2 THEN 'Seg'
		WHEN 3 THEN 'Ter'
		WHEN 4 THEN 'Qua'
		WHEN 5 THEN 'Qui' 
		WHEN 6 THEN 'Sex'
		WHEN 7 THEN 'Sáb'
		END AS DIA,
		A.NFSCLICOD AS NFSCLICOD,
		A.NFSCLINOM AS NFSCLINOM,
		CONVERT(CHAR (10), NFSDATEMI,103) AS NFSDATEMI,
		SUM(ROUND(dbo.NFSTOTITEST(B.NFSEMPCOD,B.NFSNUM,B.SNEEMPCOD,B.SNESER,B.NFSITE),2)) AS NFSTOTITEST,
		COUNT(DISTINCT A.NFSNUM) AS QTDNOT
		
		FROM 
		TBS0671 B (nolock)
		LEFT JOIN TBS067 A (nolock) ON A.NFSEMPCOD = B.NFSEMPCOD and A.NFSNUM = B.NFSNUM and A.SNEEMPCOD = B.SNEEMPCOD and A.SNESER = B.SNESER
		LEFT JOIN TBS080 D (NOLOCK) ON B.NFSNUM = D.ENFNUM AND B.SNEEMPCOD = D.SNEEMPCOD AND B.SNESER = D.SNESER
			  
		WHERE 
		NFSDATEMI between @dateDe and @dateAte and

		B.NFSCFOP IN ('5.102','5.202','5.405','5.411','5.922','5.929','6.102','6.202','6.108','6.404','6.411','6.922')

		GROUP BY 
		NFSDATEMI, A.NFSCLICOD, A.NFSCLINOM

		WHEN 4 THEN 'Qua'
		WHEN 5 THEN 'Qui' 
		WHEN 6 THEN 'Sex'
		WHEN 7 THEN 'Sáb'

		SUM(ROUND(dbo.NFSTOTITEST(B.NFSEMPCOD,B.NFSNUM,B.SNEEMPCOD,B.SNESER,B.NFSITE),2)) AS NFSTOTITEST,
		COUNT(DISTINCT A.NFSNUM) AS QTDNOT
		
		FROM 
		TBS0671 B (nolock)

		LEFT JOIN TBS080 D (NOLOCK) ON B.NFSNUM = D.ENFNUM AND B.SNEEMPCOD = D.SNEEMPCOD AND B.SNESER = D.SNESER
			  
		WHERE 
		NFSDATEMI between @dateDe and @dateAte and 
		NFSTIP IN('N','C') and 
		D.ENFSIT = 6 AND
		ENFTIPDOC = 1 AND 
		                                -- BEST                       -- PAPELYNA                    -- MISASPEL                -- TANBYS

		A.NFENUM = B.NFENUM AND A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND A.SEREMPCOD = B.SEREMPCOD
		LEFT JOIN TBS080 D (NOLOCK) ON A.NFENUM = D.ENFNUM AND A.NFECOD = D.ENFCODDES
