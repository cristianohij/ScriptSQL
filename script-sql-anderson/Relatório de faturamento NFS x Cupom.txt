if object_id('TempDB.dbo.#NFS') is not null
   drop table #NFS
   
		SELECT
		CASE DATEPART(w,NFSDATEMI) 
		WHEN 1 THEN 'Dom'
		WHEN 2 THEN 'Seg'
		WHEN 3 THEN 'Ter'
		WHEN 4 THEN 'Qua'
		WHEN 5 THEN 'Qui' 
		WHEN 6 THEN 'Sex'
		WHEN 7 THEN 'Sáb'
		END AS DIA,
		CONVERT(CHAR (10), NFSDATEMI,103) AS NFSDATEMI,
		SUM(round(dbo.NFSTOTITEST(B.NFSEMPCOD,B.NFSNUM,B.SNEEMPCOD,B.SNESER,B.NFSITE),2)) AS NFSTOTITEST,
		COUNT(DISTINCT A.NFSNUM) AS QTDNOT
		
		INTO #NFS
		FROM 
		TBS0671 B (nolock)
		LEFT JOIN TBS067 A (nolock) ON A.NFSEMPCOD = B.NFSEMPCOD and A.NFSNUM = B.NFSNUM and A.SNEEMPCOD = B.SNEEMPCOD and A.SNESER = B.SNESER
		LEFT JOIN TBS080 D (NOLOCK) ON B.NFSNUM = D.ENFNUM AND B.SNEEMPCOD = D.SNEEMPCOD AND B.SNESER = D.SNESER
			  
		WHERE 
		NFSDATEMI between @dateDe and @dateAte and 
		NFSTIP IN('N','C') and 
		D.ENFSIT = 6 AND 
		D.ENFTIPDOC = 1 AND
		A.NFSCLINOM NOT LIKE ('TANBY%') AND
		B.NFSCFOP IN ('5.102','5.202','5.405','5.411','5.922','5.929','6.102','6.202','6.108','6.404','6.411','6.922')
		
		GROUP BY 
		NFSDATEMI

-------------------------------------------------------------------------------------------------------------------------------------------------------------------

if object_id('TempDB.dbo.#NFSLOJA') is not null
   drop table #NFSLOJA
		
		SELECT 
		DISTINCT 
		C.NFSNFRCOO AS COO ,
		C.NFSNFRECF AS ECF
		
		INTO #NFSLOJA
		FROM 
		TBS0674 C (NOLOCK) 		
		LEFT JOIN TBS080 D (NOLOCK) ON C.NFSNUM = D.ENFNUM AND C.SNEEMPCOD = D.SNEEMPCOD AND C.SNESER = D.SNESER
		LEFT JOIN TBS067 A (nolock) ON A.NFSEMPCOD = C.NFSEMPCOD and A.NFSNUM = C.NFSNUM and A.SNEEMPCOD = C.SNEEMPCOD and A.SNESER = C.SNESER
		
		WHERE
		C.NFSNFRTIP = 'CUP' AND 
		A.NFSDATEMI BETWEEN @dateDe and @dateAte + 10 and
		D.ENFSIT = 6 AND -- VERIFICA SE ESTÁ AUTORIZADA
		D.ENFTIPDOC = 1 
		
-------------------------------------------------------------------------------------------------------------------------------------------------------------------

if object_id('TempDB.dbo.#CUPOM') is not null
   drop table #CUPOM
    
		SELECT 
		CONVERT(CHAR(10),datas,103) AS DATAS,
		
		CASE ISNULL(B.COO,0) WHEN 0 
			THEN 1
			ELSE 0 
		END AS QTDCUPSEMNFS,
		
		CASE ISNULL(B.COO,0) WHEN 0 
			THEN ROUND(valortot,2) 
			ELSE 0 
		END AS VALORTOT,
		
		CASE ISNULL(B.COO,0) WHEN 0 
			THEN 0
			ELSE 1
		END AS QTDCUPNFS,
		
		CASE ISNULL(B.COO,0) WHEN 0 
			THEN 0
			ELSE ROUND(valortot,2)  
		END AS VALORTOTNFS,
		
		ROUND(valortot,2) AS TTCUPOM
		
		INTO #CUPOM
		FROM 
		#MSL002 A
		LEFT JOIN #NFSLOJA B ON A.coo = B.COO AND A.ecf = B.ECF
		
-------------------------------------------------------------------------------------------------------------------------------------------------------------------

if object_id('TempDB.dbo.#CUP') is not null
   drop table #CUP
					
		SELECT
		CASE DATEPART(w,DATAS) 
		WHEN 1 THEN 'Dom'
		WHEN 2 THEN 'Seg'
		WHEN 3 THEN 'Ter'
		WHEN 4 THEN 'Qua'
		WHEN 5 THEN 'Qui' 
		WHEN 6 THEN 'Sex'
		WHEN 7 THEN 'Sáb'
		END AS DIA, 
		DATAS,
		SUM(QTDCUPSEMNFS) AS QTDCUPSEMNFS,
		SUM(VALORTOT) AS VALORTOT,
		SUM(QTDCUPNFS) AS QTDCUPNFS,
		SUM(VALORTOTNFS) AS VALORTOTNFS,
		SUM(TTCUPOM) AS TTCUPOM
		
		into #CUP
		FROM 
		#CUPOM
		
		GROUP BY 
		DATAS

if object_id('TempDB.dbo.#DEVGRU') is not null
   drop table #DEVGRU

SELECT 
@UTEL AS UTEL,
ISNULL(ISNULL(ISNULL(A.NFSDATEMI,B.DATAS),C.NFEDATEFE),E.NFSDATEMI) as NFSDATEMI,
ISNULL(ISNULL(ISNULL(A.DIA,B.DIA),C.DIA),E.DIA) AS DIA,
ISNULL(A.QTDNOT,0) AS QTDNOT,
ISNULL(A.NFSTOTITEST,0) AS NFSVAL,
ISNULL(QTDCUPNFS,0) AS QTDNOTCUP,
ISNULL(VALORTOTNFS,0) AS NFSVALCUP,
ISNULL(QTDCUPSEMNFS,0) AS QTDCUPSEMNFS,
ISNULL(VALORTOT,0) AS CUPNAONFS,
ISNULL(TTCUPOM,0) AS CUPVAL ,
ISNULL(A.NFSTOTITEST,0) + ISNULL(TTCUPOM,0) AS VALBRU,
ISNULL(C.VALDEV,0) AS VALDEV,
ISNULL(C.QTDNOTDEV,0) AS QTDNOTDEV,
ISNULL(QTDCUPNFS,0) + ISNULL(A.QTDNOT,0) as QTDNOTTOT,
ISNULL(A.NFSTOTITEST,0) + ISNULL(TTCUPOM,0) - ISNULL(C.VALDEV,0) AS VALLIQ,
ISNULL(A.NFSTOTITEST,0) + ISNULL(TTCUPOM,0) - (ISNULL(C.VALDEV,0) - ISNULL(F.VALDEV,0)) - ISNULL(E.NFSTOTITEST,0)  AS VALLIQSEMGRU,
ISNULL(E.QTDNOT,0) AS QTDNOTGRU,
ISNULL(E.NFSTOTITEST,0) AS NFSVALGRU,
ISNULL(F.VALDEV,0) AS VALDEVGRU,
ISNULL(F.QTDNOTDEV,0) AS QTDNOTDEVGRU

FROM 
#NFS A 
FULL OUTER JOIN #CUP B ON A.NFSDATEMI = B.DATAS
FULL OUTER JOIN #DEV C ON ISNULL(A.NFSDATEMI,B.DATAS) = C.NFEDATEFE
FULL OUTER JOIN #NFSGRU E ON ISNULL(ISNULL(A.NFSDATEMI,B.DATAS),C.NFEDATEFE) = E.NFSDATEMI
FULL OUTER JOIN #DEVGRU F ON ISNULL(ISNULL(ISNULL(A.NFSDATEMI,B.DATAS),C.NFEDATEFE),E.NFSDATEMI) = F.NFEDATEFE
