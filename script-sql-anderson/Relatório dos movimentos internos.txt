-- TRANSFERNCIA ENTRE ESTOQUES VIA MOVIMENTOS INTERNOS

-- FILTROS

DECLARE @DATADE DATETIME, @DATAATE DATETIME, @OPE VARCHAR (15), @MVIDOC INT, @PROCOD CHAR(15) , @OPEDOC VARCHAR(15)
SET @DATADE = '20170711'
SET @DATAATE = '20170711'
SET @OPE = 'FURADO'
SET @MVIDOC = 0 
SET @PROCOD = '0'
SET @OPEDOC = 'EFETIVADO'


----------------------------------------------------------------------------------------------------------------------------------------------------------------------

-- TABELA DE MOVIMENTOS INTERNOS

if object_id('TempDB.dbo.#TBS037') is not null 
	drop table #TBS037

SELECT 
Identity(INT,1,1) as ordem,
CASE WHEN B.PROCOD IS NULL 
	THEN 'SEM ITEM'
	ELSE
		CASE WHEN CONVERT(DATE,A.MVIDATEFE) ='17530101' 
			THEN 'EM ABERTO'
			ELSE 'EFETIVADO'
		END 
END AS SITUACAODOC,

CASE WHEN B.PROCOD IS NULL 
	THEN 'SEM ITEM'
	ELSE
		CASE WHEN CONVERT(DATE,A.MVIDATEFE) <> '17530101'  AND B.MVIQTDPED <> MVIQTDATD 
			THEN 'FALTANTE'
			ELSE 
				CASE WHEN CONVERT(DATE,A.MVIDATEFE) ='17530101' 
					THEN 'EM ABERTO'
					ELSE 'OK'
				END
		END 
END AS SITUACAO,			
A.MVIDOC, 
A.MVIDATLAN, 
A.MVIDATEFE,
CASE WHEN A.MVICCSCOD = 0 THEN dbo.HORASTRABA(A.MVIDATLAN,A.MVIDATEFE) ELSE '' END AS DIAABER,
CASE WHEN A.MVICCSCOD <> 0 THEN dbo.HORASTRABA(A.MVIDATLAN,A.MVIDATEFE) ELSE '' END AS DIAATEN,
substring(dbo.HORASTRABA(A.MVIDATLAN,A.MVIDATEFE),1,4) * 60 + substring(dbo.HORASTRABA(A.MVIDATLAN,A.MVIDATEFE),6,2) as MINUTOS,
RTRIM(LTRIM(STR(A.CCSCOD))) + ' - ' + ISNULL((SELECT CCSNOM FROM TBS036 C (NOLOCK) WHERE C.CCSCOD = A.CCSCOD),'') AS USUCUS,
-- RTRIM(LTRIM(STR(A.TMVCOD))) + ' - ' + (SELECT TMVDES FROM TBS033 D (NOLOCK) WHERE A.TMVCOD = D.TMVCOD) AS OPERAC ,
RTRIM(LTRIM(STR(A.MVILOCORI))) + ' - ' + (SELECT LESDES FROM TBS034 D (NOLOCK) WHERE A.MVILOCORI = D.LESCOD) AS LOCORI ,
RTRIM(LTRIM(STR(A.MVILOCDES))) + ' - ' + (SELECT LESDES FROM TBS034 D (NOLOCK) WHERE A.MVILOCDES = D.LESCOD) AS LOCDES,
CASE WHEN A.MVICCSCOD = 0 
	THEN '' 
	ELSE RTRIM(LTRIM(STR(A.MVICCSCOD))) + ' - ' + RTRIM(LTRIM(A.MVICCSNOM))
END AS USUEFE,
A.MVIOBS,
isnull(B.MVIITE,0) AS MVIITE,
isnull(B.PROCOD,'') AS PROCOD,
isnull(B.MVIPRODES,'') AS MVIPRODES,
isnull(B.MVIQTDPED,'') AS MVIQTDPED,
isnull(B.MVIPROUNI,'') AS MVIPROUNI,
isnull(B.MVIQTDEMB,'') AS MVIQTDEMB,
isnull(B.MVIQTDATD,'') AS MVIQTDATD

INTO #TBS037
FROM TBS037 A (NOLOCK)
LEFT JOIN TBS0371 B (NOLOCK) ON A.MVIDOC = B.MVIDOC AND A.MVIEMPCOD = B.MVIEMPCOD

WHERE 
CONVERT(DATE,MVIDATLAN) BETWEEN @DATADE AND @DATAATE AND 
A.TMVCOD = 501 AND 
A.MVIDOC = CASE WHEN @MVIDOC = 0 THEN A.MVIDOC ELSE @MVIDOC END AND 
ISNULL(B.PROCOD,'T') = CASE WHEN @PROCOD = '0' THEN ISNULL(B.PROCOD,'T') ELSE @PROCOD END /*AND
CASE WHEN B.PROCOD IS NULL 
	THEN 'SEM ITEM'
	ELSE
		CASE WHEN CONVERT(DATE,A.MVIDATEFE) <> '17530101'  AND B.MVIQTDPED <> MVIQTDATD 
			THEN 'FALTANTE'
			ELSE 
				CASE WHEN CONVERT(DATE,A.MVIDATEFE) ='17530101' 
					THEN 'EM ABERTO'
					ELSE 'OK'
				END
		END 
END IN (@OPE) AND
CASE WHEN B.PROCOD IS NULL 
	THEN 'SEM ITEM'
	ELSE
		CASE WHEN CONVERT(DATE,A.MVIDATEFE) ='17530101' 
			THEN 'EM ABERTO'
			ELSE 'EFETIVADO'
		END 
END IN (@OPEDOC)
*/

ORDER BY 
A.MVIDOC, B.MVIITE
----------------------------------------------------------------------------------------------------------------------------------------------------------------------

-- TABELA FINAL 



SELECT 
CASE WHEN  MVIDOC in (SELECT DISTINCT B.MVIDOC FROM #TBS037 B WHERE SITUACAO = 'FALTANTE') AND
		   MVIITE = (SELECT MIN(MVIITE) FROM #TBS037 B WHERE SITUACAO = 'FALTANTE' AND A.MVIDOC = B.MVIDOC)		   
	THEN 1
	ELSE 0
END AS DOCFURADO,
CASE WHEN  MVIDOC in (SELECT DISTINCT B.MVIDOC FROM #TBS037 B WHERE SITUACAO = 'EM ABERTO') AND
		   MVIITE = (SELECT MIN(MVIITE) FROM #TBS037 B WHERE SITUACAO = 'EM ABERTO' AND A.MVIDOC = B.MVIDOC)		   
	THEN 1
	ELSE 0
END AS DOCEMABERTO,
CASE WHEN  MVIDOC in (SELECT DISTINCT B.MVIDOC FROM #TBS037 B WHERE SITUACAO = 'SEM ITEM') AND
		   MVIITE = (SELECT MIN(MVIITE) FROM #TBS037 B WHERE SITUACAO = 'SEM ITEM' AND A.MVIDOC = B.MVIDOC)		   
	THEN 1
	ELSE 0
END AS DOCSEMITEM,

CASE WHEN SITUACAO = 'FALTANTE'	   
	THEN 1
	ELSE 0
END AS FURADO,
CASE WHEN SITUACAO = 'EM ABERTO'	   
	THEN 1
	ELSE 0
END AS EMABERTO,
CASE WHEN SITUACAO = 'SEM ITEM' OR (SELECT B.MVIDOC FROM #TBS037 B WHERE B.ordem = A.ordem -1) = MVIDOC
	THEN 0
	ELSE MINUTOS
END AS MINU,
* 
FROM #TBS037 A 
-- WHERE SITUACAO IN (@OPE) AND SITUACAODOC IN (@OPEDOC) -- NÃO VAI NO RELATORIO
