SET @anoAte ='2017'

SET @PROCODATE =''

cdprod ,

-- TABLELA VIRTUAL TBS010
if object_id('TempDB.dbo.#TBS010') is not null
   drop table #TBS010
SELECT 
PROLOCFIS,
PROSTATUS,
case when len(MARCOD) = 4 
	then rtrim(MARCOD) + ' - ' + rtrim(MARNOM) 
	else right(('00' + ltrim(str(MARCOD))),3) + ' - ' + rtrim(MARNOM)
end collate Latin1_General_BIN as MARCOD ,
PROCOD,
PRODES,
CASE WHEN PROUM1QTD > 1 
	THEN RTRIM(LTRIM(PROUM1)) + ' ' + RTRIM(CONVERT(DECIMAL, PROUM1QTD, 0)) + '' + 
		CASE WHEN PROUMV = '' 
			THEN PROUM1 
			ELSE PROUMV 
		END 
	ELSE RTRIM(CONVERT(DECIMAL, PROUM1QTD, 0)) + '' + 
		CASE WHEN PROUMV = '' 
			THEN PROUM1 
			ELSE PROUMV 
		END 
END AS PROUM1

INTO #TBS010 
FROM TBS010 (NOLOCK)
WHERE 
PROCOD BETWEEN @PROCODDE AND CASE WHEN @PROCODATE ='' THEN 'Z' ELSE @PROCODATE END AND
--PROCOD IN (SELECT DISTINCT COD FROM #MSL002) AND 
MARCOD = CASE WHEN @MARCOD = 0 THEN MARCOD ELSE @MARCOD END AND 
MARNOM LIKE (RTRIM(LTRIM(upper(@MARNOM)))+'%') AND 
PRODES LIKE (RTRIM(LTRIM(upper(@PRODES)))+'%')

-- TABELA DE SALDO

if object_id('TempDB.dbo.#TBS032') is not null
   drop table #TBS032
SELECT 
PROCOD,
SUM(ESTQTDCMP) AS ESTQTDCMP,
ISNULL((SELECT SUM(ESTQTDATU-ESTQTDRES) FROM TBS032 B (nolock) WHERE ESTLOC =1 AND A.PROCOD = B.PROCOD GROUP BY PROCOD),0) AS EST,
ISNULL((SELECT SUM(ESTQTDATU-ESTQTDRES) FROM TBS032 B (nolock) WHERE ESTLOC =2 AND A.PROCOD = B.PROCOD GROUP BY PROCOD),0) AS LOJA	

INTO #TBS032
FROM TBS032 A (nolock) 

WHERE 
ESTLOC IN (1,2) and ( 
ESTQTDATU - ESTQTDRES <> 0 OR  ESTQTDCMP <> 0)

GROUP BY PROCOD

	drop table #NFS

1 AS EMPCOD, 
NF.PROCOD AS PROCOD,
NF.NFSNUM AS NFSNUM,
NF.QTD AS QTD

INTO #NFS

WHERE

D.ENFCNPJCPF = ''05118717000156'' AND

GROUP BY 
PROCOD,NFSNUM') AS NF

NF.PROCOD AS PROCOD,
NF.NFSNUM AS NFSNUM,

NFSNUM,

PROCOD,NFSNUM') AS NF

-- NOTAS DE ENTRADA NA BEST BAG, PERIODO DE 30 DIAS ATÉ HOJE

IF OBJECT_ID ('TempDB.dbo.#NFE') is not null
	drop table #NFE

SELECT 
CASE WHEN NFENOM LIKE('%PAPELYNA%')
	THEN 1 
	ELSE 2
END AS EMPCOD,
A.NFENUM AS NFENUM, 
PROCOD AS PROCOD

INTO #NFE	
FROM TBS0591 A (NOLOCK)
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND 
A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND  A.SEREMPCOD = B.SEREMPCOD
LEFT JOIN TBS080 D (NOLOCK) ON A.NFENUM = D.ENFNUM AND A.NFECOD = D.ENFCODDES

WHERE 
((D.ENFSIT = 6 AND D.ENFFINEMI = 4 AND D.ENFTIPDOC = 0) OR (NFENOSFOR <> 'S' AND B.NFECAN <> 'S' AND NFEDATEFE <> '' )) AND
B.NFEDATEFE BETWEEN GETDATE() - CONVERT(DATETIME, 30) AND GETDATE() AND 
(NFENOM LIKE('%PAPELYNA%') OR  NFENOM LIKE('%MISASPEL%'))

----------------------------------------------------------------------------------------------------------------------------------------------------------------------

-- TABELA DE PRODUTOS EM TRANSITO PARA BEST BAG, DA PAPELYNA OU DA MISASPEL

IF OBJECT_ID ('TempDB.dbo.#TRA') is not null
	drop table #TRA
	
SELECT 
A.PROCOD ,
SUM(QTD) AS QTD

INTO #TRA
FROM 
#NFS A 
LEFT JOIN #NFE B ON A.PROCOD COLLATE DATABASE_DEFAULT = B.PROCOD AND A.NFSNUM = B.NFENUM AND A.EMPCOD = B.EMPCOD

WHERE 
B.NFENUM IS NULL

GROUP BY 
A.PROCOD

IF OBJECT_ID ('TempDB.dbo.#REPOR') is not null
	drop table #REPOR

SELECT 
A.PROCOD,
PROLOCFIS,
PROSTATUS,
MARCOD ,
PRODES,
PROUM1,
ISNULL(quant,0) AS TOTVEN,
ISNULL(ROUND(quant / @DIASUTEIS,2),0) as media,
ISNULL(CEILING((quant / @DIASUTEIS) * @PERIODO),0)  AS SEG_LOJA,
ISNULL(C.ESTQTDCMP,0) AS CMPND, 	-- COMPRAS BEST BAG
ISNULL(C.EST,0) AS ESTND, 		-- Disponivel EST BEST BAG
ISNULL(C.LOJA,0) AS LOJA, 		-- Disponivel Loja BEST BAG
ISNULL(D.ESTQTDRES,0) AS RESCD, -- Reservado PAPELYNA
ISNULL(D.EST,0) AS ESTCD,		-- Disponivel PAPELYNA
ISNULL(F.ESTQTDRES,0) AS RESTT, -- Reservado MISASPEL
ISNULL(F.EST,0) AS ESTTT,		-- Disponivel MISASPEL
ISNULL(QTD,0) AS QTD_TRA,		-- Em transito

CASE WHEN ISNULL(C.LOJA,0) >=0 
	THEN 
		CASE WHEN ISNULL(CEILING((quant / @DIASUTEIS) * @PERIODO),0) - ISNULL(C.LOJA,0) > 0 
				THEN ISNULL(CEILING((quant / @DIASUTEIS) * @PERIODO),0) - ISNULL(C.LOJA,0)
				ELSE 0
		END
	ELSE ISNULL(CEILING((quant / @DIASUTEIS) * @PERIODO),0)
END as qtdrepor,

@DATINI AS  dataini,
@DATFIM AS datafim,
@DIASUTEIS AS diasuteis

INTO #REPOR
FROM #TBS010 A
LEFT JOIN #MSL002 B 	ON convert(numeric, A.PROCOD) = B.cdprod		-- TABELA DE QUANTIDADE VENDIDA NA LOJA
LEFT JOIN #TBS032 C 	ON A.PROCOD = C.PROCOD							-- TABELA DE SALDO DA BEST BAG
LEFT JOIN #TBS032PAP D 	ON A.PROCOD COLLATE DATABASE_DEFAULT = D.PROCOD -- TABELA DE SALDO DA PAPELYNA
LEFT JOIN #TBS032MI F 	ON A.PROCOD COLLATE DATABASE_DEFAULT = F.PROCOD -- TABELA DE SALDO DA MISASPEL
LEFT JOIN #TRA E 		ON A.PROCOD COLLATE DATABASE_DEFAULT = E.PROCOD -- TABELA DE PRODUTOS EM TRANSITO

WHERE 
ISNULL(quant,0) <> 0 OR
ISNULL(C.LOJA,0) <> 0 OR 
ISNULL(C.EST,0) <> 0 OR 
ISNULL(C.ESTQTDCMP,0) <> 0 OR 
ISNULL(D.EST,0) <> 0 OR 
ISNULL(D.ESTQTDRES,0) <> 0 OR
ISNULL(F.EST,0) <> 0 OR 
ISNULL(F.ESTQTDRES,0) <> 0 OR
ISNULL(QTD,0) <> 0 



SELECT
CASE WHEN qtdrepor <= ESTND + CMPND + RESCD + RESTT + ESTCD + QTD_TRA + CASE WHEN ESTTT > 0 THEN ESTTT ELSE 0 END AND qtdrepor > 0
	then '1.TEM PARA REPOR OU ESTÁ CHEGANDO'
	ELSE
		CASE WHEN TOTVEN = 0 AND LOJA <= 0 AND
				  ESTND + CMPND + RESCD + RESTT + ESTCD + QTD_TRA + CASE WHEN ESTTT > 0 THEN ESTTT ELSE 0 END > 0
			THEN '2.PRODUTOS SEM VENDAS, POREM NÃO TEM NA LOJA! VERIFICAR AONDE TEM'
			ELSE
				CASE WHEN qtdrepor > 0 AND 
				  (ESTND + CMPND + RESCD + RESTT + ESTCD + QTD_TRA + CASE WHEN ESTTT > 0 THEN ESTTT ELSE 0 END = 0 OR
				  ESTND + CMPND + RESCD + RESTT + ESTCD + QTD_TRA + CASE WHEN ESTTT > 0 THEN ESTTT ELSE 0 END < qtdrepor)
					THEN '3.PRODUTOS COM VENDAS MAS NÃO TEM QUANTIDADE SUFICIENTE PARA REPOR, É NECESSARIO COMPRAR?'
					ELSE 
						CASE WHEN TOTVEN = 0 AND LOJA > 0 
							THEN '4.PRODUTOS SEM VENDA, POREM COM SALDO NA LOJA'
							ELSE
								CASE WHEN qtdrepor <= LOJA AND TOTVEN > 0 
									THEN '5.PRODUTOS COM REPOSIÇÃO OK'
									ELSE ''
								END 
						END
				END 
		END
END AS SITUACAO,
*

FROM #REPOR
