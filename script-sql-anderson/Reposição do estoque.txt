if object_id('TempDB.dbo.#REPOR') is not null
   drop table #REPOR

SELECT 
RTRIM(A.PROLOCFIS) AS 'LOC', 
RTRIM(A.PROSTATUS) AS 'STA', 
RTRIM(LTRIM(A.PROCOD)) AS 'CODIGO', 
A.PRODES AS 'DESCRICAO', 
CASE WHEN PROUM1QTD > 1 
	THEN RTRIM(LTRIM(PROUM1)) + ' ' + RTRIM(CONVERT(DECIMAL, PROUM1QTD, 0)) + '' + 
		CASE WHEN PROUMV = '' 
			THEN PROUM1 
			ELSE PROUMV 
		END 
	ELSE RTRIM(CONVERT(DECIMAL, PROUM1QTD, 0)) + '' + 
		CASE WHEN PROUMV = '' 
			THEN PROUM1 
			ELSE PROUMV 
		END 
END AS UN1,
 
CASE WHEN PROUM2QTD > 0 
	THEN RTRIM(PROUM2) + ' ' + RTRIM(CONVERT(DECIMAL, PROUM2QTD, 0)) + '' + RTRIM(LTRIM(PROUM1)) 
	ELSE '' 
END AS UN2,
 
CASE WHEN PROUM3QTD > 0 
	THEN RTRIM(PROUM3) + ' ' + RTRIM(CONVERT(DECIMAL, (PROUM3QTD / PROUM2QTD), 0)) + '' + RTRIM(PROUM2) 
    ELSE '' 
END AS UN3,

CASE WHEN PROUM4QTD > 0 
	THEN RTRIM(PROUM4) + ' ' + RTRIM(CONVERT(DECIMAL, (PROUM4QTD / PROUM3QTD), 0)) + '' + RTRIM(PROUM3) 
	ELSE '' 
END AS UN4,
 
CASE WHEN len(A.MARCOD) = 4 
	THEN rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) 
	ELSE RIGHT(('00' + ltrim(str(A.MARCOD))), 3) + ' - ' + rtrim(A.MARNOM) 
END AS 'MARCA', 

A.PROUM1 AS UNMED, 
ISNULL(F.EST,0) AS DISCD, 
ISNULL(F.ESTQTDRES,0) AS RESCD, 
ISNULL(G.QTD,0) AS TRANS, 
ISNULL(B.EST1,0) AS DISEST, 
ISNULL(B.ESTQTDRES,0) AS RESEST, 
ISNULL(B.EST2,0) AS DISLOJA, 
CASE WHEN CEILING((ISNULL(D.QTD, 0) + ISNULL(quant, 0) - ISNULL(E.QTD,0)) / 								 -- (INT + GZ - DEV) / DIAS UTEIS * PERIODO
		  dbo.FCNDIASUTEIS ((select min(datini) from #MSL002), (select max(datfim) from #MSL002)) * @PERIODO) --    - EST - LOJA - TRANS - RESCD > 0 
		  - isnull(B.EST1,0) - CASE WHEN isnull(B.EST2,0) < 0 THEN 0 ELSE isnull(B.EST2,0) END - ISNULL(G.QTD,0) - ISNULL(F.ESTQTDRES,0) > 0 	
	THEN
		CASE WHEN CEILING((ISNULL(D.QTD, 0) + ISNULL(quant, 0) - ISNULL(E.QTD,0)) / 
				  dbo.FCNDIASUTEIS ((select min(datini) from #MSL002), (select max(datfim) from #MSL002)) * @PERIODO) 
				  - isnull(B.EST1,0) - CASE WHEN isnull(B.EST2,0) < 0 THEN 0 ELSE isnull(B.EST2,0) END - ISNULL(G.QTD,0) - ISNULL(F.ESTQTDRES,0) <= ISNULL(F.EST,0)
				  
			THEN CEILING((ISNULL(D.QTD, 0) + ISNULL(quant, 0) - ISNULL(E.QTD,0)) / 
				 dbo.FCNDIASUTEIS ((select min(datini) from #MSL002), (select max(datfim) from #MSL002)) * @PERIODO) 
				 - isnull(B.EST1,0) - CASE WHEN isnull(B.EST2,0) < 0 THEN 0 ELSE isnull(B.EST2,0) END - ISNULL(G.QTD,0) - ISNULL(F.ESTQTDRES,0)
			ELSE ISNULL(F.EST,0)
		END
	ELSE 0
END AS SEG_EST-- , 

-- CEILING((isnull(quant,0) - ISNULL(E.QTDCUP,0)) / dbo.FCNDIASUTEIS ((select min(datini) from #MSL002), (select max(datfim) from #MSL002)) * @PERIODO)  AS SEG_LOJA

INTO #REPOR
FROM            
#TBS010 A (NOLOCK)
LEFT JOIN #SALDO B ON A.PROCOD = B.PROCOD
LEFT JOIN #MSL002 C ON CONVERT(INT,A.PROCOD) = C.cdprod
LEFT JOIN #CORP D ON A.PROCOD = D.PROCOD
LEFT JOIN #DEV E ON A.PROCOD = E.PROCOD
LEFT JOIN #TBS032CD F ON F.PROCOD collate Latin1_General_BIN = A.PROCOD
LEFT JOIN #TRA G ON A.PROCOD collate Latin1_General_BIN = G.PROCOD

							   
WHERE 
(D.QTD <> 0 OR C.quant <> 0 OR G.QTD <> 0 OR F.EST <> 0 OR F.ESTQTDRES <> 0 OR B.EST1 <> 0 OR B.EST2 <> 0 OR  B.ESTQTDRES <> 0) and 
A.PROCOD IS NOT NULL AND
A.PRODES LIKE(RTRIM(UPPER(@PRODES))+'%') 

ORDER BY A.PROCOD


SELECT *, 
CASE WHEN SEG_EST > 0 THEN 1 ELSE 0 END AS COMREPOR
 FROM #REPOR WHERE SEG_EST > CASE WHEN @REPOR = 1 THEN 0 ELSE -1 END
