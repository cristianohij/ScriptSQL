
-- SOMENTE REPOSIÇÃO DE LOJA

-- VARIAVEIS
DECLARE @anoDe varchar (4), @mesDe varchar (2), @anoAte varchar (4), @mesAte varchar (2), 
@PROCODDE CHAR(15) , @PROCODATE CHAR(15) ,@MARCOD INT , @MARNOM CHAR(60), @PRODES CHAR(60) ,@PERIODO INT , @comando varchar(1000)

SET @anoDe ='2017'
SET @mesDe= '04'
SET @anoAte ='2017'
SET @mesAte= '06'
SET @PROCODDE =''
SET @PROCODATE =''
SET @MARCOD = 0
SET @MARNOM = ''
SET @PRODES =''
SET @PERIODO = 6

DECLARE @DIASUTEIS INT , @DATINI DATE , @DATFIM DATE




SET @comando = 'EXECUTE(''
select 
cdprod , 
sum(quant) as quant,
MIN(data) AS datini,
max(data) as datfim

FROM movcaixa
where
CAST(data AS CHAR(7)) between "'+@anoDe+'-'+@mesDe+'" and "'+@anoAte+'-'+@mesAte+'" and 
status = 01 and 
cancelado <> "S"
group by cdprod '') at MYSQLGZ' 


if object_id('TempDB.dbo.#MSL002') is not null
   drop table #MSL002
   
create table #MSL002 (cdprod INT, quant decimal(10,4), datini date, datfim date)
INSERT INTO #MSL002
exec(@comando)



SET @DIASUTEIS = (SELECT dbo.FCNDIASUTEIS ((select min(datini) from #MSL002), (select max(datfim) from #MSL002)))
SET @DATINI = (select min(datini) from #MSL002)
SET @DATFIM = (select max(datfim) from #MSL002)
----------------------------------------------------------------------------------------------------------------------------------------------

-- TABLELA VIRTUAL TBS010
if object_id('TempDB.dbo.#TBS010') is not null
   drop table #TBS010
SELECT 
PROLOCFIS,
PROSTATUS,
case when len(MARCOD) = 4 
	then rtrim(MARCOD) + ' - ' + rtrim(MARNOM) 
	else right(('00' + ltrim(str(MARCOD))),3) + ' - ' + rtrim(MARNOM)
end collate Latin1_General_BIN as MARCOD ,
PROCOD,
PRODES,
CASE WHEN PROUM1QTD > 1 
	THEN RTRIM(LTRIM(PROUM1)) + ' ' + RTRIM(CONVERT(DECIMAL, PROUM1QTD, 0)) + '' + 
		CASE WHEN PROUMV = '' 
			THEN PROUM1 
			ELSE PROUMV 
		END 
	ELSE RTRIM(CONVERT(DECIMAL, PROUM1QTD, 0)) + '' + 
		CASE WHEN PROUMV = '' 
			THEN PROUM1 
			ELSE PROUMV 
		END 
END AS PROUM1

INTO #TBS010 
FROM TBS010 (NOLOCK)
WHERE 
PROCOD BETWEEN @PROCODDE AND CASE WHEN @PROCODATE ='' THEN 'Z' ELSE @PROCODATE END AND
--PROCOD IN (SELECT DISTINCT COD FROM #MSL002) AND 
MARCOD = CASE WHEN @MARCOD = 0 THEN MARCOD ELSE @MARCOD END AND 
MARNOM LIKE (RTRIM(LTRIM(upper(@MARNOM)))+'%') AND 
PRODES LIKE (RTRIM(LTRIM(upper(@PRODES)))+'%')

-------------------------------------------------------------------------------------------------------------------------------------------------
	
-- tabela de saldo 

if object_id('TempDB.dbo.#TBS032') is not null
   drop table #TBS032
SELECT 
PROCOD,
SUM(ESTQTDCMP) AS ESTQTDCMP,
ISNULL((SELECT SUM(ESTQTDATU-ESTQTDRES) FROM TBS032 B (nolock) WHERE ESTLOC =1 AND A.PROCOD = B.PROCOD GROUP BY PROCOD),0) AS EST,
ISNULL((SELECT SUM(ESTQTDATU-ESTQTDRES) FROM TBS032 B (nolock) WHERE ESTLOC =2 AND A.PROCOD = B.PROCOD GROUP BY PROCOD),0) AS LOJA	

INTO #TBS032
FROM TBS032 A (nolock) 

WHERE 
ESTLOC IN (1,2) and ( 
ESTQTDATU <> 0 OR  ESTQTDCMP <> 0)

GROUP BY PROCOD

----------------------------------------------------------------------------------------------------------------------------------------------------------------------

-- TABELA DE MOVIMENTOS INTERNOS ITENS QUE ESTÃO RESERVADOS PARA LOJA

-- PESQUISAR SOMENTE DO 1 PARA O ESTOQUE 2 

if object_id('TempDB.dbo.#TBS037') is not null 
	drop table #TBS037

SELECT 
B.PROCOD AS PROCOD,
SUM(B.MVIQTDPED*B.MVIQTDEMB) AS QTDRES

INTO #TBS037
FROM TBS037 A (NOLOCK)
LEFT JOIN TBS0371 B (NOLOCK) ON A.MVIDOC = B.MVIDOC AND A.MVIEMPCOD = B.MVIEMPCOD

WHERE 
CONVERT(DATE,MVIDATLAN) BETWEEN GETDATE() - CONVERT(DATETIME, 30) AND GETDATE() AND 
A.TMVCOD = 501 AND 
CONVERT(DATE,A.MVIDATEFE) ='17530101' AND 
B.PROCOD IS NOT NULL AND
A.MVILOCORI = 1 AND 
A.MVILOCDES = 2

GROUP BY 
B.PROCOD

-- NOTAS DE ENTRADA NA TANBY, PERIODO DE 30 DIAS ATÉ HOJE

	drop table #NFE

SELECT 
CASE WHEN B.NFECOD = 669
	THEN 1 
	ELSE 2
END AS EMPCOD,
A.NFENUM AS NFENUM, 
PROCOD AS PROCOD

INTO #NFE	
FROM TBS0591 A (NOLOCK)

((D.ENFSIT = 6 AND D.ENFFINEMI = 4 AND D.ENFTIPDOC = 0 ) OR (NFENOSFOR <> 'S' AND B.NFECAN <> 'S' AND NFEDATEFE <> '' )) AND
B.NFEDATEFE BETWEEN GETDATE() - CONVERT(DATETIME, 30) AND GETDATE() AND 
NFENOM LIKE('%TANBY%') AND
B.NFECOD NOT IN (590,411)

----------------------------------------------------------------------------------------------------------------------------------------------------------------------

-- TABELA DE PRODUTOS EM TRANSITO PARA BEST BAG, DA PAPELYNA OU DA MISASPEL

IF OBJECT_ID ('TempDB.dbo.#TRA') is not null
	drop table #TRA
	
SELECT 
A.PROCOD ,
SUM(QTD) AS QTD

INTO #TRA
FROM 
#NFS A 
LEFT JOIN #NFE B ON A.PROCOD COLLATE DATABASE_DEFAULT = B.PROCOD AND A.NFSNUM = B.NFENUM AND A.EMPCOD = B.EMPCOD

WHERE 
B.NFENUM IS NULL

GROUP BY 
A.PROCOD

-- TABELA FINAL DE REPOSIÇÃO DA LOJA

IF OBJECT_ID ('TempDB.dbo.#REPOR') is not null
	drop table #REPOR

SELECT 
A.PROCOD,
PROLOCFIS,
PROSTATUS,
MARCOD ,
PRODES,
PROUM1,
ISNULL(quant,0) AS TOTVEN,
ISNULL(ROUND(quant / @DIASUTEIS,2),0) as media,
ISNULL(CEILING((quant / @DIASUTEIS) * @PERIODO),0)  AS SEG_LOJA,
ISNULL(C.ESTQTDCMP,0) AS CMPND, 	-- COMPRAS ND
ISNULL(C.EST,0) AS ESTND, 		-- Disponivel EST ND
ISNULL(C.LOJA,0) AS LOJA, 		-- Disponivel Loja ND
ISNULL(QTDRES,0) AS QTDRES, 	-- RESERVADO NO ESTOQUE PARA LOJA
ISNULL(D.ESTQTDRES,0) AS RESCD, -- Reservado CD
ISNULL(D.EST,0) AS ESTCD,		-- Disponivel CD
ISNULL(F.ESTQTDRES,0) AS RESTT, -- Reservado TAUBATÉ
ISNULL(F.EST,0) AS ESTTT,		-- Disponivel EST TAUBATÉ
ISNULL(F.LOJA,0) AS LOJATT, 	-- Disponivel Loja ND
ISNULL(QTD,0) AS QTD_TRA,		-- Em transito

CASE WHEN ISNULL(C.LOJA,0) >=0 
	THEN 
		CASE WHEN ISNULL(CEILING((quant / @DIASUTEIS) * @PERIODO),0) - ISNULL(C.LOJA,0) - ISNULL(QTDRES,0) > 0 
				THEN ISNULL(CEILING((quant / @DIASUTEIS) * @PERIODO),0) - ISNULL(C.LOJA,0) - ISNULL(QTDRES,0)
				ELSE 0
		END
	ELSE 
		CASE WHEN ISNULL(CEILING((quant / @DIASUTEIS) * @PERIODO),0) - ISNULL(QTDRES,0) > 0 
			THEN ISNULL(CEILING((quant / @DIASUTEIS) * @PERIODO),0) - ISNULL(QTDRES,0)
			ELSE 0
		END 
END as qtdrepor,

@DATINI AS  dataini,
@DATFIM AS datafim,
@DIASUTEIS AS diasuteis

INTO #REPOR 
FROM #TBS010 A
LEFT JOIN #MSL002 B 	ON convert(numeric, A.PROCOD) = B.cdprod		-- TABELA DE QUANTIDADE VENDIDA NA LOJA
LEFT JOIN #TBS032 C 	ON A.PROCOD = C.PROCOD							-- TABELA DE SALDO DA ND
LEFT JOIN #TBS032CD D 	ON A.PROCOD COLLATE DATABASE_DEFAULT = D.PROCOD -- TABELA DE SALDO DO CD
LEFT JOIN #TBS032TT F 	ON A.PROCOD COLLATE DATABASE_DEFAULT = F.PROCOD -- TABELA DE SALDO DE TAUBATÉ
LEFT JOIN #TRA E 		ON A.PROCOD COLLATE DATABASE_DEFAULT = E.PROCOD -- TABELA DE PRODUTOS EM TRANSITO
LEFT JOIN #TBS037 G		ON A.PROCOD COLLATE DATABASE_DEFAULT = G.PROCOD -- TABELA DE PRODUTOS EM RESERVA NO ESTOQUE PARA LOJA (MOVIMENTO INTERNOS)

WHERE 
ISNULL(quant,0) <> 0 OR
ISNULL(C.LOJA,0) <> 0 OR 
ISNULL(C.EST,0) <> 0 OR 
ISNULL(C.ESTQTDCMP,0) <> 0 OR 
ISNULL(D.EST,0) <> 0 OR 
ISNULL(D.ESTQTDRES,0) <> 0 OR
ISNULL(F.EST,0) <> 0 OR 
ISNULL(F.ESTQTDRES,0) <> 0 OR
ISNULL(QTD,0) <> 0 OR 
ISNULL(QTDRES,0) <> 0





SELECT
CASE WHEN qtdrepor <= ESTND + CMPND + RESCD + RESTT + ESTCD + QTD_TRA + CASE WHEN LOJATT >0 THEN LOJATT ELSE 0 END + CASE WHEN ESTTT > 0 THEN ESTTT ELSE 0 END AND 
qtdrepor > 0
	then '1.TEM PARA REPOR OU ESTÁ CHEGANDO'
	ELSE
		CASE WHEN TOTVEN = 0 AND LOJA <= 0 AND QTDRES <= 0 AND 
				  ESTND + CMPND + RESCD + RESTT + ESTCD + QTD_TRA + CASE WHEN LOJATT >0 THEN LOJATT ELSE 0 END + CASE WHEN ESTTT > 0 THEN ESTTT ELSE 0 END > 0
			THEN '2.PRODUTOS SEM VENDAS, POREM NÃO TEM NA LOJA! VERIFICAR AONDE TEM'
			ELSE
				CASE WHEN qtdrepor > 0 AND 
				  (ESTND + CMPND + RESCD + RESTT + ESTCD + QTD_TRA + CASE WHEN LOJATT >0 THEN LOJATT ELSE 0 END + CASE WHEN ESTTT > 0 THEN ESTTT ELSE 0 END = 0 OR
				   ESTND + CMPND + RESCD + RESTT + ESTCD + QTD_TRA + CASE WHEN LOJATT >0 THEN LOJATT ELSE 0 END + CASE WHEN ESTTT > 0 THEN ESTTT ELSE 0 END < qtdrepor)
					THEN '3.PRODUTOS COM VENDAS MAS NÃO TEM QUANTIDADE SUFICIENTE PARA REPOR, É NECESSARIO COMPRAR?'
					ELSE 
						CASE WHEN TOTVEN = 0 AND LOJA > 0 
							THEN '4.PRODUTOS SEM VENDA, POREM COM SALDO NA LOJA'
							ELSE
								CASE WHEN qtdrepor <= LOJA AND TOTVEN > 0 AND QTDRES = 0
									THEN '6.PRODUTOS COM REPOSIÇÃO OK'
									ELSE '5.PRODUTOS OK, PORÉM EM SEPARAÇÃO NO ESTOQUE'
								END 
						END
				END 
		END
END AS SITUACAO,
*

FROM #REPOR
