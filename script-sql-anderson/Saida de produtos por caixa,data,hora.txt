SELECT A.M2_PROCOD AS COD, 
RTRIM(LTRIM(PRODES)) AS DESCRIÇÃO,

REPLACE(STR(ISNULL(SUM(M2_QTD),0),10,2),'.',',') AS QTD,
REPLACE(STR(SUM(M2_VALTOT)-SUM(M2_ABT),10,3),'.',',') AS VALOR,
M2_CXA AS CAIXA,
CONVERT(VARCHAR(10),M2_DAT,3)AS 'DATA',
M2_HOR AS HORA,
M2_CONCUPFIS as CONTADORCUPOM, 
M2_NUMDOC AS CUPOM

FROM MSL002 A (nolock) INNER JOIN TBS010 B ON A.M2_PROCOD = B.PROCOD 
WHERE M2_DAT between '20160601' and GETDATE() 
and M2_TIPREG = '01' 
AND M2_REGCAN = 'F'  
-- AND M2_NUMDOC = '83833'
AND M2_PROCOD = '1173019'

 GROUP BY M2_PROCOD,M2_CXA,M2_HOR,M2_DAT,M2_CONCUPFIS,PRODES,M2_NUMDOC
 
 ORDER BY M2_DAT  , M2_HOR
