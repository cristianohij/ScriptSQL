SELECT PROCOD AS COD ,
ISNULL((SELECT saldo FROM nd.SIBD.dbo.TT B (NOLOCK) WHERE A.PROCOD collate Latin1_General_BIN = B.codigo ),0) AS EST1,
ISNULL((SELECT saldo FROM nd.SIBD.dbo.TTLOJA B (NOLOCK) WHERE A.PROCOD collate Latin1_General_BIN = B.codigo ),0) AS EST2,
ISNULL((SELECT saldo FROM nd.SIBD.dbo.TTLOJA B (NOLOCK) WHERE A.PROCOD collate Latin1_General_BIN = B.codigo ),0) + 
ISNULL((SELECT saldo FROM nd.SIBD.dbo.TT B (NOLOCK) WHERE A.PROCOD collate Latin1_General_BIN = B.codigo ),0) AS SALDO

INTO SALDOEST2015-- USANDO ESSE
FROM TBS010 A (NOLOCK)

where 
ISNULL((SELECT saldo FROM nd.SIBD.dbo.TTLOJA B (NOLOCK) WHERE A.PROCOD collate Latin1_General_BIN = B.codigo ),0) + 
ISNULL((SELECT saldo FROM nd.SIBD.dbo.TT B (NOLOCK) WHERE A.PROCOD collate Latin1_General_BIN = B.codigo ),0) <> 0
