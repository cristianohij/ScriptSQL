INTO #TES 
FROM TBS051 (NOLOCK) 
WHERE LMEDATHOR < '20151220' --AND LMEDATHOR < '20150102'
AND LMEINFALT = 'E'  -- SIGNIFICA ESTOQUE
ORDER BY PROCOD, LMEREG DESC, LMEDATHOR DESC
