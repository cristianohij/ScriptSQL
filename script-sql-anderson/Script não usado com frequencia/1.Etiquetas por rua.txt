DECLARE @OPCAO INT, @Loces char(8), @Locger char(8), @saldo char(1)

IF @OPCAO = 1 -- UMA LOCALIZAÇÃO ESPECIFICA

SELECT  rank() OVER (ORDER BY PROLOCFIS,A.PROCOD) AS [RANK],
A.PROSTATUS AS STS, 
RTRIM(LTRIM(A.PROCOD)) AS CÓDIGO, 
RTRIM(LTRIM(A.PRODES)) AS DESCRIÇÃO, 
RTRIM(LTRIM(MARNOM)) AS MARCA, 
RTRIM(LTRIM(PROLOCFIS)) AS LOC, 
RTRIM(LTRIM(PROLOCFIS2)) AS LOC2,
'*'+RTRIM(A.PROCOD)+'*'  AS CODBAR1,
CASE WHEN PROUM2QTD > 0 
		THEN '*'+RTRIM(LTRIM(A.PROCOD))+'-2*' ELSE '' END AS  CODBAR2,
CASE WHEN PROUM1QTD = 1 THEN RTRIM(PROUM1) ELSE
	CASE WHEN PROUM1QTD > 1 THEN rtrim(PROUM1) + ' C/ ' + rtrim(CAST(PROUM1QTD AS DECIMAL(10,0))) +' '+ RTRIM(PROUMV) ELSE '' END END  as UN1,
CASE WHEN PROUM2QTD > 0 THEN rtrim(PROUM2) + ' C/ ' + rtrim(CAST(PROUM2QTD AS DECIMAL(10,0)))+' '+ RTRIM(PROUM1) ELSE '' END  as UN2

FROM  TBS010 A (NOLOCK)
LEFT JOIN TBS032 B (NOLOCK) ON A.PROCOD = B.PROCOD

WHERE 
PROLOCFIS IN(@Loces) AND
B.ESTLOC = 1 AND
B.ESTQTDATU > CASE WHEN @saldo = 'S' THEN 0 ELSE -999999 END

ORDER BY [RANK]
