SELECT
B.PROSTATUS,
PDPCOD AS CÓDIGO,
A.PRODES AS DESCRIÇÃO,
PROUM1 AS UN1,
A.PDPPRECOR1,
A.PDPREDCOR2,
A.PDPPRELOJ1,
A.PDPPRELOJ2,
SUM(ESTQTDATU - ESTQTDRES) AS DIS,
SUM(ESTQTDCMP) AS CMP

FROM TBS015 A (NOLOCK)
LEFT JOIN TBS010 B (NOLOCK) ON A.PDPCOD = B.PROCOD
LEFT JOIN TBS032 C (NOLOCK) ON A.PDPCOD = C.PROCOD

WHERE 
B.PRODES LIKE ('%LAPIS%P%') AND
ESTLOC IN (1,2)

GROUP BY 
B.PROSTATUS,
PDPCOD ,
A.PRODES ,
PROUM1,
A.PDPPRECOR1,
A.PDPREDCOR2,
A.PDPPRELOJ1,
A.PDPPRELOJ2

ORDER BY 
A.PDPPRELOJ2
