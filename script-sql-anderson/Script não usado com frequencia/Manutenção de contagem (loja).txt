DECLARE @Loc CHAR(17), @Sta char(1), @Marnom CHAR(30),@somenteComSaldo CHAR (1), @Cod CHAR(15), @Descr CHAR(60), @Opcao INT
SET @Loc = ''
set @Sta ='T'
SET @Marnom = ''
SET @somenteComSaldo ='N'
SET @Cod = ''
SET @Descr = ''
SET @Opcao = 1


IF @Opcao = 1 -- TODAS AS LOCALIZAÇOES 
SELECT 
GETDATE () AS 'DATA / HORA',
A.PROSTATUS AS 'ST', 
RTRIM(LTRIM(A.PROCOD)) AS 'COD', 
RTRIM(A.PRODES) AS DESCRIÇÃO, 
CASE WHEN len(A.MARCOD) = 4 
		THEN rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) 
		ELSE RIGHT(('00' + ltrim(str(A.MARCOD))), 3) + ' - ' + rtrim(A.MARNOM) END AS 'MARCA', 
REPLACE(STR(ESTQTDATU,10,4),'.',',') AS 'QTD ATUAL',
CASE WHEN PROUM1QTD > 1 THEN RTRIM(LTRIM(PROUM1))+' '+ 
	RTRIM(CONVERT(DECIMAL,PROUM1QTD,0))+ '' + CASE WHEN PROUMV = '' THEN PROUM1 ELSE PROUMV END ELSE 
	RTRIM(CONVERT(DECIMAL,PROUM1QTD,0))+ '' + CASE WHEN PROUMV = '' THEN PROUM1 ELSE PROUMV END END AS 'UN1',
'' AS 'QTD'

FROM TBS010 AS A (nolock) INNER JOIN
                 TBS032 AS B (nolock) ON A.PROCOD = B.PROCOD
				 
WHERE (A.PROLOCFIS LIKE(CASE WHEN @Loc = '' THEN '% %' ELSE RTRIM(UPPER(@Loc)) + '%'END) ) AND (B.ESTLOC = '2') 
AND A.PROSTATUS IN(CASE WHEN @Sta = 'T' THEN A.PROSTATUS ELSE UPPER(@Sta) END )
AND A.MARNOM LIKE (CASE WHEN @Marnom = '' THEN '% %' ELSE RTRIM(UPPER(@Marnom))+ '%' END ) 
AND (B.ESTQTDATU-B.ESTQTDRES > case when @somenteComSaldo='N' then -999999 else 0 end )
AND A.PROCOD IN ('1640054','2540001','1179933','6660029','17230003','1070029','8478633','6482073','1179932','1080067','1179903','8429995','1172995','1640089',
'1170172','7900732','7900782','1641239','1173019','9750048','7810380','9040001','10030002','6530150','6597197','1640038','0650001','1172922','6522923','2000016') 
AND A.PRODES LIKE (CASE WHEN @Descr = '' THEN '% %' ELSE RTRIM(UPPER(@Descr)) + '%' END)

ORDER BY A.PROCOD
