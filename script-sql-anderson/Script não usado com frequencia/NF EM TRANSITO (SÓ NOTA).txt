DECLARE @Data_de DATETIME,@Data_ate DATETIME, @Opcao int
SET @Data_de = '20160201'
SET @Data_ate= '20160225'
SET @Opcao = 5	




IF @Opcao = 1 -- MISASPEL PARA ND

SELECT 
NFS.NFS AS NFTRANSIÇÃO, 
ISNULL(NFE.NFE, 0) AS NFENTR, 
NFS.DATAS COLLATE Latin1_General_CI_AS AS DATA_SAÍDA, 
RTRIM(NFS.OBS_NF) AS OBS_NOTA, 
RTRIM(NFS.NUMSERIE) AS NUM_SERIE, 
RTRIM(NFS.CHAVE) AS CHAVE_ACESSO

FROM  
				  (SELECT A.NFSNUM AS NFS, 
				   CONVERT(NVARCHAR, B.NFSDATEMI, 103) AS DATAS, 
				   B.NFSOBS AS OBS_NF, 
                   D.ENFCNPJCPF AS CNPJ, D.ENFSIT AS SITUAÇÃO, 
				   D.SNESER AS NUMSERIE, 
				   D.ENFCHAACE AS CHAVE
                   
				   FROM  mi.SIBD.dbo.TBS0671 AS A (NOLOCK) 
				   					INNER JOIN mi.SIBD.dbo.TBS067 AS B (NOLOCK) ON A.NFSEMPCOD = B.NFSEMPCOD AND A.NFSNUM = B.NFSNUM AND 
																				  A.SNEEMPCOD = B.SNEEMPCOD AND A.SNESER = B.SNESER 
									INNER JOIN mi.SIBD.dbo.TBS080 AS D (NOLOCK) ON D.SNEEMPCOD = A.SNEEMPCOD AND D.SNESER = A.SNESER AND D.SNEEMPCOD = B.SNEEMPCOD AND 
																				  D.SNESER = B.SNESER AND B.NFSNUM = D.ENFNUM AND D.ENFNUM = A.NFSNUM
                   
				   WHERE (D.ENFCNPJCPF = '65069593000198') AND (D.ENFSIT = '6') AND --(B.NFSDATEMI BETWEEN '20160101' AND GETDATE()) 
				   (B.NFSDATEMI BETWEEN @Data_de AND @Data_ate)
                   
				   GROUP BY A.NFSNUM, B.NFSCLINOM, B.NFSOBS,
				   			B.NFSDATEMI,D.ENFCNPJCPF, D.ENFSIT, D.SNESER, D.ENFCHAACE) AS NFS
						 
		LEFT JOIN(SELECT B.NFENUM AS NFE, 
				  CONVERT(NVARCHAR, B.NFEDATENT, 103) AS DATAE, 
				  RTRIM(B.NFEOBS) AS OBS_NF,
				  CONVERT(NVARCHAR, B.NFEDATEMI, 103) AS DATAEMI
				  --dbo.NFETOTLIQ(B.NFETIP,B.NFENUM,B.NFECOD,B.SEREMPCOD,B.SERCOD) AS LIQ
                  
				  FROM TBS059 AS B (NOLOCK) LEFT JOIN TBS006 AS A (NOLOCK) ON A.FORNOM = B.NFENOM
                  
				  WHERE  (B.NFECAN = 'N') AND  NFENOM LIKE('%MISASPEL%') AND
				  		 ((A.FORCGC = '52080207000117') OR A.FORNOM IS NULL)) AS NFE ON NFE.NFE = NFS.NFS AND NFE.DATAEMI = NFS.DATAS
						  
WHERE (NFE.NFE IS NULL)
GROUP BY NFS.NFS, NFE.NFE, NFS.DATAS,NFS.OBS_NF, NFS.NUMSERIE, NFS.CHAVE
ORDER BY NFTRANSIÇÃO
