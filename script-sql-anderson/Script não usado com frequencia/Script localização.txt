select PROLOCFIS,TBS010.PROSTATUS,TBS010.PROCOD,TBS010.PRODES,TBS010.MARNOM,
replace(str(isnull(ESTQTDATU-ESTQTDRES,0),10,4),'.',','),
replace(str(isnull(ESTQTDRES,0),10,4),'.',',')
  from TBS010 inner join TBS032 on TBS010.PROCOD=TBS032.PROCOD
 where TBS010.PROLOCFIS Like('02%') and ESTLOC = '1'AND PROLOCFIS <>'' 
Order by TBS010.PROLOCFIS
