SELECT
RTRIM(LTRIM(P.PROCOD)) AS 'Cód', 
P.PROLOCFIS, 
P.PROSTATUS AS 'ST', 
P.PRODES AS 'Descrição', 
RTRIM(P.MARCOD) + ' - ' + RTRIM(P.MARNOM) AS 'Marca', P.PROUM1 AS 'Un1', 
ISNULL(TBS032.ESTQTDATU - TBS032.ESTQTDRES, 0) AS 'Disp. Estoque', 
ISNULL(TBS032.ESTQTDRES, 0) AS 'Reservado', 
' ' AS 'Físico Estoque', 
ISNULL(SaldoLoja.SDISLOJA, 0) AS 'Disponivel loja', 
' ' AS 'Fisico loja', 
ISNULL(RESER.QTD, '') AS QTD, 
ISNULL(RESER.PED, '') AS NUM_PED

FROM TBS010 AS P INNER JOIN TBS032 ON P.PROCOD = TBS032.PROCOD 
                 FULL OUTER JOIN (SELECT PRPSIT AS NUM, PROCOD AS COD, PRPQTD * PRPQTDEMB AS QTD, PRPNUM AS PED
                               FROM TBS058
                               WHERE (PRPSIT = 'R') AND (PRPQTDCONF = 0)
                               GROUP BY PROCOD, PRPSIT, PRPQTD, PRPQTDEMB, PRPNUM) AS RESER ON RESER.COD = P.PROCOD 
                
                 LEFT JOIN (SELECT A.PROCOD AS PRO, SUM(A.ESTQTDATU - A.ESTQTDRES) AS SDISLOJA, A.ESTQTDRES AS SRESLOJA
                               FROM TBS032 AS A INNER JOIN TBS010 AS B ON A.PROCOD = B.PROCOD
                               WHERE (A.ESTLOC IN (2)) AND (A.ESTQTDATU <> 0) OR (A.ESTLOC IN (2))
                               AND (A.ESTQTDRES <> 0) OR (A.ESTLOC IN (2)) 
                               AND (A.ESTQTDPEN <> 0) OR (A.ESTLOC IN (2)) 
                               AND (A.ESTQTDCMP <> 0)
                               GROUP BY A.PROCOD, A.ESTQTDRES) AS SaldoLoja ON P.PROCOD = SaldoLoja.PRO

WHERE (P.PROLOCFIS LIKE RTRIM(@RUA) + '%') 
AND (TBS032.ESTLOC = '1') 
AND (P.PROLOCFIS <> '')

GROUP BY P.PROLOCFIS, P.PROSTATUS, P.PROCOD, P.PRODES, P.MARCOD, P.MARNOM, P.PROUM1, TBS032.ESTQTDATU, TBS032.ESTQTDRES, SaldoLoja.SDISLOJA, RESER.QTD, RESER.PED

ORDER BY P.PROLOCFIS

--- ETIQUETAS DA LOJA COM COD BARRAS

SELECT  rank() OVER (ORDER BY A.PROCOD) AS [RANK],
A.PROSTATUS AS STS, 
RTRIM(LTRIM(A.PROCOD)) AS CÓDIGO, 
RTRIM(LTRIM(A.PRODES)) AS DESCRIÇÃO, 
RTRIM(LTRIM(A.MARNOM)) AS MARCA, 
CASE WHEN B.TDPVALPROI<=GETDATE() AND B.TDPVALPROF >= GETDATE() AND B.TDPPROLOJ = 'S' THEN B.TDPPREPRO1 ELSE B.TDPPRELOJ1 END AS PRECO1,
CASE WHEN PROUM2QTD > 0 
		THEN CASE WHEN B.TDPVALPROI<=GETDATE() AND B.TDPVALPROF >= GETDATE() AND B.TDPPROLOJ = 'S'
				THEN B.TDPPREPRO2*PROUM2QTD 
				ELSE B.TDPPRELOJ2*PROUM2QTD END
		ELSE 0 END AS PRECO2,
'*'+RTRIM(A.PROCOD)+'*'  AS CODBAR1,
CASE WHEN PROUM1QTD = 1 THEN RTRIM(PROUM1) ELSE
	CASE WHEN PROUM1QTD > 1 THEN rtrim(PROUM1) + ' C/' + rtrim(CAST(PROUM1QTD AS DECIMAL(10,0))) +''+ RTRIM(PROUMV) ELSE '' END END  as UN1,
CASE WHEN PROUM2QTD > 0 THEN rtrim(PROUM2) + ' C/' + rtrim(CAST(PROUM2QTD AS DECIMAL(10,0)))+''+ RTRIM(PROUM1) ELSE '' END  as UN2,
ISNULL(VND_GZ.SEMANAL,1) AS PP,
A.PROUM1 AS UNpp

FROM  TBS010 A(NOLOCK) JOIN TBS031 B (NOLOCK) ON A.PROCOD = B.TDPPROCOD

FULL OUTER JOIN (SELECT M2_PROCOD,Ceiling(SUM(M2_QTD)/3/4.35) as SEMANAL 
		FROM MSL002 (NOLOCK)
		WHERE MSL002.M2_DAT between getdate() - CONVERT(DATETIME,91) and getdate()
		and MSL002.M2_TIPREG='01' 
		and MSL002.M2_REGCAN='F'
		GROUP BY M2_PROCOD) AS VND_GZ ON A.PROCOD = VND_GZ.M2_PROCOD AND B.TDPPROCOD = VND_GZ.M2_PROCOD

FULL OUTER JOIN TBS032 C (nolock) ON A.PROCOD = C.PROCOD

WHERE A.PROCOD LIKE(CASE WHEN @COD = '' THEN '% %' ELSE RTRIM(@COD)+'%' END)
AND A.MARNOM LIKE(CASE WHEN @MARNON = '' THEN '% %' ELSE '%'+UPPER(RTRIM(@MARNON))+'%' END)
AND A.PROSTATUS IN( CASE WHEN @STA = 'T' THEN A.PROSTATUS ELSE @STA END )
AND TDPDATATU BETWEEN CASE WHEN @ATL = 'N' THEN '17530101' ELSE @dataDE END AND CASE WHEN @dataAte='' THEN GETDATE() ELSE @dataDE END
AND (C.ESTQTDATU-C.ESTQTDRES >= case when @somenteComSaldo='N' then -9999 else 0.001 end AND C.ESTLOC = 2)
					
ORDER BY [RANK]
