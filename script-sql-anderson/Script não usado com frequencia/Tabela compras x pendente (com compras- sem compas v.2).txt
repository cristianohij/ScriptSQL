

if object_id('sp_PENxCMP') is not null
   drop procedure sp_PENxCMP
go

create procedure sp_PENxCMP @OPCAO INT, @COD CHAR(8), @DESCRI CHAR(60), @PEDV INT,@PEDC INT, @DIAS INT,@DIAS2 INT, @DATPEN1 DATE, @DATAPEN2 DATE, @LOCAL CHAR (3)

as

begin

-- CRIANDO TABELA , PARA UNIÃO DE TODOS OS PEDIDOS PENDENTES

if object_id('TempDB.dbo.#PEN1') is not null
   drop table #PEN1

CREATE TABLE 
	#PEN1 ( 
	--PENIDE int , 	-- INCREMENTO DE 1 A 1 
	PENNOM CHAR (3) NOT NULL, 		-- SE É UM PDV OU SDC 
	PENNUM INT NOT NULL, 			-- NUMERO DO PEDIDO 
	PENCOD CHAR (15) NOT NULL,		-- CODIGO DO PRODUTO
	PENDES CHAR (60) NOT NULL,		-- DESCRIÇÃO DO PRODUTO
	PENUN1 CHAR (2) NOT NULL,		-- MENOR UNIDADE DE EMBALAGEM
	PENQTDITE DECIMAL (12,3) NOT NULL,	-- QTD DO ITEM PENDENTE
	PENVALITE MONEY NOT NULL,		-- VALOR DO ITEM PENDENTE
	PENDAT DATE NOT NULL,			-- DATA DA PENDENCIA
	PENDIA INT NOT NULL,			-- DIAS NA PENDENCIA
	SIT CHAR (1) NOT NULL ,			-- SITUAÇÃO PENDENTE OU RESERVADO
	ESTLOC INT NOT NULL )   		-- LOCAL DO ESTOQUE
	
--BEGIN TRAN 
INSERT INTO 
#PEN1 (
	PENNOM , 		-- SE É UM PDV OU SDC 
	PENNUM , 		-- NUMERO DO PEDIDO 
	PENCOD ,		-- CODIGO DO PRODUTO
	PENDES ,		-- DESCRIÇÃO DO PRODUTO
	PENUN1 ,		-- MENOR UNIDADE DE EMBALAGEM
	PENQTDITE ,		-- QTD DO ITEM PENDENTE
	PENVALITE ,		-- VALOR DO ITEM PENDENTE
	PENDAT ,		-- DATA DA PENDENCIA
	PENDIA ,		-- DIAS NA PENDENCIA
	SIT ,			-- SITUAÇÃO PENDENTE OU RESERVADO
	ESTLOC )
	
SELECT
	'PDV',
	PRPNUM AS PED , 
	A.PROCOD AS COD,
	F.PRODES AS DESCRICAO,
	F.PROUM1 AS UNI1,
	(PRPQTD*PRPQTDEMB) AS QTD ,
	(PRPPRELIQ/PRPQTDEMB)*(PRPQTD*PRPQTDEMB) as VAL,
	PDVDATCAD AS DATCAD ,
	DATEDIFF(DAY,PDVDATCAD,getdate()) AS DIAS_PEN,
	PRPSIT AS SIT,
	PRPESTLOC AS ESTLOC	
	
	FROM TBS058 A (NOLOCK) 
	LEFT JOIN TBS055 B (nolock) on A.PRPNUM=B.PDVNUM
	LEFT JOIN TBS010 F (NOLOCK) ON A.PROCOD = F.PROCOD
	
	WHERE 
	PRPSIT='P' and
	PRPMOVEST='S'
	
	--ORDER BY A.PROCOD, B.PDVDATCAD 
	
	UNION
			
SELECT
	'SDC',
	B.SDCNUM AS PED, 
	B.PROCOD AS COD,
	F.PRODES AS DESCRICAO,
	F.PROUM1 AS UNI1,
	(SDCQTDPED-SDCQTDATD-SDCQTDRES) * SDCQTDEMB AS QTD, 
	G.TDPPRELOJ1 * (SDCQTDPED-SDCQTDATD-SDCQTDRES) * SDCQTDEMB AS VAL,
	SDCDATCAD AS DATCAD,
	DATEDIFF(DAY,SDCDATCAD,getdate()) AS DIAS_PEN,
	CASE WHEN SDCPEN = 'S' THEN 'P' ELSE 'N' END AS SIT, 
	LESCOD AS ESTLOC
	
			
	FROM  TBS0761 B (NOLOCK) 
	LEFT JOIN TBS076 C (NOLOCK) ON C.SDCNUM = B.SDCNUM
	LEFT JOIN TBS031 G (NOLOCK) on B.PROCOD = G.TDPPROCOD
	LEFT JOIN TBS010 F (NOLOCK) ON B.PROCOD = F.PROCOD
	
			
	WHERE
	SDCPEN = 'S' AND 
	SDCQTDPED > (SDCQTDBAI +SDCQTDRES)
	
	--ORDER BY B.PROCOD, SDCDATCAD

-- CRIANDO TABELA DE PRODUTOS PENDENTES EM ABERTO , VALOR TOTAL e PENIDE (RANK)
if object_id('TempDB.dbo.#P') is not null
	drop table #P
		
SELECT 
rank() OVER (ORDER BY PENCOD,PENDAT,PENNUM) AS PENIDE, A.*,
SUM(ESTQTDPEN) AS ESTQTDPEN,
(SELECT SUM(PENVALITE) FROM #PEN1 C WHERE C.PENCOD = A.PENCOD GROUP BY C.PENCOD) AS VALTOTPEN,
SUM(ESTQTDCMP) AS ESTQTDCMP ,
(SELECT SUM(dbo.PDCTOTITE(F.PDCEMPCOD,F.PDCNUM,PDCITE)) 
	FROM 
	TBS0451 F (NOLOCK) 
	LEFT JOIN TBS045 G (NOLOCK) ON  F.PDCNUM = G.PDCNUM
	
	WHERE
	(F.PDCQTD - F.PDCQTDENT - F.PDCQTDRES ) > 0 AND 
	G.PDCBLQ ='N' AND 
	F.PROCOD COLLATE Latin1_General_BIN = A.PENCOD
	
	GROUP BY
	F.PROCOD) AS VALTOTCMP
	
INTO #P  
FROM #PEN1 A 
LEFT JOIN TBS032 B (NOLOCK) ON A.PENCOD COLLATE Latin1_General_BIN = B.PROCOD 

WHERE
B.ESTLOC IN(1,2)

GROUP BY 
PENNOM,PENNUM ,PENCOD,PENDES ,PENUN1 ,PENQTDITE ,PENVALITE ,PENDAT ,PENDIA ,SIT ,A.ESTLOC

-- ACUMULANDO VENDAS
if object_id('TempDB.dbo.#PEN') is not null
	drop table #PEN

SELECT *,
ESTQTDCMP - (SELECT SUM(PENQTDITE) FROM #P AS TI WHERE TI.PENIDE <= TA.PENIDE AND TI.PENCOD = TA.PENCOD ) AS PENACU,
CASE WHEN ESTQTDCMP - (SELECT SUM(PENQTDITE) FROM #P AS TI WHERE TI.PENIDE <= TA.PENIDE AND TI.PENCOD = TA.PENCOD ) >= 0
	THEN 'ATENDE'
	ELSE 'NAO ATENDE'
	END AS SITUACAO
INTO #PEN
FROM #P	AS TA


-- CRIANDO TABELA DE COMPRAS EM ABERTO E COM VALOR TOTAL 
if object_id('TempDB.dbo.#CMP') is not null
   drop table #CMP
	
SELECT 
E.PDCNUM,
D.PROCOD,
SUM(dbo.PDCTOTITE(D.PDCEMPCOD,D.PDCNUM,PDCITE)) AS VALOR_ITEM,
SUM((PDCQTD * PDCQTDEMB)) AS QTD_COMPRAS,
E.PDCDATCAD AS DATA_CADAS,
E.PDCDATPEN AS PRE_ENT

INTO #CMP 
FROM 
TBS0451 D (NOLOCK) 
LEFT JOIN TBS045 E (NOLOCK) ON  D.PDCNUM = E.PDCNUM
--LEFT JOIN TBS032 A (NOLOCK) ON A.PROCOD = D.PROCOD AND A.ESTLOC = D.LESCOD

WHERE
(PDCQTD - PDCQTDENT - PDCQTDRES ) > 0 AND PDCBLQ ='N' 
GROUP BY
E.PDCNUM,D.PROCOD,E.PDCDATCAD,E.PDCDATPEN

----------------------------------------------------------------------------------------------------------------------------------------------------

	QTD_COMPRAS AS PDCQTD,

	PRE_ENT AS PDCDATENT,
	ESTQTDCMP,
	VALTOTCMP,
	ESTQTDPEN,
	VALTOTPEN,
	PENACU,
	SITUACAO
	
	FROM #PEN I
	--LEFT JOIN TBS032 A (NOLOCK) ON I.PENCOD COLLATE Latin1_General_BIN = A.PROCOD AND I.ESTLOC = A.ESTLOC
	LEFT JOIN #CMP E (NOLOCK) ON I.PENCOD COLLATE Latin1_General_BIN = E.PROCOD
	
	WHERE 
	(PENDES LIKE(CASE WHEN @DESCRI = '' THEN '% %' ELSE Upper(RTRIM(@DESCRI))+'%' END)) AND
	PENNUM =(CASE WHEN @PEDV = 0 THEN PENNUM ELSE @PEDV END) AND
	PENDIA BETWEEN @DIAS AND CASE WHEN @DIAS2 = 0 THEN 10000 ELSE @DIAS2 END AND
	PENDAT BETWEEN @DATPEN1 AND @DATAPEN2 AND
	PENCOD LIKE(CASE WHEN @COD = '' THEN '% %' ELSE Upper(RTRIM(@COD))+'%' END) AND
	PENNOM = (CASE WHEN @LOCAL = '' THEN PENNOM ELSE @LOCAL END) AND
	--SITUACAO = (CASE WHEN @SIT = '' THEN SITUACAO ELSE @SIT END) AND

	I.ESTLOC,

			 WHERE TI.PENIDE <= TA.PENIDE AND TI.PENCOD = TA.PENCOD )

#T AS TA LEFT JOIN TBS032 A (NOLOCK) ON TA.PENCOD COLLATE Latin1_General_BIN = A.PROCOD AND TA.ESTLOC = A.ESTLOC

--------------------------------------------------------------------------------------------------------------------------------------

PRE_ENT AS PDCENT,*/
isnull((SELECT SUM(PDCQTD) FROM #CMP C WHERE A.PENCOD COLLATE Latin1_General_BIN = C.PROCOD),0) AS ESTQTDCMP,
isnull((SELECT SUM(PDCVAL) FROM #CMP C WHERE A.PENCOD COLLATE Latin1_General_BIN = C.PROCOD),0) as ESTVALCMP

-- ACUMULANDO VENDAS
if object_id('TempDB.dbo.#PEN') is not null
	drop table #PEN

SELECT *,
ESTQTDCMP - (SELECT SUM(PENQTDITE) FROM #P AS TI WHERE TI.PENIDE <= TA.PENIDE AND TI.PENCOD = TA.PENCOD ) AS PENACU,
CASE WHEN ESTQTDCMP - (SELECT SUM(PENQTDITE) FROM #P AS TI WHERE TI.PENIDE <= TA.PENIDE AND TI.PENCOD = TA.PENCOD ) >= 0
	THEN 'ATENDE'
	ELSE 'NAO ATENDE'
	END AS SITUACAOitem,

CASE WHEN ESTQTDCMP - ESTQTDPEN >= 0
	THEN 'ATENDE'
	ELSE 'NAO ATENDE'
	END AS SITUACAO

INTO #PEN
FROM #P	AS TA
