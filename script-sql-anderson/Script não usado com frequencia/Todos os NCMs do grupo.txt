SELECT 
rtrim(ltrim(PROCLAFIS)) as PROCLAFIS,
PROSTBPIS,
PROSTBCOFINS

INTO #TBS010
FROM TBS010 A(NOLOCK)
WHERE 
(PROSTBPIS = '06' OR 
PROSTBCOFINS = '06') AND 
PROCLAFIS <> '' 

GROUP BY 
PROCLAFIS,
PROSTBPIS,
PROSTBCOFINS

UNION

(SELECT TOP 1 PROCOD FROM TBS010 B WHERE A.PROCLAFIS = B.PROCLAFIS ORDER BY PROCOD ) AS PROCOD ,
(SELECT TOP 1 PRODES FROM TBS010 B WHERE A.PROCLAFIS = B.PROCLAFIS ORDER BY PROCOD ) AS PRODES ,
(SELECT TOP 1 MARNOM FROM TBS010 B WHERE A.PROCLAFIS = B.PROCLAFIS ORDER BY PROCOD ) AS MARNOM,

FROM #TBS010 A
