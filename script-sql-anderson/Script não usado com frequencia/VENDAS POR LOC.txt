DECLARE @dataDe DATE, @dataAte DATE, @produtoDe CHAR(15),@produtoAte CHAR(15) , @Loc CHAR(8)
SET @dataDe = '20151101'
SET @dataAte ='20151111'
SET @produtoDe = ''
SET @produtoAte = ''
SET @Loc = ''

SELECT        A.PROSTATUS AS ST, 
RTRIM(LTRIM(A.PROCOD)) AS CÓD, 
RTRIM(A.PRODES) AS DESCRIÇÃO, 
CASE WHEN len(A.MARCOD) = 4 THEN rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) 
                         ELSE RIGHT(('00' + ltrim(str(A.MARCOD))), 3) + ' - ' + rtrim(A.MARNOM) END AS MARCA, 
A.PROLOCFIS AS 'LOC 1', 
A.PROLOCFIS2 AS 'LOC 2',
isnull((select TBS032.ESTQTDATU-TBS032.ESTQTDRES from TBS032 (nolock) where TBS032.PROCOD=A.PROCOD and ESTLOC=2),0) as 'Disponivel_LOJA',
isnull((select TBS032.ESTQTDATU-TBS032.ESTQTDRES from TBS032 (nolock) where TBS032.PROCOD=A.PROCOD and ESTLOC=1),0) as 'Disponivel_EST',
isnull((select ESTQTDCMP from TBS032 (nolock) where TBS032.PROCOD=A.PROCOD and ESTLOC=1),0) as 'QuantidadeComprada',
sum(MSL002.M2_QTD) as 'QuantidadeVendida',
avg(MSL002.M2_VALUNI-MSL002.M2_ABT) as 'ValorMedioDoItem',
sum(MSL002.M2_VALTOT-MSL002.M2_ABT) as 'TotalVendido', 
PROCODBAR1  as 'CodigoDeBarras1',
PROCODBAR2  as 'CodigoDeBarras2'
FROM            TBS010 AS A INNER JOIN
                         TBS032 AS B ON A.PROCOD = B.PROCOD
						 LEFT JOIN MSL002 ON A.PROCOD=MSL002.M2_PROCOD
						 
WHERE (A.PROLOCFIS LIKE(CASE WHEN @Loc = '' THEN A.PROLOCFIS ELSE RTRIM(UPPER(@Loc)) + '%'END) ) 
AND MSL002.M2_DAT between @dataDe and @dataAte 
and A.PROCOD between @produtoDe and case when @produtoAte='' then 'Z' else @produtoAte end 
and MSL002.M2_TIPREG='01' 
and MSL002.M2_REGCAN='F'
	   
GROUP BY A.PROSTATUS, A.PROCOD, A.PRODES , A.MARCOD, A.MARNOM , A.PROLOCFIS, A.PROLOCFIS2,MSL002.M2_PROCOD,PROCODBAR1,PROCODBAR2
ORDER BY 'LOC 1'

DECLARE @dataDe DATE, @dataAte DATE, @produtoDe CHAR(15),@produtoAte CHAR(15) , @Loc CHAR(20)
SET @dataDe = '20151101'
SET @dataAte ='20151111'
SET @produtoDe = ''
SET @produtoAte = ''
SET @Loc = '02'
--SELECT RTRIM((@Loc))+'%'

SELECT        A.PROSTATUS AS ST, 
RTRIM(LTRIM(A.PROCOD)) AS CÓD, 
RTRIM(A.PRODES) AS DESCRIÇÃO, 
CASE WHEN len(A.MARCOD) = 4 THEN rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) 
                         ELSE RIGHT(('00' + ltrim(str(A.MARCOD))), 3) + ' - ' + rtrim(A.MARNOM) END AS MARCA, 
A.PROLOCFIS AS 'LOC 1', 
A.PROLOCFIS2 AS 'LOC 2',
A.PROUM1 AS 'UN1',
SAL_EST.QTD AS SALEST,
SAL_LOJ.QTD AS SALLOJA,
SAL_EST.ESTQTDCMP AS COMPRAS,
PROCODBAR1  as 'CodigoDeBarras1',
PROCODBAR2  as 'CodigoDeBarras2',
ISNULL(VND_LOJA.VENDAS,0) AS VENDASLOJ,
ISNULL(VND_LOJA.VALMED,VND_INT.VALOR) AS VALORMED,
PRECOS.PRE AS VALOR_ATUAL,
VND_INT.VENDAS AS VENDASINT


FROM TBS010 A 
FULL OUTER JOIN  (SELECT PROCOD,ESTQTDATU-ESTQTDRES AS QTD,ESTQTDCMP FROM TBS032 NOLOCK WHERE ESTLOC = 1) AS SAL_EST ON A.PROCOD = SAL_EST.PROCOD

FULL OUTER JOIN  (SELECT PROCOD,ESTQTDATU-ESTQTDRES AS QTD FROM TBS032 NOLOCK WHERE ESTLOC = 2) AS SAL_LOJ ON A.PROCOD = SAL_LOJ.PROCOD

FULL OUTER JOIN  (SELECT M2_PROCOD, SUM(M2_QTD) AS VENDAS, avg(M2_VALUNI-M2_ABT) AS VALMED FROM MSL002 (NOLOCK)
WHERE MSL002.M2_DAT between @dataDe and @dataAte 
and MSL002.M2_TIPREG='01' 
and MSL002.M2_REGCAN='F'
GROUP BY M2_PROCOD) AS VND_LOJA ON A.PROCOD = VND_LOJA.M2_PROCOD

FULL OUTER JOIN (SELECT A.PROCOD, SUM(A.NFSQTD*NFSQTDEMB)AS VENDAS,AVG(NFSPRE/NFSQTDEMB) AS VALOR
FROM TBS0671 A (NOLOCK) JOIN TBS067 B ON 
A.NFSEMPCOD = B.NFSEMPCOD and 
A.NFSNUM = B.NFSNUM and 
A.SNEEMPCOD = B.SNEEMPCOD and 
A.SNESER = B.SNESER

WHERE NFSCLINOM NOT LIKE('TANBY%') AND
NFSDATEMI between @dataDe and @dataAte 
AND NFSTIP = 'N' and NFSCAN = 'N' 
GROUP BY A.PROCOD) VND_INT ON A.PROCOD = VND_INT.PROCOD	

FULL OUTER JOIN  (SELECT TDPPROCOD , TDPPRELOJ1 AS PRE FROM TBS031 nolock ) AS PRECOS ON A.PROCOD = PRECOS.TDPPROCOD
 
WHERE A.PROLOCFIS LIKE(CASE WHEN @Loc = '' THEN '% %' ELSE RTRIM((@Loc))+'%' END) AND A.PROLOCFIS <> ''
