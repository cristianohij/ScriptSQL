DECLARE @Data_de DATE , @Data_ate DATE , @NFETIP CHAR (1)
SET @Data_de = '20160813'
SET @Data_ate = GETDATE()
SET @NFETIP = 'T' 

SELECT 
B.NFEDATENT AS DATAENT,
A.NFENUM,
A.NFETIP ,
NFECAN,
PROCOD,
RTRIM(NFEDESXML),
NFEFATOR,
NFECALC,
NFEQTDXML,
NFEQTD,
NFEUNI



--SELECT A.* 
FROM TBS0591 A (NOLOCK)
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND  A.SEREMPCOD = B.SEREMPCOD
WHERE (B.NFEDATENT BETWEEN @Data_de AND @Data_ate) AND B.NFETIP IN (@NFETIP) --AND A.NFENUM IN(39036,3811,57633)

--GROUP BY A.NFENUM, B.NFETOTBAS , B.NFETOTICMS , B.NFEDATENT, PROCOD

ORDER BY 
B.NFEDATENT DESC
