SELECT DISTINCT NFSCFOP 
FROM 
TBS0671 B (nolock)
LEFT JOIN TBS067 A (nolock) ON A.NFSEMPCOD = B.NFSEMPCOD and A.NFSNUM = B.NFSNUM and A.SNEEMPCOD = B.SNEEMPCOD and A.SNESER = B.SNESER
LEFT JOIN TBS080 D (NOLOCK) ON B.NFSNUM = D.ENFNUM AND B.SNEEMPCOD = D.SNEEMPCOD AND B.SNESER = D.SNESER
	  
WHERE 
NFSDATEMI >'20150101' AND
D.ENFSIT = 6 AND 
D.ENFTIPDOC = 1 

UNION
