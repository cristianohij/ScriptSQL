SELECT 'ENTRDADA' AS TIPO, FORCOD, FORNOM,UFESIG, NFEESTORI, NFENUM , CONVERT(DATE,NFEDATEFE) AS NFEDATEFE

FROM TBS006 A (NOLOCK)
LEFT JOIN TBS059 B (NOLOCK) ON A.FORCOD = B.NFECOD

WHERE 
NFETIP = 'N' AND
A.UFESIG <> B.NFEESTORI 

UNION

SELECT 'SAIDA' AS TIPO, CLICOD, CLINOM, A.UFESIG, B.UFESIG, NFSNUM , CONVERT(DATE,NFSDATEMI) AS NFSDATEMI

FROM TBS002 A (NOLOCK)
LEFT JOIN TBS067 B (NOLOCK) ON A.CLICOD = B.NFSCLICOD

WHERE 
A.UFESIG <> B.UFESIG
