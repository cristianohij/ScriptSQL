-- ESTORNANDO O PEDIDO DE COMPRAS 

-- 1° CRIAR TABELA TEMP. DE ESTORNO
if object_id('TempDB.dbo.#PDCEST') is not null 
	drop table #PDCEST
	
DECLARE @NFEEMP SMALLINT , @NFETIP CHAR(1), @NFENUM DECIMAL(10,0), @NFECOD INT ,@SEREMP SMALLINT, @SERCOD char(3), @NFEITE SMALLINT,
@PDCEMP SMALLINT, @PDCNUM INT, @PDCITE smallint

SET @NFEEMP = 0
SET @NFETIP = 'N'
SET @NFENUM = 165530
SET @NFECOD = 332
SET @SEREMP = 0
SET @SERCOD = 'CAN' -- 'CAN'
--SET @NFEITE = 0
SET @PDCEMP = 0 
SET @PDCNUM = 43348
--SET @PDCITE = 0 

SELECT 
PDCEMPCOD ,
A.NFENUM,
B.PROEMPCOD, 
C.PROCOD,
PDCITE, 
sum(NFEATEQTD) QTDPDC ,
ROUND(sum(NFEATEQTD*B.PDCQTDEMB),0) SAL_TBS051,
PDCNUM, 
--PDCQTDENT+NFEATEQTD AS SALDO, 
--(PDCQTDENT*B.PDCQTDEMB) + NFEQTD AS SAL_TBS051,
--NFEQTD,
C.LESCOD, 
C.NFEUNI

INTO #PDCEST
FROM TBS0591 C (NOLOCK)
LEFT JOIN TBS0592 A (NOLOCK) ON A.NFEEMPCOD = A.NFEEMPCOD AND A.NFETIP = C.NFETIP AND C.NFENUM = A.NFENUM AND C.NFECOD = A.NFECOD AND 
C.SEREMPCOD = A.SEREMPCOD AND C.SERCOD = A.SERCOD AND C.NFEITE = A.NFEITE
LEFT JOIN TBS0451 B (NOLOCK) ON A.NFEPEDITE = B.PDCITE AND A.NFEPEDNUM = B.PDCNUM AND A.NFEPEDEMP  = B.PDCEMPCOD
WHERE 
A.NFEEMPCOD = @NFEEMP AND 
A.NFETIP = @NFETIP AND 
A.NFENUM = @NFENUM AND 
A.NFECOD = @NFECOD AND 
A.SEREMPCOD = @SEREMP AND
A.SERCOD = @SERCOD AND
--NFEITE IN (CASE WHEN @NFEITE = 0 THEN NFEITE ELSE @NFEITE END) AND
A.NFETIPPED = 'C' AND
A.NFEPEDEMP = @PDCEMP AND 
A.NFEPEDNUM = @PDCNUM --AND
--NFEPEDITE IN (CASE WHEN @PDCITE = 0 THEN PDCITE ELSE @PDCITE END)
GROUP BY 
PDCEMPCOD ,
A.NFENUM,
B.PROEMPCOD, 
C.PROCOD,
PDCITE, 
PDCNUM, 
C.LESCOD, 
C.NFEUNI

-- 2° ESTORNAR PEDIDO DE COMPRAS, DE ACORDO COM A TABELA DE ESTORNO
BEGIN TRAN 
UPDATE TBS0451
SET PDCQTDENT = PDCQTDENT - ( SELECT QTDPDC FROM #PDCEST B WHERE TBS0451.PDCITE = B.PDCITE AND TBS0451.PDCNUM = B.PDCNUM AND TBS0451.PDCEMPCOD = B.PDCEMPCOD ) 
WHERE TBS0451.PDCITE IN (SELECT PDCITE FROM #PDCEST) AND 
TBS0451.PDCNUM IN (SELECT PDCNUM FROM #PDCEST) AND 
TBS0451.PDCEMPCOD IN (SELECT PDCEMPCOD FROM #PDCEST)

select PDCEMPCOD, --0 LMEEMPCOD
	   ROW_NUMBER() OVER(ORDER BY A.PDCITE )+(SELECT MAX(LMEREG) FROM TBS051 (NOLOCK)), -- 1 LMEREG
	   NFENUM, -- 2 LMEDOC
	   'SQL', -- 3 LMEROT
	   'ESTORNO PEDIDO DE COMPRAS', -- 4 LMEDESROT
	   'E', --5 LMEACA
	   GETDATE(), -- 6 LMEDATHOR
	   'ANDERSONL', -- 7 LMEUSU
	   'NENHUM', -- 8 LMEMOD
	   'C', -- 9 LMEINFALT
	   A.PROCOD, -- 10 PROCOD
	   A.PROEMPCOD, -- 11 PROEMPCOD
	   SAL_TBS051+ESTQTDCMP, -- 12 LMEQTDSAL
	   SAL_TBS051, -- 13 LMEQTDMOV
	   ESTQTDATU, -- 14 LMEQTDATU
	   ESTQTDRES, --15 LMEQTDRES
	   ESTQTDPEN, -- 16 LMEQTDPEN
	   ESTQTDCMP, -- 17 LMEQTDCMP
	   NFEUNI, -- 18 LMEUNI
	   ESTQTDATU - ESTQTDRES, -- 19 LMEQTDDIS
	   LESCOD 
	   from #PDCEST A
	   LEFT JOIN TBS032 B (NOLOCK) ON A.PROCOD = B.PROCOD AND A.LESCOD = B.ESTLOC
	   ORDER BY A.PDCITE

-- 5° GRAVAR SEQUENCIAL NA TBS024
UPDATE TBS024 
SET TBSVALSEQ = (SELECT MAX(LMEREG) FROM TBS051 (NOLOCK))
WHERE TBSNOM = 'TBS051'

PROCOD IN (SELECT A.PROCOD FROM #PDCEST A (NOLOCK) WHERE TBS032.ESTLOC = A.LESCOD AND TBS032.PROCOD = A.PROCOD)
