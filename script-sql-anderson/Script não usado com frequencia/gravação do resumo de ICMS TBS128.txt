if object_id('TempDB.dbo.#NFS') is not null
BEGIN
   drop table #NFS
END

SELECT 
A.NFSCFOP AS CFOP, 
-- SUM(dbo.NFSTOTPRO(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE)) AS NFSTOTPRO,  					-- TOTAL DOS PRODUTOS
SUM(ROUND(dbo.NFSTOTITEST(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) AS NFSTOTITEST, 		-- TOTAL DA NOTA
SUM(round(dbo.NFSBASICMS(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) AS NFSBASICMS, 		-- BASE ICMS
SUM(round(dbo.NFSVALICMS(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) AS NFSVALICMS, 		-- VALOR ICMS 

CASE WHEN SUBSTRING(NFSCST,2,2) IN ('30','40','41','50')
	THEN SUM(ROUND(dbo.NFSTOTITEST(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2))
	ELSE 0
END AS ICMSISENAOTRI 		-- AS VALOR ICMS ISENTO/NÃO TRIBUTADO

INTO #NFS
FROM 
TBS0671 A (NOLOCK)

LEFT JOIN TBS080 D (NOLOCK) ON A.NFSNUM = D.ENFNUM AND A.SNEEMPCOD = D.SNEEMPCOD AND A.SNESER = D.SNESER 

WHERE
SUBSTRING(CONVERT(VARCHAR(12),B.NFSDATEMI,111),1,7) = @anoDe+'/'+@mesDe AND
NFSTIP IN('N','C') and 
D.ENFSIT = 6 AND
D.ENFTIPDOC = 1

A.NFSCFOP , A.NFSCST 

----------------------------------------------------------------------\\------------------------------------------------------------------------------------------

-- CRIANDO TABELA DE DEVOLUÇÃO

if object_id('TempDB.dbo.#NFD') is not null
BEGIN
   drop table #NFD
END
   
SELECT 
A.NFDCFOP AS CFOP, 
-- SUM(dbo.NFDTOTPROITE(A.NFDEMPCOD , A.SNEEMPCOD, A.SNESER, A.NFDNUM, A.NFDNFEEMP, A.NFDNFETIP, A.NFDNFENUM, A.NFDSEREMP, A.NFDSERCOD, A.NFDITE)) AS NFSTOTPRO, -- TOTAL DOS PRODUTOS 
SUM(ROUND(dbo.NFDTOTITE(A.NFDEMPCOD, A.SNEEMPCOD, A.SNESER, A.NFDNUM, A.NFDNFEEMP, A.NFDNFETIP, A.NFDNFENUM,A.NFDSEREMP, A.NFDSERCOD, A.NFDITE),2)) AS NFSTOTITEST,-- TOTAL DA NOTA
SUM(round(A.NFDBASICMS,2)) AS NFSBASICMS, 		-- BASE ICMS
SUM(round(A.NFDVALICMS,2)) AS NFSVALICMS, 		-- VALOR ICMS

CASE WHEN SUBSTRING(NFDCSTCSOSN,2,2) IN ('30','40','41','50')
	THEN SUM(ROUND(dbo.NFDTOTITE(A.NFDEMPCOD, A.SNEEMPCOD, A.SNESER, A.NFDNUM, A.NFDNFEEMP, A.NFDNFETIP, A.NFDNFENUM,A.NFDSEREMP, A.NFDSERCOD, A.NFDITE),2))
	ELSE 0
END AS ICMSISENAOTRI 		-- AS VALOR ICMS ISENTO/NÃO TRIBUTADO

INTO #NFD
FROM
TBS1172 A (NOLOCK)
LEFT JOIN TBS117 B (NOLOCK) ON A.NFDEMPCOD = B.NFDEMPCOD AND A.NFDNUM = B.NFDNUM AND A.SNEEMPCOD = B.SNEEMPCOD AND A.SNESER = B.SNESER
LEFT JOIN TBS080 D (NOLOCK) ON A.NFDNUM = D.ENFNUM AND A.SNEEMPCOD = D.SNEEMPCOD AND A.SNESER = D.SNESER

WHERE
SUBSTRING(CONVERT(VARCHAR(12),B.NFDDATEMI,111),1,7) = @anoDe+'/'+@mesDe AND
D.ENFSIT = 6 AND
D.ENFFINEMI = 4 AND
D.ENFTIPDOC = 1

GROUP BY 
A.NFDCFOP, A.NFDCSTCSOSN 

------------------------------------------------------------------------------------------------//-------------------------------------------------------------------

	-- AGRUPANDO A TABELA DE VENDAS, TENHO QUE AGRUPAR 1° AS VENDAS, PARA DEPOIS UNIR COM A TBL DE DEVOLUÇÃO

	BEGIN 
	   drop table #NFSICMS
	END
	
	SELECT 
	CFOP, 
	-- SUM(NFSTOTPRO) 	AS NFSTOTPRO, 					-- TOTAL DOS PRODUTOS

	SUM(NFSBASICMS) AS NFSBASICMS, 						-- BASE ICMS
	SUM(NFSVALICMS) AS NFSVALICMS, 						-- VALOR ICMS
	SUM(ICMSISENAOTRI) AS ICMSISENAOTRI 				-- AS VALOR ICMS ISENTO/NÃO TRIBUTADO
	-- SUM(NFSTOTITEST) - SUM(NFSBASICMS) - SUM(ICMSISENAOTRI) AS ICMSOUTROS	-- TOTAL DA NOTA - BASE CALCULO ICMS = ICMSOUTROS
	
	INTO #NFSICMS
	FROM #NFS
	
	GROUP BY 
	CFOP

   drop table #MSLAUX

create table #MSLAUX (M2_STATUS INT, M2_LOJ INT, M2_CUP INT, M2_COO INT, M2_ECF INT, M2_CXA INT ,M2_ITE INT, M2_PROCOD INT, M2_DESCUP DECIMAL(10,4),
					  M2_ACRCUP DECIMAL(10,4), M2_VALBRU DECIMAL(10,4), M2_ABT DECIMAL(10,4), M2_CST CHAR(3), M2_TRI VARCHAR(6))
INSERT INTO #MSLAUX
exec(@comando)

-- CRIANDO TABELA TEMP COM OS VALORES REAIS DE DESCONTO
if object_id('TempDB.dbo.#TMP') is not null
BEGIN 
   drop table #TMP
END
   
	select
	'5.929' AS M2_CFOP,	
	A.M2_TRI,
	A.M2_CST,
	A.M2_ITE,
	A.M2_LOJ, 
	A.M2_CUP, 
	A.M2_COO, 
	A.M2_ECF, 
	A.M2_CXA ,
	M2_PROCOD,
	M2_VALBRU,
	M2_VALBRU - M2_ABT + M2_ACRCUP - 
	CASE WHEN  A.DESCONTO <> ISNULL(B.DESCONTO,0) AND M2_ITE = (SELECT TOP 1 M2_ITE 
																FROM #MSLAUX E 
																
																WHERE 
																A.M2_COO = E.M2_COO and A.M2_ECF = E.M2_ECF AND A.M2_CUP = E.M2_CUP 
																
																ORDER BY 
																M2_DESCUP DESC)
		THEN
			CASE WHEN A.DESCONTO > ISNULL(B.DESCONTO,0) 
				THEN ROUND(M2_DESCUP,2) - (A.DESCONTO - ISNULL(B.DESCONTO,0))
				ELSE ROUND(M2_DESCUP,2) + (ISNULL(B.DESCONTO,0) - A.DESCONTO)
			END
		ELSE ROUND(M2_DESCUP,2)
	END AS M2_LIQUIDO
	
	INTO #TMP	
	FROM (
		SELECT 
		M2_ITE,M2_PROCOD,M2_TRI,M2_CST,
		M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , 
		M2_DESCUP ,
		M2_ACRCUP,
		M2_ABT ,
		M2_VALBRU,
		SUM(ISNULL(ROUND(M2_DESCUP,2),0)) OVER (PARTITION BY M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA ) AS DESCONTO
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 1 ) AS A 
			
		LEFT JOIN (
		SELECT 
		M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , M2_VALBRU AS DESCONTO 
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 2 ) AS B ON A.M2_LOJ = B.M2_LOJ AND A.M2_CUP = B.M2_CUP AND A.M2_COO= B.M2_COO AND 
														A.M2_ECF = B.M2_ECF AND A.M2_CXA = B.M2_CXA  

	
-------------------------------------------------------------------------------------------------------------------------------------------------

-- CRIANDO TABELA DE ITENS COM OS TRIBUTOS NECESSARIOS

if object_id('TempDB.dbo.#LIQUIDO') is not null
BEGIN
   drop table #LIQUIDO
END

		SELECT
		C.M2_CFOP,
		-- C.M2_VALBRU AS NFSTOTPRO ,					-- TOTAL DOS PRODUTOS,		
		C.M2_LIQUIDO AS NFSTOTITEST,				-- TOTAL DAS NOTAS, NO CASO DOS CUPONS NÃO TEM NOTA, MAS COLQUEI LQUIDO
		
		CASE WHEN (CONVERT(DECIMAL(10,2),SUBSTRING(C.M2_TRI,2,5)) / 100) > 0 
			THEN C.M2_LIQUIDO  						-- valor liquido
			ELSE 0 
		END AS NFSBASICMS, 							-- BASE ICMS
		
		C.M2_LIQUIDO * (CONVERT(DECIMAL(10,2),SUBSTRING(C.M2_TRI,2,5)) / 100) AS NFSVALICMS ,  -- VALOR LIQUIDO * PORCENTAGEM TRIBUTADA -- VALOR ICMS				 
		
		CASE WHEN SUBSTRING(M2_CST,2,2) IN ('30','40','41','50')
			THEN C.M2_LIQUIDO
			ELSE 0
		END AS ICMSISENAOTRI							-- AS VALOR ICMS ISENTO/NÃO TRIBUTADO
		
		INTO #LIQUIDO
		FROM #TMP C 
		LEFT JOIN TBS010 B (NOLOCK) ON C.M2_PROCOD = CONVERT(INT,B.PROCOD)
		
----------------------------------------------------------------------------------------------------------------------------------------------------------------

if object_id('TempDB.dbo.#CUP') is not null
BEGIN
   drop table #CUP
END
   
		SELECT 
		M2_CFOP AS CFOP,
		-- SUM(NFSTOTPRO)					AS NFSTOTPRO,		-- TOTAL DOS PRODUTOS,
		SUM(NFSTOTITEST)					AS NFSTOTITEST,		-- TOTAL DAS NOTAS, NO CASO DOS CUPONS NÃO TEM NOTA, MAS COLQUEI O VALOR LIQUIDO
		SUM(NFSBASICMS) 					AS NFSBASICMS, 		-- BASE ICMS
		SUM(NFSVALICMS) 					AS NFSVALICMS,		-- VALOR ICMS
		SUM(ICMSISENAOTRI)					AS ICMSISENAOTRI	-- AS VALOR ICMS ISENTO/NÃO TRIBUTADO
		-- SUM(NFSTOTITEST) - SUM(NFSBASICMS)	AS ICMSOUTROS		-- ICMS OUTROS 
		
		INTO #CUP
		FROM #LIQUIDO
		
		GROUP BY
		M2_CFOP

NFSVALICMS, 	-- VALOR ICMS

if object_id('TempDB.dbo.#NFE') is not null
BEGIN
   drop table #NFE

SELECT
A.NFECFOP AS CFOP,
-- sum(dbo.NFETOTITE(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE)) AS NFETOTITE,
sum(ROUND(NFETOTOPEITE,2)) AS NFETOTOPEITE,
SUM(round(A.NFEBASICMS,2)) AS NFEBASICMS, 
SUM(round(A.NFEVALICMS,2)) AS NFEVALICMS,

CASE WHEN SUBSTRING(NFECST,2,3) IN ('30','40','103','203','300','41','50','400','102')
	THEN sum(ROUND(NFETOTOPEITE,2))
	ELSE 0
END AS ICMSISENAOTRI		-- ICMS ISENTO / NÃO TRIBUTADO

INTO #NFE
FROM TBS0591 A (NOLOCK)
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND 
A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND  A.SEREMPCOD = B.SEREMPCOD
LEFT JOIN TBS080 D (NOLOCK) ON A.NFENUM = D.ENFNUM AND A.NFECOD = D.ENFCODDES

WHERE 
((D.ENFSIT = 6 AND D.ENFFINEMI = 4 AND D.ENFTIPDOC = 0) OR ((NFENOSFOR = 'N' OR NFENOSFOR = '') AND (B.NFECAN = 'N' OR B.NFECAN = '') AND NFEDATEFE <> '' )) AND
SUBSTRING(CONVERT(VARCHAR(12),B.NFEDATEFE,111),1,7) = @anoDe+'/'+@mesDe 

GROUP BY 
A.NFECFOP, A.NFECST

ORDER BY 
A.NFECFOP, A.NFECST

   drop table #TBS057

0 AS NFEVALIPI,

ROUND(SUM(ROUND(CPABASCAL,2)) *  CPAALIICMS / 100 ,2) AS NFEVALICMS,
SUM(ROUND(CPAVAL,2)) - SUM(ROUND(CPABASCAL,2)) AS ICMSOUTROS

INTO #TBS057

TBS057 NOLOCK 

WHERE 
CPAREFMES = @mes and  
CPAREFANO = @ano and  
CPACFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E' AND PAGAR = 'S') and
CPACFOP COLLATE DATABASE_DEFAULT IN (SELECT NFECFOP FROM #CFOP1)

CPAALIICMS

END AS NFSTOTITESTCNT, 		-- TOTAL DA NOTA, CONTRIBUINTE

CASE WHEN CLIINDIE = 1 
	THEN SUM(round(dbo.NFSBASICMS(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) 
	ELSE 0
END AS NFSBASICMSCNT, 		-- BASE ICMS CONTRIBUINTE

CASE WHEN CLIINDIE = 9 
	THEN SUM(ROUND(dbo.NFSTOTITEST(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2))
	ELSE 0
END AS NFSTOTITEST, 		-- TOTAL DA NOTA NÃO CONTRIBUINTE

CASE WHEN CLIINDIE = 9 
	THEN SUM(round(dbo.NFSBASICMS(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) 
	ELSE 0
END AS NFSBASICMS, 			-- BASE ICMS NÃO CONTRIBUINTE

SUM(round(dbo.NFSVALICMS(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) AS NFSVALICMS, 		-- VALOR ICMS 

CASE WHEN SUBSTRING(NFSCST,2,2) IN ('30','40','41','50')
	THEN SUM(ROUND(dbo.NFSTOTITEST(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2))
	ELSE 0
END AS ICMSISENAOTRI, 		-- AS VALOR ICMS ISENTO/NÃO TRIBUTADO

SUM(ROUND(dbo.NFSVALICMSST(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) AS NFSVALICMSST

LEFT JOIN TBS067 B (NOLOCK) ON A.NFSEMPCOD = B.NFSEMPCOD AND A.NFSNUM = B.NFSNUM AND A.SNEEMPCOD = B.SNEEMPCOD AND A.SNESER = B.SNESER
LEFT JOIN TBS080 D (NOLOCK) ON A.NFSNUM = D.ENFNUM AND A.SNEEMPCOD = D.SNEEMPCOD AND A.SNESER = D.SNESER 
LEFT JOIN TBS002 C (NOLOCK) ON B.NFSCLICOD = C.CLICOD

WHERE
SUBSTRING(CONVERT(VARCHAR(12),B.NFSDATEMI,111),1,7) = @anoDe+'/'+@mesDe AND
NFSTIP IN('N','C') and 
D.ENFSIT = 6 AND
D.ENFTIPDOC = 1 AND 
B.UFESIG <> 'SP'

GROUP BY 
A.NFSCFOP , A.NFSCST ,B.UFESIG, CLIINDIE

-- CRIANDO TABELA DE DEVOLUÇÃO

if object_id('TempDB.dbo.#NFD1') is not null
BEGIN
   drop table #NFD1
END
   
SELECT 
B.NFDESTDES AS UF,
A.NFDCFOP AS CFOP, 
-- SUM(dbo.NFDTOTPROITE(A.NFDEMPCOD , A.SNEEMPCOD, A.SNESER, A.NFDNUM, A.NFDNFEEMP, A.NFDNFETIP, A.NFDNFENUM, A.NFDSEREMP, A.NFDSERCOD, A.NFDITE)) AS NFSTOTPRO, -- TOTAL DOS PRODUTOS

CASE WHEN F.FORTIPPES= 'J' AND F.FORIES <> '' 
	THEN SUM(ROUND(dbo.NFDTOTITE(A.NFDEMPCOD, A.SNEEMPCOD, A.SNESER, A.NFDNUM, A.NFDNFEEMP, A.NFDNFETIP, A.NFDNFENUM,A.NFDSEREMP, A.NFDSERCOD, A.NFDITE),2))
	ELSE 0
END AS NFSTOTITESTCNT,	-- TOTAL DA NOTA CONTRIBUINTE

CASE WHEN F.FORTIPPES= 'J' AND F.FORIES <> '' 
	THEN SUM(round(A.NFDBASICMS,2))
	ELSE 0
END AS NFSBASICMSCNT,	-- BASE ICMS CONTRIBUINTE

CASE WHEN (F.FORTIPPES= 'J' OR F.FORTIPPES = 'F') AND F.FORIES = '' 
	THEN SUM(ROUND(dbo.NFDTOTITE(A.NFDEMPCOD, A.SNEEMPCOD, A.SNESER, A.NFDNUM, A.NFDNFEEMP, A.NFDNFETIP, A.NFDNFENUM,A.NFDSEREMP, A.NFDSERCOD, A.NFDITE),2))

LAIANO = @anoDe AND

------------------------------------------------------------------------------------------------------------------------------------------

-- GRAVAR LAIQ18 TBS1282, QUANTIDADE DE REGITRO DA TBS1283 (ANO, MES, CFOP, UF)

			  WHERE TBS1282.LAICFOP = A.LAICFOP AND TBS1282.LAIUF = A.LAIUF AND LAIANO = @anoDe AND LAIMES = @mesDe GROUP BY A.LAICFOP,A.LAIUF)
WHERE 
LAIANO = @anoDe AND 
LAIMES = @mesDe AND
LAICFOP IN (SELECT LAICFOP FROM TBS1283 A (NOLOCK) WHERE TBS1282.LAICFOP = A.LAICFOP AND LAIANO = @anoDe AND LAIMES = @mesDe) AND
LAIUF COLLATE Latin1_General_CI_AS IN (SELECT LAIUF FROM TBS1283 A (NOLOCK) WHERE TBS1282.LAIUF COLLATE Latin1_General_CI_AS = A.LAIUF AND LAIANO = @anoDe AND LAIMES = @mesDe)

WHERE 
LAIANO = @anoDe AND 
LAIMES = @mesDe AND
LAIUF COLLATE Latin1_General_CI_AS IN (SELECT UFESIG FROM TBS001 A (NOLOCK) WHERE TBS1282.LAIUF COLLATE Latin1_General_CI_AS = A.UFESIG)
