data,

data between "'+convert(char(8), @data1de, 112)+'" and "'+convert(char(8), @data1ate, 112)+'"  and

-- contabilizar vendas televendas

if object_id('TempDB.dbo.#CORP') is not null
   drop table #CORP

SELECT 
PROCOD,
sum(B.NFSQTD * B.NFSQTDEMB) AS NFSQTDVEN,
NFSDATEMI AS DATA

into #CORP
FROM 
TBS0671 B (nolock)
LEFT JOIN TBS067 A (nolock) ON A.NFSEMPCOD = B.NFSEMPCOD and A.NFSNUM = B.NFSNUM and A.SNEEMPCOD = B.SNEEMPCOD and A.SNESER = B.SNESER
INNER JOIN #TBS080 D (NOLOCK) ON B.NFSNUM = D.ENFNUM AND B.SNEEMPCOD = D.SNEEMPCOD AND B.SNESER = D.SNESER
			  
WHERE 
NFSDATEMI between @data1de and @data1ate and 
NFSTIP IN('N','C') and 
D.ENFTIPDOC = 1 AND
B.NFSCFOP IN ('5.101','5.102','5.116','5.117','5.118','5.119','5.123','5.405','5.551','5.929','6.102','6.108','6.117','6.123','6.403','6.404','6.929') AND
A.NFSCLINOM NOT LIKE ('TANBY%') AND 		-- TIRO AS TRANSFERENCIAS
NFSCLICOD NOT IN (SELECT CLICOD FROM #GRUPOCLI) 

GROUP BY 
PROCOD, NFSDATEMI

if object_id('TempDB.dbo.#TBS010') is not null
   drop table #TBS010

SELECT 
A.PROCOD,
PRODES,
CASE WHEN len(A.MARCOD) = 4 
	then rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) 
	else right(('00' + ltrim(str(A.MARCOD))),3) + ' - ' + rtrim(A.MARNOM) 
end as 'MARCA',
PROUM1

INTO #TBS010
FROM TBS010 A (NOLOCK) 

--------------------------------------------------------------------------------------------------------------------------------------------------------------------

if object_id('TempDB.dbo.#TBS032') is not null
   drop table #TBS032

SELECT 
A.PROCOD,
(SELECT ESTQTDATU - ESTQTDRES FROM TBS032 B (NOLOCK) WHERE ESTLOC = 1 AND A.PROCOD= B.PROCOD) AS ESTQTD,
(SELECT ESTQTDATU - ESTQTDRES FROM TBS032 C (NOLOCK) WHERE ESTLOC = 2 AND A.PROCOD= C.PROCOD) AS LOJQTD

INTO #TBS032
FROM TBS032 A (NOLOCK) 

WHERE 
ESTQTDATU <> 0 AND ESTLOC IN (1,2)


-------------------------------------------------------------------------------------------------------------------------------------------------

SELECT
RTRIM(LTRIM(A.PROCOD)) AS PROCOD,
RTRIM(LTRIM(PRODES)) AS PRODES,
RTRIM(LTRIM(MARCA)) AS MARCA,
RTRIM(LTRIM(PROUM1)) AS PROUM1,
REPLACE(ISNULL(LOJQTD,0),'.',',') AS LOJ,
REPLACE(ISNULL(ESTQTD,0),'.',',') AS EST,
ISNULL((SELECT TOP 1 DATA FROM #COMPRA D WHERE A.PROCOD = D.PROCOD ORDER BY DATA DESC),'') AS DATACOMPRA,
CASE WHEN ISNULL((SELECT TOP 1 DATA FROM #CORP E WHERE A.PROCOD = E.PROCOD ORDER BY DATA DESC),'') >= 
		  ISNULL((SELECT TOP 1 DATA FROM #LOJA F WHERE CONVERT(INT,A.PROCOD) = F.PROCOD ORDER BY DATA DESC),'')
	THEN ISNULL((SELECT TOP 1 DATA FROM #CORP E WHERE A.PROCOD = E.PROCOD ORDER BY DATA DESC),'') 
	ELSE ISNULL((SELECT TOP 1 DATA FROM #LOJA F WHERE CONVERT(INT,A.PROCOD) = F.PROCOD ORDER BY DATA DESC),'')
END AS DATAVENDA,
ISNULL((SELECT TOP 1 DATA FROM #COMPRAGRU G WHERE A.PROCOD = G.PROCOD ORDER BY DATA DESC),'') AS DATACOMPRAGRU,
ISNULL((SELECT TOP 1 DATA FROM #CORPGRU H WHERE A.PROCOD = H.PROCOD ORDER BY DATA DESC),'') AS DATAVENDA

FROM 
#TBS010 A 
LEFT JOIN #TBS032 B ON A.PROCOD = B.PROCOD

WHERE 
ISNULL((SELECT TOP 1 DATA FROM #COMPRA D WHERE A.PROCOD = D.PROCOD ORDER BY DATA DESC),'') > '19000101' OR 
ISNULL((SELECT TOP 1 DATA FROM #CORP E WHERE A.PROCOD = E.PROCOD ORDER BY DATA DESC),'') > '19000101' OR
ISNULL((SELECT TOP 1 DATA FROM #LOJA F WHERE CONVERT(INT,A.PROCOD) = F.PROCOD ORDER BY DATA DESC),'') > '19000101' OR
ISNULL(LOJQTD,0) <> 0 OR
ISNULL(ESTQTD,0) <> 0 OR
ISNULL((SELECT TOP 1 DATA FROM #COMPRAGRU G WHERE A.PROCOD = G.PROCOD ORDER BY DATA DESC),'') <> '19000101' OR
ISNULL((SELECT TOP 1 DATA FROM #CORPGRU H WHERE A.PROCOD = H.PROCOD ORDER BY DATA DESC),'') <> '19000101'
