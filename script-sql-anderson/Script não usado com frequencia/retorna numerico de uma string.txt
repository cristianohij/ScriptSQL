CREATE FUNCTION FNUMEROS (@string VARCHAR(100))

RETURNS VARCHAR(100)AS 

	BEGIN 
	DECLARE @MAX INT, @NUM VARCHAR(100), @CARAC CHAR(1)
	
	SET @NUM = ''
	SET @MAX = 0
	
	WHILE @MAX<=(SELECT LEN(RTRIM(LTRIM(@string))))
		BEGIN 
			
			SET @CARAC = (SELECT SUBSTRING(@CARAC,@MAX,1))--(SELECT RIGHT(LEFT(@STRING,LEN(@STRING)-@MAX+1),1))
			
			IF @CARAC <> ''
				BEGIN 
					IF ISNUMERIC(@CARAC) = 1
						BEGIN
							SET @NUM = STR(@NUM) + STR(@CARAC)
						END
				END
			SET @MAX = @MAX + 1
		END
	RETURN STR(@NUM)
END 

DROP FUNCTION  dbo.FNUMEROS
