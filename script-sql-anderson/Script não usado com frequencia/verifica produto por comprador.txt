PROCOD

INTO #COMPRA
FROM #TBS045 A
INNER JOIN (SELECT MAX(QTD) AS QTD , PROCOD FROM #TBS045 GROUP BY PROCOD) B ON A.QTD = B.QTD AND A.PROCOD = B.PROCOD 


UPDATE #COMPRA SET
RANK = (SELECT MAX(RANK) FROM #COMPRA A WHERE #COMPRA.PROCOD = A.PROCOD GROUP BY PROCOD)
