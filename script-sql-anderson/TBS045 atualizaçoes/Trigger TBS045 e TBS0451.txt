CREATE TRIGGER dbo.trg_AtualizaTBS045CD					-- CRIA NOVA TRIGGER
	ON dbo.TBS045									-- TABELA ONDE A TRIGGER SERÁ INSERIDA (TBS045)
	FOR UPDATE, INSERT , DELETE						-- CASO FOR UPDATE, INSERT OU DELETE FAÇA...
AS 
BEGIN
	INSERT into TBS045AUX 							-- INSERIR O RESULTADO NA TABLEA AUX TBS045 AUX
	
-- PEGAR SOMENTE O QUE REALMENTE FOI INSERIDO	
	SELECT host_name(),system_user,GETDATE(),'INSERT',A.*  -- GRAVANDO NOME DA MAQUINA, USUER , DATA E HORA, AÇÃO E OS DADOS DO PEDIDO
	
	FROM INSERTED A (NOLOCK)
	LEFT JOIN DELETED B (NOLOCK) ON A.PDCEMPCOD = B.PDCEMPCOD AND A.PDCNUM = B.PDCNUM
	
	WHERE 
	B.PDCEMPCOD IS NULL AND 						-- PARA SABER O QUE FOI INSERIDO OS CAMPOS CHAVES DA TABELA DELETE TEM QUE ESTÁ NULL
	B.PDCNUM IS NULL 

	UNION 											-- USEI UNION PARA UNIR OS RESULTADOS DE TODAS AS AÇÕES
	
-- PEGAR SOMENTE O QUE REALMENTE FOI EXCLUIDO 	
	SELECT host_name(),system_user, GETDATE(),'DELETE',A.* -- GRAVANDO NOME DA MAQUINA, USUER , DATA E HORA, AÇÃO E OS DADOS DO PEDIDO
	
	FROM DELETED A (NOLOCK)
	LEFT JOIN INSERTED B (NOLOCK) ON A.PDCEMPCOD = B.PDCEMPCOD AND A.PDCNUM = B.PDCNUM 
	
	WHERE 
	B.PDCEMPCOD IS NULL AND							-- PARA SABER O QUE FOI DELETEDO OS CAMPOS CHAVES DA TABELA INSERT TEM QUE ESTÁ NULL 
	B.PDCNUM IS NULL 

	UNION
	
-- PEGAR REALMENTE O QUE FOI ALTERADO 
	SELECT host_name(),system_user, GETDATE(),'UPDATE',A.* -- GRAVANDO NOME DA MAQUINA, USUER , DATA E HORA, AÇÃO E OS DADOS DO PEDIDO
	FROM INSERTED A (NOLOCK)
	INNER JOIN DELETED B (NOLOCK) ON A.PDCEMPCOD = B.PDCEMPCOD AND A.PDCNUM = B.PDCNUM 
	
	WHERE 
	A.PDCDATALT+A.PDCHORALT <> B.PDCDATALT+B.PDCHORALT OR -- PARA SABER O QUE FOI ALTERADO OS CAMPOS DE DATA DE ALTERÇÃO E CODIGO DO FORNECEDOR NA TABLE INSERTED
	A.FORCOD <> B.FORCOD							-- DEVE ESTAR DIFERENTE COM A TABLE DELETED
		
	UNION
	
-- PEGAR REALMENTE O QUE FOI ELIMINADO RESIDUO
	SELECT host_name(),system_user,GETDATE(),'RESIDO',A.* -- GRAVANDO NOME DA MAQUINA, USUER , DATA E HORA, AÇÃO E OS DADOS DO PEDIDO 
	FROM INSERTED A (NOLOCK)
	INNER JOIN DELETED B (NOLOCK) ON A.PDCEMPCOD = B.PDCEMPCOD AND A.PDCNUM = B.PDCNUM 
	
	WHERE 
	A.PDCDATRES+A.PDCHORRES <> B.PDCDATRES+B.PDCHORRES	-- PARA SABER O QUE FOI ELIMINDO RESIDUO OS CAMPOS DE DATA DE RESIDUO NA TABLE INSERTED DEVEM ESTAR DIFERENTE 
													-- DA TABLE DELETED
end
