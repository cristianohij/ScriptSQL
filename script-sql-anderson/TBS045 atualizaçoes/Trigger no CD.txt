IF OBJECT_ID ('trg_baixaTBS0451', 'TR') IS NOT NULL  
   DROP TRIGGER trg_baixaTBS0451;  
GO 

CREATE TRIGGER dbo.trg_baixaTBS0451
	ON dbo.TBS0592
	FOR INSERT , UPDATE
AS
BEGIN
	INSERT into TBS0451AUX
-- PEGAR SOMENTE O QUE REALMENTE FOI INSERIDO, NÃO CANCELADO , MAIOR QUE ZERO E TIPO COMPRAS (C)
	SELECT
	host_name(),
	system_user,
	GETDATE(),
	'ENTROU', 
	C.PDCEMPCOD,
	C.PDCNUM,
	C.PDCITE,
	C.PROCOD,
	C.PROEMPCOD,
	C.PDCDES,
	C.PDCQTD,
	C.PDCUNI,
	C.PDCPRE,
	C.PDCPDDITE,
	C.LESCOD,
	C.LESEMPCOD,
	C.PDCQTDEMB,
	A.NFEATEQTD,
	C.PDCDATPRE,
	C.PDCDATFAT,
	C.PDCQTDRES,
	C.PDCPDVNUM,
	C.PDCPDVITE,
	C.PDCPROFOR,
	C.PDCPORST,
	C.PDCIPI,
	C.PDCPROIVA,
	C.PDCPREFOR,
	C.PDCPORFREITE,
	C.PDCPOROUTITE,
	C.PDCPORSEGITE 
	
	FROM INSERTED A (NOLOCK)
	LEFT JOIN DELETED B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFETIP = B.NFETIP AND A.NFENUM = B.NFENUM AND A.NFECOD = B.NFECOD AND A.SEREMPCOD = B.SEREMPCOD AND
	A.SERCOD = B.SERCOD AND A.NFEITE = B.NFEITE AND A.NFETIPPED = B.NFETIPPED AND A.NFEPEDEMP = B.NFEPEDEMP AND A.NFEPEDNUM = B.NFEPEDNUM AND A.NFEPEDITE = B.NFEPEDITE
	LEFT JOIN TBS0451 C (NOLOCK) ON A.NFEPEDEMP = C.PDCEMPCOD AND A.NFEPEDNUM = C.PDCNUM AND A.NFEPEDITE = C.PDCITE
	
	WHERE 
	B.NFEEMPCOD IS NULL AND
	B.NFETIP IS NULL AND
	B.NFENUM IS NULL AND
	B.NFECOD IS NULL AND
	B.SEREMPCOD IS NULL AND
	B.SERCOD IS NULL AND
	B.NFEITE IS NULL AND
	B.NFETIPPED IS NULL AND
	B.NFEPEDEMP IS NULL AND
	B.NFEPEDNUM IS NULL AND
	B.NFEPEDITE IS NULL AND 
	A.NFETIPPED = 'C' AND 
	A.NFEATEQTD > 0 AND 
	A.SERCOD <> 'CAN' 	
	
	UNION 
	
-- GRAVAR O QUE CANCELOU, PARA ESTRONAR O PEDIDO, CASO A NOTA SEJA CANCELADA 
	SELECT
	host_name(),
	system_user, 
	GETDATE(),
	'CANCEL',
	C.PDCEMPCOD,
	C.PDCNUM,
	C.PDCITE,
	C.PROCOD,
	C.PROEMPCOD,
	C.PDCDES,
	C.PDCQTD,
	C.PDCUNI,
	C.PDCPRE,
	C.PDCPDDITE,
	C.LESCOD,
	C.LESEMPCOD,
	C.PDCQTDEMB,
	A.NFEATEQTD,
	C.PDCDATPRE,
	C.PDCDATFAT,
	C.PDCQTDRES,
	C.PDCPDVNUM,
	C.PDCPDVITE,
	C.PDCPROFOR,
	C.PDCPORST,
	C.PDCIPI,
	C.PDCPROIVA,
	C.PDCPREFOR,
	C.PDCPORFREITE,
	C.PDCPOROUTITE,
	C.PDCPORSEGITE 
	
	FROM INSERTED A (NOLOCK)
	LEFT JOIN DELETED B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFETIP = B.NFETIP AND A.NFENUM = B.NFENUM AND A.NFECOD = B.NFECOD AND A.SEREMPCOD = B.SEREMPCOD AND
	A.SERCOD = B.SERCOD AND A.NFEITE = B.NFEITE AND A.NFETIPPED = B.NFETIPPED AND A.NFEPEDEMP = B.NFEPEDEMP AND A.NFEPEDNUM = B.NFEPEDNUM AND A.NFEPEDITE = B.NFEPEDITE
	LEFT JOIN TBS0451 C (NOLOCK) ON A.NFEPEDEMP = C.PDCEMPCOD AND A.NFEPEDNUM = C.PDCNUM AND A.NFEPEDITE = C.PDCITE
	
	WHERE 
	B.NFEEMPCOD IS NULL AND
	B.NFETIP IS NULL AND
	B.NFENUM IS NULL AND
	B.NFECOD IS NULL AND
	B.SEREMPCOD IS NULL AND
	B.SERCOD IS NULL AND
	B.NFEITE IS NULL AND
	B.NFETIPPED IS NULL AND
	B.NFEPEDEMP IS NULL AND
	B.NFEPEDNUM IS NULL AND
	B.NFEPEDITE IS NULL AND 
	A.NFETIPPED = 'C' AND 
	A.NFEATEQTD > 0 AND 
	A.SERCOD = 'CAN'
	  
end
