-- 1° TABELA PARA VERIFICAR AS QUANTIDADES QUE ENTROU NOS PDC ND PARA ATUALIZAR O ND 
begin tran 
IF OBJECT_ID ('TempDB.dbo.#cmp') is not null
drop table #cmp

SELECT 
'CD' AS EMPRESA, 
A.PROCOD, 
NFEPEDNUM, 
NFEPEDITE, 
NFEATEQTD AS NFEATEQTD ,
NFEPEDEMB , 
case when D.PROPESAVEL = 'N' 
	THEN ROUND(NFEATEQTD*NFEPEDEMB,0) 
	ELSE NFEATEQTD*NFEPEDEMB
END AS NFEQTDUNI1, 
NFEATEDAT as NFEATEDAT,
A.LESCOD,
B.NFEEMPCOD,
PROUM1
		
into #cmp	
FROM cd.SIBD.dbo.TBS0591 A (NOLOCK)
LEFT JOIN cd.SIBD.dbo.TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND A.NFETIP = B.NFETIP AND 
A.SERCOD = B.SERCOD AND  A.SEREMPCOD = B.SEREMPCOD

LEFT JOIN cd.SIBD.dbo.TBS0592 C (NOLOCK) ON A.NFEEMPCOD = C.NFEEMPCOD AND A.NFETIP = C.NFETIP AND A.NFENUM = C.NFENUM AND A.SEREMPCOD = C.SEREMPCOD AND
A.SERCOD = C.SERCOD  AND A.NFEITE = C.NFEITE AND A.NFECOD = C.NFECOD

LEFT JOIN TBS010 D (NOLOCK) ON A.PROCOD collate Latin1_General_CI_AS= D.PROCOD AND A.PROEMPCOD = D.PROEMPCOD

WHERE 
NFECAN = 'N' AND 
B.NFETIP = 'N' AND 
NFEATEDAT > (SELECT ULTDAT FROM TEMPO) AND
C.NFETIPPED = 'C' and 
NFEATEQTD > 0 
		
--GROUP BY 
--A.PROCOD, NFEPEDNUM, NFEPEDITE,NFEPEDEMB ,A.LESCOD,B.NFEEMPCOD, D.PROPESAVEL, D.PROUM1 

--------------------------------------------------------------------------------------------------------------------------------
		
DELETE TEMPO

INSERT INTO TEMPO 
SELECT 
CASE WHEN COUNT(NFEATEDAT)=0
	THEN GETDATE()
	ELSE MAX(NFEATEDAT)
END
FROM #cmp


----------------------------------------------------------------------------------------------------------------------------------

-- 2° Atualiza saldo em compras do CD pelas notas de entrada
--BEGIN TRAN ;

MERGE TBS0451 A -- DESTINO
USING #cmp B WITH (NOLOCK) -- ORIGEM 
ON A.PDCNUM = B.NFEPEDNUM AND A.PDCEMPCOD = B.NFEEMPCOD AND A.PDCITE = B.NFEPEDITE AND A.PROCOD collate Latin1_General_CI_AS = B.PROCOD AND A.LESCOD = B.LESCOD

WHEN MATCHED AND A.PDCQTD - A.PDCQTDENT - A.PDCQTDRES > 0 and  A.PDCQTDENT +A.PDCQTDRES + NFEATEQTD <= A.PDCQTD 
		
THEN UPDATE SET 
A.PDCQTDENT = A.PDCQTDENT + NFEATEQTD
							
OUTPUT 
GETDATE() ,
UPPER($action),
INSERTED.*,
DELETED.* --;;
INTO TmpTBS0451 ;;

-- 3° RESEVAR NUMERO PARA OS LOGS NA TBS051

IF OBJECT_ID ('TempDB.dbo.#TmpTBS0451') is not null
drop table #TmpTBS0451
select PROCOD INTO #TmpTBS0451 FROM TmpTBS0451 GROUP BY PROCOD

UPDATE TBS024 
SET TBSVALSEQ = (SELECT MAX(LMEREG) FROM TBS051) + (SELECT COUNT(*) FROM #TmpTBS0451)
WHERE TBSNOM = 'TBS051'

IF OBJECT_ID ('TempDB.dbo.#T246') is not null
drop table #T246
SELECT TBSVALSEQ - (SELECT COUNT(*) FROM #TmpTBS0451) as TBSVALSEQ INTO #T246 FROM TBS024 WHERE TBSNOM = 'TBS051'

-- 4° GRAVAR LOG DE MOVIMENTAÇÃO TBS051
--begin tran 
INSERT INTO 
TBS051( 
	LMEEMPCOD, -- 0
	LMEREG, --1
	LMEDOC, --2
	LMEROT, --3
	LMEDESROT, --4
	LMEACA, --5
	LMEDATHOR, --6
	LMEUSU, --7
	LMEMOD, --8
	LMEINFALT, --9
	PROCOD, --10
	PROEMPCOD, --11
	LMEQTDSAL, -- 12
	LMEQTDMOV, -- 13
	LMEQTDATU, -- 14
	LMEQTDRES, -- 15
	LMEQTDPEN, -- 16
	LMEQTDCMP,  -- 17
	LMEUNI, -- 18
	LMEQTDDIS, -- 19
	LMELOCEST) -- 20

-- gerar saldo de log, somente dos itens que foram feito a entrada e não atualizaram ainda, na outra unidade.
select	
	A.PDCEMPCOD, 	--0 LMEEMPCOD
	ROW_NUMBER() OVER(ORDER BY A.PROCOD )+(SELECT TBSVALSEQ FROM #T246(NOLOCK)), -- 1 LMEREG
	0, 								-- 2 LMEDOC
	'SQL', 							-- 3 LMEROT
	'SAIDA PEDIDO DE COMPRAS NFE', 		-- 4 LMEDESROT
	'S', 							--5 LMEACA
	GETDATE(), 						-- 6 LMEDATHOR
	'ANDERSONL', 					-- 7 LMEUSU
	'NENHUM', 						-- 8 LMEMOD
	'C', 							-- 9 LMEINFALT
	A.PROCOD, 						-- 10 PROCOD
	A.PROEMPCOD, 					-- 11 PROEMPCOD
	ESTQTDCMP - CASE WHEN sum(NFEQTDUNI1) - ESTQTDCMP > 0 
					THEN ESTQTDCMP
					ELSE SUM(NFEQTDUNI1)
				END , 				-- 12 LMEQTDSAL
	CASE WHEN sum(NFEQTDUNI1) - ESTQTDCMP > 0
		THEN ESTQTDCMP
		ELSE SUM(NFEQTDUNI1)
	END , 							-- 13 LMEQTDMOV
	ESTQTDATU, 						-- 14 LMEQTDATU
	ESTQTDRES, 						--15 LMEQTDRES
	ESTQTDPEN, 						-- 16 LMEQTDPEN
	ESTQTDCMP, 						-- 17 LMEQTDCMP
	PROUM1, 						-- 18 LMEUNI
	ESTQTDATU - ESTQTDRES, 			-- 19 LMEQTDDIS
	A.LESCOD 						-- LMELESCOD

	from TmpTBS0451 A
	LEFT JOIN #cmp D (NOLOCK) ON A.PROCOD collate Latin1_General_CI_AS = D.PROCOD AND A.LESCOD = D.LESCOD
	LEFT JOIN TBS032 B (NOLOCK) ON A.PROCOD collate Latin1_General_CI_AS = B.PROCOD AND A.LESCOD = B.ESTLOC
		
	GROUP BY 
	A.PDCEMPCOD,A.PROCOD,A.PROEMPCOD,ESTQTDCMP,ESTQTDRES,ESTQTDATU,ESTQTDPEN,PROUM1,A.LESCOD--,PDCNUM

	ORDER BY 
	A.PROCOD

-- 5° Atualizar saldo compras CD

-- BEGIN TRAN 
UPDATE TBS032
SET ESTQTDCMP = ESTQTDCMP - (SELECT CASE WHEN sum(NFEQTDUNI1) - TBS032.ESTQTDCMP > 0
											THEN TBS032.ESTQTDCMP
											ELSE SUM(NFEQTDUNI1)
										END
							 FROM TmpTBS0451 A (NOLOCK) 
							 LEFT JOIN #cmp B (NOLOCK) ON A.PROCOD COLLATE Latin1_General_CI_AS = B.PROCOD AND A.LESCOD = B.LESCOD
							 WHERE TBS032.ESTLOC = A.LESCOD AND TBS032.PROCOD COLLATE Latin1_General_CI_AS = A.PROCOD
							 group by A.PROCOD)
WHERE ESTLOC IN (SELECT DISTINCT A.LESCOD 
				 FROM TmpTBS0451 A 
				 WHERE TBS032.ESTLOC = A.LESCOD AND TBS032.PROCOD COLLATE Latin1_General_CI_AS = A.PROCOD ) AND 
PROCOD COLLATE Latin1_General_CI_AS IN (SELECT DISTINCT A.PROCOD 
										FROM TmpTBS0451 A (NOLOCK) 
										WHERE TBS032.ESTLOC = A.LESCOD AND TBS032.PROCOD COLLATE Latin1_General_CI_AS = A.PROCOD)


DELETE TmpTBS0451
