if not exists(select name from sysobjects where name = 'tmpcmp' and type='U')
   begin
      create table tmpcmp
         (empresa tinyint not null default 0, localEstoque tinyint not null default 0, codigoProduto char(15) not null default '',
          qCompraAnterior money default 0, qCompraPedidos money default 0,
          constraint itmpcmp primary key(empresa, localEstoque, codigoProduto))
   end
      else delete from tmpcmp
	  
	  
	  
	  
insert into tmpcmp
select A.PROEMPCOD,A.ESTLOC,A.PROCOD,A.ESTQTDCMP,isNull(sum((B.PDCQTD - (B.PDCQTDENT + B.PDCQTDRES)) * B.PDCQTDEMB),0)
  from TBS032 A (nolock) join TBS0451 B (nolock) on B.PROEMPCOD = A.PROEMPCOD and B.PROCOD = A.PROCOD and B.LESCOD = A.ESTLOC
 group by A.PROEMPCOD,A.ESTLOC,A.PROCOD,A.ESTQTDCMP
having A.ESTQTDCMP <> isNull(sum((B.PDCQTD - (B.PDCQTDENT + B.PDCQTDRES)) * B.PDCQTDEMB),0)



update TBS032 set ESTQTDCMP = qCompraPedidos from tmpcmp where empresa = PROEMPCOD and localEstoque = ESTLOC and codigoProduto = PROCOD



if object_id('TempDB.dbo.#cmplog') is not null
   drop table #cmplog
select 
Identity(INT,1,1) as ordem,
A.PROEMPCOD,
A.ESTLOC,
A.PROCOD,
A.ESTQTDCMP,
A.ESTQTDATU,
A.ESTQTDRES,
A.ESTQTDPEN,
C.PROUM1,
A.ESTQTDATU-A.ESTQTDRES AS DISPONIVEL,
isNull(sum((B.PDCQTD - (B.PDCQTDENT + B.PDCQTDRES)) * B.PDCQTDEMB),0) AS CMPQTDATU,
isNull(sum((B.PDCQTD - (B.PDCQTDENT + B.PDCQTDRES)) * B.PDCQTDEMB),0) - A.ESTQTDCMP AS QTDMOV

into #cmplog
from TBS032 A (nolock) 
join TBS0451 B (nolock) on B.PROEMPCOD = A.PROEMPCOD and B.PROCOD = A.PROCOD and B.LESCOD = A.ESTLOC
left join TBS010 C (nolock) on A.PROCOD = C.PROCOD

group by 
A.PROEMPCOD,
A.ESTLOC,
A.PROCOD,
A.ESTQTDCMP,
A.ESTQTDATU,
A.ESTQTDRES,
A.ESTQTDPEN,
C.PROUM1

having A.ESTQTDCMP <> isNull(sum((B.PDCQTD - (B.PDCQTDENT + B.PDCQTDRES)) * B.PDCQTDEMB),0) 

select * from #cmplog
