SELECT FORREGTRI,FORCOD, A.NFECOD ,NFENOM,NFECST,NFECHAACE, NFECFOP,NFEPISENT,NFECOFINSE, PROCOD, 
dbo.NFETOTITE(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE) AS VALITEM, 
dbo.NFEVALCOFINSE(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE) AS COFINS ,
dbo.NFEVALPISENT(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE)  AS PIS

FROM 
TBS0591 AS A ( nolock) 
LEFT JOIN
TBS059 AS B ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND A.SEREMPCOD = B.SEREMPCOD
LEFT JOIN 
TBS006 C (NOLOCK) ON A.NFECOD = C.FORCOD

WHERE
(B.NFECAN = 'N' OR B.NFECAN = '') AND 
(B.NFEDATENT BETWEEN '20160101' AND '20160130')  

--GROUP BY A.NFECFOP, A.NFECST, A.NFENUM, B.NFECHAACE

SELECT * FROM TBS006 WHERE FORCOD  =15974
