-- VERIFICAR QUAL BASE DUPLIQUEI, PARA INSERIR NA OUTRA BASE 

BEGIN TRAN 
INSERT INTO 
#NDTESTE
(PROSTATUS,
PROCOD,
PRODES,
PROCODANT,
MARCOD,
MARNOM)

SELECT  
B.PROSTATUS,
-- VERIFICAR SE É NECESSARIO ACRESCENTAR 0 (ZERO) NO INICIO

B.PROCODANT,
B.MARCOD,
B.MARNOM
	   
		FROM #NDTESTE A FULL OUTER JOIN #PYTESTE B ON A.PROCOD = B.PROCOD

		AND B.MARCOD = 281 -- AND LEN(A.PROCOD) <=7

	   
ROLLBACK TRAN 
COMMIT TRAN 

--- ZERAR REFERENCIAS DOS CODIGOS ANTIGOS , SETANDO OS CAMPOS NECESSARIOS.

BEGIN TRAN 
UPDATE #NDTESTE
SET PROCODANT = ''
WHERE PROCOD COLLATE Latin1_General_CI_AS IN (SELECT PROCODANT FROM #NDTESTE 
				 WHERE PROCOD like('281%') AND LEN(PROCOD) <=7 
				 AND PROCOD COLLATE Latin1_General_CI_AS <> PROCODANT)
				 --AND PROCODREF IS NULL)
				 
ROLLBACK TRAN 
COMMIT TRAN

-- CRIANDO COLUNA DE CODIGO UNICO, JUNTO COM A DESCRIÇÃO O MATERIAL FICA UNICO
SELECT 
rank() OVER (ORDER BY PROCOD) AS [RANKCOD],
 PROCOD , PRODES + CONVERT(CHAR(10),rank() OVER (ORDER BY PROCOD)) AS [DES+RANK] 
 FROM TBS010 (NOLOCK) WHERE PRODES IN (SELECT PRODES FROM TBS010(NOLOCK) GROUP BY PRODES HAVING COUNT(PRODES)>1 )
