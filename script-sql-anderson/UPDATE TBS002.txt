-- ATUALIZAR CAMPOS CLIULTENDCOD ERRADOS

-- VERIFICANDO
SELECT B.CLICOD , MAX(CLIENDCOD) AS ULTCOD
--INTO #TBS0021
FROM TBS002 A 
INNER JOIN TBS0021 B ON A.CLICOD = B.CLICOD

WHERE CLIULTENDCOD <>  (SELECT MAX(CLIENDCOD) FROM TBS0021 C WHERE C.CLICOD = A.CLICOD GROUP BY CLICOD)

GROUP BY B.CLICOD


-- UPDATE 
BEGIN TRAN 
UPDATE TBS002
SET CLIULTENDCOD =  ULTCOD FROM #TBS0021 B INNER JOIN TBS002 A ON A.CLICOD = B.CLICOD
WHERE CLIULTENDCOD <> ULTCOD 

COMMIT TRAN
