-- criar URL dos relatorios

-- tanby 182 - Matriz

SELECT 
Name ,
'http://192.168.1.182/Reports/Pages/Report.aspx?ItemPath=%2fRelat%c3%b3rios+Tanby+Matriz'+
REPLACE(REPLACE(SUBSTRING(PATH,25,200),'/','%2f'),' ','+') as URL

FROM Catalog
 
WHERE 
SUBSTRING(PATH,1,25) = '/Relatórios Tanby Matriz/' AND 
Name <> '' and 
Name not in ('Estoque','Loja','Compras','Financeiro','Contábil','Vendas')
