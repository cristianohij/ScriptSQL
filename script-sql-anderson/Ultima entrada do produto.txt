WITH 
ND as  (SELECT 
	A.PROCOD collate Latin1_General_CI_AS AS 'CÓDIGO PRODUTO',
	'ND' collate Latin1_General_CI_AS  as 'Unidade',
	NFEDES collate Latin1_General_CI_AS AS 'DESCRIÇÃO',
	replace(str(isnull(A.NFEQTD*A.NFEQTDEMB,0),10,4),'.',',')collate Latin1_General_CI_AS AS 'QUANTIDADE',
	C.PROUM1 collate Latin1_General_CI_AS AS 'UNI MED.',
	B.NFENUM  AS 'NF',
	B.NFENOM collate Latin1_General_CI_AS AS 'NOME FORNECEDOR',
	CONVERT(NVARCHAR,B.NFEDATENT,2)collate Latin1_General_CI_AS AS 'DATA ENTRADA'
	
	FROM nd.SIBD.dbo.TBS010 C 
	JOIN nd.SIBD.dbo.TBS0591 A ON
	C.PROCOD = A.PROCOD
	join nd.SIBD.dbo.TBS059 B  on 
	A.NFEEMPCOD = B.NFEEMPCOD AND
	A.NFECOD = B.NFECOD AND
	A.NFENUM = B.NFENUM AND
	A.NFETIP = B.NFETIP AND
	A.SERCOD = B.SERCOD AND
	A.SEREMPCOD = B.SEREMPCOD 

	WHERE B.NFETIP = 'N' AND B.NFECAN = 'N'
	GROUP BY A.PROCOD,NFEDES,C.PROUM1,B.NFENUM,B.NFENOM,A.NFEQTD,A.NFEQTDEMB,B.NFEDATENT ),

CD as ( SELECT 
	A.PROCOD collate Latin1_General_CI_AS AS 'CÓDIGO PRODUTO',
	'CD' collate Latin1_General_CI_AS as 'Unidade',
	NFEDES collate Latin1_General_CI_AS AS 'DESCRIÇÃO',
	replace(str(isnull(A.NFEQTD*A.NFEQTDEMB,0),10,4),'.',',')collate Latin1_General_CI_AS AS 'QUANTIDADE',
	C.PROUM1 collate Latin1_General_CI_AS AS 'UNI MED.',
	B.NFENUM  AS 'NF',
	B.NFENOM collate Latin1_General_CI_AS AS 'NOME FORNECEDOR',
	CONVERT(NVARCHAR,B.NFEDATENT,2)collate Latin1_General_CI_AS AS 'DATA ENTRADA'
	
	FROM cd.SIBD.dbo.TBS010 C 
	JOIN cd.SIBD.dbo.TBS0591 A ON
	C.PROCOD = A.PROCOD
	join cd.SIBD.dbo.TBS059 B  on
	A.NFEEMPCOD = B.NFEEMPCOD AND
	A.NFECOD = B.NFECOD AND
	A.NFENUM = B.NFENUM AND
	A.NFETIP = B.NFETIP AND
	A.SERCOD = B.SERCOD AND
	A.SEREMPCOD = B.SEREMPCOD
	
	WHERE B.NFETIP = 'N' AND B.NFECAN = 'N'
	GROUP BY A.PROCOD,NFEDES,C.PROUM1,B.NFENUM,B.NFENOM,A.NFEQTD,A.NFEQTDEMB,B.NFEDATENT),

TTÉ AS (SELECT 
	A.PROCOD collate Latin1_General_CI_AS AS 'CÓDIGO PRODUTO',
	'TTÉ' collate Latin1_General_CI_AS as 'Unidade',
	NFEDES collate Latin1_General_CI_AS AS 'DESCRIÇÃO',
	replace(str(isnull(A.NFEQTD*A.NFEQTDEMB,0),10,4),'.',',')collate Latin1_General_CI_AS AS 'QUANTIDADE',
	C.PROUM1 collate Latin1_General_CI_AS AS 'UNI MED.',
	B.NFENUM  AS 'NF',
	B.NFENOM collate Latin1_General_CI_AS AS 'NOME FORNECEDOR',
	CONVERT(NVARCHAR,B.NFEDATENT,2)collate Latin1_General_CI_AS AS 'DATA ENTRADA'
	
	FROM tt.SIBD.dbo.TBS010 C 
	JOIN tt.SIBD.dbo.TBS0591 A ON
	C.PROCOD = A.PROCOD
	join tt.SIBD.dbo.TBS059 B on
	A.NFEEMPCOD = B.NFEEMPCOD AND
	A.NFECOD = B.NFECOD AND
	A.NFENUM = B.NFENUM AND
	A.NFETIP = B.NFETIP AND
	A.SERCOD = B.SERCOD AND
	A.SEREMPCOD = B.SEREMPCOD
  
	WHERE B.NFETIP = 'N' AND B.NFECAN = 'N'
	GROUP BY A.PROCOD,NFEDES,C.PROUM1,B.NFENUM,B.NFENOM,A.NFEQTD,A.NFEQTDEMB,B.NFEDATENT)


select * from ND union 
select * from CD union
select * from TTÉ

order by  Unidade

declare @dataNFde datetime, @dataNFate datetime, @numNF int
set @dataNFde ='20150814'
set @dataNFate ='20150814'

SELECT 
	A.PROCOD  AS 'CÓDIGO PRODUTO',
	A.NFEDES AS 'DESCRIÇÃO',
	REPLACE(STR(ISNULL(A.NFEQTD*A.NFEQTDEMB,0),10,4),'.',',') AS 'QTD',
	C.PROUM1 AS 'UNI1',
	B.NFENUM  AS 'NF',
	B.NFECOD AS 'CÓD FOR',
	B.NFENOM AS 'NOME FORNECEDOR',
	CONVERT(NVARCHAR,B.NFEDATENT,2) AS 'DATA ENTRADA',
	B.NFEOBS AS 'OBS_NF'
	
FROM TBS010 C JOIN TBS0591 A ON
	C.PROCOD = A.PROCOD
	join TBS059 B  on 
	A.NFEEMPCOD = B.NFEEMPCOD AND
	A.NFECOD = B.NFECOD AND
	A.NFENUM = B.NFENUM AND
	A.NFETIP = B.NFETIP AND
	A.SERCOD = B.SERCOD AND
	A.SEREMPCOD = B.SEREMPCOD 
	
	
	
WHERE  B.NFECAN = 'N' AND B.NFECOD IN (8,669)AND
	B.NFETIP IN ('N','T') AND B.NFEDATENT  between @dataNFde and @dataNFate 
	or (@dataNFde = '' and @dataNFate = '')  
	
	
	
GROUP BY A.PROCOD,A.NFEDES,A.NFEQTD,A.NFEQTDEMB,C.PROUM1,B.NFENUM,B.NFENOM,B.NFEDATENT,B.NFEOBS,B.NFECOD
ORDER BY B.NFEDATENT
