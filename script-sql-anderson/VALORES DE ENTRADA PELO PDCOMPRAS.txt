SELECT 

convert(int,subString(convert(char(6),PDCDATCAD,112),1,4)) AS ANO,
convert(int,subString(convert(char(6),PDCDATCAD,112),5,2))AS MES ,
PROCOD AS CODIGO,


CASE WHEN PDCQTDENT > 0 THEN 
round(AVG(convert(decimal(10,4),(dbo.PDCTOTITE(TBS0451.PDCEMPCOD,TBS0451.PDCNUM,PDCITE) / PDCQTD )/ PDCQTDEMB)),2)
ELSE 0 END AS VALOR_ENT,
TBS0451.PDCNUM AS PEDCOMPRAS




FROM TBS0451(NOLOCK)FULL OUTER JOIN TBS045 (NOLOCK)ON TBS0451.PDCNUM = TBS045.PDCNUM
				
				

WHERE 
PDCDATCAD BETWEEN '20150101' AND '20151231' --AND PROCOD IS NULL
GROUP BY convert(char(6),PDCDATCAD,112), PROCOD,TBS0451.PDCNUM,PDCQTDENT

ORDER BY PROCOD,convert(char(6),PDCDATCAD,112)

SELECT  
RTRIM(LTRIM(PROCOD)) AS COD,
A.PRODES AS DES,
CASE WHEN len(A.MARCOD) = 4 
	THEN rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) 
	ELSE RIGHT(('00' + ltrim(str(A.MARCOD))), 3) + ' - ' + rtrim(A.MARNOM) 
END AS MARCA,		
A.PROUM1 AS  UNI1,
ANO,
MES,
REPLACE(STR(VALORMD,10,4),'.',',') AS VALORMD
--PEDCOMPRAS

FROM TBS010 A 
JOIN ENTRS B ON A.PROCOD = B.COD

WHERE 
VALORMD > 0

ORDER BY MES,PROCOD

SELECT convert(int,subString(convert(char(6),NFEDATENT,112),1,4)) AS ANO,
convert(int,subString(convert(char(6),NFEDATENT,112),5,2))AS MES ,
PROCOD , ISNULL(SUM(C.NFENFDQTD),0), A.NFENUM ,A.NFEITE, C.NFENUM,C.NFEITE, C.NFENFDNUM



FROM TBS0591 A (NOLOCK) 
						LEFT JOIN  TBS059 B(nolock)  on A.SERCOD=B.SERCOD and A.NFETIP=B.NFETIP and A.NFECOD=B.NFECOD and A.NFENUM=B.NFENUM
	  					LEFT JOIN  TBS0597 C(NOLOCK) ON A.NFENUM = C.NFENUM AND A.SERCOD=C.SERCOD and A.NFETIP=C.NFETIP AND A.NFECOD=C.NFECOD AND A.NFEITE=C.NFEITE
						  									
				

WHERE 
A.NFETIP='N' and NFEDATENT between '20150101' and '20151231' and
       NFENOM not like('BEST BAG%') and NFENOM not like('BEST OFFICE%') and 
	   NFENOM not like('MISASPEL%') and NFENOM not like('%PAPELYNA%') and NFENOM not like('TANBY%')
	   AND A.SERCOD NOT IN('CAN') 
GROUP BY convert(char(6),NFEDATENT,112),PROCOD,A.NFENUM ,A.NFEITE, C.NFENUM,C.NFEITE,C.NFENFDNUM
