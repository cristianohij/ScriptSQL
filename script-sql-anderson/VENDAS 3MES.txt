--CRIANDO %ACULMULADO 

WITH Res (ordem, CODIGO, DESCRICAO, SOMATT, total)
As
(SELECT ordem, CODIGO, DESCRICAO, SOMATT, SUM(SOMATT) OVER (PARTITION BY 1) As total FROM #ABC),

QPerc(ordem, CODIGO, DESCRICAO, SOMATT, total, Perc)
As
(SELECT ordem, CODIGO, DESCRICAO, SOMATT, total, SOMATT / total As Perc FROM Res),

QItens(itens)
as
(SELECT max(ordem) as itens FROM Res)

SELECT
    ordem, CODIGO, DESCRICAO, SOMATT, total, Perc,
    ROUND((SELECT SUM(TInt.Perc) FROM QPerc As TInt
        WHERE TInt.ordem <= TOut.ordem),4) As PercAcum,
    (select itens from QItens) as qtdeItens
into #VENDAS2        
FROM QPerc As TOut 

--SELECT * FROM #VENDAS2 ORDER BY SOMATT DESC 

-- CRIANDO ABC

;WITH Intervalos (grupo) As (
SELECT 0.80 UNION ALL	-- no exemplo estava 0.65
SELECT 0.95 UNION ALL	-- 0.90
SELECT 1.00)

SELECT
    ordem, CODIGO, DESCRICAO, SOMATT, Perc*100 as porcent, PercAcum*100 as porcent_acumulado,
    MIN(grupo) As grupo, CASE
        WHEN MIN(grupo) = 0.80 AND SOMATT > 0 THEN 'A'
        WHEN MIN(grupo) = 0.95 AND SOMATT > 0 THEN 'B'
        ELSE 'C' END As Intervalo,
    qtdeItens as itens
into #VENDAS3
FROM #VENDAS2 As C
INNER JOIN Intervalos As I ON C.PercAcum <= I.grupo
GROUP BY ordem, CODIGO, DESCRICAO, SOMATT, Perc, PercAcum, qtdeItens

--SELECT * FROM #VENDAS3 WHERE CODIGO = '0052434'

-- CRIANDO PORCENTAGENS DE VENDAS POR GRUPO ABC
select Intervalo,
       count(*)*100/itens AS PORCENT_ITENS,
       count(*) AS QTD_ITENS,
       case when Intervalo='A' then max(porcent_acumulado) when Intervalo='B' then max(porcent_acumulado) - 80 else max(porcent_acumulado) - 95 end AS PORCEN_VALOR,
       sum(SOMATT) AS VALOR,
       max(porcent_acumulado) AS '%ACU'
  from #VENDAS3
 group by Intervalo, itens
 order by Intervalo

										  where (Sg_UF = 'SJ' or Tp_Feriado = 1 )

										  and Nr_Dia between substring(convert(char(10),@data2de,103),1,2) and  substring(convert(char(10),@data2ate,103),1,2)) AS DIA2,

dbo.FCNDIASUTEIS (@data3de, @data3ate) - (SELECT count(*) FROM #FER
										  where (Sg_UF = 'SJ' or Tp_Feriado = 1 ) 
										  and (Nr_Ano between substring(convert(char(10),@data3de,103),7,4) and substring(convert(char(10),@data3ate,103),7,4) OR

										  and Nr_Dia between substring(convert(char(10),@data3de,103),1,2) and  substring(convert(char(10),@data3ate,103),1,2)) AS DIA3


into #ABC

where ((ISNULL(L.VL1,0) + ISNULL(L.VL2,0) + ISNULL(L.VL3,0)) <> 0 OR ISNULL(L.VND1,0) + ISNULL(L.VND2,0) + ISNULL(L.VND3,0) <> 0 OR

   drop table #ABCD

        WHERE TInt.ordem <= TOut.ordem),4) As PercAcum,

           WHERE TInt.ordem <= TOut.ordem),4)) < @pA

case when len(A.MARCOD) = 4 then rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) else right(('00' + ltrim(str(A.MARCOD))),3) + ' - ' + rtrim(A.MARNOM) end as 'MARCA',
A.PROUM1 collate Latin1_General_BIN as 'UN.1',

ISNULL(REPLACE(STR(SALDO.LOJA,10,4),'.',','),0) AS'SALDO LOJA' ,

ISNULL(REPLACE(STR(SALDO.RES,10,4),'.',','),0) AS RESER,

ISNULL(REPLACE(STR(SALDO.CMP,10,4),'.',','),0) AS COMPRAS,
ISNULL(REPLACE(STR(CD.SALTT,10,4),'.',','),0) AS 'SALDO CD',

-- SOMA AS QTDS VENDIDAS / PELA QUANTIDADE DE DIAS UTEIS, CASO >0 , SOMA SALDO TOTAL + COMPRAS + SALDO CD - PENDENTE / (VEN_UTEIS) 
CASE WHEN ISNULL((SOMAQTD / 
				((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0)> 0 
										  
	THEN str(round(((ISNULL(SALDO.SALTT,0)+ ISNULL(SALDO.CMP,0) + ISNULL(CD.SALTT,0)) - ISNULL(SALDO.PEN,0))/
	ISNULL((SOMAQTD / 
				((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0),0),10,0)
	ELSE 0 END AS 'TEMPO RESTANTE',


-- USUARIO EESCOLHE A QUANTIDADE DE DIAS UTEIS PARA REPOSIÇÃO 
-- @DIAS - (SOMA AS QTDS VENDIDAS / PELA QUANTIDADE DE DIAS UTEIS, CASO >0 , 
-- SOMA SALDO TOTAL + COMPRAS + SALDO CD - PENDENTE / (QTDS VENDAS / PELA QUANTIDADE DE DIAS UTEIS)) * VEN_UTEIS CASO <=0.9 É 0 
-- ELSE @DIAS - (SOMA AS QTDS VENDIDAS / PELA QUANTIDADE DE DIAS UTEIS, CASO >0 , 
-- SOMA SALDO TOTAL + COMPRAS + SALDO CD - PENDENTE / (QTDS VENDAS / PELA QUANTIDADE DE DIAS UTEIS)) (

CASE WHEN ((@DIAS - (CASE WHEN ISNULL((SOMAQTD / 
					((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0) > 0 
										  
					THEN str(round(((ISNULL(SALDO.SALTT,0)+ ISNULL(SALDO.CMP,0)+ ISNULL(CD.SALTT,0)) - ISNULL(SALDO.PEN,0))/
					ISNULL((SOMAQTD / 
					((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0),0),10,0)
					ELSE 0 END)) * ISNULL((SOMAQTD / 
								((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0)) <=0.9 
	THEN 0 
	ELSE ((@DIAS - (CASE WHEN ISNULL((SOMAQTD / 
					((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0) > 0 
										  
					THEN str(round(((ISNULL(SALDO.SALTT,0)+ ISNULL(SALDO.CMP,0) + ISNULL(CD.SALTT,0)) - ISNULL(SALDO.PEN,0))/
					ISNULL((SOMAQTD / 
					((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0),0),10,0)
					ELSE 0 END)) * ISNULL((SOMAQTD / 
								((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0))
					
END AS 'QTD PARA REPOR' ,

		OR B.SOMATT <> 0 OR CD.SALTT <> 0 OR CD.RES <> 0 OR CD.CMP <> 0 OR CD.PEN <>0) AND PROCOD IS NOT NULL

	AND(A.PRODES LIKE(CASE WHEN @DESCRI = '' THEN '% %' ELSE Upper(RTRIM(@DESCRI))+'%' END))
