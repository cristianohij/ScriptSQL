EST3.COD AS "3", 
EST4.COD AS "4",

  FULL OUTER JOIN (SELECT PROCOD AS COD FROM TBS032 (NOLOCK) WHERE ESTLOC = 2 ) AS EST2 ON EST2.COD = A.PROCOD
  FULL OUTER JOIN (SELECT PROCOD AS COD FROM TBS032 (NOLOCK) WHERE ESTLOC = 3 ) AS EST3 ON EST3.COD = A.PROCOD
  FULL OUTER JOIN (SELECT PROCOD AS COD FROM TBS032 (NOLOCK) WHERE ESTLOC = 4 ) AS EST4 ON EST4.COD = A.PROCOD
  FULL OUTER JOIN (SELECT PROCOD AS COD FROM TBS032 (NOLOCK) WHERE ESTLOC = 5 ) AS EST5 ON EST5.COD = A.PROCOD
  FULL OUTER JOIN (SELECT PROCOD AS COD FROM TBS032 (NOLOCK) WHERE ESTLOC = 6 ) AS EST6 ON EST6.COD = A.PROCOD
  FULL OUTER JOIN (SELECT PROCOD AS COD FROM TBS032 (NOLOCK) WHERE ESTLOC = 7 ) AS EST7 ON EST7.COD = A.PROCOD
 
WHERE ESTLOC = 1 AND (EST2.COD IS NULL 
OR EST3.COD IS NULL 
OR EST4.COD IS NULL 
OR EST5.COD IS NULL 
OR EST5.COD IS NULL 
OR EST6.COD IS NULL 
OR EST7.COD IS NULL )
