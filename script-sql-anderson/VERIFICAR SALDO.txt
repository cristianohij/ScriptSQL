WITH 
-- QUANTIDADE DE SAIDA
SAIDA AS (SELECT PROCOD AS COD ,
SUM((C.NFSQTD-C.NFSQTDDEV)*NFSQTDEMB) AS QTDSAI  FROM SIBDEST.dbo.TBS0671 C (nolock) JOIN SIBDEST.dbo.TBS067 B (nolock)  ON 
										C.NFSEMPCOD = B.NFSEMPCOD and 
										C.NFSNUM = B.NFSNUM and 
										C.SNEEMPCOD = B.SNEEMPCOD and 
										C.SNESER = B.SNESER
										WHERE NFSDATEMI between '20160101' and getdate() 
										and  NFSCAN = 'N' 
										
										GROUP BY PROCOD) ,
-- QUANTIDADE DE DEVOLUÇÃO
DEV AS ( SELECT A.PROCOD AS COD , SUM(NFDQTD*NFDQTDEMB) AS QTDDEV


		 FROM SIBDEST.dbo.TBS1172 A INNER JOIN SIBDEST.dbo.TBS117 B ON 
		 A.NFDEMPCOD = B.NFDEMPCOD AND
		 A.NFDNUM = B.NFDNUM AND
		 A.SNEEMPCOD = A.SNEEMPCOD AND
		 A.SNESER = B.SNESER
	 	 WHERE B.NFDDATEMI BETWEEN '20160101' AND GETDATE() AND B.NFDSTATUS = 'A' 

		 GROUP BY A.PROCOD )  ,
										
-- QUANTIDADE DE ENTRADA
ENTR AS (SELECT A.PROCOD AS COD,
		 SUM(A.NFEQTD*A.NFEQTDEMB) AS QTDENT
	
		 from SIBDEST.dbo.TBS0591 A
		 join SIBDEST.dbo.TBS059 B  on
		 A.NFEEMPCOD = B.NFEEMPCOD AND
		 A.NFECOD = B.NFECOD AND
		 A.NFENUM = B.NFENUM AND
		 A.NFETIP = B.NFETIP AND
		 A.SERCOD = B.SERCOD AND
		 A.SEREMPCOD = B.SEREMPCOD
	
		 WHERE B.NFECAN = 'N' AND NFEDATENT between '20160101' and getdate()-- AND PROCOD= '0128782'
		 GROUP BY A.PROCOD),
	
-- QUANTIDADE DAS MOVIMETAÇOES DO ESTOQUE	
MVI AS ( SELECT PROCOD AS COD,

		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM SIBDEST.dbo.TBS0371 P JOIN SIBDEST.dbo.TBS037 Q ON P.MVIDOC = Q.MVIDOC
		WHERE Q.MVIDATEFE BETWEEN '20160101' AND GETDATE() AND TMVCOD <=499 AND MVILOCDES = 1 AND MVILOCORI = 0 AND Z.PROCOD = P.PROCOD),0)-

		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM SIBDEST.dbo.TBS0371 R JOIN SIBDEST.dbo.TBS037 S ON R.MVIDOC = S.MVIDOC
		WHERE S.MVIDATEFE BETWEEN '20160101' AND GETDATE() AND TMVCOD >=500 AND MVILOCORI = 1 AND MVILOCDES= 0 AND Z.PROCOD = R.PROCOD),0) AS MVIEST ,-- MOVIMENTOS DO ESTOQUE
		
		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM SIBDEST.dbo.TBS0371 T JOIN SIBDEST.dbo.TBS037 U ON T.MVIDOC = U.MVIDOC
		WHERE U.MVIDATEFE BETWEEN '20160101' AND GETDATE() AND TMVCOD <=499 AND MVILOCDES = 2 AND MVILOCORI = 0 AND Z.PROCOD = T.PROCOD),0)- 

		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM SIBDEST.dbo.TBS0371 V JOIN SIBDEST.dbo.TBS037 X ON V.MVIDOC = X.MVIDOC
		WHERE X.MVIDATEFE BETWEEN '20160101' AND GETDATE() AND TMVCOD >=500 AND MVILOCORI = 2 AND MVILOCDES= 0 AND Z.PROCOD = V.PROCOD),0)AS MVILOJA ,-- MOVIMENTOS DA LOJA
		
		
		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM SIBDEST.dbo.TBS0371 P JOIN SIBDEST.dbo.TBS037 Q ON P.MVIDOC = Q.MVIDOC
		WHERE Q.MVIDATEFE BETWEEN '20160101' AND GETDATE() AND TMVCOD <=499 AND MVILOCDES = 4 AND MVILOCORI = 0 AND Z.PROCOD = P.PROCOD),0)-

		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM SIBDEST.dbo.TBS0371 R JOIN SIBDEST.dbo.TBS037 S ON R.MVIDOC = S.MVIDOC
		WHERE S.MVIDATEFE BETWEEN '20160101' AND GETDATE() AND TMVCOD >=500 AND MVILOCORI = 4 AND MVILOCDES= 0 AND Z.PROCOD = R.PROCOD),0) AS MVI4 ,-- MOVIMENTOS DO ESTOQUE 4
		
		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM SIBDEST.dbo.TBS0371 T JOIN SIBDEST.dbo.TBS037 U ON T.MVIDOC = U.MVIDOC
		WHERE U.MVIDATEFE BETWEEN '20160101' AND GETDATE() AND TMVCOD <=499 AND MVILOCDES = 5 AND MVILOCORI = 0 AND Z.PROCOD = T.PROCOD),0)- 

		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM SIBDEST.dbo.TBS0371 V JOIN SIBDEST.dbo.TBS037 X ON V.MVIDOC = X.MVIDOC
		WHERE X.MVIDATEFE BETWEEN '20160101' AND GETDATE() AND TMVCOD >=500 AND MVILOCORI = 5 AND MVILOCDES= 0 AND Z.PROCOD = V.PROCOD),0)AS MVI5 , -- MVI EST 5
		
		
		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM SIBDEST.dbo.TBS0371 P JOIN SIBDEST.dbo.TBS037 Q ON P.MVIDOC = Q.MVIDOC
		WHERE Q.MVIDATEFE BETWEEN '20160101' AND GETDATE() AND TMVCOD <=499 AND MVILOCDES = 6 AND MVILOCORI = 0 AND Z.PROCOD = P.PROCOD),0)-

		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM SIBDEST.dbo.TBS0371 R JOIN SIBDEST.dbo.TBS037 S ON R.MVIDOC = S.MVIDOC
		WHERE S.MVIDATEFE BETWEEN '20160101' AND GETDATE() AND TMVCOD >=500 AND MVILOCORI = 6 AND MVILOCDES= 0 AND Z.PROCOD = R.PROCOD),0) AS MVI6 ,-- MVI EST 6
		
		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM SIBDEST.dbo.TBS0371 T JOIN SIBDEST.dbo.TBS037 U ON T.MVIDOC = U.MVIDOC
		WHERE U.MVIDATEFE BETWEEN '20160101' AND GETDATE() AND TMVCOD <=499 AND MVILOCDES = 7 AND MVILOCORI = 0 AND Z.PROCOD = T.PROCOD),0)- 

		ISNULL((SELECT SUM(MVIQTDATD*MVIQTDEMB) FROM SIBDEST.dbo.TBS0371 V JOIN SIBDEST.dbo.TBS037 X ON V.MVIDOC = X.MVIDOC
		WHERE X.MVIDATEFE BETWEEN '20160101' AND GETDATE() AND TMVCOD >=500 AND MVILOCORI = 7 AND MVILOCDES= 0 AND Z.PROCOD = V.PROCOD),0)AS MVI7 -- MVI EST 7
		
		FROM SIBDEST.dbo.TBS010 Z )
	
		
--SELECT PROCOD, ESTQTDATU FROM TBS032 WHERE ESTLOC = 2 AND ESTQTDATU > 0
---	SELECT PROCOD ,ESTLOC,SUM(ESTQTDATU) FROM SIBDEST.dbo.TBS032 WHERE PROCOD = '0128782' GROUP BY PROCOD,ESTLOC

SELECT PROCOD AS COD ,--COUNT(PROCOD) 
SUM(ESTQTDATU) ATU, ISNULL(QTDSAI,0) SAI, ISNULL(QTDENT,0) ENT, ISNULL(MVIEST,0) MVEST, ISNULL(MVILOJA,0) MVLOJ, ISNULL(QTDDEV,0) DEV,
ISNULL(MVI4,0) AS MVI4, ISNULL(MVI5,0) AS MVI5, ISNULL(MVI6,0) AS MVI6, ISNULL(MVI7,0) AS MVI7,
SUM(ESTQTDATU) + ISNULL(QTDSAI,0) - ISNULL(QTDENT,0) - ISNULL(MVIEST,0) - ISNULL(MVILOJA,0) + ISNULL(QTDDEV,0)-
ISNULL(MVI4,0) - ISNULL(MVI5,0) - ISNULL(MVI6,0) - ISNULL(MVI7,0) AS SALDO
--INTO SALDOEST2015
FROM SIBDEST.dbo.TBS032 T FULL OUTER JOIN ENTR E ON T.PROCOD = E.COD
		 	  FULL OUTER JOIN SAIDA S ON T.PROCOD = S.COD
			  FULL OUTER JOIN MVI M ON T.PROCOD = M.COD
			  FULL OUTER JOIN DEV D ON T.PROCOD = D.COD
			  

--WHERE PROCOD = '4250435'
GROUP BY PROCOD,QTDSAI,QTDENT,MVILOJA,MVIEST,QTDDEV,MVI4,MVI5,MVI6, MVI7
