-- CRIANDO TABELA TEMP COM OS VALORES REAIS DE DESCONTO
if object_id('TempDB.dbo.#TMP') is not null
   drop table #TMP
   
	select
	M2_PROCOD,
	M2_VALBRU - M2_ABT + M2_ACRCUP - 
	CASE WHEN  A.DESCONTO <> ISNULL(B.DESCONTO,0) AND M2_ITE = (SELECT TOP 1 M2_ITE 
								    FROM #MSLAUX E 
									
								    WHERE 
								    A.M2_COO = E.M2_COO and A.M2_ECF = E.M2_ECF AND A.M2_CUP = E.M2_CUP 
																
								    ORDER BY 
								    M2_DESCUP DESC)
		THEN
			CASE WHEN A.DESCONTO > ISNULL(B.DESCONTO,0) 
				THEN ROUND(M2_DESCUP,2) - (A.DESCONTO - ISNULL(B.DESCONTO,0))
				ELSE ROUND(M2_DESCUP,2) + (ISNULL(B.DESCONTO,0) - A.DESCONTO)
			END
		ELSE ROUND(M2_DESCUP,2)
	END AS M2_LIQUIDO,
	M2_QTD
	
	INTO #TMP	
	FROM (
		SELECT 
		M2_ITE,M2_PROCOD,M2_QTD,
		M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , 
		M2_DESCUP ,
		M2_ACRCUP,
		M2_ABT ,
		M2_VALBRU,
		SUM(ISNULL(ROUND(M2_DESCUP,2),0)) OVER (PARTITION BY M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA ) AS DESCONTO
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 1 ) AS A 
			
		LEFT JOIN (
		SELECT 
		M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , M2_VALBRU AS DESCONTO 
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 2 ) AS B ON A.M2_LOJ = B.M2_LOJ AND A.M2_CUP = B.M2_CUP AND A.M2_COO= B.M2_COO AND 
														A.M2_ECF = B.M2_ECF AND A.M2_CXA = B.M2_CXA

   drop table #TBS010

SELECT 
A.PROCOD,
PRODES,
CASE WHEN len(A.MARCOD) = 4 
	then rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) 
	else right(('00' + ltrim(str(A.MARCOD))),3) + ' - ' + rtrim(A.MARNOM) 
end as 'MARCA',
PROUM1,
GRUCOD,
SUBGRUCOD,
CONVERT(INT,RTRIM(LTRIM(STR(GRUCOD,2)))+RTRIM(LTRIM(STR(SUBGRUCOD,2)))) as SUBGRUPO

INTO #TBS010
FROM TBS010 A (NOLOCK) 

WHERE 
A.MARCOD =(CASE WHEN @marcod = 0 THEN A.MARCOD ELSE Upper(RTRIM(@marcod)) END) AND 
A.MARNOM LIKE(CASE WHEN @marcanom = '' THEN '% %' ELSE Upper(RTRIM(@marcanom))+'%' END) AND
CONVERT(INT,A.PROCOD) IN (
SELECT DISTINCT PROCOD COLLATE DATABASE_DEFAULT 
FROM #CORP 
UNION 
SELECT DISTINCT PROCOD 
FROM #DEV 
UNION 
SELECT DISTINCT PROCOD FROM #NFSGRU 
UNION 
SELECT DISTINCT PROCOD FROM #LOJA)

select
A.PROCOD,
PRODES,
MARCA,
rtrim(E.GRUDES)+' ('+Ltrim(str(E.GRUCOD,3))+')' as 'grupo',
ISNULL(rtrim(F.SUBGRUDES),'')+' ('+ISNULL(Ltrim(str(F.SUBGRUCOD,3)),0)+')' as 'subgrupo',
PROUM1,
ISNULL(B.PREMED,0) AS LOJPREMED,
ISNULL(B.QTD,0) AS LOJQTDVEN,
ISNULL(B.VAL,0) AS LOJVALVEN,
ISNULL(C.NFSPREMED,0) AS NFSPREMED, 
ISNULL(C.NFSQTDVEN,0) AS NFSQTDVEN,
ISNULL(C.NFSTOTITEST,0) AS NFSTOTITEST,
ISNULL(G.NFSPREMED,0) AS NFSPREMEDGRU, 
ISNULL(G.NFSQTDVEN,0) AS NFSQTDVENGRU,
ISNULL(G.NFSTOTITEST,0) AS NFSTOTITESTGRU,
ISNULL(D.NFEPREMED,0) AS NFEPREMED,
ISNULL(D.QTD,0) AS NFDQTDDEV,
ISNULL(D.VAL,0) AS NFDVALDEV

FROM #TBS010 A 
LEFT JOIN #LOJA B ON CONVERT(INT,A.PROCOD) = B.PROCOD
LEFT JOIN #CORP C ON A.PROCOD = C.PROCOD
LEFT JOIN #DEV D ON A.PROCOD COLLATE DATABASE_DEFAULT  = D.PROCOD
LEFT JOIN TBS012 E (NOLOCK) ON A.GRUCOD = E.GRUCOD
LEFT JOIN TBS0121 F (NOLOCK) ON A.GRUCOD = F.GRUCOD AND A.SUBGRUCOD = F.SUBGRUCOD
LEFT JOIN #NFSGRU G ON A.PROCOD COLLATE DATABASE_DEFAULT = G.PROCOD

WHERE 
(ISNULL(B.QTD,0) <> 0 OR ISNULL(C.NFSQTDVEN,0) <> 0 OR ISNULL(D.QTD,0) <> 0 OR ISNULL(G.NFSQTDVEN,0) <> 0) AND
A.PROCOD LIKE( CASE WHEN @PROCOD = '' THEN '% %' ELSE RTRIM(@PROCOD) + '%' END) AND
-- E.GRUCOD IN (@GRUCOD) AND
-- SUBGRUPO IN (@SUBGRUCOD) AND
A.PRODES LIKE (CASE WHEN @PRODES = '' THEN '% %' ELSE UPPER(RTRIM(@PRODES))+'%' END)
