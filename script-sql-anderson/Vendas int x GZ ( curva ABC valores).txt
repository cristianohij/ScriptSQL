declare @DATA1 DATE, @DATE2 DATE
SET @DATA1 = '20150801'
SET @DATE2 = '20151031'

WITH 
VENDASGZ AS (select PROCOD,
SUM(M2_VALTOT)-SUM(M2_ABT) AS VALOR_VENDAS_GZ
from MSL002 A join TBS010 B ON A.M2_PROCOD = B.PROCOD

where M2_DAT between @DATA1 and @DATE2 and
M2_PROCOD LIKE ('%') AND 
M2_TIPREG = '01' AND 
M2_REGCAN = 'F'

GROUP BY M2_PROCOD,PRODES,PROCOD,MARCOD),

VENDASINTEGROS AS (SELECT A.PROCOD,
SUM(CASE WHEN B.NFSCLINOM NOT LIKE('TANBY%') THEN dbo.NFSTOTITE(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE) ELSE 0 END) AS VALOR_VENDAS_INT
FROM TBS0671 A 
	JOIN TBS067 B ON 
	A.NFSEMPCOD = B.NFSEMPCOD and 
	A.NFSNUM = B.NFSNUM and 
	A.SNEEMPCOD = B.SNEEMPCOD and 
	A.SNESER = B.SNESER

WHERE A.PROCOD LIKE ('%')AND 
NFSDATEMI between @DATA1 and @DATE2 and
NFSTIP = 'N' and NFSCAN = 'N' 


GROUP BY A.PROCOD)

SELECT rtrim(ltrim(A.PROCOD)) AS CÓD,
replace(str(isnull(rtrim(B.VALOR_VENDAS_GZ),0),10,2),'.',',') as 'Valor GZ',
replace(str(isnull(rtrim(C.VALOR_VENDAS_INT),0),10,2),'.',',') as 'Valor Int',
replace(str(isnull(rtrim(C.VALOR_VENDAS_INT+B.VALOR_VENDAS_GZ),0),10,2),'.',',') as 'Valor Total'
FROM TBS010 A

Full Outer Join VENDASGZ B ON A.PROCOD = B.PROCOD 
Full Outer Join VENDASINTEGROS C ON A.PROCOD = C.PROCOD

WHERE A.PROCOD LIKE ('%') 

GROUP BY A.PROCOD,B.VALOR_VENDAS_GZ,
C.VALOR_VENDAS_INT

order by A.PROCOD
