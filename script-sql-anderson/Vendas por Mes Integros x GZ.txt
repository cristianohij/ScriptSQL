-- contabilizar vendas CORPORATIVO

if object_id('TempDB.dbo.#CORP') is not null
   drop table #CORP

SELECT 
CASE WHEN A.NFSCLICOD IN(SELECT CLICOD FROM #GRUPOCLI)
	THEN 0
	ELSE 1
END AS GRUPO,
SUBSTRING(CONVERT(CHAR (10), A.NFSDATEMI,103),4,7) AS NFSDAT,
CONVERT(CHAR(7),A.NFSDATEMI,102) AS DATA,
PROCOD,
sum(B.NFSQTD * B.NFSQTDEMB) AS NFSQTDVEN

into #CORP
FROM 
TBS0671 B (nolock)
LEFT JOIN TBS067 A (nolock) ON A.NFSEMPCOD = B.NFSEMPCOD and A.NFSNUM = B.NFSNUM and A.SNEEMPCOD = B.SNEEMPCOD and A.SNESER = B.SNESER
INNER JOIN #TBS080 D (NOLOCK) ON B.NFSNUM = D.ENFNUM AND B.SNEEMPCOD = D.SNEEMPCOD AND B.SNESER = D.SNESER
			  
WHERE 
NFSDATEMI between @data1de and @data1ate and 
NFSTIP IN('N','C') and 
D.ENFTIPDOC = 1 AND
B.NFSCFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S') AND
A.NFSCLINOM NOT LIKE ('TANBY%') 		-- TIRO AS TRANSFERENCIAS

GROUP BY 
PROCOD, NFSCLICOD, CONVERT(CHAR(7),A.NFSDATEMI,102), SUBSTRING(CONVERT(CHAR (10), A.NFSDATEMI,103),4,7)

-- DEVOLUÇÃO

if object_id('TempDB.dbo.#DEV') is not null
   drop table #DEV

SELECT
CASE WHEN B.NFECOD IN (SELECT CLICOD FROM #GRUPOCLI WHERE CLICOD NOT IN(@EMPCOD))
	THEN 0
	ELSE 1
END AS GRUPO,
SUBSTRING(CONVERT(CHAR (10), B.NFEDATEFE,103),4,7) AS NFEDAT,
CONVERT(CHAR(7),B.NFEDATEFE,102) AS DATA,
PROCOD,
SUM(NFEQTD*NFEQTDEMB) AS QTD

INTO #DEV		
FROM TBS0591 A (NOLOCK) 
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND 
							   A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND A.SEREMPCOD = B.SEREMPCOD
LEFT JOIN #TBS080 D (NOLOCK) ON A.NFENUM = D.ENFNUM AND A.NFECOD = D.ENFCODDES		

WHERE 
NFEDATEFE between @data1de and @data1ate and 
A.NFETIP = 'D' and 
((D.ENFFINEMI = 4 AND D.ENFTIPDOC = 0) OR (NFENOSFOR <> 'S' AND B.NFECAN <> 'S' AND NFEDATEFE <> '' )) AND
A.NFECFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E') 		

GROUP BY 
PROCOD, B.NFECOD, CONVERT(CHAR(7),B.NFEDATEFE,102), SUBSTRING(CONVERT(CHAR (10), B.NFEDATEFE,103),4,7)

if object_id('TempDB.dbo.#TBS010') is not null
   drop table #TBS010

SELECT 
A.PROCOD,
PRODES,
CASE WHEN len(A.MARCOD) = 4 
	then rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) 
	else right(('00' + ltrim(str(A.MARCOD))),3) + ' - ' + rtrim(A.MARNOM) 
end as 'MARCA',
PROUM1,
CONVERT(INT,RTRIM(LTRIM(STR(GRUCOD,2)))+RTRIM(LTRIM(STR(SUBGRUCOD,2)))) as SUBGRUPO

INTO #TBS010
FROM TBS010 A (NOLOCK) 

WHERE 
A.MARCOD =(CASE WHEN @marcod = 0 THEN A.MARCOD ELSE Upper(RTRIM(@marcod)) END) AND 
A.MARNOM LIKE(CASE WHEN @marcanom = '' THEN '% %' ELSE Upper(RTRIM(@marcanom))+'%' END) AND
CONVERT(INT,A.PROCOD) IN (
SELECT DISTINCT CONVERT(INT,PROCOD)
FROM #TBS067 
UNION 
SELECT DISTINCT PROCOD FROM #LOJA
UNION
SELECT DISTINCT PROCOD FROM #TBS059)



select DISTINCT DATA INTO #DATA FROM #LOJA
UNION 
select DISTINCT DATA FROM #TBS067
UNION 
select DISTINCT DATA FROM #TBS059


SELECT * FROM #TBS010,#DATA


-------------------------------------------------------------------------------------------------------------------------------------------------

select
ISNULL(ISNULL(B.NFSDAT, C.NFSDAT),D.NFEDAT) AS MESANO,
ISNULL(ISNULL(B.DATA, C.DATA),D.DATA) AS DATA,
A.PROCOD,
PRODES,
MARCA,
PROUM1,
ISNULL(B.QTD,0) AS LOJQTDVEN,
ISNULL(C.NFSQTDVEN,0) AS NFSQTDVEN,
ISNULL(D.QTD,0) AS DEVQTD

FROM #TBS010 A 
full outer join #LOJA B ON CONVERT(INT,A.PROCOD) = B.PROCOD
full outer join #TBS067 C ON A.PROCOD = C.PROCOD AND B.DATA = C.DATA
full outer join #TBS059 D ON A.PROCOD = D.PROCOD AND ISNULL(B.DATA,C.DATA) = D.DATA

WHERE 
(ISNULL(B.QTD,0) <> 0 OR ISNULL(C.NFSQTDVEN,0) <> 0 OR ISNULL(D.QTD,0) <> 0) AND
A.PROCOD LIKE( CASE WHEN @PROCOD = '' THEN '% %' ELSE RTRIM(@PROCOD) + '%' END) AND
A.PRODES LIKE (CASE WHEN @PRODES = '' THEN '% %' ELSE UPPER(RTRIM(@PRODES))+'%' END)

ORDER BY 
PROCOD , ISNULL(ISNULL(B.DATA, C.DATA),D.DATA)
