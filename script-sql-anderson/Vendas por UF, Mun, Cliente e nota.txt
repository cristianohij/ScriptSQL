-- CRIANDO TABELA TEMP COM OS VALORES REAIS DE DESCONTO

if object_id('TempDB.dbo.#TMP') is not null
   drop table #TMP
   
select
A.M2_CXA,
A.M2_ECF,
A.M2_CUP,
A.M2_COO,
M2_PROCOD,
M2_PRE,
M2_QNT,
M2_VALBRU,
M2_VALBRU - M2_ABT + M2_ACRCUP - 
CASE WHEN  A.DESCONTO <> ISNULL(B.DESCONTO,0) AND M2_ITE = (SELECT TOP 1 M2_ITE 
															FROM #MSLAUX E 
																
															WHERE 
															A.M2_COO = E.M2_COO AND 
															A.M2_ECF = E.M2_ECF AND 
															A.M2_CUP = E.M2_CUP 
																
															ORDER BY 
															M2_DESCUP DESC)
																
	THEN
		CASE WHEN A.DESCONTO > ISNULL(B.DESCONTO,0) 
			THEN ROUND(M2_DESCUP,2) - (A.DESCONTO - ISNULL(B.DESCONTO,0))
			ELSE ROUND(M2_DESCUP,2) + (ISNULL(B.DESCONTO,0) - A.DESCONTO)
		END
	ELSE ROUND(M2_DESCUP,2)
END AS M2_LIQUIDO
	
INTO #TMP	
FROM (
	SELECT
	M2_DAT, 
	M2_ITE,M2_PROCOD,M2_TRI,M2_CST,
	M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , 
	M2_DESCUP , M2_PRE, M2_QNT,
	M2_ACRCUP,
	M2_ABT ,
	M2_VALBRU,
	SUM(ISNULL(ROUND(M2_DESCUP,2),0)) OVER (PARTITION BY M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA ) AS DESCONTO
				
	FROM 
	#MSLAUX 
				
	WHERE
	M2_STATUS = 1 ) AS A 
		
LEFT JOIN (
	SELECT 
	M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , M2_VALBRU AS DESCONTO 
				
	FROM 
	#MSLAUX 
				
	WHERE
	M2_STATUS = 2 ) AS B ON A.M2_LOJ = B.M2_LOJ AND A.M2_CUP = B.M2_CUP AND A.M2_COO= B.M2_COO AND 
														A.M2_ECF = B.M2_ECF AND A.M2_CXA = B.M2_CXA  
		
LEFT JOIN #NFSLOJA D (NOLOCK) ON A.M2_COO = D.COO AND A.M2_ECF = D.ECF
	
WHERE 
D.COO IS NULL

-- contabilizar vendas CORPORATIVO

if object_id('TempDB.dbo.#CORP') is not null
   drop table #CORP

SELECT
A.UFESIG AS UF,
C.MUNCOD,
(SELECT MUNNOM FROM TBS003 E (NOLOCK) WHERE E.MUNCOD = C.MUNCOD) AS MUNNOM,
NFSCLICOD,
NFSCLINOM,
CONVERT(VARCHAR(10),A.NFSNUM) AS NFSNUM,
B.PROCOD,
B.NFSPRODES,
F.PROUM1,
SUM(ROUND(dbo.NFSTOTITEST(B.NFSEMPCOD,B.NFSNUM,B.SNEEMPCOD,B.SNESER,B.NFSITE),2)) /sum(B.NFSQTD * B.NFSQTDEMB) AS NFSPREMED,
SUM(B.NFSQTD * B.NFSQTDEMB) AS NFSQTDVEN,
SUM(ROUND(dbo.NFSTOTITEST(B.NFSEMPCOD,B.NFSNUM,B.SNEEMPCOD,B.SNESER,B.NFSITE),2)) AS NFSTOTITEST,
SUM(ROUND(dbo.NFSVDDITE(B.NFSEMPCOD, B.NFSNUM, B.SNEEMPCOD, B.SNESER,B.NFSITE),2)) AS NFSVDDTOT,
SUM(NFSQTDDEV) AS NFSQTDDEV, 
SUM(ROUND(dbo.NFSVALDEV(B.NFSEMPCOD, B.NFSNUM, B.SNEEMPCOD, B.SNESER,B.NFSITE),2)) AS NFSTOTDEV

into #CORP
FROM 
TBS0671 B (nolock)
LEFT JOIN TBS067 A (nolock) ON A.NFSEMPCOD = B.NFSEMPCOD and A.NFSNUM = B.NFSNUM and A.SNEEMPCOD = B.SNEEMPCOD and A.SNESER = B.SNESER
INNER JOIN #TBS080 D (NOLOCK) ON B.NFSNUM = D.ENFNUM AND B.SNEEMPCOD = D.SNEEMPCOD AND B.SNESER = D.SNESER
LEFT JOIN TBS002 C (NOLOCK)   ON A.NFSCLICOD = C.CLICOD
LEFT JOIN TBS010 F (NOLOCK) ON B.PROCOD = F.PROCOD
		  
WHERE 
NFSDATEMI between @data1de and @data1ate and 
-- NFSTIP IN('N','C') and 
B.NFSCFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S') AND
A.NFSCLINOM NOT LIKE ('TANBY%') AND 		-- TIRO AS TRANSFERENCIAS
NFSCLICOD NOT IN ((SELECT CLICOD FROM #GRUPOCLI)) --AND 
--A.UFESIG IN (@UFESIG) AND 
--C.MUNCOD IN (@MUNCOD) 

GROUP BY 
A.UFESIG,
C.MUNCOD,
NFSCLICOD,
NFSCLINOM,
A.NFSNUM,
B.PROCOD,
B.NFSPRODES,
F.PROUM1

UNION 

SELECT * FROM #LIQUIDO



SELECT * FROM #CORP
