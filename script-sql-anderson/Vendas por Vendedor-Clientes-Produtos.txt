if object_id('TempDB.dbo.#CLI') is not null
   drop table #CLI

SELECT 
CLICOD, 
RTRIM(LTRIM(CLINOM)) AS CLINOM,
ISNULL(RTRIM(LTRIM(STR(A.VENCOD))) + ' - ' + RTRIM(LTRIM(VENNOM)), '0 - SEM VENDEDOR') AS VENNOM,
ISNULL(RTRIM(LTRIM(STR(B.GVECOD))) + ' - ' + RTRIM(LTRIM(C.GVEDES)), '0 - SEM GRUPO') AS GVDES

INTO #CLI
FROM 
TBS002 A (NOLOCK)
LEFT JOIN 
TBS004 B (NOLOCK) ON A.VENCOD = B.VENCOD AND A.VENEMPCOD = B.VENEMPCOD
LEFT JOIN 
TBS091 C (NOLOCK) ON B.GVECOD = C.GVECOD AND B.GVEEMPCOD = C.GVEEMPCOD

WHERE
CLINOM LIKE(CASE WHEN @nomecli = '' THEN CLINOM ELSE rtrim(ltrim(upper(@nomecli)))+'%' END) AND 
CLICOD = CASE WHEN @codcli = 0 THEN CLICOD ELSE rtrim(ltrim(@codcli)) END AND
CLICGC NOT LIKE ('65069593000%') AND
CLICGC NOT LIKE('05118717000%') AND
CLICGC NOT LIKE('44125185000%') AND
CLICGC NOT LIKE('52080207000%') AND 
ISNULL(RTRIM(LTRIM(STR(A.VENCOD))) + ' - ' + RTRIM(LTRIM(VENNOM)), '0 - SEM VENDEDOR') 
LIKE (CASE WHEN @VENNOM = '' 
		THEN ISNULL(RTRIM(LTRIM(STR(A.VENCOD))) + ' - ' + RTRIM(LTRIM(VENNOM)), '0 - SEM VENDEDOR')
		ELSE upper(@VENNOM)
	END) /*AND 
ISNULL(B.GVECOD,0) IN (@GVECOD)		*/ -- uso no reports

ORDER BY 
CLICOD

SELECT
E.GVDES, 
E.VENNOM,
NFSCLICOD AS 'COD CLIENTE',
E.CLINOM AS 'NOME CLIENTE',
RTRIM(LTRIM(A.PROCOD)) AS 'CODIGO',
RTRIM(LTRIM(NFSPRODES)) AS 'DESCRIÇÃO',
SUM(A.NFSQTD*NFSQTDEMB) AS 'QTD',
RTRIM(LTRIM(C.PROUM1)) AS 'UN1',
ROUND(SUM(dbo.NFSTOTITE(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER,A.NFSITE)),2) AS VALITE,
-- ROUND(SUM(dbo.NFSVDDITE(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER,A.NFSITE)),2) AS NFSVDDTOT,
SUM(A.NFSQTDDEV) AS 'QTDDEV', 
ROUND(SUM(dbo.NFSVALDEVMEN(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE)),2) AS NFSTOTDEV

FROM 
TBS0671 A (NOLOCK)
LEFT JOIN TBS067 B (NOLOCK) ON A.NFSEMPCOD = B.NFSEMPCOD AND A.NFSNUM = B.NFSNUM AND A.SNEEMPCOD = B.SNEEMPCOD AND A.SNESER = B.SNESER
LEFT JOIN TBS080 D (NOLOCK) ON A.NFSNUM = D.ENFNUM AND A.SNEEMPCOD = D.SNEEMPCOD AND A.SNESER = D.SNESER 
LEFT JOIN TBS010 C ON A.PROCOD = C.PROCOD 
LEFT JOIN #CLI   E ON B.NFSCLICOD = E.CLICOD

WHERE 
NFSCLICOD IN (SELECT CLICOD FROM #CLI) AND
B.NFSDATEMI between @DATADE and @DATAATE and 
D.ENFSIT = 6 AND
D.ENFTIPDOC = 1 AND 
A.NFSCFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S')

GROUP BY 
NFSCLICOD,
E.CLINOM ,
E.VENNOM,
E.GVDES,
RTRIM(LTRIM(A.PROCOD)),
RTRIM(LTRIM(NFSPRODES)),
RTRIM(LTRIM(C.PROUM1))
