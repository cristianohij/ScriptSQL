-- TABELA DE FILTROS

if object_id('TempDB.dbo.#TBS010') is not null
   drop table #TBS010

SELECT 
A.PROCOD,
PRODES,
rtrim(A.MARNOM)+' ('+Ltrim(str(MARCOD,4))+')' as 'MarcaDoProduto',
PROUM1,
GRUCOD,
SUBGRUCOD,
CONVERT(INT,RTRIM(LTRIM(STR(GRUCOD,2)))+RTRIM(LTRIM(STR(SUBGRUCOD,2)))) as SUBGRUPO

INTO #TBS010
FROM TBS010 A (NOLOCK) 

WHERE 
A.MARCOD =(CASE WHEN @codigoDaMarca = 0 THEN A.MARCOD ELSE Upper(RTRIM(@codigoDaMarca)) END) AND 
A.MARNOM LIKE(CASE WHEN @nomeDaMarca = '' THEN '% %' ELSE Upper(RTRIM(@nomeDaMarca))+'%' END) AND
A.PROCOD between @produtoDe and case when @produtoAte='' then 'Z' else @produtoAte end and
GRUCOD IN (@codigoDoGrupo) AND 
CONVERT(INT,RTRIM(LTRIM(STR(GRUCOD,2)))+RTRIM(LTRIM(STR(SUBGRUCOD,2)))) IN (@codigoDoSubgrupo)

if object_id('TempDB.dbo.#CORP') is not null
   drop table #CORP

SELECT 
CASE WHEN A.NFSTIP = 'N' 
	THEN 'NORMAL'
	ELSE 
		CASE WHEN A.NFSTIP = 'L' 
			THEN 'LOJA'
			ELSE
				CASE WHEN A.NFSTIP = 'C' 
					THEN 'COMPLEMENTAR'
					ELSE 'TRANSF. DE SALDO'
				END
		END
END AS NFSTIP,
LTRIM(STR(NFSCLICOD)) + ' - ' + RTRIM(NFSCLINOM) AS CLIENTE,
LTRIM(RTRIM(STR(A.SNESER))) + '/'+ LTRIM(RTRIM(STR(A.NFSNUM))) AS 'SerieENumeroDaNF',
PROCOD,
NFSPRODES,
NFSUNI,
NFSQTD,
dbo.NFSPRELIQ(B.NFSEMPCOD,B.NFSNUM,B.SNEEMPCOD,B.SNESER,B.NFSITE) as 'ValorDoItem',
dbo.NFSTOTITEST(B.NFSEMPCOD,B.NFSNUM,B.SNEEMPCOD,B.SNESER,B.NFSITE) as 'ValorTotal',
B.NFSQTDDEV AS 'QtdDevolvida'

into #CORP
FROM 
TBS0671 B (nolock)
LEFT JOIN TBS067 A (nolock) ON A.NFSEMPCOD = B.NFSEMPCOD and A.NFSNUM = B.NFSNUM and A.SNEEMPCOD = B.SNEEMPCOD and A.SNESER = B.SNESER
INNER JOIN #TBS080 D (NOLOCK) ON B.NFSNUM = D.ENFNUM AND B.SNEEMPCOD = D.SNEEMPCOD AND B.SNESER = D.SNESER
			  
WHERE 
NFSDATEMI between @dataDe and @dataAte and 
D.ENFTIPDOC = 1 AND
B.NFSCFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S') AND
NFSCLICOD NOT IN (SELECT CLICOD FROM #GRUPOCLI) AND 
NFSCLINOM LIKE(CASE WHEN @CLIENTE='' THEN NFSCLINOM ELSE RTRIM(LTRIM(@CLIENTE)) + '%' END) AND
NFSCLICOD = CASE WHEN @codcli = 0 THEN NFSCLICOD ELSE @codcli END AND 
PROCOD IN (SELECT PROCOD FROM #TBS010) AND 
A.NFSTIP IN (@NFSTIP)

------------------------------------------------------------------------------------------------------------------------------------------------------------------

-- TABELA FINAL

SELECT  
NFSTIP,	
ISNUlL(rtrim(GRUDES)+' ('+Ltrim(str(C.GRUCOD,3))+')',0) as 'grupo',
ISNULL(rtrim(SUBGRUDES)+' ('+Ltrim(str(D.SUBGRUCOD,3))+')',0) as 'subGrupo',
MarcaDoProduto,
SerieENumeroDaNF,
CLIENTE,
B.PROCOD as 'CodigoDoProduto',
B.NFSPRODES as 'DescricaoDoProduto',
NFSUNI as 'UnidadeDeMedida',
NFSQTD as 'QuantidadeVendida',
ValorDoItem,
ValorTotal,
QtdDevolvida
 
FROM #TBS010 A 
INNER JOIN #CORP B ON A.PROCOD = B.PROCOD
LEFT JOIN TBS012  C (nolock) on C.GRUCOD=A.GRUCOD
LEFT JOIN TBS0121 D (nolock) on D.GRUEMPCOD=C.GRUEMPCOD and D.GRUCOD=C.GRUCOD and D.SUBGRUCOD=A.SUBGRUCOD
