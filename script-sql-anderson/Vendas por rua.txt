-- VENDAS DO COORPORATIVO

if object_id('TempDB.dbo.#CORP') is not null
   drop table #CORP
   
SELECT 
PROCOD,
SUM(B.NFSQTD*NFSQTDEMB) AS QTD,
sum(dbo.NFSTOTITEST(B.NFSEMPCOD,B.NFSNUM,B.SNEEMPCOD,B.SNESER,B.NFSITE)) as VAL,
sum(dbo.NFSTOTITEST(B.NFSEMPCOD,B.NFSNUM,B.SNEEMPCOD,B.SNESER,B.NFSITE)) / SUM(B.NFSQTD*NFSQTDEMB) AS PRE

into #CORP
FROM 
TBS0671 B (nolock)
LEFT JOIN TBS067 A (nolock) ON A.NFSEMPCOD = B.NFSEMPCOD and A.NFSNUM = B.NFSNUM and A.SNEEMPCOD = B.SNEEMPCOD and A.SNESER = B.SNESER
INNER JOIN #TBS080 D (NOLOCK) ON B.NFSNUM = D.ENFNUM AND B.SNEEMPCOD = D.SNEEMPCOD AND B.SNESER = D.SNESER
			  
WHERE 
NFSDATEMI between @dataDe and @dataAte and
NFSTIP IN('N','C') and 
D.ENFTIPDOC = 1 AND
B.NFSCFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S') AND
NFSCLICOD NOT IN (SELECT CLICOD FROM #GRUPOCLI) 

GROUP BY 
PROCOD

SELECT
PROCOD,

INTO #DEV
FROM TBS0591 A (NOLOCK) 
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND 
							   A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND A.SEREMPCOD = B.SEREMPCOD
LEFT JOIN #TBS080 D (NOLOCK) ON A.NFENUM = D.ENFNUM AND A.NFECOD = D.ENFCODDES

NFEDATEFE between @dataDe and @dataAte and 
A.NFETIP = 'D' and 
((D.ENFFINEMI = 4 AND D.ENFTIPDOC = 0) OR (NFENOSFOR <> 'S' AND B.NFECAN <> 'S' AND NFEDATEFE <> '' )) AND
A.NFECFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E') and
B.NFECOD NOT IN (SELECT CLICOD FROM #GRUPOCLI WHERE CLICOD NOT IN (590) )

PROCOD

PROCOD, 
EST1 = (SELECT ESTQTDATU - ESTQTDRES FROM TBS032 B (NOLOCK) WHERE A.PROCOD = B.PROCOD AND B.ESTLOC = 1 ),
