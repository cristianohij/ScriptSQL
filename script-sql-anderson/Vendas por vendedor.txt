

declare @dateDe datetime, @dateAte datetime , @nomecli char (50), @codcli smallint,
@codven smallint, @nomven char(60)

set @dateDe = '20160601'
set @dateAte = '20170120'
set @codcli = ''
set @nomecli = ''
set @codven = 6
set @nomven = ''

;with 
-- TABCLI as (select CLICOD from TBS002 where VENCOD IN (CASE WHEN @codven = 0 THEN VENCOD ELSE @codven END)), 
TABCLI as (
		SELECT CLICOD
		FROM TBS002 (NOLOCK)
		WHERE CLICGC LIKE ('65069593000%') OR CLICGC LIKE('05118717000%') OR CLICGC LIKE('44125185000%') OR CLICGC LIKE('52080207000%')) ,

--select CLICOD, CLINOM from TBS002 where VENCOD = '32'

TABVEN as (
		SELECT 
		VENCOD,
		-- ISNULL((SELECT VENCOD FROM TBS002 E (NOLOCK) WHERE B.NFSCLICOD = E.CLICOD),0) AS VENCODCLI,
		ISNULL((SELECT VENNOM FROM TBS004 D (NOLOCK) WHERE B.VENCOD = D.VENCOD),'') AS VENNOM,
		SUBSTRING(CONVERT(CHAR(12),NFSDATEMI,103),4,7) AS DATA,
		SUBSTRING(CONVERT(CHAR(12),NFSDATEMI,111),1,7) AS DATA2,
		NFSCLICOD,
		NFSCLINOM,
		CASE WHEN NFSTIP = 'N'
			THEN SUM(dbo.NFSTOTITEST(C.NFSEMPCOD,C.NFSNUM,C.SNEEMPCOD,C.SNESER,C.NFSITE))
			ELSE 0
		END AS VALITENFE,
		CASE WHEN NFSTIP = 'L'
			THEN SUM(dbo.NFSTOTITEST(C.NFSEMPCOD,C.NFSNUM,C.SNEEMPCOD,C.SNESER,C.NFSITE))
			ELSE 0
		END AS VALITELOJA,
		CASE WHEN NFSTIP = 'C'
			THEN SUM(dbo.NFSTOTITEST(C.NFSEMPCOD,C.NFSNUM,C.SNEEMPCOD,C.SNESER,C.NFSITE))
			ELSE 0
		END AS VALITECOM,		
		CASE WHEN NFSQTDDEV > 0
			THEN (SUM(dbo.NFSTOTITEST(C.NFSEMPCOD,C.NFSNUM,C.SNEEMPCOD,C.SNESER,C.NFSITE))/NFSQTD/NFSQTDEMB)*NFSQTDDEV
			ELSE 0
		END AS DEVOLUCAO
		
		FROM 
		TBS067 B (NOLOCK)
		left join TBS0671 C (NOLOCK) ON C.NFSEMPCOD = B.NFSEMPCOD and C.NFSNUM = B.NFSNUM and C.SNEEMPCOD = B.SNEEMPCOD and C.SNESER = B.SNESER 
		LEFT JOIN TBS080 D (NOLOCK) ON B.NFSNUM = D.ENFNUM AND B.SNEEMPCOD = D.SNEEMPCOD AND B.SNESER = D.SNESER 
			  
		WHERE 
		NFSDATEMI BETWEEN @dateDe AND @dateAte AND
		D.ENFSIT = 6 AND
		D.ENFTIPDOC = 1 AND
		NFSFINNFE NOT IN (4) AND 
		VENCOD in (@codven) AND
		NFSCLICOD NOT IN (SELECT CLICOD FROM TABCLI)
		
		GROUP BY 
		NFSCLICOD,
		NFSCLINOM,
		NFSTIP,
		NFSQTD,
		NFSQTDEMB,
		NFSQTDDEV,
		SUBSTRING(CONVERT(CHAR(12),NFSDATEMI,103),4,7),
		SUBSTRING(CONVERT(CHAR(12),NFSDATEMI,111),1,7),
		VENCOD )

SELECT * FROM TABVEN
