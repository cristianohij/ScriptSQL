DECLARE @DATADE DATETIME, @DATAATE DATETIME, @EMPRESA VARCHAR(MAX)

SET @DATADE = '20170101'
SET @DATAATE = '20171110'

-----------------------------------------------------------------------------------------------------------------------------------------------------------------

-- CRIAR A BASE PARA PROCURAR OS PRODUTOS

IF OBJECT_ID('TempDB.dbo.#BASE') IS NOT NULL 
DROP TABLE #BASE

SELECT 
NFENUM,
CASE WHEN B.NFETIP <> 'D' 
	THEN RTRIM(LTRIM(STR((SELECT UFECODIBGE FROM TBS001 A (NOLOCK) WHERE A.UFESIG = B.NFEESTORI))))
		+REPLACE(CONVERT(CHAR(5),B.NFEDATEMI,11),'/','')+RTRIM(LTRIM((SELECT FORCGC FROM TBS006 C (NOLOCK) WHERE C.FORCOD = B.NFECOD)))
	ELSE RTRIM(LTRIM(STR((SELECT UFECODIBGE FROM TBS001 A (NOLOCK) WHERE A.UFESIG = B.NFEESTORI))))
		+REPLACE(CONVERT(CHAR(5),B.NFEDATEMI,11),'/','')+RTRIM(LTRIM((SELECT CLICGC FROM TBS002 C (NOLOCK) WHERE C.CLICOD = B.NFECOD)))
END AS NFECHAACE

INTO #BASE
FROM
TBS059 B (NOLOCK)

WHERE 
B.NFECAN <> 'S' AND 
B.NFEDATEMI BETWEEN @DATADE AND @DATAATE AND 
B.NFEDATEFE <> '' AND
B.NFETIP <> 'D'

UNION 

SELECT 
A.CPATIT,
RTRIM(LTRIM(STR((SELECT UFECODIBGE FROM TBS001 C (NOLOCK) WHERE C.UFESIG = B.UFESIG))))
+REPLACE(CONVERT(CHAR(5),A.CPADATEMI,11),'/','')+RTRIM(LTRIM(FORCGC))

FROM TBS057 A (NOLOCK)
INNER JOIN TBS006 B (NOLOCK) ON A.FORCOD = B.FORCOD

WHERE 
A.CPADATEMI BETWEEN @DATADE AND @DATAATE AND 
A.PFXCOD <> 'CAN' AND 
LEN(RTRIM(LTRIM(STR((SELECT UFECODIBGE FROM TBS001 C (NOLOCK) WHERE C.UFESIG = B.UFESIG))))
+REPLACE(CONVERT(CHAR(5),A.CPADATEMI,11),'/','')+RTRIM(LTRIM(FORCGC)))>=20

-- SELECT *, LEN(NFECHAACE) FROM #BASE where NFENUM = 107703

---------------------------------------------------------------------------------------------------------------------------------------------------------------------

SELECT 
substring(NEECHAACE,1,20), 
NEECHAACE,
NEENUM,
NEECGCCPF,
NEENOM,
NEEDOWNLOAD,
NEEDATEMI,
NEEVALTOT,
NEEEVEOBS,
NEENATOPE,
CHARINDEX('6922',NEENATOPE,1) ,
CHARINDEX('6.922',NEENATOPE,1)

FROM TBS099 A (NOLOCK) 
LEFT JOIN #BASE B ON substring(NEECHAACE,1,20) COLLATE Latin1_General_CI_AS = B.NFECHAACE AND A.NEENUM = B.NFENUM 

WHERE
convert(char(10),NEEDATEMI,103) BETWEEN @DATADE AND @DATAATE AND
NEESITNFE = 1 AND 		-- SITUAÇÃO DA NOTA 1 = AUTORIZADA, 2 = DENEGADA, 3 = CANCELADA
-- NEENFEENT = 'N' AND		-- NOTA DEU ENTRADA? 
NEENFEEFE = 'N' AND		-- NOTA EFETIVADA?
--CHARINDEX('6922',NEENATOPE,1) = 0 AND 
--CHARINDEX('6.922',NEENATOPE,1) = 0 AND
--CHARINDEX('5922',NEENATOPE,1) = 0 AND
--CHARINDEX('5.922',NEENATOPE,1) = 0 AND
-- NEEDOWNLOAD = 'S' AND
NEETIPOPE = 1 AND -- SAIDA , 0 - ENTRADA
B.NFENUM IS NULL 

ORDER BY 
NEECGCCPF,
NEEDATEMI
