SELECT
TDPPROCOD, TDPHISDAT, TDPHISVAL , TDPHISCPO 

-- INTO #TBS0311
FROM TBS0311 (nolock)

WHERE  
TDPHISDAT<'20171020' AND 
TDPPROCOD IN (SELECT DISTINCT TDPPROCOD FROM TBS0311 NOLOCK WHERE TDPHISDAT>='20171020' and TDPPROCOD IN (SELECT PROCOD FROM TBS010 NOLOCK WHERE MARCOD = 880)) 
-- AND TDPPROCOD = '8800815'

ORDER BY TDPPROCOD, TDPHISDAT DESC


SELECT 
TDPPROCOD, TDPPRECOR1, TDPPRECOR2,TDPPRECOR3,TDPPRECOR4, TDPPRELOJ1,TDPPRELOJ2,TDPPRELOJ3,TDPPRELOJ4 
FROM TBS031 NOLOCK
WHERE
TDPPROCOD IN (SELECT DISTINCT TDPPROCOD FROM TBS0311 NOLOCK WHERE TDPHISDAT>='20171020' and TDPPROCOD IN (SELECT PROCOD FROM TBS010 NOLOCK WHERE MARCOD = 880)) -- AND

-- TDPPROCOD = '8804520'


SELECT 
*
FROM TBS0311 B (NOLOCK)
			 
WHERE 
TDPHISDAT<'20171020' AND 
B.TDPPROCOD IN (SELECT DISTINCT TDPPROCOD FROM TBS0311 NOLOCK WHERE TDPHISDAT>='20171020' and TDPPROCOD IN (SELECT PROCOD FROM TBS010 NOLOCK WHERE MARCOD = 880)) AND
TDPHISCPO = 'TDPPRECOR1' 

ORDER BY TDPHISDAT DESC
