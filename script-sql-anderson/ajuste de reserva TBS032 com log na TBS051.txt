-- CRIA TABELA TEMPORARIA, COM AS CHAVES EMPRESA, LOCAL ESTOQUE E CODIGO
BEGIN TRAN 
IF OBJECT_ID('TempDB.dbo.#RES') is not null 
	BEGIN 
		drop table #RES
	END

CREATE TABLE #RES(
EMPCOD tinyint not null default 0, 
LESCOD tinyint not null default 0, 
PROCOD char(15) not null default '',
PROUM1 varchar(3),
ESTQTDATU money default 0,
ESTQTDRES money default 0,
ESTQTDPEN money default 0,
ESTQTDCMP money default 0, 
PDVQTD money default 0,
SDCQTD money default 0,
MVIQTD money default 0,

constraint i#RES primary key(EMPCOD, LESCOD, PROCOD))

-- SELECT * FROM #RES
-------------------------------------------------------------------------------------------------------------------------------------------------------------------

-- PEDIDO DE VENDAS, PEGA OS PRODUTOS QUE ESTÃO RESERVADOS E MOVIMENTANDO O ESTOQUE

if object_id('TempDB.dbo.#PDV') is not null
   drop table #PDV  

SELECT 
PRPEMP AS PROEMPCOD,
PRPESTLOC AS LESCOD,
PROCOD,
sum(PRPQTD * PRPQTDEMB) AS QTDRES

INTO #PDV
FROM 
TBS058 NOLOCK

WHERE 
PRPSIT = 'R' AND 
PRPMOVEST = 'S' 

GROUP BY 
PRPEMP,
PRPESTLOC,
PROCOD 

-------------------------------------------------------

-- SOLICITAÇÃO DE COMPRAS, PEGA OS PRODUTOS QUE ESTÃO RESERVADO

if object_id('TempDB.dbo.#SDC') is not null
   drop table #SDC

select
SDCEMPCOD AS PROEMPCOD, 
LESCOD,
PROCOD,
sum((SDCQTDATD - SDCQTDBAI) * SDCQTDEMB) AS QTDRES

INTO #SDC
from 
TBS0761 (nolock)

where 
SDCQTDRES = 0 AND -- SE ESTIVER COM RESIDUO, SIGNIFICA QUE NÃO ESTÁ RESERVADO
SDCQTDATD - SDCQTDBAI > 0 -- QUANTIDADE ATENDIDA - QUANTIDADE BAIXADA > 0 , SIGNIFICA QUE TEM ALGO RESERVADO

GROUP BY 
SDCEMPCOD, 
LESCOD,
PROCOD

---------------------------------------------------------------------------------------

-- MOVIMENTOS INTERNOS, TODOS OS PRODUTOS QUE AINDA NÃO FORAM EFETIVADOS, CONTINUAM RESERVADO

if object_id('TempDB.dbo.#MVI') is not null
   drop table #MVI 

select 
A.MVIEMPCOD AS PROEMPCOD,
B.MVILOCORI AS LESCOD, 
PROCOD ,
sum(A.MVIQTDPED*A.MVIQTDEMB) AS QTDRES

INTO #MVI
FROM TBS0371 A (nolock) 
JOIN TBS037 B (nolock) on A.MVIEMPCOD=B.MVIEMPCOD and A.MVIDOC=B.MVIDOC and B.MVIDATEFE ='17530101'  -- DATA '17530101', SEM EFETIVAÇÃO
JOIN TBS033 C (nolock) on B.TMVEMPCOD=C.TMVEMPCOD and B.TMVCOD=C.TMVCOD and C.TMVMOVAUT='N' and C.TMVTIP='S' -- TMVMOVAUT='N' - GERA UMA RESERVA, TMVTIP='S' - SAIDA
 
GROUP BY 
A.MVIEMPCOD,
B.MVILOCORI, 
PROCOD 

-------------------------------------------------------------------------------------------------------------------

-- INSERE NA TABELA #RES, OS ITENS DIVERGENTES 

insert into #RES 

SELECT  
A.PROEMPCOD,
A.ESTLOC,
A.PROCOD,
E.PROUM1,
A.ESTQTDATU,
A.ESTQTDRES,
A.ESTQTDPEN,
A.ESTQTDCMP, 
CASE WHEN PROPESAVEL = 'N' 
	THEN ROUND(CASE WHEN ISNULL(B.QTDRES,0) < 0 THEN 0 ELSE ISNULL(B.QTDRES,0) END,0)
	ELSE CASE WHEN ISNULL(B.QTDRES,0) < 0 THEN 0 ELSE ISNULL(B.QTDRES,0) END
END AS PDV,
CASE WHEN PROPESAVEL = 'N' 
	THEN ROUND(CASE WHEN ISNULL(C.QTDRES,0) < 0 THEN 0 ELSE ISNULL(C.QTDRES,0) END,0)
	ELSE CASE WHEN ISNULL(C.QTDRES,0) < 0 THEN 0 ELSE ISNULL(C.QTDRES,0) END
END AS SDC,
CASE WHEN PROPESAVEL = 'N' 
	THEN ROUND(CASE WHEN ISNULL(D.QTDRES,0) < 0 THEN 0 ELSE ISNULL(D.QTDRES,0) END,0)
	ELSE CASE WHEN ISNULL(D.QTDRES,0) < 0 THEN 0 ELSE ISNULL(D.QTDRES,0) END
END AS MVI

FROM TBS032 A (NOLOCK)
LEFT JOIN #PDV B ON A.PROEMPCOD = B.PROEMPCOD AND A.ESTLOC = B.LESCOD AND A.PROCOD = B.PROCOD
LEFT JOIN #SDC C ON A.PROEMPCOD = C.PROEMPCOD AND A.ESTLOC = C.LESCOD AND A.PROCOD = C.PROCOD
LEFT JOIN #MVI D ON A.PROEMPCOD = D.PROEMPCOD AND A.ESTLOC = D.LESCOD AND A.PROCOD = D.PROCOD
LEFT JOIN TBS010 E (NOLOCK) ON A.PROCOD = E.PROCOD AND A.PROEMPCOD = E.PROEMPCOD

WHERE 
A.ESTQTDRES <> 	CASE WHEN PROPESAVEL = 'N' 
					THEN ROUND(CASE WHEN ISNULL(B.QTDRES,0) < 0 THEN 0 ELSE ISNULL(B.QTDRES,0) END,0)
					ELSE CASE WHEN ISNULL(B.QTDRES,0) < 0 THEN 0 ELSE ISNULL(B.QTDRES,0) END
				END +
				CASE WHEN PROPESAVEL = 'N' 
					THEN ROUND(CASE WHEN ISNULL(C.QTDRES,0) < 0 THEN 0 ELSE ISNULL(C.QTDRES,0) END,0)
					ELSE CASE WHEN ISNULL(C.QTDRES,0) < 0 THEN 0 ELSE ISNULL(C.QTDRES,0) END
				END +
				CASE WHEN PROPESAVEL = 'N' 
					THEN ROUND(CASE WHEN ISNULL(D.QTDRES,0) < 0 THEN 0 ELSE ISNULL(D.QTDRES,0) END,0)
					ELSE CASE WHEN ISNULL(D.QTDRES,0) < 0 THEN 0 ELSE ISNULL(D.QTDRES,0) END
				END
