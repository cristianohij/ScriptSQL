BEGIN TRAN ; 
	MERGE TBS010  AS A  -- DESTINO
	
	USING nd.SIBD.dbo.TBS010 AS B (NOLOCK) -- ORIGEM
	
	ON A.PROCOD collate Latin1_General_CI_AS  = B.PROCOD
	
	WHEN MATCHED AND 
	(A.PROCODBAR1 collate Latin1_General_CI_AS <> B.PROCODBAR1 OR 
	A.PROCODBAR2 collate Latin1_General_CI_AS <> B.PROCODBAR2 OR
	A.PROCODBAR3 collate Latin1_General_CI_AS <> B.PROCODBAR3 OR 
	A.PROCODBAR4 collate Latin1_General_CI_AS <> B.PROCODBAR4)
	
THEN	
	UPDATE SET 
A.PROCODBAR1 = B.PROCODBAR1,
A.PROCODBAR2 = B.PROCODBAR2,
A.PROCODBAR3 = B.PROCODBAR3,
A.PROCODBAR4 = B.PROCODBAR4

OUTPUT 
INSERTED.PROCOD,INSERTED.PROCODBAR1,INSERTED.PROCODBAR2,INSERTED.PROCODBAR3,INSERTED.PROCODBAR4,
$action,
DELETED.PROCODBAR1,DELETED.PROCODBAR2,DELETED.PROCODBAR3,DELETED.PROCODBAR4 ;
