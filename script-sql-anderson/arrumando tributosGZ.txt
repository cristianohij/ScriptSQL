SELECT DISTINCT PROSTBB FROM TBS010 NOLOCK 

SELECT 
PROSTATUS AS S,
PROCOD as Código, 
PRODES as Descrição, 
PROSTBB as 'ST int',
st as 'ST GZ' ,
replace(PROREDBASICMS,'.',',') as 'Red. Base ICMS int',
replace(reducao,'.',',') as 'Red. Base ICMS GZ' ,
TGZCOD as 'Cod. GZ',
descricao AS Descrição

FROM TBS010 A (NOLOCK)  
full outer join #tributa B ON A.TGZCOD =B.codigo

WHERE PROSTBB COLLATE Latin1_General_CI_AS <> st OR
PROREDBASICMS <> reducao
