	SELECT 
	'MISASPEL' as 'UNIDADE',
	rtrim(ltrim(A.PROCOD)) collate Latin1_General_BIN as 'CODIGO',
	A.PRODES collate Latin1_General_BIN as 'DESCRICAO',							
	case when len(A.MARCOD) = 4 then rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) else right(('00' + ltrim(str(A.MARCOD))),3) + ' - ' + rtrim(A.MARNOM) end collate Latin1_General_BIN as 'MARCA',
	A.PROUM1 collate Latin1_General_BIN as 'UNIDMEDIDA',
	ISNULL(SALDO.LOJA,0) AS 'LOJA',						
	ISNULL(SALDO.EST,0) AS 'DISPONIVEL',
	ISNULL(SALDO.RES,0) AS 'RESERVADO',
	ISNULL(SALDO.PEN,0) AS 'PENDENTE',
	ISNULL(CMP.QTD,0) AS 'COMPRAS',
	CMP.DAT AS 'PRE_ENT',
	ISNULL(SALDO.CMP,0) AS 'COMPRASTT',
	CMP.PDCDES AS 'DESCRICAOPDC',
	GETDATE()AS DATULTATU,
	SALDO.ALTERADO AS ALTERADO
	into #misaldo
	from mi.SIBD.dbo.TBS010 A (nolock)	LEFT JOIN (SELECT PROCOD AS COD,SUM(ESTQTDATU-ESTQTDRES) AS SALTT, SUM(ESTQTDRES) AS RES , SUM(ESTQTDCMP) AS CMP, SUM(ESTQTDPEN) AS PEN,SUM(ESTQTDATU) AS ATU,
								  				   ISNULL((SELECT SUM(ESTQTDATU-ESTQTDRES) FROM mi.SIBD.dbo.TBS032 B (nolock) WHERE ESTLOC =1 AND A.PROCOD = B.PROCOD GROUP BY PROCOD),0) AS EST,
								  				   ISNULL((SELECT SUM(ESTQTDATU-ESTQTDRES) FROM mi.SIBD.dbo.TBS032 B (nolock) WHERE ESTLOC =2 AND A.PROCOD = B.PROCOD GROUP BY PROCOD),0) AS LOJA,
												   max(ESTDATALT) AS ALTERADO	
								  				   FROM mi.SIBD.dbo.TBS032 A (nolock) WHERE ESTLOC IN (1,2) and (ESTQTDATU <> 0 OR ESTQTDPEN <> 0 OR ESTQTDCMP <>0) GROUP BY PROCOD)
												   AS SALDO ON SALDO.COD = A.PROCOD
													
								FULL OUTER JOIN(SELECT B.PROCOD AS COD ,B.PDCDES, (B.PDCQTD - B.PDCQTDENT) * B.PDCQTDEMB AS QTD , PDCDATPRE AS DAT
													FROM mi.SIBD.dbo.TBS0451 B (NOLOCK) 
													JOIN mi.SIBD.dbo.TBS045 C (NOLOCK) ON B.PDCNUM = C.PDCNUM
													WHERE ((B.PDCQTD - B.PDCQTDENT - B.PDCQTDRES ) > 0) AND (C.PDCBLQ ='N')) AS CMP ON A.PROCOD = CMP.COD
	
	where (SALDO.ATU <> 0 OR SALDO.PEN <> 0 OR SALDO.CMP <> 0)

drop table misaldoatual 	

SELECT * INTO misaldoatual from #misaldo
