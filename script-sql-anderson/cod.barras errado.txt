SELECT A.PROSTATUS,A.PROCOD,A.PRODES,
CASE WHEN PROUM1QTD > 0  THEN PROCODBAR1 ELSE 'NÃO' END ,
CASE WHEN PROUM2QTD > 0 THEN PROCODBAR2 ELSE 'NÃO' END,
(ESTQTDATU - ESTQTDRES) AS DISPONIVEL,PROLOCFIS FROM TBS010 A JOIN TBS032 B ON A.PROCOD = B.PROCOD
WHERE ESTLOC in (1,2) AND ESTQTDATU > 0 AND (PROCODBAR1 = '' OR PROCODBAR2 = '')
ORDER BY PROLOCFIS

CASE WHEN PROUM1QTD > 0 THEN PROCODBAR1 ELSE 'NÃO' END , 
CASE WHEN PROUM2QTD > 0 THEN PROCODBAR2 ELSE 'NÃO' END , 
CASE WHEN PROUM3QTD > 0 THEN PROCODBAR3 ELSE 'NÃO' END , 
CASE WHEN PROUM4QTD > 0 THEN PROCODBAR4 ELSE 'NÃO' END ,
PROLOCFIS
