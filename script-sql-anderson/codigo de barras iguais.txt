-- TABELA FINAL COM SALDO DISPONIVEL

IF OBJECT_ID('TempDB.dbo.#FIM') IS NOT NULL 
DROP TABLE #FIM

SELECT 
A.*, 
isnull((SELECT SUM(ESTQTDATU-ESTQTDRES) FROM TBS032 B (NOLOCK) WHERE ESTLOC IN(@ESTLOC) AND B.PROCOD = A.PROCOD GROUP BY B.PROCOD),0)AS ESTQTDDIS

INTO #FIM
FROM #T10 A


SELECT * FROM #FIM WHERE PROCODBAR IN (SELECT PROCODBAR FROM #FIM WHERE ESTQTDDIS > CASE WHEN @SALDO = 1 THEN 0 ELSE -99999999 END)
