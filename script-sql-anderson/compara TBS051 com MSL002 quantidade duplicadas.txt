SELECT M2_PROCOD AS COD, SUM(M2_QTD) AS QTDSAILOJ ,M2_NUMDOC  --CONVERT(DATE,M2_DAT) AS DAT , 
			--INTO #SAILOJ
			FROM MSL002 B (nolock) 
			WHERE M2_DAT between '20160301' and '20160331' and M2_TIPREG = '01' AND M2_REGCAN = 'F' 
			--AND M2_PROCOD = '1640054'
			GROUP BY M2_PROCOD,M2_NUMDOC
			ORDER BY  M2_PROCOD,M2_NUMDOC
