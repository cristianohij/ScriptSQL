BEGIN TRAN

SELECT 
0 AS LMEEMPCOD,
A.NFENUM AS LEMDOC,
'SQL' AS LMEROT,
'AJUSTE DE SALDO NF CAN' AS LMEDESROT,
'S' AS LMEACAO,
'ANDERSONL' AS LMEUSU, -- 7
'NENHUM' AS LMEMOD, -- 8
'E' AS LMEINFALT ,
A.PROCOD , 
0 AS PROEMPCOD,
A.NFEQTD*A.NFEQTDEMB AS LMEQTDMOV

INTO #TBS0591
FROM TBS0591 A (NOLOCK) 
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFETIP = B.NFETIP AND A.NFENUM = B.NFENUM AND A.NFECOD = B.NFECOD AND A.SERCOD = B.SERCOD

WHERE 
A.NFENUM = 121019 AND 
B.NFECAN = 'S'
