-- PROCURA O QUE FOI GERADO RESERVA NOS MOVIMENTOS INTERNOS ATRAVES DE UMA SOLICITAÇÃO DE COMPRA
-- CASO COUNT(*) MAIOR QUE ZERO, FAZER UPDATE, TBS0371 ZERAR A QUANTIADE ATENDIDA, TBS037 ZERAR USUARIO QUE "EFETIVOU"

begin tran 

IF (SELECT COUNT(*)
	FROM TBS037 A (NOLOCK)
	LEFT JOIN TBS0371 B (NOLOCK) ON A.MVIDOC = B.MVIDOC AND A.MVIEMPCOD = B.MVIEMPCOD
	
	WHERE 
	A.MVIDATEFE = '17530101' AND 
	B.MVIQTDPED = B.MVIQTDATD AND
	A.TMVCOD = 501 AND 
	A.MVIDOC IN(SELECT DISTINCT LMEDOC 
				FROM TBS051 NOLOCK 
				WHERE
				LMEDATHOR IN (SELECT LMEDATHOR FROM TBS051 NOLOCK WHERE LMEDATHOR > CONVERT(DATE,GETDATE()) AND LMEROT = 'PCOM055' AND LMEACA = 'S') AND 
				LMEROT = 'PEST005' AND LMEACA = 'E')
	) > 0 
	
	BEGIN 
	UPDATE TBS0371 SET
	MVIQTDATD = 0 
	FROM TBS037 A (NOLOCK)
	LEFT JOIN TBS0371 B (NOLOCK) ON A.MVIDOC = B.MVIDOC AND A.MVIEMPCOD = B.MVIEMPCOD

	WHERE 
	A.MVIDATEFE = '17530101' AND 
	B.MVIQTDPED = B.MVIQTDATD AND
	A.TMVCOD = 501 AND 
	A.MVIDOC IN(SELECT DISTINCT LMEDOC 
				FROM TBS051 NOLOCK 
				WHERE
				LMEDATHOR IN (SELECT LMEDATHOR FROM TBS051 NOLOCK WHERE LMEDATHOR > CONVERT(DATE,GETDATE()) AND LMEROT = 'PCOM055' AND LMEACA = 'S') AND 
				LMEROT = 'PEST005' AND LMEACA = 'E')

	UPDATE TBS037 SET
	MVICCSCOD = 0 ,
	MVICCSNOM = ''
	FROM TBS037 A (NOLOCK)
	LEFT JOIN TBS0371 B (NOLOCK) ON A.MVIDOC = B.MVIDOC AND A.MVIEMPCOD = B.MVIEMPCOD

	WHERE 
	A.MVIDATEFE = '17530101' AND 
	B.MVIQTDPED = B.MVIQTDATD AND
	A.TMVCOD = 501 AND 
	A.MVIDOC IN(SELECT DISTINCT LMEDOC 
				FROM TBS051 NOLOCK 
				WHERE
				LMEDATHOR IN (SELECT LMEDATHOR FROM TBS051 NOLOCK WHERE LMEDATHOR > CONVERT(DATE,GETDATE()) AND LMEROT = 'PCOM055' AND LMEACA = 'S') AND 
				LMEROT = 'PEST005' AND LMEACA = 'E')
	END

ELSE SELECT 0 

commit tran

-- VERIFICA SES TEM ITENS NO PEDIDO E DA UPDATE NA TBS037 COLOCANDO PELO MENOS A QUANTIDADE 1 NO CAMPO DE ULTIMO ITEM, ASSIM O USUARIO PODE EFETIVAR

begin tran 
 
IF (SELECT COUNT(*) 
	FROM TBS037 A (NOLOCK)
	LEFT JOIN TBS0371 B (NOLOCK) ON A.MVIDOC = B.MVIDOC AND A.MVIEMPCOD = B.MVIEMPCOD
	 
	WHERE 
	MVIDATEFE = '17530101' AND 
	TMVCOD = 501 AND 
	MVIULTITE = 0 AND
	B.PROCOD IS NOT NULL ) > 0 
	
	BEGIN 
	UPDATE TBS037 SET 
	MVIULTITE  = 1 
	FROM TBS037 A (NOLOCK)
	LEFT JOIN TBS0371 B (NOLOCK) ON A.MVIDOC = B.MVIDOC AND A.MVIEMPCOD = B.MVIEMPCOD
	WHERE 
	MVIDATEFE = '17530101' AND 
	TMVCOD = 501 AND 
	MVIULTITE = 0 AND
	B.PROCOD IS NOT NULL 
	END

ELSE SELECT 0

commit tran
