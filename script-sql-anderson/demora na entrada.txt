SELECT 
NFETIP, 
NFENUM , 
NFEDATEMI, 
NFEDATEFE ,
DATEDIFF(d,NFEDATEMI,NFEDATEFE) AS DIAS

FROM TBS059 NOLOCK 

WHERE 
NFEDATEFE > '20170101'





-- F.A

SELECT 
B.NFENUM , B.NFEDATEFE

FROM TBS059 B
LEFT JOIN TBS080 D (NOLOCK) ON B.NFENUM = D.ENFNUM AND B.NFECOD = D.ENFCODDES
LEFT JOIN TBS0596 C (NOLOCK) ON C.NFEEMPCOD = B.NFEEMPCOD AND C.NFECOD = B.NFECOD AND 
		C.NFENUM = B.NFENUM AND C.NFETIP = B.NFETIP AND C.SERCOD = B.SERCOD AND C.SEREMPCOD = B.SEREMPCOD
		
WHERE 
NFEDATEFE between '20170101' and '20170331' and 
B.NFETIP = 'D' and 
((D.ENFSIT = 6 AND D.ENFFINEMI = 4 AND D.ENFTIPDOC = 0) OR ((NFENOSFOR = 'N' OR NFENOSFOR = '') AND (B.NFECAN = 'N' OR B.NFECAN = '') AND NFEDATEFE <> '' )) AND
C.NFENFRTIP = 'NFE'
