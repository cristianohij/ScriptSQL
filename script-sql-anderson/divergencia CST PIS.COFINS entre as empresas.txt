-- IINDICE 

SELECT PROCOD INTO #IND FROM #ND
UNION
SELECT PROCOD FROM #PP
UNION
SELECT PROCOD FROM #MI
UNION
SELECT PROCOD FROM #BB


SELECT 
A.PROCOD AS PROCOD,
ISNULL(ISNULL(ISNULL(B.PRODES, C.PRODES), D.PRODES), E.PRODES) AS PRODES,
ISNULL(ISNULL(ISNULL(B.PROCLAFIS, C.PROCLAFIS), D.PROCLAFIS), E.PROCLAFIS) AS PROCLAFIS,
B.PROSTBPIS AS PISND, 
B.PROSTBCOFINS AS COFINSND,
C.PROSTBPIS AS PISPP,  
C.PROSTBCOFINS AS COFINSPP,
D.PROSTBPIS AS PISMI, 
D.PROSTBCOFINS AS COFINSMI,
E.PROSTBPIS AS PISBB, 
E.PROSTBCOFINS AS COFINSBB

FROM #IND A
LEFT JOIN #ND B ON A.PROCOD = B.PROCOD 
LEFT JOIN #PP C ON A.PROCOD = C.PROCOD 
LEFT JOIN #MI D ON A.PROCOD = D.PROCOD 
LEFT JOIN #BB E ON A.PROCOD = E.PROCOD 

WHERE 
B.PROSTBPIS IS NULL OR B.PROSTBCOFINS IS NULL OR 
C.PROSTBPIS IS NULL OR C.PROSTBCOFINS IS NULL OR 
D.PROSTBPIS IS NULL OR D.PROSTBCOFINS IS NULL OR 
E.PROSTBPIS IS NULL OR E.PROSTBCOFINS IS NULL
