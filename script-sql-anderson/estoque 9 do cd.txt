select 
PROEMPCOD ,
ESTLOC,
PROCOD,
REPLACE(ESTQTDATU,'.',',') AS ESTQTDATU,
REPLACE(ESTQTDRES,'.',',') AS ESTQTDRES,
REPLACE(ESTQTDPEN,'.',',') AS ESTQTDPEN,
REPLACE(ESTQTDCMP,'.',',') AS ESTQTDCMP,
ESTDATCAD,
ESTUSUNOM,
ESTROTEXE,
ESTROTDES,
ESTSEL,
PRODES,
FORCOD,
MARCOD,
PROSTATUS,
ESTQTDSEG,
ESTCONMED,
ESTTEMREP,
ESTQTDMIN,
ESTPONPED,
ESTDCMDE,
ESTDCMATE,
ESTFATSEG,
ESTLOTCMP,
ESTMAXCMP,
ESTDATALT

from TBS032 
WHERE ESTLOC = 9 AND (ESTQTDATU <> 0 OR ESTQTDPEN <> 0 OR ESTQTDCMP <> 0)

TBS051( LMEEMPCOD, -- 1

0 AS LMEEMPCOD, 							-- 1 LMEEMPCOD
(select MAX(LMEREG) FROM TBS051 NOLOCK) + ROW_NUMBER() OVER(ORDER BY A.PROCOD ) AS ORDEM,

'SQL' AS LMEROT, 							-- 3 LMEROT

'NENHUM' AS LMEMOD,	 						-- 8 LMEMOD
'R' AS LMEINFALT, 							-- 9 LMEINFALT
A.PROCOD AS  PROCOD, 						-- 10 PROCOD
0  AS PROEMPCOD, 							-- 11 PROEMPCOD
0  AS LMEQTDSAL, 							-- 12 LMEQTDSAL
ESTQTDRES AS LMEQTDMOV, 					-- 13 LMEQTDMOV
ESTQTDATU AS LMEQTDATU, 					-- 14 LMEQTDATU
ESTQTDRES AS LMEQTDRES, 					-- 15 LMEQTDRES
ESTQTDPEN AS LMEQTDPEN, 					-- 16 LMEQTDPEN
ESTQTDCMP AS LMEQTDCMP, 					-- 17 LMEQTDCMP
PROUM1 AS LMEUNI, 							-- 18 LMEUNI
ESTQTDATU - ESTQTDRES AS LMEQTDDIS, 		-- 19 LMEQTDDIS
9 AS LMELOCEST 								-- 20 LMELOCEST -- igaul ao estoque na condição where
	  	    
FROM 
TBS032 A (NOLOCK) 
LEFT JOIN TBS010 B (NOLOCK) ON A.PROCOD = B.PROCOD 
WHERE ESTLOC = 9 AND ESTQTDRES > 0

SET TBSVALSEQ = (SELECT MAX(LMEREG) FROM TBS051 (NOLOCK))
WHERE TBSNOM = 'TBS051'
