DECLARE @COD varchar (500), @MARNON CHAR(20) , @STA CHAR(1), @ATL CHAR(1), @dataDE DATETIME , @dataAte DATETIME, @somenteComSaldo CHAR(1),
@sql varchar (500), @horaDe nvarchar(8) , @horaAte nvarchar(8)

SET @COD = ''
SET @MARNON =''
SET @STA ='T' 
SET @ATL = 'S'
SET @dataDE = '20160711'
SET @dataAte = '20160711'
SET @somenteComSaldo = 'N'
set @horaDe = '12:30'
set @horaAte = '' 

if object_id('TempDB.dbo.##T') is not null
		drop table ##T
 
IF @COD <> ''
BEGIN 
set @sql = 'SELECT RTRIM(PROCOD) PROCOD INTO ##T FROM TBS010 T (NOLOCK) WHERE PROCOD IN ('+@COD+')'
exec(@sql) 
END

IF @COD = ''
BEGIN 
set @sql = 'SELECT RTRIM(PROCOD) PROCOD INTO ##T FROM TBS010 T (NOLOCK)'
exec(@sql) 
END


SELECT  rank() OVER (ORDER BY A.PROCOD) AS [RANK],
A.PROSTATUS AS STS, 
RTRIM(LTRIM(A.PROCOD)) AS CÓDIGO, 
RTRIM(LTRIM(A.PRODES)) AS DESCRIÇÃO, 
RTRIM(LTRIM(A.MARNOM)) AS MARCA,

convert(decimal (12,2),
CASE WHEN B.TDPVALPROI<=GETDATE() AND B.TDPVALPROF >= GETDATE() AND B.TDPPROLOJ = 'S' 
	THEN ROUND(B.TDPPREPRO1,2) 
	ELSE ROUND(B.TDPPRELOJ1,2) 
END) AS PRECO1,

convert(decimal (12,2),
CASE WHEN PROUM2QTD > 0 
	THEN 
		CASE WHEN B.TDPVALPROI<=GETDATE() AND B.TDPVALPROF >= GETDATE() AND B.TDPPROLOJ = 'S'
			THEN ROUND(B.TDPPREPRO2,2)*round(PROUM2QTD,2)
			ELSE ROUND(B.TDPPRELOJ2,2)*round(PROUM2QTD,2) 
		END
	ELSE 0 
END) AS PRECO2,
		

CASE WHEN PROUM1QTD = 1
	THEN RTRIM((SELECT UNIDES FROM TBS011 D (NOLOCK) WHERE A.PROUM1 = D.UNICOD))   
	ELSE
		CASE WHEN PROUM1QTD > 1 
		THEN rtrim(PROUM1) + ' C/' + rtrim(CAST(PROUM1QTD AS DECIMAL(10,0))) +''+ RTRIM(PROUMV) 
		ELSE '' 
	END 
END  as UN1,
'R$' sifra,

CASE WHEN PROUM2QTD > 0 and TDPPRELOJ2 > 0
	THEN rtrim(PROUM2) + ' C/' + rtrim(CAST(PROUM2QTD AS DECIMAL(10,0)))+''+ RTRIM(PROUM1) 
	ELSE '' 
END  as UN2


FROM  TBS010 A(NOLOCK) LEFT JOIN TBS031 B (NOLOCK) ON A.PROCOD = B.TDPPROCOD

FULL OUTER JOIN TBS032 C (nolock) ON A.PROCOD = C.PROCOD

WHERE A.PROCOD IN ((SELECT PROCOD FROM ##T ))
AND A.MARNOM LIKE(CASE WHEN @MARNON = '' THEN '% %' ELSE '%'+UPPER(RTRIM(@MARNON))+'%' END)
AND A.PROSTATUS IN( CASE WHEN @STA = 'T' THEN A.PROSTATUS ELSE @STA END )
AND convert(DATE,TDPDATATU) BETWEEN CASE WHEN @ATL = 'N' THEN '17530101' ELSE @dataDE END AND CASE WHEN @dataAte='' THEN GETDATE()+1 ELSE @dataAte END
and convert(nvarchar(8),TDPDATATU,108) BETWEEN CASE WHEN @ATL = 'N' THEN '00:00:00' ELSE @horaDe END AND CASE WHEN @horaAte='' THEN '23:59:59' ELSE @horaAte END
AND (C.ESTQTDATU-C.ESTQTDRES >= case when @somenteComSaldo='N' then -9999 else 0.001 end AND C.ESTLOC = 2)
					
ORDER BY [RANK]
