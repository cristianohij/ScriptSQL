DECLARE @MESDE INT, @ANODE INT, @MESATE INT, @ANOATE INT
SET @MESDE = 5
SET @ANODE = 2017
SET @MESATE = 8
SET @ANOATE = 2017

-- SELECT LTRIM(STR(@ANODE+1))

SELECT 
SINDAT,
CASE SUBSTRING(CONVERT(CHAR(10),SINDAT,103),4,2)
	WHEN '01' THEN 'DEZ'
	WHEN '02' THEN 'JAN'
	WHEN '03' THEN 'FEV'
	WHEN '04' THEN 'MAR'
	WHEN '05' THEN 'ABR'
	WHEN '06' THEN 'MAI'
	WHEN '07' THEN 'JUN'
	WHEN '08' THEN 'JUL'
	WHEN '09' THEN 'AGO'
	WHEN '10' THEN 'SET'
	WHEN '11' THEN 'OUT'
	WHEN '12' THEN 'NOV'
END AS MES,
CASE SUBSTRING(CONVERT(CHAR(10),SINDAT,103),4,2)
	WHEN '01' THEN '12'
	WHEN '02' THEN '01'
	WHEN '03' THEN '02'
	WHEN '04' THEN '03'
	WHEN '05' THEN '04'
	WHEN '06' THEN '05'
	WHEN '07' THEN '06'
	WHEN '08' THEN '07'
	WHEN '09' THEN '08'
	WHEN '10' THEN '09'
	WHEN '11' THEN '10'
	WHEN '12' THEN '11'
END AS MES,
SUBSTRING(CONVERT(CHAR(10),SINDAT,103),7,4) AS ANO,
RTRIM(LTRIM(str(LESCOD))) AS LESCOD1,
RTRIM(LTRIM(str(LESCOD)))+' - '+(SELECT RTRIM(LTRIM(LESDES)) FROM TBS034 D (NOLOCK) WHERE A.LESCOD=D.LESCOD) AS LESCOD,
SINPROCOD,
(SELECT RTRIM(LTRIM(PRODES)) FROM TBS010 B (NOLOCK) WHERE A.SINPROCOD = B.PROCOD) AS PRODES,
(select case when len(B.MARCOD) = 4 
			then rtrim(B.MARCOD) + ' - ' + rtrim(B.MARNOM) 
			else right(('00' + ltrim(str(B.MARCOD))),3) + ' - ' + rtrim(B.MARNOM) 
		end FROM TBS010 B (NOLOCK) WHERE A.SINPROCOD = B.PROCOD) AS MARNOM,
SINUNI,
SINQTDEMB,
SINQTD,
CASE WHEN SINCUSAQU = 0 
	THEN ISNULL((SELECT TDPCUSBAS FROM TBS031 C (NOLOCK) WHERE A.SINPROCOD= C.TDPPROCOD),0)
	ELSE SINCUSAQU 
END AS SINCUSAQU,
CASE WHEN ROUND(SINQTD*SINCUSAQU,2) = 0 
	THEN ISNULL(ROUND(SINQTD*(SELECT TDPCUSBAS FROM TBS031 C (NOLOCK) WHERE A.SINPROCOD= C.TDPPROCOD),2),0)
	ELSE ROUND(SINQTD*SINCUSAQU,2) 
END AS SINCUSTOT,
CASE WHEN ROUND(SINQTD*SINCUSAQU,2) = 0 
	THEN 'POL'
	ELSE 'NFE'
END AS SINCUSTOT

FROM TBS124 A (NOLOCK)

WHERE 
SUBSTRING(CONVERT(VARCHAR(12),SINDAT,111),1,7) BETWEEN CASE WHEN @MESDE = 12 THEN LTRIM(STR(@ANODE+1)) ELSE LTRIM(STR(@ANODE)) END + '/' + 
													   CASE WHEN @MESDE = 12 THEN RIGHT(('0'+LTRIM(STR(1))),2) ELSE RIGHT(('0'+LTRIM(STR(@MESDE+1))),2) END AND
													   CASE WHEN @MESATE = 12 THEN LTRIM(STR(@ANOATE+1)) ELSE LTRIM(STR(@ANOATE)) END + '/' + 
													   CASE WHEN @MESATE = 12 THEN RIGHT(('0'+LTRIM(STR(1))),2) ELSE RIGHT(('0'+LTRIM(STR(@MESATE+1))),2) END AND
A.SINQTD <> 0
