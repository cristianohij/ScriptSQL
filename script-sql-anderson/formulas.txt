-- formulas da nota fiscal de saída de DEVOLUÇÃO

-- NFDQTD = quantidade do produto
-- NFDPRE = preco do produto

-- NFDPERDES = percentual do desconto por item
-- NFDTOTPRO = valor total do produto = NFSPRE * NFSQTD

drop function NFDTOTPROITE -- PARA CANCELAR A FUNÇÃO
go

create function NFDTOTPROITE(@empresa smallint ,@seremp smallint ,@serie smallint ,@nfd int,@empnfent smallint ,@nfetipent char(3),@nfeent int,
@empsernfeent smallint,@sernfeent char(3),@item smallint) returns decimal(11,4) as
   begin
      declare @retorno decimal(11,4)
      set @retorno = (select NFDPRE * NFDQTD from TBS1172 (nolock)
                       where NFDEMPCOD = @empresa and SNEEMPCOD = @seremp and SNESER = @serie and NFDNUM = @nfd and
					   NFDNFEEMP = @empnfent and NFDNFETIP = @nfetipent and NFDNFENUM = @nfeent and 
					   NFDSEREMP = @empsernfeent and NFDSERCOD = @sernfeent and NFDITE = @item )
      return @retorno
   end
go

--CPADIAATR - DIAS EM ATRASO NO PAGAMENTO
create function CPADIAATR(@emptit smallint ,@empprefix smallint ,@empfor smallint ,@prefi char(3) ,@titulo decimal(10,0) ,@parcela char(2) ,@fornece int)
        returns int as
   begin
      declare @retorno int ,@datatu datetime
      set @datatu = (select top 1 data from DataAtual)
      set @retorno = (select case 
	  							when CPADATBAI = '17530101' then
								case 
									when datediff(day,CPADATVENREA,@datatu) < 0 then 0
										else
                                        	datediff(day,CPADATVENREA,@datatu)
                                 	end
								   		else
                                   			case
                                     			when datediff(day,CPADATVENREA,CPADATBAI) < 0 then
                                         0
                                      else
                                         datediff(day,CPADATVENREA,CPADATBAI)
                                   end
                                end
								FROM TBS057 (nolock)
								where CPAEMPCOD = @emptit and PFXEMPCOD = @empprefix and FOREMPCOD = @empfor and PFXCOD = @prefi and CPATIT = @titulo and
                             CPAPAR = @parcela and FORCOD = @fornece)
      return @retorno
   end
go

DROP FUNCTION dbo.FCNDIASUTEIS

CREATE FUNCTION dbo.FCNDIASUTEIS (@DATAINICIAL DATETIME, @DATAFINAL DATETIME)
RETURNS INT
AS
BEGIN

DECLARE @CONT INT, @DIASUTEIS INT 

SET @CONT = 0

DECLARE @TAB AS TABLE (SEMANA INT, DATA DATETIME, CONT INT )

WHILE @CONT <= (SELECT DATEDIFF(DD, @DATAINICIAL, @DATAFINAL ))

BEGIN 
INSERT INTO @TAB SELECT DATEPART(DW,(DATEADD(DD, @CONT, @DATAINICIAL))), DATEADD(DD, @CONT, @DATAINICIAL), @CONT

SET @CONT = @CONT+1

END

SET @DIASUTEIS =( SELECT COUNT(SEMANA) FROM @TAB WHERE SEMANA NOT IN (1) ) - 
(SELECT count(*) 
FROM 
FERIADO -- CRIEI ESSA TABELA EM CADA UNIDADE, NA FORMULA MUDEI O Sg_UF DE ACORDO COM O LOCAL DA UNIDADE (08/08/2017)

WHERE
(Sg_UF = 'SJ' or Tp_Feriado = 1 ) and 
case when Nr_Ano = 0 
	then convert(date,(convert(char(4),YEAR(GETDATE()))+convert(char(2),right(('0' + ltrim(str(Nr_Mes))),2))+convert(char(2),right(('0' + ltrim(str(Nr_Dia))),2))))
	else convert(date,(convert(char(4),Nr_Ano)+convert(char(2),right(('0' + ltrim(str(Nr_Mes))),2))+convert(char(2),right(('0' + ltrim(str(Nr_Dia))),2))))
end between @DATAINICIAL and @DATAFINAL)

RETURN @DIASUTEIS
end

-- FUNÇÃO PARA PEGAR VALOR DO ITEM COM TODOS OS IMPOSTOS + OS DESCONTOS
-- PARA GERAR O DESCONTO CORRETO NAS NOTAS DE LOJA, POIS NO INTEGROS ESTÁ DIFERENTE

create function NFSTOTITESTSEMDES(@empresa smallint ,@nf int ,@seremp smallint ,@serie smallint ,@item smallint) returns decimal(11,4) as
   begin
      declare @retorno decimal(11,4)
      set @retorno = (select case
                                when NFSPERICMSST > 0
                                   then
                                      dbo.NFSTOTITE(@empresa,@nf,@seremp,@serie,@item) + dbo.NFSVALICMSSTRET(@empresa,@nf,@seremp,@serie,@item) + 
									  dbo.NFSVDDITE(@empresa,@nf,@seremp,@serie,@item)
                                   else
                                      dbo.NFSTOTITE(@empresa,@nf,@seremp,@serie,@item) + dbo.NFSVDDITE(@empresa,@nf,@seremp,@serie,@item)
                             end
                        from TBS0671 (nolock)
                       where NFSEMPCOD = @empresa and NFSNUM = @nf and SNEEMPCOD = @seremp and SNESER = @serie and NFSITE = @item)
      return @retorno
   end
go

DROP FUNCTION dbo.DIAHORMIN

CREATE FUNCTION dbo.DIAHORMIN (@DATAINICIAL DATETIME, @DATAFINAL DATETIME)

RETURNS VARCHAR(20)
AS 
BEGIN 
	DECLARE @TEMPO VARCHAR(20)
	
SET @TEMPO = (select rtrim(ltrim(str((DATEDIFF(minute, @DATAINICIAL, @DATAFINAL) / 60 / 24)))) + 'D ' + 
right('00' +rtrim(ltrim(str(CONVERT(INT,((convert(money,DATEDIFF(minute, @DATAINICIAL, @DATAFINAL)) / 60 / 24) - (DATEDIFF(minute, @DATAINICIAL, @DATAFINAL) / 60 / 24)) * 24)))),2) +'H '+
right('00' +rtrim(ltrim(str(round(((convert(money,DATEDIFF(minute, @DATAINICIAL, @DATAFINAL)) / 60) - (DATEDIFF(minute, @DATAINICIAL, @DATAFINAL) / 60)) * 60,2)))),2)+'M' )

RETURN @TEMPO
END

-- NFSVALDEV = valor da devolução NA MENOR UNIDADE

drop function NFSVALDEVMEN
go

create function NFSVALDEVMEN(@empresa smallint ,@nf int ,@seremp smallint ,@serie smallint ,@item smallint) returns decimal(11,4) as
   begin
      declare @retorno decimal(11,4)
      set @retorno = (select NFSQTDDEV * (NFSPRE/NFSQTDEMB)
                        from TBS0671 (nolock)
                       where NFSEMPCOD = @empresa and NFSNUM = @nf and SNEEMPCOD = @seremp and SNESER = @serie and NFSITE = @item)
      return @retorno
   end
go

-- NFSTOTDEV = valor total das devolucoes = SUM(NFSVALDEV) NA MENOR UNIDADE

drop function NFSTOTDEVMEN
go

create function NFSTOTDEVMEN(@empresa smallint ,@nf int ,@seremp smallint ,@serie smallint) returns decimal(11,4) as
   begin
      declare @retorno decimal(11,4)
      set @retorno = (select isnull(sum(dbo.NFSVALDEVMEN(@empresa,@nf,@seremp,@serie,NFSITE)),0) from TBS0671 (nolock)
                       where NFSEMPCOD = @empresa and NFSNUM = @nf and SNEEMPCOD = @seremp and SNESER = @serie)
      return @retorno
   end
go
