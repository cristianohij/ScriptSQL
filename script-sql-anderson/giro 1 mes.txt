SELECT PROSTATUS, 
rtrim(ltrim(A.PROCOD))  as 'CODIGO',
A.PRODES  as 'DESCRICAO',							
case when len(A.MARCOD) = 4 then rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) else right(('00' + ltrim(str(A.MARCOD))),3) + ' - ' + rtrim(A.MARNOM) end as 'MARCA',
A.PROUM1 collate Latin1_General_BIN as 'UNID. MEDIDA',
REPLACE(STR(ISNULL(VLOJA.VND1,0),10,4),'.',',') AS LOJA1,
REPLACE(STR(ISNULL(VINT.VND1,0),10,4),'.',',') AS INT1,
REPLACE(STR(ISNULL(VLOJA.VND1,0)+ISNULL(VINT.VND1,0),10,3),'.',',') AS TOTAL,
REPLACE(STR(ISNULL(VLOJA.VALOR,0)+ ISNULL(VINT.VALOR,0),10,3),'.',',') AS SOMATT
--REPLACE(STR(ISNULL(SUM(VLOJA.VALOR),0),10,4),'.',',') + REPLACE(STR(ISNULL(SUM(VINT.VALOR),0),10,4),'.',',')) OVER (PARTITION BY 1) AS VALOR

FROM TBS010 A (NOLOCK)

FULL OUTER JOIN   
(SELECT A.M2_PROCOD AS COD, 
ISNULL((SELECT SUM(M2_QTD) AS QTD FROM MSL002 B (nolock) WHERE M2_DAT between '20151201' and '20151231' and M2_TIPREG = '01' AND M2_REGCAN = 'F' AND A.M2_PROCOD = B.M2_PROCOD GROUP BY M2_PROCOD ),0) AS VND1,
--ISNULL(SUM(M2_QTD),0) AS TT,
SUM(M2_VALTOT)-SUM(M2_ABT) AS VALOR

FROM MSL002 A (nolock) WHERE M2_DAT between '20151001' and '20151231' and M2_TIPREG = '01' AND M2_REGCAN = 'F'   GROUP BY M2_PROCOD ) AS VLOJA ON A.PROCOD = VLOJA.COD


FULL OUTER JOIN   
(SELECT PROCOD AS COD ,
ISNULL((SELECT SUM(C.NFSQTD*NFSQTDEMB) AS QTD  FROM TBS0671 C (nolock) JOIN TBS067 B (nolock)  ON 
										C.NFSEMPCOD = B.NFSEMPCOD and 
										C.NFSNUM = B.NFSNUM and 
										C.SNEEMPCOD = B.SNEEMPCOD and 
										C.SNESER = B.SNESER
										WHERE NFSDATEMI between '20151201' and '20151231' 
										and NFSTIP = 'N' and NFSCAN = 'N' and NFSDEV='N'
										and  NFSCLINOM not Like('%BEST BAG%') and NFSCLINOM not Like('%BEST OFFICE%') 
										and NFSCLINOM not Like('%MISASPEL%') and NFSCLINOM not Like('%PAPELYNA%') 
										and  NFSCLINOM not Like('%TANBY%')
										AND C.PROCOD = A.PROCOD
										GROUP BY PROCOD),0) AS VND1,



--ISNULL(SUM(A.NFSQTD*NFSQTDEMB),0) AS TT,
sum(dbo.NFSTOTITEST(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE)) as VALOR

FROM TBS0671 A (nolock) JOIN TBS067 B (nolock)  ON 
										A.NFSEMPCOD = B.NFSEMPCOD and 
										A.NFSNUM = B.NFSNUM and 
										A.SNEEMPCOD = B.SNEEMPCOD and 
										A.SNESER = B.SNESER
										WHERE NFSDATEMI between '20151001' and '20151231' 
										and NFSTIP = 'N' and NFSCAN = 'N' and NFSDEV='N'
										and  NFSCLINOM not Like('%BEST BAG%') and NFSCLINOM not Like('%BEST OFFICE%') 
										and NFSCLINOM not Like('%MISASPEL%') and NFSCLINOM not Like('%PAPELYNA%') 
										and  NFSCLINOM not Like('%TANBY%')
										GROUP BY PROCOD) AS VINT ON VINT.COD = A.PROCOD





ORDER BY 'SOMATT' DESC
