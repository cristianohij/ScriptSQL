A.NFSEMPCOD = B.NFSEMPCOD and

join TBS010 D on A.PROCOD = D.PROCOD

WHERE A.PROCOD LIKE ('%')AND 
NFSDATEMI BETWEEN '20150101' AND '20150630' AND
NFSTIP = 'N' and NFSCAN = 'N' and TESCNTVEN = 'S'


GROUP BY A.PROCOD	







---GZ E INT


WITH 
VENDASGZ AS (select PROCOD,PRODES,MARCOD,
REPLACE(STR(ISNULL(SUM(A.M2_QTD),0),10,4),'.',',') AS VENDAS_GZ
from MSL002 A join TBS010 B ON A.M2_PROCOD = B.PROCOD

where M2_DAT between '20150601'and '20150630' and
M2_PROCOD LIKE ('%') AND 
M2_TIPREG = '01' AND 
M2_REGCAN = 'F'

GROUP BY M2_PROCOD,PRODES,PROCOD,MARCOD),

VENDASINTEGROS AS (SELECT D.PROCOD,D.PRODES,
REPLACE(STR(ISNULL(SUM(A.NFSQTD*NFSQTDEMB),0),10,4),'.',',') AS VENDAS_INT 
FROM TBS0671 A JOIN TBS067 B ON 
A.NFSEMPCOD = B.NFSEMPCOD and 
A.NFSNUM = B.NFSNUM and 
A.SNEEMPCOD = B.SNEEMPCOD and 
A.SNESER = B.SNESER
join TBS042 C on C.TESCOD = A.TESCOD
join TBS010 D on A.PROCOD = D.PROCOD

WHERE A.PROCOD LIKE ('%')AND 
NFSDATEMI between '20150601'and '20150630' and
NFSTIP = 'N' and NFSCAN = 'N' and TESCNTVEN = 'S'


GROUP BY D.PROCOD,D.PRODES)

SELECT PROSTATUS,
RTRIM(LTRIM(A.PROCOD)),
A.PRODES,
rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) as 'MARCA',
PROUM1,
isnull(B.VENDAS_GZ,0)as VendasGZ,
isnull(C.VENDAS_INT,0) as VendasINT

FROM TBS010 A

Full Outer Join VENDASGZ B ON A.PROCOD = B.PROCOD 
Full Outer Join VENDASINTEGROS C ON A.PROCOD = C.PROCOD

WHERE A.PROCOD LIKE ('%') 

GROUP BY PROSTATUS,A.PROCOD, A.PRODES,A.MARCOD,A.MARNOM,PROUM1,B.VENDAS_GZ,C.VENDAS_INT


order by A.PROCOD
