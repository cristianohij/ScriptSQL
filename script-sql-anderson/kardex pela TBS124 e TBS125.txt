DECLARE @DATADE DATETIME , @DATAATE DATETIME, @PROCOD CHAR(15), @ESTLOC INT

SET @ESTLOC = 1

if object_id('TempDB.dbo.#TBS124') is not null
   drop table #TBS124

SELECT
LESCOD,
SINPROCOD,
SINQTD,
SINDAT, 
SINUNI

INTO #TBS124
FROM TBS124 NOLOCK 

WHERE 
SINDAT BETWEEN CONVERT(VARCHAR, @DATADE - DAY(@DATADE) + 1, 103) AND @DATAATE AND
LESCOD in (@ESTLOC) AND 
SINPROCOD = CASE WHEN @PROCOD = 0 THEN SINPROCOD ELSE @PROCOD END 

GROUP BY 
LESCOD,
SINPROCOD,
SINQTD,
SINDAT, 
SINUNI

ORDER BY 
SINDAT

DELETE #TBS124
WHERE 
(SELECT MIN(SINDAT) FROM #TBS124 A WHERE A.SINPROCOD = #TBS124.SINPROCOD AND A.LESCOD = #TBS124.LESCOD GROUP BY LESCOD,SINPROCOD) <> SINDAT

if object_id('TempDB.dbo.#KAR') is not null
   drop table #KAR

SELECT IDENTITY(INT,1,1) AS ORDEM, MAX(KESDAT) AS KESDAT, KESPROCOD, LESCOD INTO #KAR FROM #KARDEX1 GROUP BY KESPROCOD, LESCOD

-- ACREENTAR ULTIMO DIA DA PESQUISA (DATAATE) CASO O PRODUTO NÃO TENHA MOVIMENTO NESSSE DIA, PARA QUE O SALDO DO ITEM POSSA APARECER

DECLARE @A INT

SET @A = 1

WHILE @A <= (SELECT MAX(ORDEM) FROM #KAR) 

	IF(SELECT 1 FROM #KAR WHERE ORDEM = @A AND KESDAT = @DATAATE) = 1
		BEGIN 
			SET @A = @A + 1
		END 
	ELSE
		BEGIN 
			INSERT INTO #KARDEX1
			
			SELECT
			0			AS SINQTD,
			CONVERT(DATETIME,@DATAATE) 	AS KESDAT,
			LESCOD 		AS LESCOD,
			KESPROCOD 	AS KESPROCOD,
			0 			AS KESNFENT,
			0			AS KESENTDEV,
			0			AS KESCANNFSAI,
			0			AS KESCANCUPFIS,
			0			AS KESCANNFDEV,
			0			AS KESMOVENT,
			0			AS KESSALENT,
			0			AS KESNFSAI,
			0			AS KESCUPFIS,
			0			AS KESNFDEVSAI,
			0			AS KESMOVSAI,
			0			AS KESSALSAI,
			(SELECT TOP 1 A.KESUNI FROM #KARDEX1 A WHERE A.KESPROCOD = B.KESPROCOD) AS KESUNI
			
			FROM #KAR B
			
			WHERE
			ORDEM = @A 
			
			SET @A = @A + 1
		END

--------------------------------------------------------------------------------------------------

if object_id('TempDB.dbo.#KARDEX') is not null
   drop table #KARDEX

SELECT

INTO #KARDEX

DECLARE @i INT

SET @i = 2

WHILE @i <= (SELECT MAX(ORDEM) FROM #KARDEX) 

	BEGIN 
	
	UPDATE #KARDEX SET
	SINQTD = (SELECT 
			  SINQTD+			-- QUANTIDADE ANTERIOR
			  -- ENTRADAS
			  KESNFENT+			-- NOTA DE ENTRADA
			  KESENTDEV+		-- ENTRADA DE DEVOLUÇÃO
			  KESCANNFSAI+		-- CANCELAMENTO DE SAIDA DE NOTA
			  KESCANCUPFIS+		-- CANCELAMENTO DE CUPOM FISCAL
			  KESCANNFDEV+		-- CANCELAMENTO DE DEVOLUÇÃO DE SAIDA
			  KESMOVENT+		-- ENTRADAS PELA MOVIMENTAÇÃO INTERNAS
			  KESSALENT-		-- ENTRADAS PELA MANUTENÇÃO DE SALDO
			  -- SAIDAS
			  KESNFSAI-			-- NOTAS FISCAIS DE SAÍDA
			  KESCUPFIS-		-- CUPOM FISCAL
			  KESNFDEVSAI-		-- NOTAS DE DEVOLUÇÃO DE SAÍDA
			  KESMOVSAI-		-- SAÍDA PELA MOVIMENTAÇÃO INTERNA
			  KESSALSAI			-- SAÍDA PELA MANUTENÇÃO DE SALDO
			  			  
			  FROM #KARDEX B WHERE B.ORDEM = @i - 1 )
			 
	WHERE 
	ORDEM = @i AND 
	LESCOD = (SELECT LESCOD FROM #KARDEX WHERE ORDEM = @i - 1 ) AND
	KESPROCOD = (SELECT KESPROCOD FROM #KARDEX WHERE ORDEM = @i - 1)
		
	SET @i = @i + 1

END
