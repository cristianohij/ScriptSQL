
WITH ND1 AS(

SELECT 'SJC' AS 'UNIDA',
(SELECT VENCOD FROM TBS004 C (nolock) WHERE A.VENCOD = C.VENCOD) AS CODVEN,
(SELECT VENNOM FROM TBS004 D (nolock) WHERE D.VENCOD = A.VENCOD ) collate Latin1_General_CI_AS  AS NOMVEN,

AND
 A.CLICOD = NFSCLICOD) AS NFSEMITIDAS

A.CLICGC collate Latin1_General_CI_AS  AS CNPJ,

A.CLICPF collate Latin1_General_CI_AS AS CPF,

(SELECT COUNT(*) FROM cd.SIBD.dbo.TBS067 (NOLOCK) 
WHERE NFSTIP = 'N' AND NFSCAN = 'N' AND NFSDEV ='N'
AND A.CLICOD = NFSCLICOD) AS NFSEMITIDAS


FROM cd.SIBD.dbo.TBS002 A (NOLOCK)) 


SELECT A.UNIDA,A.CODVEN,A.NOMVEN,A.CODCLI,A.NOMCLI,A.NOMFAN,A.CNPJ, A.NFSEMITIDAS, A.CPF FROM TTE A FULL OUTER JOIN ND1 B ON A.CNPJ  = B.CNPJ
																									--FULL OUTER JOIN TTE C ON A.CNPJ = C.CNPJ
WHERE B.CNPJ IS NULL --(A.NOMVEN <> B.NOMVEN OR A.CODCLI <> B. CODCLI) AND (A.CPF NOT LIKE('00000000%') AND A.CPF <> '') --AND (A.CPF = '' OR A.CPF LIKE('00000000%')) 


UNION 

SELECT A.UNIDA,A.CODVEN,A.NOMVEN,A.CODCLI,A.NOMCLI,A.NOMFAN,A.CNPJ,A.NFSEMITIDAS, A.CPF FROM CD A FULL OUTER JOIN ND1 B ON A.CNPJ  = B.CNPJ
