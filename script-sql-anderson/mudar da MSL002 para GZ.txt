DECLARE @anoDe varchar (8),@anoAte varchar (8)

SET @anoDe = '20160801'
SET @anoAte= '20161025'


EXECUTE('
select 
cdprod , 
sum(quant) as quant,
cast(MIN(data) as datetime) AS datini,
max(data) as datfim

FROM movcaixa
where
cast(data as datetime) between "'+@anoDe+'" and "'+@anoAte+'" and 
status = 01 and 
cancelado <> "S"
group by cdprod ') at MYSQLGZ

CAST('2000-01-01' AS DATE);
