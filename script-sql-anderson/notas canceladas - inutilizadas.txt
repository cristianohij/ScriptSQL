DECLARE @Data_de DATETIME , @Data_ate DATETIME , @NFSTIP CHAR(1) , @CFOP CHAR(100), @NFSSUBTIP CHAR(13), @CNPJ CHAR(14)
SET @Data_de = '20170401'
SET @Data_ate = '20170531'
SET @CNPJ = '65069593000279'

SELECT 
LTRIM(RTRIM(@CNPJ)) +';'+
CONVERT(VARCHAR(10),ENFNUM) + ';'+
CONVERT(CHAR(10),ENFDATEMI,103) +';'+
CASE ENFSIT
	WHEN 7 THEN 'CANCELADA'
	WHEN 11 THEN 'INUTILIZADA'
END 
	

FROM 
TBS080 NOLOCK 

WHERE 
ENFDATEMI BETWEEN @Data_de AND @Data_ate AND 
ENFSIT  IN (7,11)

ORDER BY 
ENFDATEMI
