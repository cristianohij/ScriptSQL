-- CNPJ DA TANBY|DATA NF|NF|VALOR NF|BASE ICMS|VALOR ICMS

SELECT 
'SAIDA' AS NOME,
CONVERT(CHAR(10),B.NFSDATEMI,102)		AS DATAS,
CONVERT(CHAR(10),B.NFSDATEMI,103)		AS DAT,			-- DATA NF
A.NFSNUM 		AS NF,			-- NF 
SUM(ROUND(dbo.NFSTOTITEST(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) 	AS NFSTOTITEST, -- VALOR NF
CASE WHEN A.NFSCFOP COLLATE DATABASE_DEFAULT NOT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S')
	THEN SUM(ROUND(dbo.NFSBASICMS(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2)) 	
	ELSE 0
END AS NFSBASICMS, 			-- BASE ICMS
CASE WHEN A.NFSCFOP COLLATE DATABASE_DEFAULT NOT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S')
	THEN SUM(round(dbo.NFSVALICMS(A.NFSEMPCOD, A.NFSNUM, A.SNEEMPCOD, A.SNESER, A.NFSITE),2))
	ELSE 0
END AS NFSVALICMS 			-- VALOR ICMS 

INTO #NFS
FROM 
TBS0671 A (NOLOCK)
LEFT JOIN TBS067 B (NOLOCK) ON A.NFSEMPCOD = B.NFSEMPCOD AND A.NFSNUM = B.NFSNUM AND A.SNEEMPCOD = B.SNEEMPCOD AND A.SNESER = B.SNESER
LEFT JOIN TBS080 D (NOLOCK) ON A.NFSNUM = D.ENFNUM AND A.SNEEMPCOD = D.SNEEMPCOD AND A.SNESER = D.SNESER 
-- LEFT JOIN TBS002 C (NOLOCK) ON B.NFSCLICOD = C.CLICOD

WHERE
-- SUBSTRING(CONVERT(VARCHAR(12),B.NFSDATEMI,111),1,7) = @anoDe+'/'+@mesDe AND
B.NFSDATEMI BETWEEN @dataDe AND @dataAte AND 
NFSTIP IN('N','C') and 
D.ENFSIT = 6 AND
D.ENFTIPDOC = 1 

GROUP BY 
A.NFSCFOP ,A.NFSNUM, CONVERT(CHAR(10),B.NFSDATEMI,103), CONVERT(CHAR(10),B.NFSDATEMI,102)

-- CRIANDO TABELA TEMP COM OS VALORES REAIS DE DESCONTO
if object_id('TempDB.dbo.#TMP') is not null
BEGIN 
   drop table #TMP
END
   
SELECT
M2_DAT,
A.M2_TRI,
A.M2_CST,
--A.M2_ITE,
--A.M2_LOJ, 
A.M2_CUP, 
A.M2_COO, 
A.M2_ECF, 
A.M2_CXA ,
M2_PROCOD,
M2_VALBRU,
M2_VALBRU - M2_ABT + M2_ACRCUP - 
CASE WHEN  A.DESCONTO <> ISNULL(B.DESCONTO,0) AND M2_ITE = (SELECT TOP 1 M2_ITE 
																FROM #MSLAUX E 
																
																WHERE 
																A.M2_COO = E.M2_COO and A.M2_ECF = E.M2_ECF AND A.M2_CUP = E.M2_CUP 
																
																ORDER BY 
																M2_DESCUP DESC)
	THEN
		CASE WHEN A.DESCONTO > ISNULL(B.DESCONTO,0) 
			THEN ROUND(M2_DESCUP,2) - (A.DESCONTO - ISNULL(B.DESCONTO,0))
			ELSE ROUND(M2_DESCUP,2) + (ISNULL(B.DESCONTO,0) - A.DESCONTO)
		END
	ELSE ROUND(M2_DESCUP,2)
END AS M2_LIQUIDO
	
INTO #TMP	
FROM (
	SELECT 
	M2_DAT,
	M2_ITE,M2_PROCOD,M2_TRI,M2_CST,
	M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , 
	M2_DESCUP ,
	M2_ACRCUP,
	M2_ABT ,
	M2_VALBRU,
	SUM(ISNULL(ROUND(M2_DESCUP,2),0)) OVER (PARTITION BY M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA ) AS DESCONTO
					
	FROM 
	#MSLAUX 
					
	WHERE
	M2_STATUS = 1 ) AS A 
			
	LEFT JOIN (
	SELECT 
	M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , M2_VALBRU AS DESCONTO 
					
	FROM 
	#MSLAUX 
					
	WHERE
	M2_STATUS = 2 ) AS B ON A.M2_LOJ = B.M2_LOJ AND A.M2_CUP = B.M2_CUP AND A.M2_COO= B.M2_COO AND A.M2_ECF = B.M2_ECF AND A.M2_CXA = B.M2_CXA

-- CNPJ DA TANBY|DATA NF|NF|VALOR NF|BASE ICMS|VALOR ICMS 

SELECT
'ENTRADA' AS NOME,
CONVERT(CHAR(10),B.NFEDATEFE,102) AS DATAS,
CONVERT(CHAR(10),B.NFEDATEFE,103) AS DAT,
B.NFENUM AS NF,
sum(ROUND(NFETOTOPEITE,2)) AS NFETOTOPEITE,
CASE WHEN A.NFECFOP COLLATE DATABASE_DEFAULT NOT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E')
	THEN SUM(ROUND(A.NFEBASICMS,2)) 
	ELSE 0
END AS NFEBASICMS, 
CASE WHEN A.NFECFOP COLLATE DATABASE_DEFAULT NOT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E')
	THEN SUM(round(A.NFEVALICMS,2)) 
	ELSE 0
END AS NFEVALICMS

INTO #NFE
FROM TBS0591 A (NOLOCK)
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND 
A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND  A.SEREMPCOD = B.SEREMPCOD
LEFT JOIN TBS080 D (NOLOCK) ON A.NFENUM = D.ENFNUM AND A.NFECOD = D.ENFCODDES

WHERE 
((D.ENFSIT = 6 AND D.ENFFINEMI = 4 AND D.ENFTIPDOC = 0) OR ((NFENOSFOR = 'N' OR NFENOSFOR = '') AND (B.NFECAN = 'N' OR B.NFECAN = '') AND NFEDATEFE <> '' )) AND
B.NFEDATEFE BETWEEN @dataDe AND @dataAte AND 
A.NFECFOP COLLATE DATABASE_DEFAULT NOT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E' and PAGAR = 'S')
-- SUBSTRING(CONVERT(VARCHAR(12),B.NFEDATEFE,111),1,7) = @anoDe+'/'+@mesDe AND

GROUP BY 
CONVERT(CHAR(10),B.NFEDATEFE,103),B.NFENUM,A.NFECFOP, CONVERT(CHAR(10),B.NFEDATEFE,102)
