-- SELECT * FROM TBS057 NOLOCK

DECLARE @DataDeE date, @DataAteE date, @TituloDe VARCHAR, @TituloAte VARCHAR, @VencDe date, 
@VencAte date, @zero int, @NomFor char(50) , @Portador int, @Opcao int , @codfor char(200), 
@mobdes char(200), @BaiDe date, @BaiAte date

set @DataDeE = '20150101'
set @DataAteE = '20161231'
set @VencDe = '20150101'
set @VencAte = '20161231'
set @BaiDe = ''
set @BaiAte = ''
set @TituloDe = ''
set @TituloAte = ''
set @zero = 0
set @NomFor = '' 
set @Portador = 0
set @Opcao = 8
set @codfor = 0


if object_id('TempDB.dbo.#CON') is not null
      drop table #CON

SELECT 
CASE WHEN CPADATBAI = '17530101' AND CPADATVENREA >= CONVERT(DATE,GETDATE())
	THEN 'EM ABERTO A VENCER'
	ELSE 
		CASE WHEN CPADATBAI = '17530101' AND CPADATVENREA < CONVERT(DATE,GETDATE())
			THEN 'EM ABERTO VENCIDOS'
			ELSE
				CASE WHEN CPADATBAI <> '17530101' AND CPADATBAI > CPADATVENREA
					THEN 'PAGOS COM ATRASO'
					ELSE
						CASE WHEN CPADATBAI <> '17530101' AND CPADATBAI < CPADATVENREA
							THEN 'PAGOS ANTES DO VENCIMENTO'
							ELSE
								CASE WHEN CPADATBAI <> '17530101' AND CPADATBAI = CPADATVENREA
									THEN 'PAGOS NO VENCIMENTO'
									ELSE ''
								END
						END
				END
		END
END AS SITUACAO,
1 AS CONTADOR,
PFXCOD AS PFX,
CPATIT AS TITULO,
CPAPAR AS PAR,
TPTCOD AS TIPO,
PORCOD AS PORT,
CPADATEMI AS DATA_EMI,
CPADATVENREA AS VENC_REAL,
CPADATBAI AS BAIXA,
subString(convert(char(8),CPADATVENREA,112),1,4) AS ANO_VEN_REAL,
subString(convert(char(8),CPADATVENREA,112),5,2) AS MES_VEN_REAL,
subString(convert(char(8),CPADATVENREA,112),7,2) AS DIA_VEN_REAL,
CASE WHEN DATEDIFF(DAY,GETDATE(),CPADATVENREA)<0 OR CPADATBAI > '17530101' THEN 0 ELSE DATEDIFF(DAY,GETDATE(),CPADATVENREA) END AS DIAS_PARA_VENCER,
dbo.CPADIAATR(CPAEMPCOD,PFXEMPCOD,A.FOREMPCOD,PFXCOD,CPATIT,CPAPAR,A.FORCOD) AS DIA_ATRA,
CPAVAL AS VALOR_ORIG,
CPAVALRES AS RESIDUO,
CPAVALABT AS ABATIMENTO,
CPAVALACR AS ACRESCIMO,
CPAVALPAG AS VALOR_PAGO,
dbo.CPAVALSDO(CPAEMPCOD,PFXEMPCOD,A.FOREMPCOD,PFXCOD,CPATIT,CPAPAR,A.FORCOD) as SALDO_ABERTO,
A.MOBCOD AS CODBAI,
ISNULL(D.MOBDES,'EM ABERTO') AS MOTBAI,
RTRIM(LTRIM(CPAOBS)) AS OBS,
RTRIM(A.FORCOD) AS COD_FOR,
RTRIM(A.FORNOM) AS FORNECEDOR,
RTRIM(LTRIM(FOREND)) +' , '+ RTRIM(LTRIM(FORNUM)) + ' , ' + RTRIM(LTRIM(FORBAI)) AS ENDFOR,
RTRIM(LTRIM(FORCEP)) AS CEPFOR,
RTRIM(LTRIM(ISNULL(C.MUNNOM,''))) AS MUNFOR

INTO #CON
FROM 
TBS057 A (NOLOCK) 
LEFT JOIN TBS074 D (NOLOCK) ON A.MOBCOD = D.MOBCOD
LEFT JOIN TBS006 B (NOLOCK) ON A.FORCOD = B.FORCOD
LEFT JOIN TBS003 C (NOLOCK) ON B.MUNCOD = C.MUNCOD
					  	
WHERE 
	CPADATEMI  BETWEEN  @DataDeE  and  @DataAteE  AND 
	CPADATVENREA  BETWEEN  @VencDe  and  @VencAte  AND 
	CPADATBAI BETWEEN CASE WHEN @BaiDe = '' THEN '17530101' ELSE @BaiDe END and CASE WHEN @BaiAte = '' THEN '20201231' ELSE @BaiAte END AND 
	LTRIM(RTRIM(CONVERT(CHAR(20),CPATIT))) = case when @TituloAte = '' then LTRIM(RTRIM(CONVERT(CHAR(20),CPATIT))) else @TituloAte end and
	PORCOD <= CASE WHEN @zero=0 THEN 9999999 ELSE 0 END  AND
	A.FORNOM LIKE(RTRIM(CASE WHEN @NomFor = '' THEN '% %' ELSE LTRIM(RTRIM(@NomFor)) +'%' END)) AND
	PORCOD = CASE WHEN @Portador = 0 THEN PORCOD ELSE @Portador END AND 
	B.FORCOD = CASE WHEN @codfor = 0 THEN B.FORCOD ELSE @codfor END


;WITH  
CON_PAGAR AS (
SELECT  
SITUACAO,
CONTADOR,
PFX,
TITULO,
PAR,
TIPO,
PORT,
DATA_EMI,
VENC_REAL,
BAIXA,
ANO_VEN_REAL,
MES_VEN_REAL,
DIA_VEN_REAL,
DIAS_PARA_VENCER,
DIA_ATRA,
VALOR_ORIG,
RESIDUO,
ABATIMENTO,
ACRESCIMO,
VALOR_PAGO,
SALDO_ABERTO,
CODBAI,
MOTBAI,
OBS,
COD_FOR,
FORNECEDOR,
ENDFOR,
CEPFOR,
MUNFOR 
FROM 
#CON)

SELECT * FROM CON_PAGAR
WHERE
MOTBAI IN ('EM ABERTO')



OPERACAO IN (@Opcao) and
MOTBAI IN (SELECT DISTINCT ISNULL(A.MOBDES,'EM ABERTO')
FROM TBS074 A (NOLOCK) 
FULL OUTER JOIN TBS057 B (NOLOCK) ON A.MOBCOD = B.MOBCOD 
ORDER BY A.MOBDES)



SELECT DISTINCT ISNULL(A.MOBDES,'EM ABERTO')
FROM TBS074 A (NOLOCK) 
FULL OUTER JOIN TBS057 B (NOLOCK) ON A.MOBCOD = B.MOBCOD
