SELECT 
CASE WHEN CREDATBAI = '17530101' AND CREDATVENREA >= CONVERT(DATE,GETDATE())
	THEN 'EM ABERTO A VENCER'
	ELSE 
		CASE WHEN CREDATBAI = '17530101' AND CREDATVENREA < CONVERT(DATE,GETDATE())
			THEN 'EM ABERTO VENCIDOS'
			ELSE
				CASE WHEN CREDATBAI <> '17530101' AND CREDATBAI > CREDATVENREA
					THEN 'RECEBIDOS COM ATRASO'
					ELSE
						CASE WHEN CREDATBAI <> '17530101' AND CREDATBAI < CREDATVENREA
							THEN 'RECEBIDOS ANTES DO VENCIMENTO'
							ELSE
								CASE WHEN CREDATBAI <> '17530101' AND CREDATBAI = CREDATVENREA
									THEN 'RECEBIDOS NO VENCIMENTO'
									ELSE ''
								END
						END
				END
		END
END AS SITUACAO,
1 AS CONTADOR,
PFXCOD AS PFX,
CRETIT AS TITULO,
CREPAR AS PAR,
TPTCOD AS TIPO,
RTRIM(CLICOD) AS COD_CLI,
RTRIM(CLINOM) AS NOM_CLI,
CREVAL AS VALOR_ORIGINAL,
CREDATEMI AS EMISAO,
CREDATVENREA AS VENC_REAL,
CASE WHEN CONVERT(CHAR(10),CREDATBAI,103) = '01/01/1753' THEN '' ELSE CONVERT(CHAR(10),CREDATBAI,103) END AS BAIXA,
dbo.CREDIAATR(CREEMPCOD,PFXEMPCOD,CLIEMPCOD,PFXCOD,CRETIT,CREPAR,CLICOD) AS DIAS_ATRASO,
CASE WHEN DATEDIFF(DAY,GETDATE(),CREDATVENREA)<0 OR CONVERT(CHAR(10),CREDATBAI,103) <> '01/01/1753' THEN 0 ELSE DATEDIFF(DAY,GETDATE(),CREDATVENREA) END AS DIAS_PARA_VENCER,
A.MOBCOD AS CODBAI,
ISNULL(D.MOBDES,'EM ABERTO') AS MOTBAI,
A.PORCOD AS PORT,
ISNULL(PORNOM,'') AS PORTNOM,
CRETAXJUR AS TAXA,
dbo.CREVALJUR(CREEMPCOD,PFXEMPCOD,CLIEMPCOD,PFXCOD,CRETIT,CREPAR,CLICOD) AS JUROS,
CREVALABT AS ABATIMENTO,
CREVALACR AS ACRESCIMO,
CRERESBAN AS ACRESCBANCO,
CREVALREC AS RECEBIDO,
CREVALRES AS RESIDUO,
CRERESBAN AS RESBANCO,
dbo.CREVALSDO(CREEMPCOD,PFXEMPCOD,CLIEMPCOD,PFXCOD,CRETIT,CREPAR,CLICOD) AS SALDO,
CRENOSNUM AS NOSSO_NUM,
CRENUMBAN AS BANNUM,
CREOBS AS OBS,
RTRIM(subString(convert(char(8),CREDATVENREA,112),1,4)) as ANO_VENC_REAL,
LTRIM(RTRIM(subString(convert(char(8),CREDATVENREA,112),5,2))) as MES_VENC_REAL,
LTRIM(RTRIM(subString(convert(char(8),CREDATVENREA,112),7,2))) as DIA_VENC_REAL

INTO #RCB
 
FROM 
TBS056 A (NOLOCK)
LEFT JOIN TBS074 D (NOLOCK) ON A.MOBCOD = D.MOBCOD
LEFT JOIN TBS063 E (NOLOCK) ON A.PORCOD = E.PORCOD


WHERE 
CREDATEMI BETWEEN @DataDeE AND @DataAteE AND 
CREDATVENREA BETWEEN @VencDe and @VencAte AND 
CREDATBAI BETWEEN CASE WHEN @BaiDe = '' THEN '17530101' ELSE @BaiDe END and CASE WHEN @BaiAte = '' THEN '20201231' ELSE @BaiAte END AND
LTRIM(RTRIM(CONVERT(CHAR(20),CRETIT))) = case when @TituloAte = '' then LTRIM(RTRIM(CONVERT(CHAR(20),CRETIT))) else @TituloAte end and
A.PORCOD <= CASE WHEN @zero=0 THEN 999999999 ELSE 0 END AND 
A.CLINOM LIKE(RTRIM(CASE WHEN @NomCli = '' THEN '% %' ELSE LTRIM(RTRIM(@NomCli)) +'%' END)) AND
A.PORCOD = CASE WHEN @Portador = 0 THEN A.PORCOD ELSE @Portador END AND 
A.CLICOD = CASE WHEN @codcli = 0 THEN A.CLICOD ELSE @codcli END
