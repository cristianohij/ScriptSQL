if object_id('TempDB.dbo.#NFE') is not null
   begin
      drop table #NFE
   end
   
		SELECT
		IDENTITY(INT,1,1) 				AS ORDEM ,
		CONVERT(CHAR (10), NFEDATEMI,103) AS NFEDATEMI,
		CONVERT(CHAR (10), NFEDATEFE,103) AS NFEDATEFE,
		-- CONVERT(CHAR (10), NFEDATEFE,102) AS DATA,
		B.NFECOD, 
		B.NFENOM,
		A.NFENUM,
		A.PROCOD,
		C.PRODES,
		C.PROUM1,
		A.NFEQTD * A.NFEQTDEMB AS QTD,
		LTRIM(RTRIM(STR(COMCOD))) + RTRIM(LTRIM(COMNOM)) AS COMNOM
		
		INTO #NFE		 
		FROM 
		TBS0591 A (NOLOCK)
		LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND 
		A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND  A.SEREMPCOD = B.SEREMPCOD
		LEFT JOIN #TBS010 C (NOLOCK) ON A.PROCOD = C.PROCOD
		  
		WHERE 
		A.LESCOD= 7 AND
		NFEDATEFE between @dateDe and @dateAte and 
		A.NFETIP ='N' and 
		NFECAN <> 'S' AND
		-- B.NFENOM NOT LIKE ('%TANBY%') AND
		--A.NFECOD NOT IN ((SELECT FORCOD FROM #GRUPOFOR)) AND 
		--A.NFECFOP IN ('1.102','2.102','1.202','2.202','1.403','2.403','2.411','1.411','1.653','2.653','1.922','2.922') AND
		A.PROCOD IN (SELECT PROCOD FROM #TBS010)

		--GROUP BY 
		--NFEDATEFE, A.PROCOD, A.NFENUM,C.PRODES,C.PROUM1, COMCOD, COMNOM
		
		ORDER BY 
		B.NFENOM, CONVERT(CHAR (10), NFEDATEMI,102)

SELECT 
NFECOD AS 'Código do Fornecedor', 
NFENOM as 'Nome do Fornecedor',
COUNT(*) AS 'Ocorrências'

FROM #NFE

GROUP BY 
NFECOD, 
NFENOM

order by 'Ocorrências' desc 


select 
ORDEM as 'Ordem',
NFEDATEMI as 'Data de Emissão',
NFEDATEFE as 'Data de Efetivação',
NFECOD as 'Código do Fornecedor', 
NFENOM as 'Nome do Fornecedor',
NFENUM as 'NF',
PROCOD as 'Código Produto',
PRODES as 'Descrição',
PROUM1 as 'Uni1',
replace(QTD,'.',',') as 'Quantidade',
COMNOM as 'Comprador'
from #NFE

SELECT

B.PRODES as 'Descrição',

WHERE 
LMELOCEST = 7 AND 
LMEDATHOR > '20170101' AND 
LMEINFALT = 'E' AND 
LMEACA = 'E' AND 
LMEQTDMOV <> 0 

GROUP BY 
A.PROCOD,
B.PRODES

order by 'Ocorrências' desc
