-- olhar os preços de entrada e olhar os preços de saida
SELECT NFECHAACE , NFEPRE,NFEPREXML , NFEDATENT, NFEQTDEMB, PROCOD ,NFEDES, NFEUNI, NFEQTD
FROM TBS0591 A (NOLOCK) LEFT JOIN TBS059 B (nolock) ON A.SERCOD=B.SERCOD and A.NFETIP=B.NFETIP and A.NFECOD=B.NFECOD and A.NFENUM=B.NFENUM 
WHERE --PROCOD = '1640054' ORDER BY NFEDATENT DESC
NFEQTDEMB = 0


SELECT C.NFSNUM, C.NFSPRE,C.NFSQTDEMB, B.NFSDATEMI , * FROM TBS0671 C (nolock) JOIN TBS067 B (nolock)  ON 
C.NFSEMPCOD = B.NFSEMPCOD and 
C.NFSNUM = B.NFSNUM and 
C.SNEEMPCOD = B.SNEEMPCOD and 
C.SNESER = B.SNESER
WHERE --PROCOD = '1640054' ORDER BY B.NFSDATEMI DESC  
NFSQTDEMB = 0
