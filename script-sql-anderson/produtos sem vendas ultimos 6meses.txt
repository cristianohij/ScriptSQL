
SELECT 
'ND' AS 'UNIDADE' ,
rtrim(A.PROLOCFIS) AS 'LOC',
RTRIM(LTRIM(A.PROSTATUS)) AS 'STA',
rtrim(ltrim(A.PROCOD))  as 'CODIGO',
A.PRODES  as 'DESCRICAO',							
case when len(A.MARCOD) = 4 then rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) else right(('00' + ltrim(str(A.MARCOD))),3) + ' - ' + rtrim(A.MARNOM) end as 'MARCA',
A.PROUM1 collate Latin1_General_BIN as 'UN.1',
ISNULL(REPLACE(STR(SALDO.EST,10,4),'.',','),0) AS 'SALDO ESTOQUE',
ISNULL(REPLACE(STR(SALDO.LOJA,10,4),'.',','),0) AS'SALDO LOJA' ,
ISNULL(REPLACE(STR(SALDO.SALTT,10,4),'.',','),0) AS 'SOMA EST+LOJ',
ISNULL(REPLACE(STR(SALDO.RES,10,4),'.',','),0) AS RESER,
ISNULL(REPLACE(STR(SALDO.PEN,10,4),'.',','),0) AS PENDENTE,
ISNULL(REPLACE(STR(SALDO.CMP,10,4),'.',','),0) AS COMPRAS,
ISNULL(REPLACE(STR(VLOJA.TT,10,4),'.',','),0) AS 'VENDAS GZ',
ISNULL(REPLACE(STR(VINT.TT,10,4),'.',','),0) AS 'VENDAS INT'


FROM TBS010 A (NOLOCK)

FULL OUTER JOIN   
(SELECT A.M2_PROCOD AS COD, 
ISNULL(SUM(M2_QTD),0) AS TT,
SUM(M2_VALTOT)-SUM(M2_ABT) AS VALOR

FROM MSL002 A (nolock) WHERE M2_DAT between '20150901' and '20160229' and M2_TIPREG = '01' AND M2_REGCAN = 'F'   GROUP BY M2_PROCOD ) AS VLOJA ON A.PROCOD = VLOJA.COD


FULL OUTER JOIN   
(SELECT PROCOD AS COD ,
ISNULL(SUM(A.NFSQTD*NFSQTDEMB),0) AS TT,
sum(dbo.NFSTOTITEST(A.NFSEMPCOD,A.NFSNUM,A.SNEEMPCOD,A.SNESER,A.NFSITE)) as VALOR

FROM TBS0671 A (nolock) JOIN TBS067 B (nolock)  ON 
										A.NFSEMPCOD = B.NFSEMPCOD and 
										A.NFSNUM = B.NFSNUM and 
										A.SNEEMPCOD = B.SNEEMPCOD and 
										A.SNESER = B.SNESER
										WHERE NFSDATEMI between '20150901' and '20160229' 
										and NFSTIP = 'N' and NFSCAN = 'N' and NFSDEV='N' 
										and  NFSCLINOM not Like('%BEST BAG%') and NFSCLINOM not Like('%BEST OFFICE%') 
										and NFSCLINOM not Like('%MISASPEL%') and NFSCLINOM not Like('%PAPELYNA%') 
										and  NFSCLINOM not Like('%TANBY%')
										GROUP BY PROCOD) AS VINT ON VINT.COD = A.PROCOD

										
FULL OUTER JOIN   
(SELECT PROCOD AS COD,SUM(ESTQTDATU-ESTQTDRES) AS SALTT, SUM(ESTQTDRES) AS RES , SUM(ESTQTDCMP) AS CMP, SUM(ESTQTDPEN) AS PEN,
ISNULL((SELECT SUM(ESTQTDATU-ESTQTDRES) FROM TBS032 B (nolock) WHERE ESTLOC =1 AND A.PROCOD = B.PROCOD GROUP BY PROCOD),0) AS EST,
ISNULL((SELECT SUM(ESTQTDATU-ESTQTDRES) FROM TBS032 B (nolock) WHERE ESTLOC =2 AND A.PROCOD = B.PROCOD GROUP BY PROCOD),0) AS LOJA	
FROM TBS032 A (nolock) WHERE ESTLOC IN (1,2) GROUP BY PROCOD) AS SALDO ON SALDO.COD = A.PROCOD

where (SALDO.SALTT <> 0 OR SALDO.RES <> 0 OR SALDO.CMP <> 0 OR SALDO.PEN <>0)
		AND ISNULL(VLOJA.TT,0) = 0 AND ISNULL(VINT.TT,0) = 0 --  AND PROCOD IS NOT NULL					
		
AND A.MARCOD IN('288','916')



ORDER BY PROCOD
