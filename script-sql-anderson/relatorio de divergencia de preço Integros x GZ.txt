if object_id ('TempDB.dbo.#T') is not null 
	drop table #T;

select 
rtrim(c.PROCOD) as Código,
rtrim(ltrim(c.PRODES)) as Descrição,
case when len(c.MARCOD) = 4 then 
	rtrim(c.MARCOD) + ' - ' + rtrim(c.MARNOM) 
	else right(('00' + ltrim(str(c.MARCOD))),3) + ' - ' + rtrim(c.MARNOM) end as 'MARCA',
CASE WHEN PROUM1QTD = 1
	THEN RTRIM((SELECT UNIDES FROM TBS011 D (NOLOCK) WHERE c.PROUM1 = D.UNICOD))   
	ELSE
		CASE WHEN PROUM1QTD > 1 
		THEN rtrim(PROUM1) + ' C/' + rtrim(CAST(PROUM1QTD AS DECIMAL(10,0))) +''+ RTRIM(PROUMV) 
		ELSE '' 
	END 
END  as UN1,
a.termvenda as 'Preço GZ 1',
CASE WHEN d.TDPVALPROI<=GETDATE() AND d.TDPVALPROF >= GETDATE() AND d.TDPPROLOJ = 'S' 
	THEN d.TDPPREPRO1 
	ELSE d.TDPPRELOJ1 
END AS 'Preço 1',

CASE WHEN PROUM2QTD > 0 and TDPPRELOJ2 > 0
	THEN rtrim(PROUM2) + ' C/' + rtrim(CAST(PROUM2QTD AS DECIMAL(10,0)))+''+ RTRIM(PROUM1) 
	ELSE '' 
END  as UN2,
b.termvenda*c.PROUM2QTD as 'Preço GZ 2' ,
CASE WHEN c.PROUM2QTD > 0 
		THEN CASE WHEN d.TDPVALPROI<=GETDATE() AND d.TDPVALPROF >= GETDATE() AND d.TDPPROLOJ = 'S'
				THEN d.TDPPREPRO2*c.PROUM2QTD 
				ELSE d.TDPPRELOJ2*c.PROUM2QTD 
			END
		ELSE 0 
END AS 'Preço 2'
--into #T  
from 
MYSQLGZ...estoque a
left join TBS010 c on CONVERT(NUMERIC,c.PROCOD) = a.cdprod
left join MYSQLGZ...barrarel b on c.PROUM2QTD = b.multiplos and CONVERT(NUMERIC,c.PROCOD) = b.cdprod
left join TBS031 d on CONVERT(NUMERIC,d.TDPPROCOD) = a.cdprod

where 
a.cdprod in (1640054,1080067,1640020,0470438)






select 
D.TDPPROCOD, 
D.TDPPRELOJ1 AS LOJA1,
D.TDPPRELOJ2 AS LOJA2,
D.TDPPRELOJ3 AS LOJA3,
D.TDPPRELOJ4 AS LOJA4,
D.TDPPREPRO1 AS PRO1,
D.TDPPREPRO2 AS PRO2,
D.TDPVALPROI AS PROINI,
D.TDPVALPROF AS PROFIN
FROM TBS031 D (NOLOCK) 
WHERE D.TDPPROLOJ = 'S' AND 
D.TDPPREPRO1 <> D.TDPPREPRO2 AND 
TDPPREPRO2 > 0
