-- Criar uma tabela unica , juntando as quantidades vendidas, seus valores e datas dos pedidos , das tabelas TBS058, TBS0761 (TBS076), usando UNION

if object_id('TempDB.dbo.#TES1') is not null
   drop table #TES1

CREATE TABLE 
#TES1 
	(PENNUM INT NOT NULL,
	
	PENCOD CHAR(15) NOT NULL,
	
	PENVALITE  MONEY NOT NULL,
	
	PENQTDITE DECIMAL(12,5) NOT NULL,
	
	PENDAT DATE NOT NULL,
	
	SIT CHAR (1) NOT NULL,
	
	ESTLOC INT NOT NULL,
	
	PENDIA INT NOT NULL,
	
	PENCMP DECIMAL(12,5) NOT NULL)

INSERT INTO 
#TES1
	(PENNUM,PENCOD,PENVALITE,PENQTDITE,PENDAT,SIT,ESTLOC,PENDIA,PENCMP)
	
	SELECT
	PRPNUM AS PED , 
	B.PROCOD AS COD, 
--	(PRPPRELIQ/PRPQTDEMB)*(PRPQTD*PRPQTDEMB) as VAL,
--	(PRPQTD*PRPQTDEMB) AS QTD ,
	SUM(PRPPRELIQ*PRPQTD) as VAL,
	SUM(PRPQTD*PRPQTDEMB) AS QTD ,
	PDVDATCAD AS DATCAD ,
	PRPSIT AS SIT,
	PRPESTLOC AS ESTLOC,
	DATEDIFF(DAY,PDVDATCAD,getdate()) AS DIAS_PEN,
	-- SUM(A.ESTQTDCMP) AS CMP
	(SELECT SUM(A.ESTQTDCMP) FROM  TBS032 A (NOLOCK) WHERE A.PROCOD = B.PROCOD) AS CMP 
	
	FROM TBS058 B (NOLOCK) 
	LEFT JOIN TBS055 (nolock) on B.PRPNUM=TBS055.PDVNUM
	-- LEFT JOIN TBS032 A (NOLOCK) ON A.PROCOD = B.PROCOD 
	
	WHERE 
	PRPSIT='P' and
	PRPMOVEST='S' 
	
	GROUP BY 
	-- PRPNUM ,B.PROCOD,PRPPRELIQ,PRPQTDEMB,PRPQTD,PDVDATCAD,PRPSIT,PRPESTLOC
	PRPNUM ,B.PROCOD,PDVDATCAD,PRPSIT,PRPESTLOC		
	
--- ACRESCENTAR O RANK 
if object_id('TempDB.dbo.#T') is not null
   drop table #T
SELECT rank() OVER (ORDER BY PENCOD,PENDAT,PENNUM) AS PENIDE,*
INTO #T
FROM #TES1 

-- ACUMULANDO VENDAS 
if object_id('TempDB.dbo.#TES') is not null
   drop table #TES

SELECT * ,PENCMP - (SELECT SUM(PENQTDITE) FROM #T AS TI WHERE TI.PENIDE <= TA.PENIDE AND TI.PENCOD = TA.PENCOD ) AS PENACU,
CASE WHEN PENCMP - (SELECT SUM(PENQTDITE) FROM #T AS TI WHERE TI.PENCOD = TA.PENCOD ) >= 0
	THEN 'ATENDE'
	ELSE 'NAO ATENDE'
	END AS SITUACAO
INTO #TES
FROM 
#T AS TA 

		
			
-- pedidos de vendas pendentes 

; WITH PEN AS(select sum(PENVALITE) as VALPEN ,count(DISTINCT PENCOD) AS CONTPEN from #TES (nolock) ),
	
-- pedidos de vendas pendentes com compras atende
PENCOM AS (select sum(PENVALITE) AS VALPENCOM ,count(DISTINCT PENCOD) AS CONTPENCOM
			from #TES A (nolock)  
						
			where 
			SITUACAO = 'ATENDE' ) , 
						
-- pedidos de vendas pendentes com compras não atende
PENCOMN AS (select sum(PENVALITE) AS VALPENCOMN ,count(DISTINCT PENCOD) AS CONTPENCOMN
			from #TES A (nolock)  
		
			where 
			SITUACAO = 'NAO ATENDE' AND PENCMP > 0 ) , 

-- pedidos de vendas pendentes sem compras
PENSEMCOM AS (select sum(PENVALITE) AS VALPENSEMCOM,count(DISTINCT PENCOD) AS CONTPENSEMCOM
			  from #TES A (nolock)  
			  			  
			  where 
			  PENCMP <= 0) , 

-- período versus número de dias
-- sem compras
