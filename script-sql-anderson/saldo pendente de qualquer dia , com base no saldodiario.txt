DECLARE @DATADE DATETIME , @DATAATE DATETIME, @PROCOD CHAR(15), @ESTLOC int, @MARCOD INT, @MARNOM CHAR(60), @PRODES CHAR(60)
SET @DATADE = '20170601'
SET @DATAATE = '20170610'
SET @PROCOD = ''
SET @ESTLOC = 1
SET @PROCOD = '1640054'
SET @MARCOD = 0
SET @MARNOM = ''
SET @PRODES = ''

if object_id('TempDB.dbo.#INICIO') is not null 
drop table #INICIO

select 
DATEADD(day,1,ESTDATSAL) AS ESTDATSAL,
ESTLOC AS LESCOD,
PROCOD,
ESTQTDPEN

INTO #INICIO
FROM SALDODIARIO WHERE ESTQTDPEN <> 0 AND ESTDATSAL = '20161231'

--------------------------------------------------------------------------------------------------------------------------------------------------

if object_id('TempDB.dbo.#MOV') is not null 
drop table #MOV

SELECT 
A.DATA,
A.LMELOCEST AS LESCOD,
A.PROCOD,
ISNULL(B.LEMQTD,0) AS LEMQTDENT,
ISNULL(C.LEMQTD,0) AS LEMQTDSAI

INTO #MOV
FROM 
(	SELECT DISTINCT CONVERT(DATE,LMEDATHOR) AS DATA, PROCOD, LMELOCEST 
	
	FROM TBS051 NOLOCK 
	
	WHERE
	LMEINFALT ='P' AND 
	CONVERT(DATE,LMEDATHOR) BETWEEN '20170101' AND CONVERT(DATE,getdate()-1)) AS A

LEFT JOIN
(
	SELECT
	 
	CONVERT(DATE,LMEDATHOR)AS DATA,
	LMELOCEST,
	PROCOD,
	SUM(LMEQTDMOV) AS LEMQTD
	
	FROM TBS051 A (NOLOCK) 
	
	WHERE
	LMEINFALT ='P' AND 
	CONVERT(DATE,LMEDATHOR) BETWEEN '20170101' AND CONVERT(DATE,getdate()-1) AND
	LMEACA = 'E'
	
	GROUP BY 
	CONVERT(DATE,LMEDATHOR),
	LMELOCEST,
	PROCOD ) AS B ON A.DATA = B.DATA AND A.PROCOD = B.PROCOD AND A.LMELOCEST = B.LMELOCEST

LEFT JOIN

(
	SELECT 
	CONVERT(DATE,LMEDATHOR)AS DATA,
	LMELOCEST,
	PROCOD,
	SUM(LMEQTDMOV) AS LEMQTD
	
	FROM TBS051 A (NOLOCK) 
	
	WHERE
	LMEINFALT ='P' AND 
	CONVERT(DATE,LMEDATHOR) BETWEEN '20170101' AND CONVERT(DATE,getdate()-1) AND
	LMEACA = 'S'
	
	GROUP BY 
	CONVERT(DATE,LMEDATHOR),
	LMELOCEST,
	PROCOD ) AS C ON A.DATA = C.DATA AND A.PROCOD = C.PROCOD AND A.LMELOCEST = C.LMELOCEST

---------------------------------------------------------------------------------------------------------------------------------------------------------


if object_id('TempDB.dbo.#PEN1') is not null
   drop table #PEN1

SELECT 
IDENTITY(INT,1,1) 				AS ORDEM ,
ISNULL(A.ESTQTDPEN,0) 			AS SINQTD,
ISNULL(B.DATA,'17530101')		AS KESDAT,
ISNULL(B.LESCOD,A.LESCOD) 		AS LESCOD,
ISNULL(B.PROCOD,A.PROCOD) 		AS KESPROCOD,
ISNULL(B.LEMQTDENT,0) 			AS KESQTDENT,
ISNULL(B.LEMQTDSAI,0)			AS KESQTDSAI, 
(SELECT PROUM1 FROM TBS010 C (NOLOCK) WHERE ISNULL(A.PROCOD,B.PROCOD) = C.PROCOD) AS KESUNI

INTO #PEN1
FROM #INICIO A 
FULL OUTER JOIN #MOV B ON A.PROCOD = B.PROCOD AND A.LESCOD = B.LESCOD 

ORDER BY 
ISNULL(B.PROCOD,A.PROCOD), ISNULL(B.LESCOD,A.LESCOD), ISNULL(B.DATA,A.ESTDATSAL)

--------------------------------------------------------------------------------------------------------------------------------------------

-- ZERRANDO O SALDO INICIAL DAS DATAS QUE NÃO É INICIAL

--BEGIN TRAN 
UPDATE #PEN1 SET 
SINQTD = 0 
WHERE 
ORDEM NOT IN (SELECT MIN(ORDEM) FROM #PEN1 GROUP BY LESCOD, KESPROCOD)

--------------------------------------------------------------------------------------------------------------------------------------------

-- PEGANDO O SALDO DAS MOVIMENTAÇÕES

if object_id('TempDB.dbo.#PEN') is not null
   drop table #PEN

SELECT
*,
SINQTD + 
KESQTDENT-
-- SAIDAS
KESQTDSAI AS SALDO

INTO #PEN 
FROM #PEN1 A 
--WHERE KESPROCOD = '8420423'
--------------------------------------------------------------------------------------------------------------------------------------------

SELECT 
KESDAT AS KESDAT,
LESCOD,
KESPROCOD,
KESUNI,
(SELECT sum(SALDO) FROM #PEN B WHERE B.ORDEM <= A.ORDEM and A.KESPROCOD = B.KESPROCOD AND B.LESCOD = A.LESCOD) - SALDO+SINQTD  AS INICIO, 
KESQTDENT AS ENTRADAS,
KESQTDSAI AS SAIDAS,
(SELECT sum(SALDO) FROM #PEN B WHERE B.ORDEM <= A.ORDEM and A.KESPROCOD = B.KESPROCOD AND B.LESCOD = A.LESCOD) AS FINAL
	
FROM #PEN A

--WHERE KESPROCOD = '8420423'

ORDER BY 
KESPROCOD, KESDAT
