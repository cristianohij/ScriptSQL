if object_id('TempDB.dbo.#TBS010') is not null
   drop table #TBS010
SELECT PROCOD,PRODES,PROSTATUS,PROUM1,PROCODBAR1,REPLACE(STR(PROUM1QTD,10,4),'.',',') AS QTD1 INTO #TBS010 FROM TBS010 (nolock)


select 'ND' AS UNIDADE ,
P.PROCOD COLLATE Latin1_General_BIN AS COD ,
P.PRODES COLLATE Latin1_General_BIN AS DESCR,
P.PROSTATUS COLLATE Latin1_General_BIN AS ST,
P.PROUM1 COLLATE Latin1_General_CI_AS AS UM1,
QTD1,
P.PROCODBAR1 COLLATE Latin1_General_CI_AS AS BAR1,
DIFFERENCE(P.PRODES,C.PRODES) AS DIFE,
SOUNDEX(P.PRODES) AS SOUND,
ISNULL(REPLACE(STR(S.SAIDAS,10,4),'.',','),0) AS SAIDAS

FROM #TBS010 P (NOLOCK) FULL OUTER JOIN bb2.SIBD2.dbo.TBS010 C (NOLOCK) ON P.PROCOD COLLATE Latin1_General_CI_AS = C.PROCOD	

FULL OUTER JOIN  (SELECT PROCOD, SUM(C.NFSQTD*NFSQTDEMB) AS SAIDAS  
			FROM TBS0671 C (nolock) JOIN TBS067 B (nolock)  ON 
										C.NFSEMPCOD = B.NFSEMPCOD and 
										C.NFSNUM = B.NFSNUM and 
										C.SNEEMPCOD = B.SNEEMPCOD and 
										C.SNESER = B.SNESER
			WHERE NFSDATEMI between '20140101' and GETDATE () AND NFSCAN = 'N'
			GROUP BY PROCOD) AS S ON P.PROCOD = S.PROCOD
				  

WHERE P.PRODES <> C.PRODES and (P.PROSTATUS = 'A' OR C.PROSTATUS = 'A')
 
UNION 

select 'BB' AS UNIDADE,
C.PROCOD COLLATE Latin1_General_BIN AS COD ,
C.PRODES COLLATE Latin1_General_BIN AS DESCR,
C.PROSTATUS COLLATE Latin1_General_BIN AS ST,
C.PROUM1 COLLATE Latin1_General_CI_AS AS UM1,
REPLACE(STR(C.PROUM1QTD,10,4),'.',',') AS QTD1,
C.PROCODBAR1 COLLATE Latin1_General_CI_AS AS BAR1,
DIFFERENCE(P.PRODES,C.PRODES) AS DIFE,
SOUNDEX(C.PRODES) AS SOUND,
ISNULL(REPLACE(STR(S.SAIDAS,10,4),'.',','),0) AS SAIDAS

FROM #TBS010 P (NOLOCK) FULL OUTER JOIN bb2.SIBD2.dbo.TBS010 C (NOLOCK) ON P.PROCOD COLLATE Latin1_General_CI_AS = C.PROCOD

FULL OUTER JOIN (SELECT PROCOD, SUM(C.NFSQTD*NFSQTDEMB) AS SAIDAS  
			FROM bb2.SIBD2.dbo.TBS0671 C (nolock) JOIN bb2.SIBD2.dbo.TBS067 B (nolock)  ON 
										C.NFSEMPCOD = B.NFSEMPCOD and 
										C.NFSNUM = B.NFSNUM and 
										C.SNEEMPCOD = B.SNEEMPCOD and 
										C.SNESER = B.SNESER
			WHERE NFSDATEMI between '20140101' and GETDATE () AND NFSCAN = 'N'
			GROUP BY PROCOD) AS S ON C.PROCOD = S.PROCOD
				  
					  

WHERE P.PRODES <> C.PRODES and (P.PROSTATUS = 'A' OR C.PROSTATUS = 'A')
