   drop table #TMP
END
   
	select
	'5.929' AS M2_CFOP,	
	A.M2_TRI,
	A.M2_CST,
	A.M2_ITE,
	A.M2_LOJ, 
	A.M2_CUP, 
	A.M2_COO, 
	A.M2_ECF, 
	A.M2_CXA ,
	M2_PROCOD,
	M2_VALBRU,
	M2_VALBRU - M2_ABT + M2_ACRCUP - 
	CASE WHEN  A.DESCONTO <> ISNULL(B.DESCONTO,0) AND M2_ITE = (SELECT TOP 1 M2_ITE 
																FROM #MSLAUX E 
																
																WHERE 
																A.M2_COO = E.M2_COO and A.M2_ECF = E.M2_ECF AND A.M2_CUP = E.M2_CUP 
																
																ORDER BY 
																M2_DESCUP DESC)
		THEN
			CASE WHEN A.DESCONTO > ISNULL(B.DESCONTO,0) 
				THEN ROUND(M2_DESCUP,2) - (A.DESCONTO - ISNULL(B.DESCONTO,0))
				ELSE ROUND(M2_DESCUP,2) + (ISNULL(B.DESCONTO,0) - A.DESCONTO)
			END
		ELSE ROUND(M2_DESCUP,2)
	END AS M2_LIQUIDO
	
	INTO #TMP	
	FROM (
		SELECT 
		M2_ITE,M2_PROCOD,M2_TRI,M2_CST,
		M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , 
		M2_DESCUP ,
		M2_ACRCUP,
		M2_ABT ,
		M2_VALBRU,
		SUM(ISNULL(ROUND(M2_DESCUP,2),0)) OVER (PARTITION BY M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA ) AS DESCONTO
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 1 ) AS A 
			
		LEFT JOIN (
		SELECT 
		M2_LOJ, M2_CUP, M2_COO, M2_ECF, M2_CXA , M2_VALBRU AS DESCONTO 
					
		FROM 
		#MSLAUX 
					
		WHERE
		M2_STATUS = 2 ) AS B ON A.M2_LOJ = B.M2_LOJ AND A.M2_CUP = B.M2_CUP AND A.M2_COO= B.M2_COO AND 
														A.M2_ECF = B.M2_ECF AND A.M2_CXA = B.M2_CXA

   drop table #NFE

SELECT
A.NFECFOP AS CFOP,
SUM(ROUND(dbo.NFETOTITE(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE),2)) AS NFETOTITE,
SUM(ROUND(NFETOTOPEITE,2)) AS NFETOTOPEITE,
SUM(ROUND(NFEVALIPI,2)) AS NFEVALIPI,
SUM(ROUND(NFETOTOPEITE,2)) - SUM(ROUND(NFEVALIPI,2)) AS NFEBASCAL,
ROUND((SUM(ROUND(NFETOTOPEITE,2)) - SUM(ROUND(NFEVALIPI,2))) * 0.0165,2) AS PIS, -- TOTAL NOTA - IPI * 0.0165 = PIS
ROUND((SUM(ROUND(NFETOTOPEITE,2)) - SUM(ROUND(NFEVALIPI,2))) * 0.076,2)  AS COFINS

INTO #NFE
FROM TBS0591 A (NOLOCK)
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND 
A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND  A.SEREMPCOD = B.SEREMPCOD
LEFT JOIN TBS080 D (NOLOCK) ON A.NFENUM = D.ENFNUM AND A.NFECOD = D.ENFCODDES

WHERE 
((D.ENFSIT = 6 AND D.ENFFINEMI = 4 AND D.ENFTIPDOC = 0) OR ((NFENOSFOR = 'N' OR NFENOSFOR = '') AND (B.NFECAN = 'N' OR B.NFECAN = '') AND NFEDATEFE <> '' )) AND
B.NFEDATEFE BETWEEN convert(date,@ano+'/'+@mes+'/'+'01') AND @ultimoDiaMes AND  -- @Data_de AND @Data_ate AND
A.NFECFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E' AND PAGAR = 'N')  -- TIRAR 1.252 PARA NÃO DUPLICAR NO CONTAS A PAGAR '1.252',

GROUP BY 
A.NFECFOP, A.NFECST

ORDER BY 
A.NFECFOP, A.NFECST

SUM(round(dbo.NFDTOTITEST(A.NFDEMPCOD,A.SNEEMPCOD,A.SNESER,A.NFDNUM,A.NFDITE),2)) AS NFSBASCAL

INTO #NFD
FROM
TBS1172 A (NOLOCK)
LEFT JOIN TBS117 B (NOLOCK) ON A.NFDEMPCOD = B.NFDEMPCOD AND A.NFDNUM = B.NFDNUM AND A.SNEEMPCOD = B.SNEEMPCOD AND A.SNESER = B.SNESER
LEFT JOIN TBS080 D (NOLOCK) ON A.NFDNUM = D.ENFNUM AND A.SNEEMPCOD = D.SNEEMPCOD AND A.SNESER = D.SNESER

WHERE
B.NFDDATEMI BETWEEN convert(date,@ano+'/'+@mes+'/'+'01') AND @ultimoDiaMes and --- @Data_de AND @Data_ate AND 
D.ENFSIT = 6 AND
D.ENFFINEMI = 4 AND
D.ENFTIPDOC = 1 AND
A.NFDCFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'S' AND PAGAR = 'N')

GROUP BY 
A.NFDCFOP, A.NFDCSTCSOSN 

------------------------------------------------------------------------------------------------//-------------------------------------------------------------------

	-- AGRUPANDO A TABELA DE VENDAS, TENHO QUE AGRUPAR 1° AS VENDAS, PARA DEPOIS UNIR COM A TBL DE DEVOLUÇÃO
	
	if object_id('TempDB.dbo.#NFSPIS') is not null
	   drop table #NFSPIS
	
	SELECT 
	CFOP, 
	SUM(NFSTOTPRO) 						AS NFSTOTPRO, 		-- TOTAL DOS PRODUTOS
	SUM(NFSTOTITEST)					AS NFSTOTITEST, 	-- TOTAL DA NOTA
	SUM(NFSTOTITEST)					AS NFSBASCAL, 		-- BASE CALCULO
	ROUND(SUM(NFSTOTITEST)* 0.0165,2) 	AS PIS, 			-- VALOR PIS
	ROUND(SUM(NFSTOTITEST)* 0.076 ,2) 	AS COFINS			-- VALOR COFINS
	
	INTO #NFSPIS
	FROM #NFS
	
	GROUP BY 
	CFOP

-- APURAÇÃO DE ICMS DE ENTRADA 

if object_id('TempDB.dbo.#NFE') is not null
   drop table #NFE

SELECT
A.NFECFOP AS CFOP,
SUM(ROUND(dbo.NFETOTITE(A.NFEEMPCOD, A.NFETIP, A.NFENUM, A.NFECOD, A.SEREMPCOD, A.SERCOD, A.NFEITE),2)) AS NFETOTITE,
SUM(ROUND(NFETOTOPEITE,2)) AS NFETOTOPEITE,
SUM(ROUND(NFEVALIPI,2)) AS NFEVALIPI,
SUM(ROUND(NFETOTOPEITE,2)) - SUM(ROUND(NFEVALIPI,2)) AS NFEBASCAL,
ROUND((SUM(ROUND(NFETOTOPEITE,2)) - SUM(ROUND(NFEVALIPI,2))) * 0.0165,2) AS PIS, -- TOTAL NOTA - IPI * 0.0165 = PIS
ROUND((SUM(ROUND(NFETOTOPEITE,2)) - SUM(ROUND(NFEVALIPI,2))) * 0.076,2)  AS COFINS

INTO #NFE
FROM TBS0591 A (NOLOCK)
LEFT JOIN TBS059 B (NOLOCK) ON A.NFEEMPCOD = B.NFEEMPCOD AND A.NFECOD = B.NFECOD AND A.NFENUM = B.NFENUM AND 
A.NFETIP = B.NFETIP AND A.SERCOD = B.SERCOD AND  A.SEREMPCOD = B.SEREMPCOD
LEFT JOIN TBS080 D (NOLOCK) ON A.NFENUM = D.ENFNUM AND A.NFECOD = D.ENFCODDES

WHERE 
((D.ENFSIT = 6 AND D.ENFFINEMI = 4 AND D.ENFTIPDOC = 0) OR ((NFENOSFOR = 'N' OR NFENOSFOR = '') AND (B.NFECAN = 'N' OR B.NFECAN = '') AND NFEDATEFE <> '' )) AND
B.NFEDATEFE BETWEEN convert(date,@ano+'/'+@mes+'/'+'01') AND @ultimoDiaMes AND  -- @Data_de AND @Data_ate AND
A.NFECFOP COLLATE DATABASE_DEFAULT IN (SELECT CFOP FROM #CFOP WHERE TIPO = 'E' AND PAGAR = 'N')  -- TIRAR 1.252 PARA NÃO DUPLICAR NO CONTAS A PAGAR '1.252',

GROUP BY 
A.NFECFOP, A.NFECST

ORDER BY 
A.NFECFOP, A.NFECST
