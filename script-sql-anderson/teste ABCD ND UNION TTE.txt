   drop table #ABC

SELECT Identity(INT,1,1) as ordem,

rtrim(ltrim(A.PROCOD))  as 'CODIGO',
L.VND1 AS GZ_1,
L.VL1 AS VLGZ_1,
I.IVND1 AS IN_1,
I.IVL1 AS VLIN_1,
L.VND2 AS GZ_2,
L.VL2 AS VLGZ_2,
I.IVND2 AS IN_2,
I.IVL2 AS VLIN_2,
L.VND3 AS GZ_3,
L.VL3 AS VLGZ_3,
I.IVND3 AS IN_3,
I.IVL3 AS VLIN_3,
L.VND1 + L.VND2+ L.VND3+ 
I.IVND1 + I.IVND2 + I.IVND3 AS SOMAQTD,
I.IVL1 +I.IVL2 + I.IVL3 + L.VL1 + L.VL2 + L.VL3 AS SOMATT,
SUM(I.IVL1 +I.IVL2 + I.IVL3 + L.VL1 + L.VL2 + L.VL3) OVER (PARTITION BY 1) AS VALOR,
(I.IVL1 +I.IVL2 + I.IVL3 + L.VL1 + L.VL2 + L.VL3) / 
SUM(I.IVL1 +I.IVL2 + I.IVL3 + L.VL1 + L.VL2 + L.VL3) OVER (PARTITION BY 1) *100 AS PORC,

dbo.FCNDIASUTEIS (@data1de, @data1ate) - (SELECT count(*) FROM #FER
										  where (Sg_UF = 'SJ' or Tp_Feriado = 1 ) 
										  and (Nr_Ano between substring(convert(char(10),@data1de,103),7,4) and substring(convert(char(10),@data1ate,103),7,4) OR 
										  	  (Nr_Ano between 0 and 0))
										  and Nr_Mes between substring(convert(char(10),@data1de,103),4,2) and substring(convert(char(10),@data1ate,103),4,2)
										  and Nr_Dia between substring(convert(char(10),@data1de,103),1,2) and  substring(convert(char(10),@data1ate,103),1,2)) AS DIA1,

dbo.FCNDIASUTEIS (@data2de, @data2ate) - (SELECT count(*) FROM #FER
										  where (Sg_UF = 'SJ' or Tp_Feriado = 1 ) 
										  and (Nr_Ano between substring(convert(char(10),@data2de,103),7,4) and substring(convert(char(10),@data2ate,103),7,4) OR 
										  	  (Nr_Ano between 0 and 0))
										  and Nr_Mes between substring(convert(char(10),@data2de,103),4,2) and substring(convert(char(10),@data2ate,103),4,2)
										  and Nr_Dia between substring(convert(char(10),@data2de,103),1,2) and  substring(convert(char(10),@data2ate,103),1,2)) AS DIA2,

dbo.FCNDIASUTEIS (@data3de, @data3ate) - (SELECT count(*) FROM #FER
										  where (Sg_UF = 'SJ' or Tp_Feriado = 1 ) 
										  and (Nr_Ano between substring(convert(char(10),@data3de,103),7,4) and substring(convert(char(10),@data3ate,103),7,4) OR 
										  	  (Nr_Ano between 0 and 0))
										  and Nr_Mes between substring(convert(char(10),@data3de,103),4,2) and substring(convert(char(10),@data3ate,103),4,2)
										  and Nr_Dia between substring(convert(char(10),@data3de,103),1,2) and  substring(convert(char(10),@data3ate,103),1,2)) AS DIA3


into #ABC

FROM TBS010 A (NOLOCK)

FULL OUTER JOIN   
#LOJA L ON A.PROCOD = L.COD

FULL OUTER JOIN   
#CORP I ON A.PROCOD = I.COD

			
where ((ISNULL(L.VL1,0) + ISNULL(L.VL2,0) + ISNULL(L.VL3,0)) <> 0 OR ISNULL(L.VND1,0) + ISNULL(L.VND2,0) + ISNULL(L.VND3,0) <> 0 OR
 
	   (ISNULL(I.IVL1,0) + ISNULL(I.IVL2,0) + ISNULL(I.IVL3,0)) <> 0 OR (ISNULL(I.IVND1,0) + ISNULL(I.IVND2,0) + ISNULL(I.IVND3,0)) <> 0) AND PROCOD IS NOT NULL
	   					
		
ORDER BY (I.IVL1 +I.IVL2 + I.IVL3 + L.VL1 + L.VL2 + L.VL3) / 
SUM(I.IVL1 +I.IVL2 + I.IVL3 + L.VL1 + L.VL2 + L.VL3) OVER (PARTITION BY 1) *100 DESC 



-- CRIANDO ABC COM VALORES DETERMINADOS PELO USUARIO #ABCD , SENDO D SEM VENDAS 
If object_id('TempDB.dbo.#ABCD') is not null
   drop table #ABCD

SELECT
ordem, CODIGO,SOMATT, VALOR, PORC,SOMAQTD,
GZ_1, VLGZ_1,
IN_1,VLIN_1,
GZ_2,VLGZ_2,
IN_2,VLIN_2,
GZ_3,VLGZ_3,
IN_3,VLIN_3,
DIA1,
DIA2,
DIA3,
ROUND((SELECT SUM(TInt.PORC) FROM #ABC As TInt
        WHERE TInt.ordem <= TOut.ordem),4) As PercAcum,
CASE WHEN (ROUND((SELECT SUM(TInt.PORC) FROM #ABC As TInt
           WHERE TInt.ordem <= TOut.ordem),4)) < @pA 
		THEN 'A' 
		ELSE 
			CASE WHEN (ROUND((SELECT SUM(TInt.PORC) FROM #ABC As TInt
        			   WHERE TInt.ordem <= TOut.ordem),4)) > @pA AND
					  (ROUND((SELECT SUM(TInt.PORC) FROM #ABC As TInt
        			   WHERE TInt.ordem <= TOut.ordem),4)) < @pB
			THEN 'B' ELSE 'C' 
			END
		END AS ABC
INTO #ABCD
FROM #ABC As TOut

SELECT

RTRIM(LTRIM(A.PROSTATUS)) AS 'STA',

case when len(A.MARCOD) = 4 then rtrim(A.MARCOD) + ' - ' + rtrim(A.MARNOM) else right(('00' + ltrim(str(A.MARCOD))),3) + ' - ' + rtrim(A.MARNOM) end as 'MARCA',

ISNULL(REPLACE(STR(SALDO.LOJA,10,4),'.',','),0) AS'SALDO LOJA' ,

ISNULL(REPLACE(STR(SALDO.CMP,10,4),'.',','),0) AS COMPRAS,

CASE WHEN ISNULL((SOMAQTD / 
				((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0)> 0 
										  
	THEN str(round(((ISNULL(SALDO.SALTT,0)+ ISNULL(SALDO.CMP,0) ) - ISNULL(SALDO.PEN,0))/
	ISNULL((SOMAQTD / 
				((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0),0),10,0)
	ELSE 0 END AS 'TEMPO RESTANTE',

-- @DIAS - (SOMA AS QTDS VENDIDAS / PELA QUANTIDADE DE DIAS UTEIS, CASO >0 , 
-- SOMA SALDO TOTAL + COMPRAS + SALDO CD - PENDENTE / (QTDS VENDAS / PELA QUANTIDADE DE DIAS UTEIS)) * VEN_UTEIS CASO <=0.9 É 0 
-- ELSE @DIAS - (SOMA AS QTDS VENDIDAS / PELA QUANTIDADE DE DIAS UTEIS, CASO >0 , 
-- SOMA SALDO TOTAL + COMPRAS + SALDO CD - PENDENTE / (QTDS VENDAS / PELA QUANTIDADE DE DIAS UTEIS)) (

CASE WHEN ((@DIAS - (CASE WHEN ISNULL((SOMAQTD / 
					((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0) > 0 
										  
					THEN str(round(((ISNULL(SALDO.SALTT,0)+ ISNULL(SALDO.CMP,0)) - ISNULL(SALDO.PEN,0))/
					ISNULL((SOMAQTD / 
					((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0),0),10,0)
					ELSE 0 END)) * ISNULL((SOMAQTD / 
								((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0)) <=0.9 
	THEN 0 
	ELSE ((@DIAS - (CASE WHEN ISNULL((SOMAQTD / 
					((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0) > 0 
										  
					THEN str(round(((ISNULL(SALDO.SALTT,0)+ ISNULL(SALDO.CMP,0)) - ISNULL(SALDO.PEN,0))/
					ISNULL((SOMAQTD / 
					((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0),0),10,0)
					ELSE 0 END)) * ISNULL((SOMAQTD / 
								((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0))
					
END AS 'QTD PARA REPOR' ,

(SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0) AS DIASU1,
ISNULL(GZ_1,0) AS 'GZ1',
ISNULL(IN_1,0) AS 'INT1',
(ISNULL(GZ_1,0) + ISNULL(IN_1,0)) / (SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0) AS 'MEDIA POR DIA1' ,
(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0) AS DIASU2,

((SELECT TOP 1 DIA1 FROM #ABCD WHERE DIA1 > 0)+(SELECT TOP 1 DIA2 FROM #ABCD WHERE DIA2 > 0)+(SELECT TOP 1 DIA3 FROM #ABCD WHERE DIA3 > 0))),0) as VEN_UTEIS,
