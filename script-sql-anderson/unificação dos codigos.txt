if object_id('TempDB.dbo.#TBS010') is not null
   drop table #TBS010
SELECT PROCOD,PRODES,PROSTATUS,PROUM1,PROCODBAR1,REPLACE(STR(PROUM1QTD,10,4),'.',',') AS QTD1,PROCLAFIS INTO #TBS010 FROM TBS010 (nolock)

if object_id('TempDB.dbo.#EMIND') is not null
   drop table #EMIND
SELECT PROCOD, SUM(C.NFSQTD*NFSQTDEMB) AS SAIDAS , NFSDATEMI AS EMI 
			INTO #EMIND
			FROM TBS0671 C (nolock) JOIN TBS067 B (nolock)  ON 
										C.NFSEMPCOD = B.NFSEMPCOD and 
										C.NFSNUM = B.NFSNUM and 
										C.SNEEMPCOD = B.SNEEMPCOD and 
										C.SNESER = B.SNESER
			WHERE NFSDATEMI between '20140101' and GETDATE () AND NFSCAN = 'N' 
			GROUP BY PROCOD,NFSDATEMI
			
			ORDER BY NFSDATEMI DESC     
			
if object_id('TempDB.dbo.#EMIPP') is not null
   drop table #EMIPP			
SELECT PROCOD, SUM(C.NFSQTD*NFSQTDEMB) AS SAIDAS , NFSDATEMI AS EMI 
			INTO #EMIPP
			FROM py.SIBD.dbo.TBS0671 C (nolock) JOIN py.SIBD.dbo.TBS067 B (nolock)  ON
										C.NFSEMPCOD = B.NFSEMPCOD and 
										C.NFSNUM = B.NFSNUM and 
										C.SNEEMPCOD = B.SNEEMPCOD and 
										C.SNESER = B.SNESER
			WHERE NFSDATEMI between '20140101' and GETDATE () AND NFSCAN = 'N' 
			GROUP BY PROCOD,NFSDATEMI
			
			ORDER BY NFSDATEMI DESC    
			


if object_id('TempDB.dbo.#VNDPP') is not null
   drop table #VNDPP	
SELECT PROCOD, COUNT(B.NFSNUM) AS SAIDASPP
			into #VNDPP  
			FROM py.SIBD.dbo.TBS0671 C (nolock) JOIN py.SIBD.dbo.TBS067 B (nolock)  ON 
										C.NFSEMPCOD = B.NFSEMPCOD and 
										C.NFSNUM = B.NFSNUM and 
										C.SNEEMPCOD = B.SNEEMPCOD and 
										C.SNESER = B.SNESER
			WHERE NFSDATEMI between '20140101' and GETDATE () AND NFSCAN = 'N'
			GROUP BY PROCOD 
					

if object_id('TempDB.dbo.#VNDND') is not null
   drop table #VNDND
SELECT PROCOD, count(B.NFSNUM) AS SAIDAS
			INTO #VNDND  
			FROM TBS0671 C (nolock) JOIN TBS067 B (nolock)  ON 
										C.NFSEMPCOD = B.NFSEMPCOD and 
										C.NFSNUM = B.NFSNUM and 
										C.SNEEMPCOD = B.SNEEMPCOD and

			WHERE NFSDATEMI between '20140101' and GETDATE () AND NFSCAN = 'N'
			GROUP BY PROCOD			
						
			

select-- 'ND' AS UNIDADE ,
DIFFERENCE(RTRIM(LTRIM(P.PRODES)),RTRIM(LTRIM(C.PRODES))) AS DIFE,
P.PROCOD COLLATE Latin1_General_BIN AS COD ,
P.PRODES COLLATE Latin1_General_BIN AS 'DESCR',
C.PRODES COLLATE Latin1_General_BIN AS 'DESCR PP',
P.PROCLAFIS AS 'NCM ND',
C.PROCLAFIS AS 'NCM PP',
'' AS DEPARA,
ISNULL(REPLACE(STR(S.SAIDAS,10,4),'.',','),0) AS 'OCOR.',
ISNULL(REPLACE(STR(B.SAIDASPP,10,4),'.',','),0) AS 'OCOR.PP',
REPLACE(STR(ISNULL(S.SAIDAS,0) + ISNULL(B.SAIDASPP,0),10,4),'.',',') AS 'SOMA OCOR',
CONVERT(NVARCHAR,ISNULL((SELECT TOP 1 EMI FROM #EMIND A WHERE P.PROCOD = A.PROCOD),''),103) AS 'DATA',
CONVERT(NVARCHAR,ISNULL((SELECT TOP 1 EMI FROM #EMIPP A WHERE P.PROCOD = A.PROCOD),''),103) AS 'DATA PP',
CASE WHEN ISNULL((SELECT TOP 1 EMI FROM #EMIND A WHERE P.PROCOD = A.PROCOD),'') > ISNULL((SELECT TOP 1 EMI FROM #EMIPP A WHERE P.PROCOD = A.PROCOD),'')
THEN CONVERT(NVARCHAR,ISNULL((SELECT TOP 1 EMI FROM #EMIND A WHERE P.PROCOD = A.PROCOD),''),103) 
ELSE CONVERT(NVARCHAR,ISNULL((SELECT TOP 1 EMI FROM #EMIPP A WHERE P.PROCOD = A.PROCOD),''),103) END AS 'MAIOR DATA',
P.PROSTATUS COLLATE Latin1_General_BIN AS 'ST',
C.PROSTATUS COLLATE Latin1_General_BIN AS 'ST PP',
P.PROUM1 COLLATE Latin1_General_CI_AS AS 'UN1',
C.PROUM1 COLLATE Latin1_General_CI_AS AS 'UN1 PP',
QTD1,
REPLACE(STR(C.PROUM1QTD,10,4),'.',',') AS 'QTD1 PP',
P.PROCODBAR1 COLLATE Latin1_General_CI_AS AS BAR1,
C.PROCODBAR1 COLLATE Latin1_General_CI_AS AS 'BAR1 PP',
SOUNDEX(P.PRODES) AS 'SOUD',
SOUNDEX(C.PRODES) AS 'SOUD PP'

FROM #TBS010 P (NOLOCK) 

LEFT JOIN py.SIBD.dbo.TBS010 C (NOLOCK) ON P.PROCOD = C.PROCOD	

LEFT JOIN  #VNDND AS S ON P.PROCOD = S.PROCOD
			
LEFT JOIN #VNDPP AS B ON P.PROCOD = B.PROCOD

WHERE P.PRODES <> C.PRODES and (P.PROSTATUS = 'A' OR C.PROSTATUS = 'A')
AND (S.SAIDAS > 0 OR B.SAIDASPP > 0)
