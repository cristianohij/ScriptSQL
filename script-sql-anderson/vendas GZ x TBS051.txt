SELECT 
ROW_NUMBER ( ) OVER (ORDER BY A.M2_PROCOD) AS [RANK],
A.M2_PROCOD AS COD,
M2_CODBARDIG, 
RTRIM(LTRIM(PRODES)) AS DESCRIÇÃO,

REPLACE(STR(ISNULL(SUM(M2_QTD),0),10,2),'.',',') AS QTD,
REPLACE(STR(SUM(M2_VALTOT)-SUM(M2_ABT),10,3),'.',',') AS VALOR,
M2_CXA AS CAIXA,
CONVERT(VARCHAR(10),M2_DAT,3)AS 'DATA',
M2_HOR AS HORA,
M2_CONCUPFIS as CONTADORCUPOM, 
M2_NUMDOC AS CUPOM

FROM MSL002 A (nolock) INNER JOIN TBS010 B ON A.M2_PROCOD = B.PROCOD 
WHERE M2_DAT between '20160801' and GETDATE() 
and M2_TIPREG = '01' 
AND M2_REGCAN = 'F'  
-- AND M2_NUMDOC = '83833'
AND M2_PROCOD IN( '14530054','14530055') AND M2_NUMDOC = 95335

 GROUP BY M2_PROCOD,M2_CXA,M2_HOR,M2_DAT,M2_CONCUPFIS,PRODES,M2_NUMDOC,M2_CODBARDIG
 
 ORDER BY M2_DAT  , M2_HOR

SELECT 
M2_PROCOD AS COD,
M2_NUMDOC ,
B.LMEDOC, 
SUM(M2_QTD) AS QTDSAILOJ ,
SUM( B.LMEQTDMOV) as MOVTBS051,
CASE WHEN M2_NUMDOC <> B.LMEDOC 
	THEN 'DOC. ERRADO'
	ELSE 'DOC. OK'
END,
CASE WHEN SUM(M2_QTD) <> SUM( B.LMEQTDMOV)
	THEN 'QTD. ERRADO'
	ELSE 'QTD. OK'
END

FROM MSL002 A (nolock)
LEFT JOIN TBS051 B (NOLOCK) ON A.M2_PROCOD = B.PROCOD AND A.M2_NUMDOC = B.LMEDOC

WHERE 
M2_DAT between '20160801' and GETDATE() and 
LMEDATHOR between '20160801' and GETDATE() and 
M2_TIPREG = '01' AND 
M2_REGCAN = 'F' AND
M2_PROCOD IN('14530054','14530055')

GROUP BY 
M2_PROCOD, M2_NUMDOC,B.LMEDOC
