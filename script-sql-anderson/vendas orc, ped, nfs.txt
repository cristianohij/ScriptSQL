-- ORÇAMENTO

if object_id('TempDB.dbo.#TBS043') is not null 
	drop table #TBS043

SELECT
A.ORCCLI AS CLICOD,
A.ORCNOM AS CLINOM,		
A.VENCOD,
VENNOM,
ROW_NUMBER() OVER (Partition by ORCNUM order by ORCNUM) AS 'QTD',
CONVERT(CHAR(12),A.ORCDATCAD,103) AS 'DATA',
ORCNUM as 'NUMERO',
CASE WHEN ORCULTITE > 0
	THEN ROUND(dbo.ORCTOTLIQ(ORCEMPCOD, ORCNUM),2)
    ELSE 0
END as 'VALOR',
SUM(CASE WHEN ORCULTITE > 0
	THEN ROUND(dbo.ORCTOTLIQ(ORCEMPCOD, ORCNUM),2)
    ELSE 0
END) OVER (PARTITION BY A.VENCOD ) AS VALTOT,
SUM(CASE WHEN ORCULTITE > 0
	THEN ROUND(dbo.ORCTOTLIQ(ORCEMPCOD, ORCNUM),2)
    ELSE 0
END) OVER (PARTITION BY 1 ) AS TOTGERAL,
A.ORCPDVNUM AS PDVNUM

INTO #TBS043								
FROM TBS043 A (NOLOCK)
INNER JOIN #CLI B ON A.VENCOD = B.VENCOD

WHERE 
ORCDATCAD between @datai and @dataf AND
ORCULTITE > 0 AND
A.ORCCLI NOT IN ((SELECT CLICOD FROM #CNPJ))
