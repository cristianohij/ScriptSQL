-- verifica a qtd de itens que saiu de algum lugar de são paulo e deu entrada aqui na ND em qtd de itens diferente
SELECT NFS.NFS 


FROM
(SELECT A.NFSNUM AS NFS,
				   COUNT(A.PROCOD) AS QTDITENS
				   
                   
				   FROM  py.SIBD.dbo.TBS0671 AS A (NOLOCK) 
				   					INNER JOIN py.SIBD.dbo.TBS067 AS B (NOLOCK) ON A.NFSEMPCOD = B.NFSEMPCOD AND A.NFSNUM = B.NFSNUM AND 
																				  A.SNEEMPCOD = B.SNEEMPCOD AND A.SNESER = B.SNESER 
									INNER JOIN py.SIBD.dbo.TBS080 AS D (NOLOCK) ON D.SNEEMPCOD = A.SNEEMPCOD AND D.SNESER = A.SNESER AND D.SNEEMPCOD = B.SNEEMPCOD AND 
																				  D.SNESER = B.SNESER AND B.NFSNUM = D.ENFNUM AND D.ENFNUM = A.NFSNUM
                   
				   WHERE (D.ENFCNPJCPF = '65069593000198') AND (D.ENFSIT = '6')
				  
				   GROUP BY A.PROCOD,A.NFSNUM)AS NFS 
				   
				  LEFT JOIN(SELECT B.NFENUM AS NFE, 
				 			COUNT(C.PROCOD) AS QTDITENS
                  
				  FROM TBS0591 AS C (NOLOCK) 
				  					LEFT JOIN TBS059 AS B (NOLOCK) ON B.NFECOD = C.NFECOD AND B.NFEEMPCOD = C.NFEEMPCOD AND B.NFENUM = C.NFENUM AND 
																	  B.NFETIP = C.NFETIP AND B.SERCOD = C.SERCOD AND B.SEREMPCOD = C.SEREMPCOD 
									LEFT JOIN TBS006 AS A (NOLOCK) ON A.FORNOM = B.NFENOM
                  
				  WHERE  (B.NFECAN = 'N') AND  NFENOM LIKE('%PAPELYNA%') AND
				  		 ((A.FORCGC = '44125185000136') OR A.FORNOM IS NULL)
						   GROUP BY B.NFENUM , C.PROCOD ) AS NFE ON  NFE.NFE = NFS.NFS
						  
WHERE (NFE.QTDITENS <> NFS.QTDITENS)
