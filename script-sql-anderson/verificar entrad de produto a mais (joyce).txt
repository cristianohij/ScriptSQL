select B.NFEDATENT,A.NFENUM,A.PROCOD,NFEDES,REPLACE(STR(NFEQTD,10,4),'.',',') AS QTDNF ,NFEUNI,REPLACE(STR(NFEQTDEMB,10,4),'.',',') AS QTDEMB,
REPLACE(STR(NFEQTD*NFEQTDEMB,10,4),'.',',') AS QTDENT 
 FROM TBS0591 A (NOLOCK) LEFT JOIN TBS010 C (NOLOCK) ON A.PROCOD = C.PROCOD
						FULL OUTER JOIN  TBS059 B(nolock)  on A.SERCOD=B.SERCOD and A.NFETIP=B.NFETIP and A.NFECOD=B.NFECOD and A.NFENUM=B.NFENUM
	  					
WHERE C.PROUM1 = A.NFEUNI AND NFEQTDEMB > 1 AND NFEDATENT > '20151201' AND NFECAN = 'N'
